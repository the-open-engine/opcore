1use std::{collections::BTreeMap, sync::Arc};
2
3use crate::{
4    identity::{ContentViewId, hash_domain},
5    model::SourceFile,
6    path::RepoPath,
7};
8
9pub struct SourceSnapshot {
10    id: ContentViewId,
11    files: BTreeMap<RepoPath, Arc<SourceFile>>,
12}
13
14impl SourceSnapshot {
15    #[must_use]
16    pub fn new(files: impl IntoIterator<Item = SourceFile>) -> Self {
17        let files: BTreeMap<_, _> = files
18            .into_iter()
19            .map(|file| (file.path.clone(), Arc::new(file)))
20            .collect();
21        let mut manifest = Vec::new();
22        for (path, file) in &files {
23            manifest.extend_from_slice(&(path.as_bytes().len() as u64).to_be_bytes());
24            manifest.extend_from_slice(path.as_bytes());
25            let language = match file.language {
26                crate::model::Language::JavaScript => b"javascript".as_slice(),
27                crate::model::Language::TypeScript => b"typescript".as_slice(),
28                crate::model::Language::Rust => b"rust".as_slice(),
29                crate::model::Language::Python => b"python".as_slice(),
30                crate::model::Language::Go => b"go".as_slice(),
31                crate::model::Language::Hcl => b"hcl".as_slice(),
32                crate::model::Language::Shell => b"shell".as_slice(),
33                crate::model::Language::Protobuf => b"protobuf".as_slice(),
34            };
35            manifest.extend_from_slice(&(language.len() as u64).to_be_bytes());
36            manifest.extend_from_slice(language);
37            manifest.extend_from_slice(&(file.language_mode.len() as u64).to_be_bytes());
38            manifest.extend_from_slice(file.language_mode.as_bytes());
39            manifest.extend_from_slice(file.content_id.as_bytes());
40        }
41        Self {
42            id: ContentViewId::from_bytes(hash_domain("source-snapshot/v2", &[&manifest])),
43            files,
44        }
45    }
46
47    #[must_use]
48    pub const fn id(&self) -> ContentViewId {
49        self.id
50    }
51
52    pub fn paths(&self) -> impl Iterator<Item = &RepoPath> {
53        self.files.keys()
54    }
55
56    pub fn files(&self) -> impl Iterator<Item = &SourceFile> {
57        self.files.values().map(AsRef::as_ref)
58    }
59
60    #[must_use]
61    pub fn len(&self) -> usize {
62        self.files.len()
63    }
64
65    #[must_use]
66    pub fn is_empty(&self) -> bool {
67        self.files.is_empty()
68    }
69
70    #[must_use]
71    pub fn read(&self, path: &RepoPath) -> Option<Arc<SourceFile>> {
72        self.files.get(path).cloned()
73    }
74}
