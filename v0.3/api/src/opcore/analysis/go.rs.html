20/// Analyzes one Go source file without invoking the Go toolchain or executing repository code.
21///
22/// # Errors
23///
24/// Returns an error when the language mode is unsupported, analysis is cancelled, or a bounded
25/// parser-safety preflight is exceeded.
26pub fn analyze(
27    file: &SourceFile,
28    limits: &RuleLimits,
29    cancel: &CancelToken,
30) -> Result<FileFacts, AnalysisError> {
31    walk::check_cancel(cancel)?;
32    let target = input::validate_language_mode(file)?;
33    let (source, byte_base) = input::source_without_bom(&file.bytes);
34    let lines = input::LineIndex::new(source, byte_base);
35    let text = match std::str::from_utf8(source) {
36        Ok(text) => text,
37        Err(error) => return Ok(input::invalid_utf8_facts(source, &lines, error, target)),
38    };
39    analyze_text(AnalysisContext {
40        file,
41        text,
42        source,
43        lines: &lines,
44        target,
45        limits,
46        cancel,
47    })
48}

50#[derive(Clone, Copy)]
51struct AnalysisContext<'a> {
52    file: &'a SourceFile,
53    text: &'a str,
54    source: &'a [u8],
55    lines: &'a input::LineIndex,
56    target: input::GoTarget<'a>,
57    limits: &'a RuleLimits,
58    cancel: &'a CancelToken,
59}
60
61fn analyze_text(context: AnalysisContext<'_>) -> Result<FileFacts, AnalysisError> {
62    let tokens = input::scan_tokens(context.text, context.cancel)?;
63    let tree = input::parse(context.text, context.cancel)?;
64    let root = tree.root_node();
65    let syntax = input::syntax_diagnostics(
66        root,
67        context.source,
68        context.lines,
69        context.target,
70        context.cancel,
71    )?;
72    if !syntax.is_empty() {
73        return Ok(input::parser_failed_facts(syntax));
74    }
75    analyze_tree(root, &tokens, &context)
76}
77
78fn analyze_tree(
79    root: Node<'_>,
80    tokens: &[input::GoToken],
81    context: &AnalysisContext<'_>,
82) -> Result<FileFacts, AnalysisError> {
83    let metric_facts = metrics::extract(
84        root,
85        metrics::MetricInput {
86            source: context.source,
87            tokens,
88            lines: context.lines,
89            limits: context.limits,
90            cancel: context.cancel,
91        },
92    )?;
93    let build_status = input::build_status(context.file, context.text, context.target);
94    let (dependencies, mut interfaces) =
95        module_facts(build_status, root, context.source, tokens, context.cancel)?;
96    if context.target.test {
97        interfaces.exports.clear();
98        interfaces.shapes.clear();
99    }
100    Ok(FileFacts {
101        diagnostics: metric_facts.diagnostics,
102        parser: PARSER_NAME.into(),
103        parser_version: GO_GRAMMAR_VERSION.into(),
104        dependencies,
105        callable_fingerprints: metric_facts.callables,
106        region_fingerprints: metric_facts.regions,
107        interfaces,
108    })
109}
110
111fn module_facts(
112    status: BuildStatus,
113    root: Node<'_>,
114    source: &[u8],
115    tokens: &[input::GoToken],
116    cancel: &CancelToken,
117) -> Result<(DependencyFacts, InterfaceFacts), AnalysisError> {
118    match status {
119        BuildStatus::Active => interfaces::extract(root, source, tokens, cancel),
120        BuildStatus::Inactive => Ok((
121            interfaces::complete_dependencies(Vec::new()),
122            interfaces::complete_interfaces(),
123        )),
124        BuildStatus::Conditional => Ok(interfaces::conditional_facts()),
125    }
126}

128fn node_text<'a>(node: Node<'_>, source: &'a [u8]) -> Option<&'a str> {
129    std::str::from_utf8(source.get(node.start_byte()..node.end_byte())?).ok()
130}

141    #[test]
142    fn parses_a_minimal_file() {
143        let path = RepoPath::from_protocol("main.go").unwrap();
144        let (language, mode) = detect_language(&path).unwrap();
145        let source = SourceFile::new(
146            path,
147            b"package main\nfunc main() {}\n".as_slice(),
148            language,
149            mode,
150        );
151        let facts = analyze(&source, &RuleLimits::default(), &CancelToken::new()).unwrap();
152        assert!(facts.diagnostics.is_empty());
153        assert_eq!(facts.parser, PARSER_NAME);
154        assert_eq!(
155            facts.dependencies.status,
156            DependencyExtractionStatus::Complete
157        );
158        assert_eq!(
159            facts.callable_fingerprints.status,
160            FingerprintExtractionStatus::Complete
161        );
