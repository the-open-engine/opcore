1//! Strict, bounded ownership declarations for source documentation.
2
3use std::collections::BTreeSet;
4
5use serde::{Deserialize, Serialize};
6use thiserror::Error;
7
8use crate::{json::parse_unique_json, path::RepoPath};
9
10pub const DOCUMENTATION_REGISTRY_PATH: &str = crate::policy::POLICY_PATH;
11pub const DOCUMENTATION_REGISTRY_SCHEMA_VERSION: u32 = 1;
12pub const MAX_DOCUMENTATION_REGISTRY_BYTES: usize = 256 * 1024;
13pub const MAX_DOCUMENTATION_BINDINGS: usize = 4_096;
14
15#[derive(Clone, Debug, Deserialize, Eq, Ord, PartialEq, PartialOrd, Serialize)]
16#[serde(deny_unknown_fields, rename_all = "camelCase")]
17pub struct DocumentationBinding {
18    pub source: RepoPath,
19    pub document: RepoPath,
20}
21
22#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
23#[serde(rename_all = "camelCase")]
24pub struct DocumentationRegistry {
25    pub schema_version: u32,
26    pub bindings: Vec<DocumentationBinding>,
27}
28
29impl DocumentationRegistry {
30    #[must_use]
31    pub fn document_for(&self, source: &RepoPath) -> Option<&RepoPath> {
32        self.bindings
33            .binary_search_by(|binding| binding.source.cmp(source))
34            .ok()
35            .map(|index| &self.bindings[index].document)
36    }
37}
38
39#[derive(Debug, Eq, Error, PartialEq)]
40pub enum DocumentationRegistryError {
41    #[error("documentation registry exceeds {MAX_DOCUMENTATION_REGISTRY_BYTES} bytes")]
42    TooLarge,
43    #[error("documentation registry is malformed: {0}")]
44    Malformed(String),
45    #[error("documentation registry has more than {MAX_DOCUMENTATION_BINDINGS} bindings")]
46    TooManyBindings,
47    #[error("documentation registry contains duplicate source path {0}")]
48    DuplicateSource(RepoPath),
49    #[error("documentation binding source and document must differ: {0}")]
50    SelfBinding(RepoPath),
51    #[error("documentation binding cannot use the registry as its document")]
52    RegistryAsDocument,
53}
54
55pub fn parse_optional_documentation_registry(
56    bytes: Option<&[u8]>,
57) -> Result<Option<DocumentationRegistry>, DocumentationRegistryError> {
58    let Some(bytes) = bytes else {
59        return Ok(None);
60    };
61    if bytes.len() > MAX_DOCUMENTATION_REGISTRY_BYTES {
62        return Err(DocumentationRegistryError::TooLarge);
63    }
64    let value = parse_unique_json(bytes)
65        .map_err(|error| DocumentationRegistryError::Malformed(error.to_string()))?;
66    if value
67        .get("schemaVersion")
68        .and_then(serde_json::Value::as_u64)
69        != Some(u64::from(DOCUMENTATION_REGISTRY_SCHEMA_VERSION))
70    {
71        return Err(DocumentationRegistryError::Malformed(format!(
72            "schemaVersion must be {DOCUMENTATION_REGISTRY_SCHEMA_VERSION}"
73        )));
74    }
75    let Some(documentation) = value.get("documentation") else {
76        return Ok(None);
77    };
78    let mut settings: crate::policy::DocumentationSettings =
79        serde_json::from_value(documentation.clone()).map_err(|error| {
80            DocumentationRegistryError::Malformed(format!("documentation: {error}"))
81        })?;
82    validate_bindings(&settings.bindings)?;
83    settings.bindings.sort();
84    Ok(Some(DocumentationRegistry {
85        schema_version: DOCUMENTATION_REGISTRY_SCHEMA_VERSION,
86        bindings: settings.bindings,
87    }))
88}

90#[cfg(test)]
91pub fn parse_documentation_registry(
92    bytes: &[u8],
93) -> Result<DocumentationRegistry, DocumentationRegistryError> {
94    Ok(
95        parse_optional_documentation_registry(Some(bytes))?.unwrap_or(DocumentationRegistry {
96            schema_version: DOCUMENTATION_REGISTRY_SCHEMA_VERSION,
97            bindings: Vec::new(),
98        }),
99    )
100}
101
102pub fn validate_bindings(
103    bindings: &[DocumentationBinding],
104) -> Result<(), DocumentationRegistryError> {
105    if bindings.len() > MAX_DOCUMENTATION_BINDINGS {
106        return Err(DocumentationRegistryError::TooManyBindings);
107    }
108    let mut seen = BTreeSet::new();
109    for binding in bindings {
110        if !seen.insert(&binding.source) {
111            return Err(DocumentationRegistryError::DuplicateSource(
112                binding.source.clone(),
113            ));
114        }
115        if binding.source == binding.document {
116            return Err(DocumentationRegistryError::SelfBinding(
117                binding.source.clone(),
118            ));
119        }
120        if binding.document.as_utf8() == Some(DOCUMENTATION_REGISTRY_PATH) {
121            return Err(DocumentationRegistryError::RegistryAsDocument);
122        }
123    }
124    Ok(())
125}

127#[cfg(test)]
128mod tests {
129    use super::*;
130
131    fn registry(bindings: &str) -> Vec<u8> {
132        format!(r#"{{"schemaVersion":1,"documentation":{{"bindings":[{bindings}]}}}}"#).into_bytes()
133    }
134
135    #[test]
136    fn absent_registry_is_distinct_from_an_empty_registry() {
137        assert_eq!(parse_optional_documentation_registry(None).unwrap(), None);
138        assert_eq!(
139            parse_optional_documentation_registry(Some(&registry("")))
140                .unwrap()
141                .unwrap()
142                .bindings,
143            Vec::new()
144        );
145    }
146
147    #[test]
148    fn parses_exact_bindings_in_deterministic_source_order() {
149        let parsed = parse_documentation_registry(&registry(
150            r#"{"source":"src/z.ts","document":"docs/z.md"},{"source":"src/a.ts","document":"docs/a.md"}"#,
151        ))
152        .unwrap();
153        assert_eq!(parsed.schema_version, 1);
154        assert_eq!(parsed.bindings[0].source.as_utf8(), Some("src/a.ts"));
155        assert_eq!(
156            parsed
157                .document_for(&RepoPath::from_protocol("src/z.ts").unwrap())
158                .and_then(RepoPath::as_utf8),
159            Some("docs/z.md")
160        );
161    }
162
163    #[test]
164    fn rejects_duplicate_keys_unknown_fields_and_duplicate_sources() {
165        for malformed in [
166            br#"{"schemaVersion":1,"schemaVersion":1,"bindings":[]}"#.as_slice(),
167            br#"{"schemaVersion":1,"documentation":{"bindings":[],"unknown":true}}"#.as_slice(),
168            br#"{"schemaVersion":1,"documentation":{"bindings":[{
169                "source":"src/a.ts","document":"docs/a.md","unknown":true
170            }]}}"#
171                .as_slice(),
172        ] {
173            assert!(matches!(
174                parse_documentation_registry(malformed),
175                Err(DocumentationRegistryError::Malformed(_))
176            ));
177        }
178        let duplicate = registry(
179            r#"{"source":"src/a.ts","document":"docs/a.md"},{"source":"src/a.ts","document":"docs/other.md"}"#,
180        );
181        assert!(matches!(
182            parse_documentation_registry(&duplicate),
183            Err(DocumentationRegistryError::DuplicateSource(_))
184        ));
185    }
186
187    #[test]
188    fn rejects_self_binding_and_registry_as_document() {
189        assert!(matches!(
190            parse_documentation_registry(&registry(
191                r#"{"source":"src/a.ts","document":"src/a.ts"}"#
192            )),
193            Err(DocumentationRegistryError::SelfBinding(_))
194        ));
195        assert!(
196            parse_documentation_registry(&registry(
197                r#"{"source":"src/a.ts","document":".opcore.json"}"#
198            ))
199            .is_err()
200        );
201    }

203    #[test]
204    fn rejects_malformed_versions_paths_and_json() {
205        assert!(parse_documentation_registry(br#"{"schemaVersion":2}"#).is_err());
206        for invalid in ["../src/a.ts", "/src/a.ts", "src//a.ts", "src\\a.ts"] {
207            let bytes = registry(&format!(
208                r#"{{"source":{},"document":"docs/a.md"}}"#,
209                serde_json::to_string(invalid).unwrap()
210            ));
211            assert!(matches!(
212                parse_documentation_registry(&bytes),
213                Err(DocumentationRegistryError::Malformed(_))
214            ));
215        }
216        assert!(matches!(
217            parse_documentation_registry(b"{"),
218            Err(DocumentationRegistryError::Malformed(_))
219        ));
220    }

222    #[test]
223    fn treats_pattern_metacharacters_as_literal_git_path_bytes() {
224        let parsed = parse_documentation_registry(&registry(
225            r#"{"source":"src/[literal]*?.ts","document":"docs/[literal]*?.md"}"#,

234    #[test]
235    fn enforces_byte_and_binding_caps() {
236        assert_eq!(
237            parse_documentation_registry(&vec![b' '; MAX_DOCUMENTATION_REGISTRY_BYTES + 1]),
238            Err(DocumentationRegistryError::TooLarge)
239        );
240        let bindings = (0..=MAX_DOCUMENTATION_BINDINGS)
241            .map(|index| format!(r#"{{"source":"s/{index}","document":"d/{index}"}}"#))
242            .collect::<Vec<_>>()
243            .join(",");
244        assert!(parse_documentation_registry(&registry(&bindings)).is_err());
245    }
