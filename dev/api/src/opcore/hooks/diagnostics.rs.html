13pub(crate) fn diagnose(agent: Agent) -> Result<Option<String>> {
14    let Some(root) = agent_root(agent)? else {
15        return Ok(None);
16    };
17    let args = inspection_args(agent, &root)?;
18    if !inspect_installation(&args).with_context(|| {
19        format!(
20            "inspect hook setup at {}; repair agent settings and rerun installation",
21            args.config.display()
22        )
23    })? {
24        return Ok(None);
25    }
26    let next = match agent {
27        Agent::Codex => {
28            "restart Codex, review and trust the command in /hooks, then perform a smoke edit"
29        }
30        Agent::Claude => "restart Claude, inspect /hooks, then perform a smoke edit",
31    };
32    Ok(Some(format!(
33        "owned PostToolUse configuration matches {}; host trust and live execution are unverified: {next}",
34        args.config.display()
35    )))
36}

38fn inspect_installation(args: &ConfigureHookArgs) -> Result<bool> {
39    let receipt = read_receipt(&args.receipt)?;
40    let enrollment = installed_enrollment(&args.receipt.with_file_name("install.receipt"))?;
41    let Some(receipt) = receipt else {
42        ensure!(
43            !matches!(enrollment.as_deref(), Some("yes" | "pending")),
44            "installed hook ownership receipt is missing at {}; rerun installation to repair it",
45            args.receipt.display()
46        );
47        return Ok(false);
48    };
49    ensure!(
50        enrollment.as_deref() != Some("no"),
51        "hook receipt conflicts with disabled installation enrollment; rerun installation"
52    );
53    validate_receipt(&receipt, args)?;
54    let owned = receipt_entries(&receipt, args)?;
55    let (config, exists) = read_config(&args.config)?;
56    ensure!(
57        exists,
58        "installed hook config is missing at {}; rerun installation",
59        args.config.display()
60    );
61    verify_owned_entries(&config, &owned)?;
62    ensure!(
63        enrollment.as_deref() != Some("pending"),
64        "hook installation is pending; rerun installation to finish enrollment"
65    );
66    Ok(true)
67}

69fn agent_root(agent: Agent) -> Result<Option<PathBuf>> {
70    let (variable, directory) = match agent {
71        Agent::Codex => ("CODEX_HOME", ".codex"),
72        Agent::Claude => ("CLAUDE_CONFIG_DIR", ".claude"),
73    };
74    let root = env::var_os(variable)
75        .filter(|value| !value.is_empty())
76        .map(PathBuf::from)
77        .or_else(|| env::var_os("HOME").map(|home| PathBuf::from(home).join(directory)));
78    if let Some(root) = &root {
79        ensure!(
80            root.is_absolute(),
81            "{variable} must resolve to an absolute agent directory"
82        );
83    }
84    Ok(root)
85}

87fn inspection_args(agent: Agent, root: &std::path::Path) -> Result<ConfigureHookArgs> {
88    Ok(ConfigureHookArgs {
89        agent,
90        config: root.join(match agent {
91            Agent::Codex => "hooks.json",
92            Agent::Claude => "settings.json",
93        }),
94        binary: env::current_exe().context("locate running Opcore binary")?,
95        receipt: root.join("opcore/hook-install.json"),
96        remove: false,
97        check: true,
98    })
99}

101fn installed_enrollment(path: &std::path::Path) -> Result<Option<String>> {
102    let Some(bytes) = read_optional_bytes(path)? else {
103        return Ok(None);
104    };
105    let values = bytes
106        .split(|byte| *byte == b'\n')
107        .filter_map(|line| line.strip_prefix(b"hooks "))
108        .collect::<Vec<_>>();
109    if values.is_empty() && legacy_install_receipt(&bytes) {
110        return Ok(None);
111    }
112    ensure!(
113        values.len() == 1 && matches!(values[0], b"yes" | b"no" | b"pending"),
114        "cannot read hook enrollment from {}; rerun installation to repair its receipt",
115        path.display()
116    );
117    Ok(Some(String::from_utf8_lossy(values[0]).into_owned()))
118}
119
120fn legacy_install_receipt(bytes: &[u8]) -> bool {
121    matches!(
122        bytes.split(|byte| *byte == b'\n').next(),
123        Some(
124            b"opcore.install.v1"
125                | b"opcore.install.v2"
126                | b"opcore.install.v3"
127                | b"opcore.install.v4"
128        )
129    )
130}

132fn verify_owned_entries(config: &Value, owned: &[OwnedHook]) -> Result<()> {
133    let groups = config
134        .get("hooks")
135        .and_then(|hooks| hooks.get("PostToolUse"))
136        .and_then(Value::as_array)
137        .context(
138            "installed PostToolUse hook configuration is missing or malformed; rerun installation",
139        )?;
140    for entry in owned {
141        let count = groups
142            .iter()
143            .filter(|group| group.get("matcher").and_then(Value::as_str) == Some(&entry.matcher))
144            .filter_map(|group| group.get("hooks").and_then(Value::as_array))
145            .flatten()
146            .filter(|configured| **configured == entry.entry)
147            .count();
148        ensure!(
149            count == 1,
150            "owned PostToolUse hook is missing, modified, or duplicated; inspect agent /hooks and rerun installation"
151        );
152    }
153    Ok(())
154}
