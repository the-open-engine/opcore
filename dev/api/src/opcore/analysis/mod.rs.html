90pub(crate) fn invalid_utf8_diagnostic(input: InvalidUtf8Diagnostic<'_>) -> RawDiagnostic {
91    let start = input.error.valid_up_to();
92    let end = start
93        .saturating_add(input.error.error_len().unwrap_or(1))
94        .min(input.source.len());
95    RawDiagnostic {
96        rule_id: input.rule_id.into(),
97        severity: Severity::Error,
98        message: format!("{} source is not valid UTF-8", input.language),
99        range: Some(input.lines.byte_range(start, end)),
100        entity_key: format!("syntax:invalid-utf8:{}", line_anchor(input.source, start)),
101        cause_key: "invalid_utf8".into(),
102        evidence: input.evidence,
103    }
104}

106pub(crate) fn source_without_utf8_bom(bytes: &[u8]) -> (&[u8], usize) {

118pub(crate) fn line_anchor(bytes: &[u8], offset: usize) -> String {
119    let offset = offset.min(bytes.len());
120    let mut start = bytes[..offset]
121        .iter()
122        .rposition(|byte| matches!(byte, b'\n' | b'\r'))
123        .map_or(0, |index| index.saturating_add(1));
124    let mut end = bytes[offset..]
125        .iter()
126        .position(|byte| matches!(byte, b'\n' | b'\r'))
127        .map_or(bytes.len(), |index| offset.saturating_add(index));
128    let mut line = trim_ascii_whitespace(
129        bytes[start..end]
130            .strip_prefix(b"\xef\xbb\xbf")
131            .unwrap_or(&bytes[start..end]),
132    );
133    while line.is_empty() && start > 0 {
134        end = start;
135        while end > 0 && matches!(bytes[end - 1], b'\n' | b'\r') {
136            end -= 1;
137        }
138        start = bytes[..end]
139            .iter()
140            .rposition(|byte| matches!(byte, b'\n' | b'\r'))
141            .map_or(0, |index| index.saturating_add(1));
142        line = trim_ascii_whitespace(&bytes[start..end]);
143    }
144    hex::encode(&crate::identity::hash_domain("source-line-anchor/v1", &[line])[..12])
145}

147pub(crate) fn next_char_end(source: &[u8], start: usize) -> usize {
148    std::str::from_utf8(source.get(start..).unwrap_or_default())
149        .ok()
150        .and_then(|tail| tail.chars().next())
151        .map_or(start, |character| {
152            start.saturating_add(character.len_utf8())
153        })
154}
155
156fn trim_ascii_whitespace(mut bytes: &[u8]) -> &[u8] {
157    while bytes.first().is_some_and(u8::is_ascii_whitespace) {
158        bytes = &bytes[1..];
159    }
160    while bytes.last().is_some_and(u8::is_ascii_whitespace) {
161        bytes = &bytes[..bytes.len().saturating_sub(1)];
162    }
163    bytes
164}

167pub enum AnalysisError {
168    #[error("analysis cancelled")]
169    Cancelled,
170    #[error("unsupported source: {0}")]
171    Unsupported(String),
172    #[error("parser failure: {0}")]
173    Parser(String),
174}

176pub fn analyze(
177    file: &SourceFile,
178    limits: &RuleLimits,
179    cancel: &CancelToken,
180) -> Result<FileFacts, AnalysisError> {
181    if cancel.is_cancelled() {
182        return Err(AnalysisError::Cancelled);
183    }
184    let mut facts = hygiene::analyze(file, limits)?;
185    let mut language_facts = analyze_language(file, limits, cancel)?;
186    facts.diagnostics.append(&mut language_facts.diagnostics);
187    facts.parser = language_facts.parser;
188    facts.parser_version = language_facts.parser_version;
189    facts.dependencies = language_facts.dependencies;
190    facts.callable_fingerprints = language_facts.callable_fingerprints;
191    facts.region_fingerprints = language_facts.region_fingerprints;
192    facts.interfaces = language_facts.interfaces;
193    Ok(facts)
194}

205fn analyze_language(
206    file: &SourceFile,
207    limits: &RuleLimits,
208    cancel: &CancelToken,
209) -> Result<FileFacts, AnalysisError> {
210    match file.language {
211        Language::JavaScript | Language::TypeScript => node::analyze(file, limits, cancel),
212        Language::Rust => rust::analyze(file, limits, cancel),
213        Language::Python => python::analyze(file, limits, cancel),
214        Language::Go => go::analyze(file, limits, cancel),
215        Language::Hcl | Language::Shell | Language::Protobuf => analyze_declarative(file, cancel),
216    }
217}
218
219fn analyze_declarative(
220    file: &SourceFile,
221    cancel: &CancelToken,
222) -> Result<FileFacts, AnalysisError> {
223    match file.language {
224        Language::Hcl => hcl::analyze(file, cancel),
225        Language::Shell => shell::analyze(file, cancel),
226        Language::Protobuf => protobuf::analyze(file, cancel),
227        _ => Err(AnalysisError::Unsupported(format!(
228            "{} is not declarative source",
229            file.language.family()
230        ))),
231    }
