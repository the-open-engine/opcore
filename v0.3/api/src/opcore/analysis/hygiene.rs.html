9pub fn analyze(file: &SourceFile, limits: &RuleLimits) -> Result<FileFacts, AnalysisError> {
10    if file.bytes.len() > crate::limits::MAX_SOURCE_BYTES {
11        return Err(AnalysisError::Unsupported(format!(
12            "file is {} bytes; limit is {}",
13            file.bytes.len(),
14            crate::limits::MAX_SOURCE_BYTES
15        )));
16    }
17
18    let mut diagnostics = Vec::new();
19    let mut line_count = 0u32;
20    let mut line_start = 0usize;
21    let mut cursor = 0usize;
22    while cursor < file.bytes.len() {
23        if !matches!(file.bytes[cursor], b'\n' | b'\r') {
24            cursor += 1;
25            continue;
26        }
27        check_line(
28            &mut diagnostics,
29            LineInput {
30                file,
31                limits,
32                number: line_count.saturating_add(1),
33                start: line_start,
34                end: cursor,
35            },
36        )?;
37        line_count = line_count.saturating_add(1);
38        if file.bytes[cursor] == b'\r' && file.bytes.get(cursor + 1) == Some(&b'\n') {
39            cursor += 1;
40        }
41        cursor += 1;
42        line_start = cursor;
43    }
44    if line_start < file.bytes.len() {
45        check_line(
46            &mut diagnostics,
47            LineInput {
48                file,
49                limits,
50                number: line_count.saturating_add(1),
51                start: line_start,
52                end: file.bytes.len(),
53            },
54        )?;
55        line_count = line_count.saturating_add(1);
56    }
57    if line_count > limits.max_file_lines {
58        diagnostics.push(metric(MetricInput {
59            rule_id: "hygiene.max-file-lines",
60            message: format!(
61                "file has {line_count} lines; configured maximum is {}",
62                limits.max_file_lines
63            ),
64            line: 1,
65            start: 0,
66            end: file.bytes.len(),
67            entity: "file".into(),
68            actual: line_count as usize,
69            limit: limits.max_file_lines,
70        }));
71    }
72    Ok(FileFacts {
73        diagnostics,
74        ..FileFacts::default()
75    })
76}

78#[derive(Clone, Copy)]
79struct LineInput<'a> {
80    file: &'a SourceFile,
81    limits: &'a RuleLimits,
82    number: u32,
83    start: usize,
84    end: usize,
85}
86
87fn check_line(
88    diagnostics: &mut Vec<RawDiagnostic>,
89    input: LineInput<'_>,
90) -> Result<(), AnalysisError> {
91    let content_start = if input.number == 1
92        && input.file.bytes[input.start..input.end].starts_with(b"\xef\xbb\xbf")
93    {
94        input.start.saturating_add(3)
95    } else {
96        input.start
97    };
98    let content_len = input.end.saturating_sub(content_start);
99    if content_len > MAX_LINE_BYTES {
100        return Err(AnalysisError::Unsupported(format!(
101            "line {} is {content_len} bytes; hard limit is {MAX_LINE_BYTES}",
102            input.number
103        )));
104    }
105    if content_len > input.limits.max_line_bytes as usize {
106        diagnostics.push(metric(MetricInput {
107            rule_id: "hygiene.max-line-bytes",
108            message: format!(
109                "line has {content_len} bytes; configured maximum is {}",
110                input.limits.max_line_bytes
111            ),
112            line: input.number,
113            start: content_start,
114            end: input.end,
115            entity: format!(
116                "line:{}",
117                crate::analysis::line_anchor(&input.file.bytes, input.start)
118            ),
119            actual: content_len,
120            limit: input.limits.max_line_bytes,
121        }));
122    }
123    Ok(())
124}

126struct MetricInput<'a> {
127    rule_id: &'a str,
128    message: String,
129    line: u32,
130    start: usize,
131    end: usize,
132    entity: String,
133    actual: usize,
134    limit: u32,
135}
136
137fn metric(input: MetricInput<'_>) -> RawDiagnostic {
138    let mut evidence = BTreeMap::new();
139    evidence.insert("actual".into(), serde_json::json!(input.actual));
140    evidence.insert("limit".into(), serde_json::json!(input.limit));
141    RawDiagnostic {
142        rule_id: input.rule_id.into(),
143        severity: Severity::Warning,
144        message: input.message,
145        range: Some(SourceRange {
146            start: Position {
147                line: input.line,
148                column: 1,
149                byte: u32::try_from(input.start).unwrap_or(u32::MAX),
150            },
151            end: Position {
152                line: input.line,
153                column: u32::try_from(input.end.saturating_sub(input.start) + 1)
154                    .unwrap_or(u32::MAX),
155                byte: u32::try_from(input.end).unwrap_or(u32::MAX),
156            },
157        }),
158        entity_key: input.entity,
159        cause_key: format!("limit:{}", input.limit),
160        evidence,
161    }
162}

179    fn line_limits_exclude_terminators_and_accept_universal_newlines() {
180        let exact = [vec![b'x'; 512], b"\n".to_vec()].concat();
181        assert!(
182            analyze(&file(exact), &RuleLimits::default())
183                .unwrap()
184                .diagnostics
185                .is_empty()
186        );
187
188        let cr_lines = b"x\r".repeat(1_501);
189        let facts = analyze(&file(cr_lines), &RuleLimits::default()).unwrap();
190        assert!(
191            facts
192                .diagnostics
193                .iter()
194                .any(|item| item.rule_id == "hygiene.max-file-lines")
195        );
196    }
