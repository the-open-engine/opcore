1use crate::model::{Position, SourceRange};
2
3pub(crate) struct LineIndex {
4    starts: Vec<usize>,
5    byte_base: usize,
6}
7
8impl LineIndex {
9    pub(crate) fn new(source: &[u8], byte_base: usize) -> Self {
10        let mut starts = Vec::with_capacity(source.len().saturating_div(32).saturating_add(1));
11        starts.push(0);
12        let mut index = 0usize;
13        while index < source.len() {
14            index = advance(source, index, &mut starts);
15        }
16        Self { starts, byte_base }
17    }
18
19    pub(crate) fn position(&self, byte: usize) -> Position {
20        let line_index = self
21            .starts
22            .partition_point(|start| *start <= byte)
23            .saturating_sub(1);
24        let line_start = self.starts.get(line_index).copied().unwrap_or(0);
25        Position {
26            line: u32::try_from(line_index.saturating_add(1)).unwrap_or(u32::MAX),
27            column: u32::try_from(byte.saturating_sub(line_start).saturating_add(1))
28                .unwrap_or(u32::MAX),
29            byte: u32::try_from(byte.saturating_add(self.byte_base)).unwrap_or(u32::MAX),
30        }
31    }
32
33    pub(crate) fn byte_range(&self, start: usize, end: usize) -> SourceRange {
34        SourceRange {
35            start: self.position(start),
36            end: self.position(end),
37        }
38    }
39
40    pub(crate) fn line_span(&self, start: usize, end: usize) -> u32 {
41        self.position(end)
42            .line
43            .saturating_sub(self.position(start).line)
44            .saturating_add(1)
45    }
46}
47
48fn advance(source: &[u8], index: usize, starts: &mut Vec<usize>) -> usize {
49    let width = if source[index] == b'\r' && source.get(index + 1) == Some(&b'\n') {
50        2
51    } else if matches!(source[index], b'\r' | b'\n') {
52        1
53    } else {
54        return index.saturating_add(1);
55    };
56    let next = index.saturating_add(width);
57    starts.push(next);
58    next
59}
