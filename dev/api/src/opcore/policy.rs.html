41/// A built-in check workflow with repository-defined settings.
42pub enum Workflow {
43    /// Fast Verify and Sense against the current worktree.
44    PostEdit,
45    /// Full staged Verify and selected native checks, plus introduced Sense.
46    PreCommit,
47    /// Full immutable-commit Verify and native checks, plus Sense against a base commit.
48    Ci,
49}

55        match self {
56            Self::PostEdit => "post-edit",
57            Self::PreCommit => "pre-commit",
58            Self::Ci => "ci",
59        }
60    }

134impl Policy {
135    #[must_use]
136    pub fn sense_options(&self) -> SenseOptions {
137        SenseOptions {
138            limits: self.verify,
139            high_impact_threshold: self.sense.important_fan_in,
140            minimum_identical_bytes: self.sense.minimum_identical_bytes,
141            max_dependency_targets: self.sense.max_dependency_targets,
142            max_edge_selectors: self.sense.max_edge_selectors,
143            max_module_exports: self.sense.max_module_exports,
144            max_shape_members: self.sense.max_shape_members,
145        }
146    }
147
148    pub fn resolve(&self, workflow: Option<Workflow>) -> Result<Self> {
149        let mut resolved = self.clone();
150        if let Some(workflow) = workflow {
151            let settings = self.workflows.selected(workflow);
152            settings.verify.apply(&mut resolved.verify);
153            settings.sense.apply(&mut resolved.sense);
154            settings.targets.apply(&mut resolved.targets);
155            settings.providers.apply(&mut resolved.providers);
156            settings.coverage.apply(&mut resolved.coverage);
157            if let Some(native) = &settings.native {
158                resolved.native.clone_from(native);
159            }
160            if workflow == Workflow::PostEdit {
161                ensure!(
162                    settings.native.as_ref().is_none_or(Vec::is_empty),
163                    "workflows.post-edit.native must be empty: automatic hooks use Fast Verify and Sense"
164                );
165                resolved.native.clear();
166            }
167        }
168        resolved.workflows = Workflows::default();
169        resolved.validate_effective()?;
170        Ok(resolved)
171    }
172
173    fn validate(&self) -> Result<()> {
174        self.validate_effective()?;
175        for workflow in [Workflow::PostEdit, Workflow::PreCommit, Workflow::Ci] {
176            self.resolve(Some(workflow))
177                .with_context(|| format!("workflows.{}", workflow.as_str()))?;
178        }
179        Ok(())
180    }
181
182    fn validate_effective(&self) -> Result<()> {
183        ensure!(
184            self.schema_version == POLICY_SCHEMA_VERSION,
185            "schemaVersion must be {POLICY_SCHEMA_VERSION}"
186        );
187        validate_verify(&self.verify)?;
188        ensure!(
189            (1..=MAX_SOURCE_BYTES).contains(&self.sense.minimum_identical_bytes),
190            "sense.minimumIdenticalBytes must be within 1..={MAX_SOURCE_BYTES}"
191        );
192        ensure!(
193            (1..=MAX_FILES).contains(&self.sense.important_fan_in),
194            "sense.importantFanIn must be within 1..={MAX_FILES}"
195        );
196        ensure!(
197            self.sense.max_dependency_targets <= MAX_GRAPH_EDGES,
198            "sense.maxDependencyTargets exceeds the graph safety bound"
199        );
200        for (name, value) in [
201            ("maxEdgeSelectors", self.sense.max_edge_selectors),
202            ("maxModuleExports", self.sense.max_module_exports),
203            ("maxShapeMembers", self.sense.max_shape_members),
204        ] {
205            ensure!(
206                value <= MAX_INTERFACE_FACTS_PER_FILE,
207                "sense.{name} exceeds the per-file fact bound"
208            );
209        }
210        self.targets.validate()?;
211        self.providers.validate()?;
212        validate_bindings(&self.documentation.bindings)?;
213        ensure!(
214            self.native.iter().collect::<BTreeSet<_>>().len() == self.native.len(),
215            "native contains a duplicate provider"
216        );
217        for provider in &self.native {
218            ensure!(
219                !self.providers.options(*provider).roots.is_empty(),
220                "providers.{}.roots cannot be empty while that native provider is required",
221                provider.as_str()
222            );
223        }
224        Ok(())
225    }
226}

228pub fn validate_verify(limits: &RuleLimits) -> Result<()> {
229    ensure!(
230        limits.max_file_lines as usize <= MAX_SOURCE_BYTES,
231        "verify.maxFileLines exceeds the hard safety bound"
232    );
233    ensure!(
234        limits.max_line_bytes as usize <= MAX_LINE_BYTES,
235        "verify.maxLineBytes exceeds the hard safety bound"
236    );
237    ensure!(
238        limits.max_function_lines <= limits.max_file_lines,
239        "verify.maxFunctionLines must not exceed maxFileLines"
240    );
241    ensure!(
242        limits.max_parameters as usize <= MAX_DEPENDENCY_FACTS_PER_FILE,
243        "verify.maxParameters exceeds the hard safety bound"
244    );
245    ensure!(
246        limits.max_nesting as usize <= MAX_STRUCTURAL_DEPTH,
247        "verify.maxNesting exceeds the parser safety bound"
248    );
249    ensure!(
250        limits.max_cyclomatic_complexity as usize <= MAX_SOURCE_BYTES,
251        "verify.maxCyclomaticComplexity exceeds the hard safety bound"
252    );
253    Ok(())
254}

256pub fn parse_policy(bytes: &[u8]) -> Result<Policy> {
257    ensure!(
258        bytes.len() <= MAX_POLICY_BYTES,
259        "{POLICY_PATH} exceeds {MAX_POLICY_BYTES} bytes"
260    );
261    let mut value =
262        parse_unique_json(bytes).with_context(|| format!("parse unique {POLICY_PATH} JSON"))?;
263    shape::validate(&value)?;
264    if let Some(object) = value.as_object_mut()
265        && let Some(schema) = object.remove("$schema")
266    {
267        ensure!(schema.is_string(), "$schema must be a schema URL string");
268    }
269    let mut policy: Policy =
270        serde_json::from_value(value).with_context(|| format!("validate {POLICY_PATH} schema"))?;
271    policy.documentation.bindings.sort();
272    policy
273        .validate()
274        .with_context(|| format!("validate {POLICY_PATH} settings"))?;
275    Ok(policy)
276}

286pub struct PolicySnapshot {
287    pub policy: Policy,
288    pub state: PolicyState,
289    pub digest: String,
290    pub workflow: Option<Workflow>,
291    pub view: ConfigView,
292    pub origins: BTreeMap<String, String>,
293    root: PathBuf,
294    bytes: Option<Vec<u8>>,
295    capture: Option<GitAuxiliaryCapture>,

298impl PolicySnapshot {
299    #[cfg(test)]
300    pub fn load(root: &Path) -> Result<Self> {
301        let bytes = stable_read(&root.join(POLICY_PATH))?;
302        Self::from_bytes(root, bytes, ConfigView::Worktree, None, None)
303    }
304
305    pub fn capture(
306        repository: &GitRepository,
307        view: ConfigView,
308        workflow: Option<Workflow>,
309    ) -> Result<Self> {
310        let path = RepoPath::from_protocol(POLICY_PATH)?;
311        let paths = BTreeSet::from([path.clone()]);
312        let capture = match &view {
313            ConfigView::Worktree => repository.capture_auxiliary(&paths)?,
314            ConfigView::Index => repository.capture_staged_auxiliary(&paths)?,
315            ConfigView::Tree(reference) => {
316                repository.capture_tree_auxiliary(&paths, reference, None)?
317            }
318        };
319        let state = capture
320            .paths
321            .get(&path)
322            .context("configuration capture omitted requested path")?;
323        let bytes = state.after.as_ref().map(|blob| blob.bytes.to_vec());
324        Self::from_bytes(repository.root(), bytes, view, workflow, Some(capture))
325    }
326
327    fn from_bytes(
328        root: &Path,
329        bytes: Option<Vec<u8>>,
330        view: ConfigView,
331        workflow: Option<Workflow>,
332        capture: Option<GitAuxiliaryCapture>,
333    ) -> Result<Self> {
334        let configured = bytes
335            .as_ref()
336            .map(|bytes| parse_policy(bytes))
337            .transpose()?;
338        let state = if configured.is_some() {
339            PolicyState::Configured
340        } else {
341            PolicyState::Default
342        };
343        let mut configuration = configured.unwrap_or_default();
344        if workflow == Some(Workflow::PostEdit) {
345            let explicit = bytes
346                .as_ref()
347                .map(|bytes| parse_unique_json(bytes))
348                .transpose()?;
349            if explicit
350                .as_ref()
351                .and_then(|value| value.get("coverage"))
352                .and_then(|value| value.get("allowPartial"))
353                .is_none()
354            {
355                configuration.coverage.allow_partial = true;
356            }
357        }
358        let policy = configuration.resolve(workflow)?;
359        let canonical = serde_json::to_vec(&policy).context("serialize effective configuration")?;
360        let mut hasher = Sha256::new();
361        hasher.update(b"opcore-configuration/v1\0");
362        hasher.update(canonical);
363        let origins = configuration_origins(&policy, bytes.as_deref(), workflow)?;
364        Ok(Self {
365            policy,
366            state,
367            digest: format!("sha256:{}", hex::encode(hasher.finalize())),
368            workflow,
369            view,
370            origins,
371            root: root.to_path_buf(),
372            bytes,
373            capture,
374        })
375    }
376
377    pub fn is_current(&self) -> Result<bool> {
378        if let Some(capture) = &self.capture {
379            let repository = GitRepository::discover(&self.root)?;
380            return Ok(repository.is_auxiliary_current(capture)?);
381        }
382        Ok(stable_read(&self.root.join(POLICY_PATH))? == self.bytes)
383    }
384
385    pub fn validate_hypothetical(&self, proposed: Option<&[u8]>) -> Result<()> {
386        let current = self
387            .bytes
388            .as_deref()
389            .map(parse_policy)
390            .transpose()?
391            .unwrap_or_default();
392        let mut candidate = proposed.map(parse_policy).transpose()?.unwrap_or_default();
393        candidate.documentation = current.documentation.clone();
394        ensure!(
395            candidate == current,
396            "hypothetical configuration may change documentation.bindings only; Verify, Sense, targets, providers and workflows must stay unchanged"
397        );
398        Ok(())
399    }

401    #[must_use]
402    pub fn bind_valid_as_of(&self, source: &str) -> String {
403        let source = serde_json::from_str::<Value>(source)
404            .unwrap_or_else(|_| Value::String(source.to_owned()));
405        serde_json::json!({"kind":"opcore-local", "source":source, "policy":self.digest,
406            "configurationView":self.view, "workflow":self.workflow})
407        .to_string()
408    }

411fn configuration_origins(
412    policy: &Policy,
413    bytes: Option<&[u8]>,
414    workflow: Option<Workflow>,
415) -> Result<BTreeMap<String, String>> {
416    let mut origins = BTreeMap::new();
417    mark_origins(&serde_json::to_value(policy)?, "", "default", &mut origins);
418    if let Some(bytes) = bytes {
419        let raw = parse_unique_json(bytes)?;
420        mark_origins(&raw, "", "repository", &mut origins);
421        if let Some(workflow) = workflow
422            && let Some(settings) = raw
423                .get("workflows")
424                .and_then(|workflows| workflows.get(workflow.as_str()))
425        {
426            mark_origins(
427                settings,
428                "",
429                &format!("workflow:{}", workflow.as_str()),
430                &mut origins,
431            );
432        }
433    }
434    origins.retain(|path, _| !path.starts_with("/workflows") && path != "/$schema");
435    Ok(origins)
436}
437
438fn mark_origins(
439    value: &Value,
440    pointer: &str,
441    origin: &str,
442    origins: &mut BTreeMap<String, String>,
443) {
444    if let Value::Object(fields) = value {
445        for (key, value) in fields {
446            mark_origins(value, &format!("{pointer}/{key}"), origin, origins);
447        }
448    } else {
449        origins.insert(pointer.to_owned(), origin.to_owned());
450    }
451}
452fn stable_read(path: &Path) -> Result<Option<Vec<u8>>> {
453    let mut previous = None;
454    for _ in 0..2 {
455        let first = read_once(path)?;
456        let second = read_once(path)?;
457        if first == second {
458            return Ok(first);
459        }
460        previous = Some("policy changed while it was read");
461    }
462    anyhow::bail!(previous.unwrap_or("policy capture did not stabilize"))
463}

465fn read_once(path: &Path) -> Result<Option<Vec<u8>>> {
466    let mut file = match OpenOptions::new()
467        .read(true)
468        .custom_flags(libc::O_CLOEXEC | libc::O_NOFOLLOW)
469        .open(path)
470    {
471        Ok(file) => file,
472        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(None),
473        Err(error) => return Err(error).with_context(|| format!("open {POLICY_PATH}")),
474    };
475    let metadata = file
476        .metadata()
477        .with_context(|| format!("inspect {POLICY_PATH}"))?;
478    ensure!(metadata.is_file(), "{POLICY_PATH} must be a regular file");
479    ensure!(
480        metadata.len() <= MAX_POLICY_BYTES as u64,
481        "{POLICY_PATH} exceeds {MAX_POLICY_BYTES} bytes"
482    );
483    let capacity = usize::try_from(metadata.len()).context("policy size fits this platform")?;
484    let mut bytes = Vec::with_capacity(capacity);
485    file.by_ref()
486        .take((MAX_POLICY_BYTES + 1) as u64)
487        .read_to_end(&mut bytes)
488        .with_context(|| format!("read {POLICY_PATH}"))?;
489    ensure!(
490        bytes.len() <= MAX_POLICY_BYTES,
491        "{POLICY_PATH} exceeds {MAX_POLICY_BYTES} bytes"
492    );
493    Ok(Some(bytes))
494}
