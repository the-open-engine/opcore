41struct WorkspaceReader<'a> {
42    session: &'a AspSession,
43    host: &'a dyn AspHost,
44    cancel: &'a CancelToken,
45    deadline: RequestDeadline,
46}
47
48impl<'a> WorkspaceReader<'a> {
49    const fn new(
50        session: &'a AspSession,
51        host: &'a dyn AspHost,
52        cancel: &'a CancelToken,
53        deadline: RequestDeadline,
54    ) -> Self {
55        Self {
56            session,
57            host,
58            cancel,
59            deadline,
60        }
61    }
62
63    async fn list_tree(
64        &self,
65        scope: &Scope,
66        paths: ListingPaths<'_>,
67    ) -> Result<ListTreeResult, RpcFailure> {
68        let mut list_params = serde_json::Map::new();
69        list_params.insert("baseline".into(), self.session.baseline.clone());
70        if !matches!(scope, Scope::Workspace) {
71            list_params.insert("paths".into(), listing_path_value(paths)?);
72        }
73        let value = self
74            .host
75            .request(
76                "workspace/listTree",
77                Value::Object(list_params),
78                self.deadline.remaining_ms()?,
79                self.cancel,
80            )
81            .await?;
82        serde_json::from_value(value).map_err(|error| {
83            RpcFailure::contract(format!("malformed workspace/listTree result: {error}"))
84        })
85    }
86
87    async fn read_blobs(
88        &self,
89        ids: &BTreeSet<String>,
90    ) -> Result<BTreeMap<String, Arc<[u8]>>, RpcFailure> {
91        let mut blobs = BTreeMap::new();
92        let mut total = 0usize;
93        for id in ids {
94            let blob = self.read_blob(id).await?;
95            let bytes = decode_blob(&blob)?;
96            total = total.saturating_add(bytes.len());
97            if total > MAX_TOTAL_SOURCE_BYTES {
98                return Err(RpcFailure::unavailable(
99                    "read blobs exceed the total request byte bound",
100                ));
101            }
102            verify_blob(&blob.id, &bytes).map_err(|failure| {
103                RpcFailure::contract(format!(
104                    "workspace/readBlob returned bytes inconsistent with their id: {}",
105                    failure.detail
106                ))
107            })?;
108            blobs.insert(blob.id, Arc::<[u8]>::from(bytes));
109        }
110        Ok(blobs)
111    }
112
113    async fn read_blob(&self, id: &str) -> Result<BlobResult, RpcFailure> {
114        let value = self
115            .host
116            .request(
117                "workspace/readBlob",
118                json!({ "blobs": [id] }),
119                self.deadline.remaining_ms()?,
120                self.cancel,
121            )
122            .await?;
123        let mut result: ReadBlobResult = serde_json::from_value(value).map_err(|error| {
124            RpcFailure::contract(format!("malformed workspace/readBlob result: {error}"))
125        })?;
126        if result.blobs.len() != 1 {
127            return Err(RpcFailure::contract(format!(
128                "workspace/readBlob did not return exactly one result for {id}"
129            )));
130        }
131        let blob = result
132            .blobs
133            .pop()
134            .ok_or_else(|| RpcFailure::contract(format!("workspace/readBlob omitted {id}")))?;
135        if blob.id != id {
136            return Err(RpcFailure::contract(format!(
137                "workspace/readBlob returned {} while requesting {id}",
138                blob.id
139            )));
140        }
141        Ok(blob)
142    }
143}

145pub(super) async fn prepare_views(
146    params: &NormalizedParams,
147    session: &AspSession,
148    host: &dyn AspHost,
149    cancel: &CancelToken,
150    deadline: RequestDeadline,
151) -> WorkspaceResult<PreparedViews> {
152    let reader = WorkspaceReader::new(session, host, cancel, deadline);
153    let affected = affected_source_paths(&params.changeset);
154    let requested_paths = requested_paths(&params.scope, &affected);
155    let preconditions =
156        scoped_change_precondition_paths(&params.changeset, &params.scope, &requested_paths);
157    let listing_paths = ListingPaths {
158        preconditions: &preconditions,
159        requested: &requested_paths,
160    };
161    let listing = reader.list_tree(&params.scope, listing_paths).await?;
162    require_complete_listing(&listing)?;
163    let baseline = tree_from_listing(listing, &params.scope, &preconditions, &requested_paths)?;
164    validate_change_preconditions(
165        &params.changeset,
166        &params.scope,
167        &requested_paths,
168        &baseline.entries,
169    )?;
170    let after = apply_changes(&baseline, &params.changeset)?;
171    let paths = assessment_paths(&params.scope, &requested_paths, &baseline, &after);
172    reject_source_symlinks(&paths, &baseline, &after)?;
173    let read_ids = required_blob_ids(&paths, &baseline, &after);
174    let blobs = reader.read_blobs(&read_ids).await?;
175    let before_files = materialize_files(&paths, &baseline.entries, &baseline.kinds, &blobs)?;
176    let after_files = materialize_files(&paths, &after.entries, &after.kinds, &blobs)?;
177    Ok(PreparedViews {
178        before: Arc::new(SourceSnapshot::new(before_files)),
179        after: Arc::new(SourceSnapshot::new(after_files)),
180        paths,
181        read_ids: read_ids.into_iter().collect(),
182    })
183}

185pub(super) async fn prepare_native_views(
186    params: &NormalizedParams,
187    session: &AspSession,
188    host: &dyn AspHost,
189    cancel: &CancelToken,
190    deadline: RequestDeadline,
191) -> WorkspaceResult<PreparedNativeViews> {
192    let reader = WorkspaceReader::new(session, host, cancel, deadline);
193    let empty = BTreeSet::new();
194    let listing = reader
195        .list_tree(
196            &Scope::Workspace,
197            ListingPaths {
198                preconditions: &empty,
199                requested: &empty,
200            },
201        )
202        .await?;
203    require_complete_listing(&listing)?;
204    let (baseline, after) = validated_native_trees(params, listing, &empty)?;
205    reject_native_symlinks(&baseline, &after)?;
206    let read_ids = native_blob_ids(&baseline, &after)?;
207    let blobs = reader.read_blobs(&read_ids).await?;
208    Ok(PreparedNativeViews {
209        before: native_files(&baseline, &blobs)?,
210        after: native_files(&after, &blobs)?,
211        read_ids: read_ids.into_iter().collect(),
212    })
213}

215fn require_complete_listing(listing: &ListTreeResult) -> WorkspaceResult<()> {
216    if listing.truncated {
217        return Err(WorkspaceFailure::Coverage(WorkspaceGap {
218            status: "incomplete",
219            detail: "workspace/listTree truncated required coverage".into(),
220            truncated: true,
221        }));
222    }
223    Ok(())
224}

226fn reject_source_symlinks(
227    paths: &BTreeSet<RepoPath>,
228    before: &TreeState,
229    after: &TreeState,
230) -> WorkspaceResult<()> {
231    let unsupported = paths.iter().find(|path| {
232        before.kinds.get(*path) == Some(&EntryKind::Symlink)
233            || after.kinds.get(*path) == Some(&EntryKind::Symlink)
234    });
235    if let Some(path) = unsupported {
236        return Err(WorkspaceFailure::Coverage(WorkspaceGap {
237            status: "unsupported",
238            detail: format!("source symlink {path} is outside fast Check coverage"),
239            truncated: false,
240        }));
241    }
242    Ok(())
243}
244
245fn reject_native_symlinks(before: &TreeState, after: &TreeState) -> WorkspaceResult<()> {
246    let unsupported = before
247        .kinds
248        .iter()
249        .chain(&after.kinds)
250        .find(|(_, kind)| **kind == EntryKind::Symlink);
251    if let Some((path, _)) = unsupported {
252        return Err(WorkspaceFailure::Coverage(WorkspaceGap {
253            status: "unsupported",
254            detail: format!("workspace symlink {path} is outside native callback coverage"),
255            truncated: false,
256        }));
257    }
258    Ok(())
259}

261fn validated_native_trees(
262    params: &NormalizedParams,
263    listing: ListTreeResult,
264    empty: &BTreeSet<RepoPath>,
265) -> Result<(TreeState, TreeState), RpcFailure> {
266    let baseline = tree_from_listing(listing, &Scope::Workspace, empty, empty)?;
267    validate_all_change_preconditions(&params.changeset, &baseline.entries)?;
268    let after = apply_changes(&baseline, &params.changeset)?;
269    validate_native_tree(&baseline)?;
270    validate_native_tree(&after)?;
271    Ok((baseline, after))
272}

306fn native_blob_ids(before: &TreeState, after: &TreeState) -> Result<BTreeSet<String>, RpcFailure> {
307    let mut ids = BTreeSet::new();
308    collect_native_blob_ids(before, &mut ids)?;
309    collect_native_blob_ids(after, &mut ids)?;
310    Ok(ids)

313fn collect_native_blob_ids(tree: &TreeState, ids: &mut BTreeSet<String>) -> Result<(), RpcFailure> {
314    for (path, id) in &tree.entries {
315        match tree.kinds.get(path) {
316            Some(EntryKind::File) => {
317                ids.insert(id.clone());
318            }
319            Some(EntryKind::Dir) => {}
320            Some(EntryKind::Symlink) | None => {
321                return Err(RpcFailure::unavailable(format!(
322                    "native workspace entry {path} has an invalid file kind"
323                )));
324            }
325        }
326    }
327    Ok(())
328}

330fn native_files(
331    tree: &TreeState,
332    blobs: &BTreeMap<String, Arc<[u8]>>,
333) -> Result<BTreeMap<RepoPath, Arc<[u8]>>, RpcFailure> {
334    let mut total = 0usize;
335    let mut files = BTreeMap::new();
336    for (path, id) in &tree.entries {
337        if tree.kinds.get(path) == Some(&EntryKind::Dir) {
338            continue;
339        }
340        let bytes = blobs.get(id).cloned().ok_or_else(|| {
341            RpcFailure::unavailable(format!("missing materialized blob {id} for {path}"))
342        })?;
343        total = total
344            .checked_add(bytes.len())
345            .ok_or_else(|| RpcFailure::unavailable("native materialized byte count overflowed"))?;
346        if total > MAX_TOTAL_SOURCE_BYTES {
347            return Err(RpcFailure::unavailable(format!(
348                "native workspace exceeds {MAX_TOTAL_SOURCE_BYTES} materialized bytes"
349            )));
350        }
351        files.insert(path.clone(), bytes);
352    }
353    Ok(files)
354}

363fn scoped_change_precondition_paths(
364    changeset: &ChangeSet,
365    scope: &Scope,
366    requested: &BTreeSet<RepoPath>,
367) -> BTreeSet<RepoPath> {
368    let mut paths = BTreeSet::new();
369    for change in &changeset.changes {
370        if !change_is_in_scope(change, scope, requested) {
371            continue;
372        }
373        paths.insert(change.path.clone());
374        paths.extend(change.from.iter().cloned());
375    }
376    paths
377}

379fn change_is_source_relevant(change: &Change) -> bool {
380    is_source_candidate(&change.path) || change.from.as_ref().is_some_and(is_source_candidate)
381}
382
383fn change_is_in_scope(change: &Change, scope: &Scope, requested: &BTreeSet<RepoPath>) -> bool {
384    match scope {
385        Scope::Paths { .. } => {
386            requested.contains(&change.path)
387                || change
388                    .from
389                    .as_ref()
390                    .is_some_and(|path| requested.contains(path))
391        }
392        Scope::Changeset | Scope::Workspace => change_is_source_relevant(change),
393    }
394}

396fn requested_paths(scope: &Scope, affected: &BTreeSet<RepoPath>) -> BTreeSet<RepoPath> {
397    match scope {
398        Scope::Changeset => affected.clone(),
399        Scope::Paths { paths } => paths.iter().cloned().collect(),
400        Scope::Workspace => BTreeSet::new(),
401    }

404fn tree_from_listing(
405    listing: ListTreeResult,
406    scope: &Scope,
407    preconditions: &BTreeSet<RepoPath>,
408    requested: &BTreeSet<RepoPath>,
409) -> Result<TreeState, RpcFailure> {
410    if listing.entries.len() > MAX_FILES {
411        return Err(RpcFailure::unavailable(format!(
412            "workspace listing exceeds {MAX_FILES} entries"
413        )));
414    }
415    let allowed = allowed_listing_paths(scope, preconditions, requested);
416    let mut tree = TreeState {
417        entries: BTreeMap::new(),
418        kinds: BTreeMap::new(),
419    };
420    for entry in listing.entries {
421        if allowed
422            .as_ref()
423            .is_some_and(|allowed| !allowed.contains(&entry.path))
424        {
425            return Err(RpcFailure::contract(format!(
426                "workspace/listTree returned unrequested path {}",
427                entry.path
428            )));
429        }
430        validate_blob_ref(&entry.blob_id).map_err(|failure| {
431            RpcFailure::contract(format!(
432                "workspace/listTree returned an invalid blob ref: {}",
433                failure.detail
434            ))
435        })?;
436        if tree
437            .entries
438            .insert(entry.path.clone(), entry.blob_id)
439            .is_some()
440        {
441            return Err(RpcFailure::contract(format!(
442                "workspace/listTree returned duplicate path {}",
443                entry.path
444            )));
445        }
446        tree.kinds.insert(entry.path, entry.kind);
447    }
448    Ok(tree)
449}

451fn allowed_listing_paths(
452    scope: &Scope,
453    preconditions: &BTreeSet<RepoPath>,
454    requested: &BTreeSet<RepoPath>,
455) -> Option<BTreeSet<RepoPath>> {
456    if matches!(scope, Scope::Workspace) {
457        return None;
458    }
459    let mut paths = requested.clone();
460    paths.extend(preconditions.iter().cloned());
461    Some(paths)
462}

464fn apply_changes(baseline: &TreeState, changeset: &ChangeSet) -> Result<TreeState, RpcFailure> {
465    let mut after = baseline.clone();
466    for change in &changeset.changes {
467        apply_change(&mut after, change)?;
468    }
469    Ok(after)
470}
471
472fn apply_change(tree: &mut TreeState, change: &Change) -> Result<(), RpcFailure> {
473    match change.kind {
474        ChangeKind::Create | ChangeKind::Modify => {
475            let Some(after) = change.after.clone() else {
476                return Err(illegal_change(change));
477            };
478            tree.entries.insert(change.path.clone(), after);
479            tree.kinds.insert(change.path.clone(), EntryKind::File);
480        }
481        ChangeKind::Delete => remove_tree_path(tree, &change.path),
482        ChangeKind::Rename => apply_rename(tree, change)?,
483    }
484    Ok(())
485}
486
487fn remove_tree_path(tree: &mut TreeState, path: &RepoPath) {
488    tree.entries.remove(path);
489    tree.kinds.remove(path);
490}
491
492fn apply_rename(tree: &mut TreeState, change: &Change) -> Result<(), RpcFailure> {
493    let Some(from) = change.from.as_ref() else {
494        return Err(illegal_change(change));
495    };
496    remove_tree_path(tree, from);
497    let Some(blob) = change.after.clone().or_else(|| change.before.clone()) else {
498        return Err(illegal_change(change));
499    };
500    tree.entries.insert(change.path.clone(), blob);
501    tree.kinds.insert(change.path.clone(), EntryKind::File);
502    Ok(())
503}

505fn assessment_paths(
506    scope: &Scope,
507    requested: &BTreeSet<RepoPath>,
508    baseline: &TreeState,
509    after: &TreeState,
510) -> BTreeSet<RepoPath> {
511    if matches!(scope, Scope::Workspace) {
512        let mut paths: BTreeSet<_> = baseline.entries.keys().cloned().collect();
513        paths.extend(after.entries.keys().cloned());
514        paths.retain(|path| {
515            is_source_candidate(path)
516                && (matches!(
517                    baseline.kinds.get(path),
518                    Some(EntryKind::File | EntryKind::Symlink)
519                ) || matches!(
520                    after.kinds.get(path),
521                    Some(EntryKind::File | EntryKind::Symlink)
522                ))
523        });
524        paths
525    } else {
526        requested.clone()
527    }
528}

530fn required_blob_ids(
531    paths: &BTreeSet<RepoPath>,
532    baseline: &TreeState,
533    after: &TreeState,
534) -> BTreeSet<String> {
535    let mut read_ids = BTreeSet::new();
536    for path in paths {
537        if detect_language(path).is_none() {
538            continue;
539        }
540        insert_blob_id(&mut read_ids, path, baseline);
541        insert_blob_id(&mut read_ids, path, after);
542    }
543    read_ids
544}

546fn insert_blob_id(ids: &mut BTreeSet<String>, path: &RepoPath, tree: &TreeState) {
547    if tree.kinds.get(path) == Some(&EntryKind::File)
548        && let Some(id) = tree.entries.get(path)
549    {
550        ids.insert(id.clone());
551    }
552}

556    for change in &changeset.changes {

565fn validate_change_preconditions(
566    changeset: &ChangeSet,
567    scope: &Scope,
568    requested: &BTreeSet<RepoPath>,
569    baseline: &BTreeMap<RepoPath, String>,
570) -> Result<(), RpcFailure> {
571    for change in &changeset.changes {
572        if !change_is_in_scope(change, scope, requested) {
573            continue;
574        }
575        validate_change_precondition(change, baseline)?;
576    }
577    Ok(())
578}
579
580fn validate_change_precondition(
581    change: &Change,
582    baseline: &BTreeMap<RepoPath, String>,
583) -> Result<(), RpcFailure> {
584    match change.kind {
585        ChangeKind::Create => require_absent(baseline, &change.path, "create destination"),
586        ChangeKind::Modify | ChangeKind::Delete => require_before(
587            baseline,
588            &change.path,
589            change.before.as_ref(),
590            "before blob",
591        ),
592        ChangeKind::Rename => validate_rename_precondition(change, baseline),
593    }
594}
595
596fn validate_rename_precondition(
597    change: &Change,
598    baseline: &BTreeMap<RepoPath, String>,
599) -> Result<(), RpcFailure> {
600    let Some(from) = change.from.as_ref() else {
601        return Err(illegal_change(change));
602    };
603    require_before(baseline, from, change.before.as_ref(), "rename before blob")?;
604    require_absent(baseline, &change.path, "rename destination")
605}

607fn require_before(
608    baseline: &BTreeMap<RepoPath, String>,
609    path: &RepoPath,
610    expected: Option<&String>,
611    label: &str,
612) -> Result<(), RpcFailure> {
613    if baseline.get(path) != expected {
614        return Err(RpcFailure::health(format!(
615            "{label} does not match baseline for {path}"
616        )));
617    }
618    Ok(())
619}

621fn require_absent(
622    baseline: &BTreeMap<RepoPath, String>,
623    path: &RepoPath,
624    label: &str,
625) -> Result<(), RpcFailure> {
626    if baseline.contains_key(path) {
627        return Err(RpcFailure::input(format!("{label} already exists: {path}")));
628    }
629    Ok(())
630}

632fn decode_blob(blob: &BlobResult) -> Result<Vec<u8>, RpcFailure> {
633    let bytes = match blob.encoding {
634        BlobEncoding::Utf8 => blob.bytes.as_bytes().to_vec(),
635        BlobEncoding::Base64 => base64::engine::general_purpose::STANDARD
636            .decode(&blob.bytes)
637            .map_err(|error| {
638                RpcFailure::contract(format!("invalid base64 blob {}: {error}", blob.id))
639            })?,
640    };
641    if bytes.len() > MAX_SOURCE_BYTES {
642        return Err(RpcFailure::unavailable(format!(
643            "blob {} exceeds {MAX_SOURCE_BYTES} bytes",
644            blob.id
645        )));
646    }
647    Ok(bytes)
648}

650fn materialize_files(
651    paths: &BTreeSet<RepoPath>,
652    entries: &BTreeMap<RepoPath, String>,
653    kinds: &BTreeMap<RepoPath, EntryKind>,
654    blobs: &BTreeMap<String, Arc<[u8]>>,
655) -> Result<Vec<SourceFile>, RpcFailure> {
656    let mut files = Vec::new();
657    for path in paths {
658        if detect_language(path).is_none() {
659            continue;
660        }
661        let Some(id) = entries.get(path) else {
662            continue;
663        };
664        if kinds.get(path) != Some(&EntryKind::File) {
665            continue;
666        }
667        let bytes = blobs
668            .get(id)
669            .ok_or_else(|| RpcFailure::unavailable(format!("missing materialized blob {id}")))?;
670        let Some((language, mode)) = resolve_language(path, bytes) else {
671            continue;
672        };
673        files.push(SourceFile::new(
674            path.clone(),
675            Arc::clone(bytes),
676            language,
677            mode,
678        ));
679    }
680    Ok(files)
681}
