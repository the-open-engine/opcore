81impl DecodedEvidence {
82    fn state(
83        &mut self,
84        fact_key: FileFactKey,
85        source: &SourceFile,
86        facts: &RegionFingerprintFacts,
87        retain: bool,
88    ) -> EvidenceState {
89        self.states.entry(fact_key).or_insert_with(|| {
90            if DecodedTokens::validate(source, facts) {
91                EvidenceState::Validated
92            } else {
93                EvidenceState::Malformed
94            }
95        });
96        if retain && matches!(self.states[&fact_key], EvidenceState::Validated) {
97            let cost = DecodedTokens::decode_cost(source, facts);
98            if cost > self.decode_remaining {
99                self.decode_exhausted = true;
100                return EvidenceState::Validated;
101            }
102            self.decode_remaining -= cost;
103            self.touch(fact_key);
104            let decoded = DecodedTokens::decode(source, facts);
105            #[cfg(test)]
106            let decoded = decoded.map(|decoded| decoded.tracked(Arc::clone(&self.live)));
107            let decoded = decoded
108                .map(Arc::new)
109                .map_or(EvidenceState::Malformed, EvidenceState::Decoded);
110            self.states.insert(fact_key, decoded);
111            #[cfg(test)]
112            {
113                self.decodes += 1;
114            }
115        }
116        if retain && matches!(self.states[&fact_key], EvidenceState::Decoded(_)) {
117            self.touch(fact_key);
118        }
119        self.states[&fact_key].clone()
120    }
121
122    fn touch(&mut self, fact_key: FileFactKey) {
123        // Exact comparisons need two files. Evict before decoding the next file.
124        if let Some(position) = self.recent.iter().position(|key| *key == fact_key) {
125            self.recent.remove(position);
126        } else if self.recent.len() == 2 {
127            let expired = self.recent.remove(0);
128            self.states.insert(expired, EvidenceState::Validated);
129        }
130        self.recent.push(fact_key);
131    }
132
133    fn anchor_order(
134        &mut self,
135        key: FileFactKey,
136        facts: &RegionFingerprintFacts,
137    ) -> Option<Arc<[u16]>> {
138        if let Some(order) = self.orders.get(&key) {
139            return Some(Arc::clone(order));
140        }
141        let mut order = (0..facts.anchors.len())
142            .map(u16::try_from)
143            .collect::<Result<Vec<_>, _>>()
144            .ok()?;
145        order.sort_by_key(|index| {
146            let anchor = &facts.anchors[usize::from(*index)];
147            (anchor.hash_one, anchor.hash_two)
148        });
149        let order: Arc<[u16]> = order.into();
150        self.orders.insert(key, Arc::clone(&order));
151        Some(order)
152    }
153}

214fn region_findings_bounded(inputs: RegionInputs<'_>, anchor_limit: usize) -> RegionResult {
215    let decoded = Rc::new(RefCell::new(DecodedEvidence::default()));
216    let mut state = EvaluationState::default();
217    let before = RegionView::new(&RegionViewInputs {
218        view: SenseView::Before,
219        snapshot: inputs.before,
220        file_facts: inputs.file_facts,
221        fact_keys: inputs.fact_keys,
222        continuity: inputs.continuity,
223        decoded_evidence: Rc::clone(&decoded),
224    });
225    let after = RegionView::new(&RegionViewInputs {
226        view: SenseView::After,
227        snapshot: inputs.after,
228        file_facts: inputs.file_facts,
229        fact_keys: inputs.fact_keys,
230        continuity: &BTreeMap::new(),
231        decoded_evidence: decoded,
232    });
233    state.comparison.observe_malformed_view(&before);
234    state.comparison.observe_malformed_view(&after);
235    state.before = before.coverage(&state.comparison);
236    state.after = after.coverage(&state.comparison);
237    if !state.comparison.incomplete {
238        batches::evaluate(inputs, anchor_limit, &mut [before, after], &mut state);
239    }
240    state
241        .before
242        .include_comparison(&state.comparison, SenseView::Before);
243    state
244        .after
245        .include_comparison(&state.comparison, SenseView::After);
246    state.issues.extend(state.comparison.issues());
247    RegionResult {
248        findings: state.findings,
249        findings_truncated: state.comparison.findings_truncated,
250        before: state.before,
251        after: state.after,
252        comparison_issues: state.issues,
253    }
254}

256pub(super) fn validate_identical_snapshot(
257    snapshot: &SourceSnapshot,
258    file_facts: &facts::FactMap,
259    fact_keys: &FactKeyLookup,
260) -> (RegionCoverage, Vec<SenseIssue>) {
261    let decoded_evidence = Rc::new(RefCell::new(DecodedEvidence::default()));
262    let view = RegionView::new(&RegionViewInputs {
263        view: SenseView::Before,
264        snapshot,
265        file_facts,
266        fact_keys,
267        continuity: &BTreeMap::new(),
268        decoded_evidence,
269    });
270    let mut comparison = ComparisonState::default();
271    comparison.observe_malformed_view(&view);
272    if comparison.malformed_evidence {
273        comparison.views.insert(SenseView::After);
274    }
275    let coverage = view.coverage(&comparison);
276    (coverage, comparison.issues())
277}

286#[derive(Default)]
287struct EvaluationState {
288    comparison: ComparisonState,
289    before: RegionCoverage,
290    after: RegionCoverage,
291    issues: Vec<SenseIssue>,
292    findings: Vec<IntroducedDuplicate>,
293    reported_classes: BTreeSet<(String, String)>,
294    covered: BTreeMap<usize, Vec<Range<usize>>>,
295}
296
297impl EvaluationState {
298    fn evaluate_group(
299        &mut self,
300        context: EvaluationContext<'_, '_>,
301        family: &'static str,
302        anchor: AnchorKey,
303        after_occurrences: &[AnchorOccurrence],
304    ) {
305        if !anchor_class_may_be_introduced(AnchorClassInputs {
306            after: after_occurrences,
307            after_view: context.after,
308            changed_paths: context.inputs.changed_paths,
309            suppressed_paths: context.inputs.suppressed_paths,
310        }) {
311            return;
312        }
313        for seed in after_occurrences {
314            self.evaluate_seed(context, family, anchor, after_occurrences, *seed);
315            if self.comparison.incomplete {
316                break;
317            }
318        }
319    }
320
321    fn evaluate_seed(
322        &mut self,
323        context: EvaluationContext<'_, '_>,
324        family: &'static str,
325        anchor: AnchorKey,
326        occurrences: &[AnchorOccurrence],
327        seed: AnchorOccurrence,
328    ) {
329        let seed_path = &context.after.files[seed.file].path;
330        if !context.inputs.changed_paths.contains(seed_path)
331            || context.inputs.suppressed_paths.contains(seed_path)
332            || interval_contains(&self.covered, seed.file, seed.token_index)
333        {
334            return;
335        }
336        let candidates = candidate_regions(
337            context.after,
338            anchor,
339            seed,
340            occurrences,
341            &mut self.comparison,
342        );
343        let Some(class) = self.best_class(context, family, candidates) else {
344            return;
345        };
346        mark_covered(&mut self.covered, &class.after_occurrences);
347        if !self
348            .reported_classes
349            .insert((family.to_owned(), class.digest.clone()))
350        {
351            return;
352        }
353        if self.findings.len() >= context.inputs.finding_limit {
354            self.comparison.findings_truncated = true;
355            return;
356        }
357        self.findings.push(class.into_finding());
358    }
359
360    fn best_class(
361        &mut self,
362        context: EvaluationContext<'_, '_>,
363        family: &'static str,
364        candidates: Vec<Candidate>,
365    ) -> Option<ExactClass> {
366        let mut best = None;
367        for candidate in candidates {
368            let Some(class) = exact_class(
369                ExactClassInputs {
370                    before: context.before,
371                    after: context.after,
372                    family,
373                    changed_paths: context.inputs.changed_paths,
374                    suppressed_paths: context.inputs.suppressed_paths,
375                    callable_findings: context.inputs.callable_findings,
376                },
377                candidate,
378                &mut self.comparison,
379            ) else {
380                if self.comparison.incomplete {
381                    break;
382                }
383                continue;
384            };
385            if class_precedes(best.as_ref(), &class) {
386                best = Some(class);
387            }
388        }
389        best
390    }
391}
392
393fn class_precedes(current: Option<&ExactClass>, candidate: &ExactClass) -> bool {
394    current.is_none_or(|current| {
395        (candidate.after_occurrences.len(), candidate.token_count)
396            > (current.after_occurrences.len(), current.token_count)
397    })
398}

408fn anchor_class_may_be_introduced(inputs: AnchorClassInputs<'_, '_>) -> bool {
409    inputs.after.len() >= 2
410        && has_distinct_anchors(inputs.after)
411        && inputs.after.iter().any(|occurrence| {
412            let path = &inputs.after_view.files[occurrence.file].path;
413            inputs.changed_paths.contains(path) && !inputs.suppressed_paths.contains(path)
414        })
415}
416
417fn has_distinct_anchors(occurrences: &[AnchorOccurrence]) -> bool {
418    let Some(first) = occurrences.first() else {
419        return false;
420    };
421    occurrences[1..]
422        .iter()
423        .any(|candidate| distinct_anchor(*first, *candidate))
424}

495fn candidate_regions(
496    view: &RegionView<'_>,
497    anchor: AnchorKey,
498    seed: AnchorOccurrence,
499    occurrences: &[AnchorOccurrence],
500    comparison: &mut ComparisonState,
501) -> Vec<Candidate> {
502    let mut candidates = Vec::new();
503    let mut candidate_ranges = BTreeSet::new();
504    for peer in occurrences
505        .iter()
506        .copied()
507        .filter(|peer| distinct_anchor(seed, *peer))
508    {
509        let Some((start, end)) = maximal_exact_region(view, seed, peer, comparison) else {
510            if comparison.incomplete {
511                break;
512            }
513            continue;
514        };
515        if end.saturating_sub(start) < MIN_DEDUP_REGION_TOKENS {
516            continue;
517        }
518        let candidate = Candidate {
519            file: seed.file,
520            start,
521            end,
522            anchor,
523            anchor_offset: usize::try_from(seed.token_index)
524                .unwrap_or(usize::MAX)
525                .saturating_sub(start),
526        };
527        if candidate_ranges.insert((start, end)) {
528            candidates.push(candidate);
529        }
530    }
531    candidates.sort_by(|left, right| (right.len(), left.start).cmp(&(left.len(), right.start)));
532    candidates
533}

535fn distinct_anchor(left: AnchorOccurrence, right: AnchorOccurrence) -> bool {
536    left.file != right.file
537        || usize::try_from(left.token_index.abs_diff(right.token_index)).unwrap_or(usize::MAX)
538            >= MIN_DEDUP_REGION_TOKENS
539}

541fn maximal_exact_region(
542    view: &RegionView<'_>,
543    left: AnchorOccurrence,
544    right: AnchorOccurrence,
545    comparison: &mut ComparisonState,
546) -> Option<(usize, usize)> {
547    let left_anchor = usize::try_from(left.token_index).ok()?;
548    let right_anchor = usize::try_from(right.token_index).ok()?;
549    if !anchor_tokens_match(
550        view,
551        TokenLocation {
552            file: left.file,
553            token: left_anchor,
554        },
555        TokenLocation {
556            file: right.file,
557            token: right_anchor,
558        },
559        comparison,
560    )? {
561        return None;
562    }
563    let left_start = extend_left(
564        view,
565        TokenLocation {
566            file: left.file,
567            token: left_anchor,
568        },
569        TokenLocation {
570            file: right.file,
571            token: right_anchor,
572        },
573        comparison,
574    )?;
575    let left_end = extend_right(
576        view,
577        TokenLocation {
578            file: left.file,
579            token: left_anchor.saturating_add(DEDUP_REGION_KGRAM_TOKENS),
580        },
581        TokenLocation {
582            file: right.file,
583            token: right_anchor.saturating_add(DEDUP_REGION_KGRAM_TOKENS),
584        },
585        comparison,
586    )?;
587    Some((left_start, left_end))
588}

590fn anchor_tokens_match(
591    view: &RegionView<'_>,
592    left: TokenLocation,
593    right: TokenLocation,
594    comparison: &mut ComparisonState,
595) -> Option<bool> {
596    for offset in 0..DEDUP_REGION_KGRAM_TOKENS {
597        if !comparison.tokens_equal(TokenPair::same_view(
598            view,
599            left.offset(offset),
600            right.offset(offset),
601        ))? {
602            return Some(false);
603        }
604    }
605    Some(true)
606}
607
608fn extend_left(
609    view: &RegionView<'_>,
610    mut left: TokenLocation,
611    mut right: TokenLocation,
612    comparison: &mut ComparisonState,
613) -> Option<usize> {
614    while left.token > 0 && right.token > 0 {
615        let prior_left = left.prior();
616        let prior_right = right.prior();
617        if !comparison.tokens_equal(TokenPair::same_view(view, prior_left, prior_right))? {
618            break;
619        }
620        left = prior_left;
621        right = prior_right;
622    }
623    Some(left.token)
624}
625
626fn extend_right(
627    view: &RegionView<'_>,
628    mut left: TokenLocation,
629    mut right: TokenLocation,
630    comparison: &mut ComparisonState,
631) -> Option<usize> {
632    while comparison.token_exists(view, left.file, left.token)
633        && comparison.token_exists(view, right.file, right.token)
634    {
635        if !comparison.tokens_equal(TokenPair::same_view(view, left, right))? {
636            break;
637        }
638        left.token = left.token.saturating_add(1);
639        right.token = right.token.saturating_add(1);
640    }
641    Some(left.token)
642}

644struct ExactClass {
645    digest: String,
646    family: &'static str,
647    before_count: usize,
648    after_occurrences: Vec<ExactOccurrence>,
649    introduced_count: usize,
650    token_count: usize,
651}
652
653impl ExactClass {
654    fn into_finding(self) -> IntroducedDuplicate {
655        let mut occurrences = self
656            .after_occurrences
657            .into_iter()
658            .map(|occurrence| DuplicateOccurrence {
659                path: occurrence.path,
660                range: Some(occurrence.range),
661                callable_kind: None,
662                changed: occurrence.changed,
663            })
664            .collect::<Vec<_>>();
665        occurrences.sort_by(|left, right| {
666            (!left.changed, &left.path, left.range).cmp(&(!right.changed, &right.path, right.range))
667        });
668        let after_count = occurrences.len();
669        let (occurrences, occurrences_truncated) = bound_occurrences(occurrences);
670        IntroducedDuplicate {
671            class_id: format!("region-{}", short_id(&self.digest)),
672            kind: DuplicateKind::TokenRegion,
673            language_family: Some(self.family.into()),
674            before_count: self.before_count,
675            after_count,
676            introduced_count: self.introduced_count,
677            bytes: None,
678            token_count: Some(self.token_count),
679            occurrences,
680            occurrences_truncated,
681        }
682    }
683}

686struct ExactClassInputs<'a> {
687    before: &'a RegionView<'a>,
688    after: &'a RegionView<'a>,
689    family: &'static str,
690    changed_paths: &'a BTreeSet<RepoPath>,

692    callable_findings: &'a [IntroducedDuplicate],

694
695fn exact_class(
696    inputs: ExactClassInputs<'_>,
697    candidate: Candidate,
698    comparison: &mut ComparisonState,
699) -> Option<ExactClass> {
700    let before_occurrences = exact_occurrences(
701        OccurrenceInputs {
702            candidate_view: inputs.after,
703            candidate,
704            target: inputs.before,
705            family: inputs.family,
706            changed_paths: inputs.changed_paths,
707        },
708        comparison,
709    )?;
710    let after_occurrences = exact_occurrences(
711        OccurrenceInputs {
712            candidate_view: inputs.after,
713            candidate,
714            target: inputs.after,
715            family: inputs.family,
716            changed_paths: inputs.changed_paths,
717        },
718        comparison,
719    )?;
720    if after_occurrences.len() < 2 || after_occurrences.len() <= before_occurrences.len() {
721        return None;
722    }
723    let before_by_path = path_counts(&before_occurrences, |occurrence| &occurrence.path);
724    let after_by_path = path_counts(&after_occurrences, |occurrence| &occurrence.path);
725    let grew_paths = paths_with_growth(&before_by_path, &after_by_path);
726    if grew_paths.is_disjoint(inputs.changed_paths)
727        || (!grew_paths.is_empty() && grew_paths.is_subset(inputs.suppressed_paths))
728        || shadowed_by_callable(&after_occurrences, &grew_paths, inputs.callable_findings)
729    {
730        return None;
731    }
732    let Some(digest) = inputs
733        .after
734        .region_digest(candidate.file, candidate.start..candidate.end)
735    else {
736        comparison.mark_malformed_location(inputs.after, candidate.file);
737        return None;
738    };
739    Some(ExactClass {
740        digest,
741        family: inputs.family,
742        before_count: before_occurrences.len(),
743        introduced_count: after_occurrences
744            .len()
745            .saturating_sub(before_occurrences.len()),
746        after_occurrences,
747        token_count: candidate.len(),
748    })
749}
750
751fn shadowed_by_callable(
752    occurrences: &[ExactOccurrence],
753    grew_paths: &BTreeSet<RepoPath>,
754    callable_findings: &[IntroducedDuplicate],
755) -> bool {
756    !grew_paths.is_empty()
757        && grew_paths.iter().all(|path| {
758            occurrences
759                .iter()
760                .filter(|occurrence| &occurrence.path == path)
761                .all(|region| {
762                    callable_findings.iter().any(|finding| {
763                        finding.kind == DuplicateKind::CallableBody
764                            && finding.occurrences.iter().any(|callable| {
765                                callable.path == region.path
766                                    && callable.range.is_some_and(|range| {
767                                        range.start.byte < region.range.end.byte
768                                            && range.end.byte > region.range.start.byte
769                                    })
770                            })
771                    })
772                })
773        })
774}

787struct OccurrenceInputs<'view, 'data> {
788    candidate_view: &'view RegionView<'data>,
789    candidate: Candidate,
790    target: &'view RegionView<'data>,
791    family: &'static str,
792    changed_paths: &'data BTreeSet<RepoPath>,

794
795fn exact_occurrences(
796    inputs: OccurrenceInputs<'_, '_>,
797    comparison: &mut ComparisonState,
798) -> Option<Vec<ExactOccurrence>> {
799    let Some(group) = inputs
800        .target
801        .groups
802        .get(inputs.family)
803        .and_then(|groups| groups.get(&inputs.candidate.anchor))
804    else {
805        return Some(Vec::new());
806    };
807    let mut occurrences = BTreeMap::<(RepoPath, usize), ExactOccurrence>::new();
808    for anchor in group {
809        let Ok(anchor_index) = usize::try_from(anchor.token_index) else {
810            comparison.mark_malformed_location(inputs.target, anchor.file);
811            return None;
812        };
813        let Some(start) = anchor_index.checked_sub(inputs.candidate.anchor_offset) else {
814            continue;
815        };
816        let end = start.saturating_add(inputs.candidate.len());
817        if !comparison.regions_equal(&RegionPair {
818            left_view: inputs.candidate_view,
819            left_file: inputs.candidate.file,
820            left: inputs.candidate.start..inputs.candidate.end,
821            right_view: inputs.target,
822            right_file: anchor.file,
823            right: start..end,
824        })? {
825            continue;
826        }
827        let file = &inputs.target.files[anchor.file];
828        let Some(range) = inputs.target.region_range(anchor.file, start..end) else {
829            comparison.mark_malformed_location(inputs.target, anchor.file);
830            return None;
831        };
832        occurrences
833            .entry((file.path.clone(), start))
834            .or_insert_with(|| ExactOccurrence {
835                file: anchor.file,
836                path: file.path.clone(),
837                start,
838                end,
839                range,
840                changed: inputs.changed_paths.contains(&file.path),
841            });
842    }
843    let mut occurrences = occurrences.into_values().collect::<Vec<_>>();
844    occurrences.sort_by(|left, right| {
845        (&left.path, left.start, left.end).cmp(&(&right.path, right.start, right.end))
846    });
847    let mut non_overlapping = Vec::<ExactOccurrence>::new();
848    for occurrence in occurrences {
849        if non_overlapping
850            .last()
851            .is_none_or(|prior| prior.path != occurrence.path || prior.end <= occurrence.start)
852        {
853            non_overlapping.push(occurrence);
854        }
855    }
856    Some(non_overlapping)
857}

859fn interval_contains(
860    covered: &BTreeMap<usize, Vec<Range<usize>>>,
861    file: usize,
862    token: u32,
863) -> bool {
864    let token = usize::try_from(token).unwrap_or(usize::MAX);
865    covered
866        .get(&file)
867        .is_some_and(|ranges| ranges.iter().any(|range| range.contains(&token)))
868}
869
870fn mark_covered(covered: &mut BTreeMap<usize, Vec<Range<usize>>>, occurrences: &[ExactOccurrence]) {
871    for occurrence in occurrences.iter().filter(|occurrence| occurrence.changed) {
872        covered
873            .entry(occurrence.file)
874            .or_default()
875            .push(occurrence.start..occurrence.end);
876    }
877}

902fn record_bounded_path(paths: &mut BTreeSet<RepoPath>, paths_truncated: &mut bool, path: RepoPath) {
903    if paths.contains(&path) {
904        return;
905    }

907        *paths_truncated = true;
908        return;
909    }

928#[expect(
929    clippy::struct_excessive_bools,
930    reason = "comparison completeness, report truncation, limit identity, and sample truncation are independent states"
931)]
932struct ComparisonState {
933    remaining: usize,
934    incomplete: bool,
935    findings_truncated: bool,
936    limit_reached: bool,
937    decode_processed: Option<(usize, usize)>,
938    malformed_evidence: bool,
939    views: BTreeSet<SenseView>,
940    limit_paths: BTreeSet<RepoPath>,
941    malformed_paths: BTreeSet<RepoPath>,
942    malformed_paths_truncated: bool,
943}
944
945impl Default for ComparisonState {
946    fn default() -> Self {
947        Self {
948            remaining: MAX_DEDUP_TOKEN_COMPARISONS,
949            incomplete: false,
950            findings_truncated: false,
951            limit_reached: false,
952            decode_processed: None,
953            malformed_evidence: false,
954            views: BTreeSet::new(),
955            limit_paths: BTreeSet::new(),
956            malformed_paths: BTreeSet::new(),
957            malformed_paths_truncated: false,
958        }
959    }
960}
961
962impl ComparisonState {
963    fn observe_malformed_view(&mut self, view: &RegionView<'_>) {
964        if view.malformed_paths.is_empty() && !view.malformed_paths_truncated {
965            return;
966        }
967        self.incomplete = true;
968        self.malformed_evidence = true;
969        self.views.insert(view.view);
970        for path in &view.malformed_paths {
971            record_bounded_path(
972                &mut self.malformed_paths,
973                &mut self.malformed_paths_truncated,
974                path.clone(),
975            );
976        }
977        self.malformed_paths_truncated |= view.malformed_paths_truncated;
978    }
979
980    fn record_limit_location(&mut self, view: &RegionView<'_>, file: usize) {
981        self.views.insert(view.view);
982        let Some(path) = view.files.get(file).map(|file| &file.path) else {
983            return;
984        };
985        if self.limit_paths.len() < MAX_OBSERVATION_PATHS {
986            self.limit_paths.insert(path.clone());
987        }
988    }
989
990    fn record_limit_pair(&mut self, pair: &TokenPair<'_, '_>) {
991        self.record_limit_location(pair.left_view, pair.left.file);
992        self.record_limit_location(pair.right_view, pair.right.file);
993    }
994
995    fn mark_malformed_location(&mut self, view: &RegionView<'_>, file: usize) {
996        let cache = view.decoded_evidence.borrow();
997        if cache.decode_exhausted {
998            self.incomplete = true;
999            self.decode_processed = Some((
1000                cache.decode_limit,
1001                cache.decode_limit - cache.decode_remaining,
1002            ));
1003            self.record_limit_location(view, file);
1004            return;
1005        }
1006        self.incomplete = true;
1007        self.malformed_evidence = true;
1008        self.views.insert(view.view);
1009        let Some(path) = view.files.get(file).map(|file| &file.path) else {
1010            return;
1011        };
1012        record_bounded_path(
1013            &mut self.malformed_paths,
1014            &mut self.malformed_paths_truncated,
1015            path.clone(),
1016        );
1017    }
1018
1019    fn issues(&self) -> Vec<SenseIssue> {
1020        let mut issues = Vec::new();
1021        if let Some(issue) = self.limit_issue() {
1022            issues.push(issue);
1023        }
1024        if self.malformed_evidence {
1025            let mut issue = SenseIssue::new(
1026                "dedup_exact_evidence_malformed",
1027                "exact token evidence was malformed; duplicate comparison could not complete",
1028            );
1029            issue.views = self.views.iter().copied().collect();
1030            issue.paths = self.malformed_paths.iter().cloned().collect();
1031            issue.paths_truncated = self.malformed_paths_truncated;
1032            issue.next_step = Some(
1033                concat!(
1034                    "Rerun with a fresh local cache. If the error repeats, report the affected ",
1035                    "files; --allow-partial cannot make malformed evidence complete."
1036                )
1037                .into(),
1038            );
1039            issues.push(issue);
1040        }
1041        issues
1042    }
1043
1044    fn limit_issue(&self) -> Option<SenseIssue> {
1045        let (code, stage, limit, processed, work) =
1046            if let Some((limit, processed)) = self.decode_processed {
1047                (
1048                    "dedup_exact_decode_limit",
1049                    DedupLimitStage::ExactEvidenceDecoding,
1050                    limit,
1051                    processed,
1052                    "exact evidence decoding",
1053                )
1054            } else if self.limit_reached {
1055                (
1056                    "dedup_exact_comparison_limit",
1057                    DedupLimitStage::ExactTokenComparison,
1058                    MAX_DEDUP_TOKEN_COMPARISONS,
1059                    MAX_DEDUP_TOKEN_COMPARISONS,
1060                    "exact token comparison",
1061                )
1062            } else {
1063                return None;
1064            };
1065        let mut issue = SenseIssue::new(
1066            code,
1067            format!(
1068                "{work} reached the fixed work limit of {limit}; candidate duplicate regions remain unchecked"
1069            ),
1070        );
1071        issue.stage = Some(stage);
1072        issue.views = self.views.iter().copied().collect();
1073        issue.limit = Some(limit);
1074        issue.processed = Some(processed);
1075        issue.paths = self.limit_paths.iter().cloned().collect();
1076        // Evaluation stops at the exhausted comparison. Paths from unchecked candidates are
1077        // intentionally unknown, so this remains a bounded sample rather than complete evidence.
1078        issue.paths_truncated = true;
1079        issue.next_step = Some(concat!(
1080            "No runtime option raises this fixed safety limit. Split unusually repetitive source ",
1081            "or, for generated/vendor trees, use literal targets.exclude entries such as ",
1082            "{\"schemaVersion\":1,\"targets\":{\"exclude\":[\"generated\",\"vendor\"]}}. See ",
1083            "https://the-open-engine.github.io/opcore/v",
1084            env!("CARGO_PKG_VERSION_MAJOR"),
1085            ".",
1086            env!("CARGO_PKG_VERSION_MINOR"),
1087            "/docs/configuration.html#select-targets."
1088        ).into());
1089        Some(issue)
1090    }
1091
1092    fn token_exists(&mut self, view: &RegionView<'_>, file_index: usize, token: usize) -> bool {
1093        let Some(tokens) = view.tokens(file_index) else {
1094            self.mark_malformed_location(view, file_index);
1095            return false;
1096        };
1097        token < tokens.tokens.len()
1098    }
1099
1100    fn tokens_equal(&mut self, pair: TokenPair<'_, '_>) -> Option<bool> {
1101        if self.remaining == 0 {
1102            self.incomplete = true;
1103            self.limit_reached = true;
1104            self.record_limit_pair(&pair);
1105            return None;
1106        }
1107        self.remaining = self.remaining.saturating_sub(1);
1108        let Some(left_tokens) = pair.left_view.tokens(pair.left.file) else {
1109            self.mark_malformed_location(pair.left_view, pair.left.file);
1110            return None;
1111        };
1112        let Some(right_tokens) = pair.right_view.tokens(pair.right.file) else {
1113            self.mark_malformed_location(pair.right_view, pair.right.file);
1114            return None;
1115        };
1116        let Some(left) = pair.left_view.exact_token(pair.left, &left_tokens) else {
1117            self.mark_malformed_location(pair.left_view, pair.left.file);
1118            return None;
1119        };
1120        let Some(right) = pair.right_view.exact_token(pair.right, &right_tokens) else {
1121            self.mark_malformed_location(pair.right_view, pair.right.file);
1122            return None;
1123        };
1124        Some(left == right)
1125    }
1126
1127    fn regions_equal(&mut self, pair: &RegionPair<'_, '_>) -> Option<bool> {
1128        if pair.left.len() != pair.right.len() {
1129            return Some(false);
1130        }
1131        if !self.token_exists(
1132            pair.left_view,
1133            pair.left_file,
1134            pair.left.end.saturating_sub(1),
1135        ) || !self.token_exists(
1136            pair.right_view,
1137            pair.right_file,
1138            pair.right.end.saturating_sub(1),
1139        ) {
1140            return if self.incomplete { None } else { Some(false) };
1141        }
1142        for offset in 0..pair.left.len() {
1143            if !self.tokens_equal(TokenPair {
1144                left_view: pair.left_view,
1145                left: TokenLocation {
1146                    file: pair.left_file,
1147                    token: pair.left.start.saturating_add(offset),
1148                },
1149                right_view: pair.right_view,
1150                right: TokenLocation {
1151                    file: pair.right_file,
1152                    token: pair.right.start.saturating_add(offset),
1153                },
1154            })? {
1155                return Some(false);
1156            }
1157        }
1158        Some(true)
1159    }
1160}
