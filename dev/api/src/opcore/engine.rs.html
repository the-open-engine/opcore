62    /// Evaluates one immutable before/after request.
63    ///
64    /// # Errors
65    ///
66    /// Returns an error when fact serialization, worker execution, or deterministic cache
67    /// publication fails.
68    pub async fn evaluate(
69        &self,
70        request: EvaluationRequest,
71        cancel: CancelToken,
72    ) -> anyhow::Result<Assessment> {
73        let started = Instant::now();
74        let cache_before = self.cache.metadata();
75        let prepared = match prepare_evaluation(&request, &cancel, started) {
76            Ok(prepared) => prepared,
77            Err(assessment) => return Ok(*assessment),
78        };
79        let PreparedEvaluation {
80            inputs,
81            path_continuity,
82            files_read,
83        } = prepared;
84        let analyzed =
85            analyze_prepared(inputs, request.limits, Arc::clone(&self.cache), cancel).await?;
86        let batch = collect_analysis(analyzed)?;
87        Ok(finalize_assessment(FinalAssessment {
88            request,
89            path_continuity,
90            files_read,
91            batch,
92            cache_before,
93            cache_after: self.cache.metadata(),
94            started,
95        }))
96    }
97}

105fn prepare_evaluation(
106    request: &EvaluationRequest,
107    cancel: &CancelToken,
108    started: Instant,
109) -> Result<PreparedEvaluation, Box<Assessment>> {
110    let paths = evaluation_paths(request);
111    if paths.len() > MAX_FILES {
112        return Err(Box::new(incomplete_assessment(
113            request.valid_as_of.clone(),
114            format!(
115                "request includes {} files; limit is {MAX_FILES}",
116                paths.len()
117            ),
118            started,
119        )));
120    }
121    let mut total_bytes = 0usize;
122    let mut files_read = 0usize;
123    let mut inputs = Vec::with_capacity(paths.len());
124    for path in paths {
125        if cancel.is_cancelled() {
126            return Err(Box::new(cancelled_assessment(
127                request.valid_as_of.clone(),
128                started,
129            )));
130        }
131        let input = evaluation_input(request, path);
132        files_read = files_read.saturating_add(input.file_count());
133        total_bytes = total_bytes.saturating_add(input.byte_count());
134        if total_bytes > MAX_TOTAL_SOURCE_BYTES {
135            return Err(Box::new(incomplete_assessment(
136                request.valid_as_of.clone(),
137                format!("request source exceeds the {MAX_TOTAL_SOURCE_BYTES}-byte limit"),
138                started,
139            )));
140        }
141        inputs.push(input);
142    }
143    let path_continuity = path_continuity(request.comparison, &inputs);
144    Ok(PreparedEvaluation {
145        inputs,
146        path_continuity,
147        files_read,
148    })
149}

151fn evaluation_paths(request: &EvaluationRequest) -> BTreeSet<RepoPath> {
152    let mut paths = request.paths.clone();
153    if !matches!(request.scope, Scope::Workspace) {
154        return paths;
155    }
156    paths.extend(request.after.paths().cloned());
157    if request.comparison != Comparison::All
158        && let Some(before) = &request.before
159    {
160        paths.extend(before.paths().cloned());
161    }
162    paths
163}

165fn evaluation_input(request: &EvaluationRequest, path: RepoPath) -> EvaluationInput {
166    let after = request.after.read(&path);
167    let before = match (&request.before, request.comparison) {
168        (Some(view), comparison) if comparison != Comparison::All => view.read(&path),
169        _ => None,
170    };
171    EvaluationInput {
172        path,
173        before,
174        after,
175    }
176}

178fn path_continuity(
179    comparison: Comparison,
180    inputs: &[EvaluationInput],

182    if comparison == Comparison::All {
183        BTreeMap::new()
184    } else {
185        exact_content_path_continuity(inputs)
186    }

189async fn analyze_prepared(
190    inputs: Vec<EvaluationInput>,
191    limits: RuleLimits,
192    cache: Arc<dyn FactCache>,
193    cancel: CancelToken,
194) -> anyhow::Result<Vec<Result<AnalyzedInput, anyhow::Error>>> {
195    let parser_options = serde_json::to_vec(&limits)?;
196    Ok(tokio::task::spawn_blocking(move || {
197        run_analysis(inputs, &limits, &parser_options, cache.as_ref(), &cancel)
198    })
199    .await?)
200}

202fn run_analysis(
203    inputs: Vec<EvaluationInput>,
204    limits: &RuleLimits,
205    parser_options: &[u8],
206    cache: &dyn FactCache,
207    cancel: &CancelToken,
208) -> Vec<Result<AnalyzedInput, anyhow::Error>> {
209    let analyze = |input| {
210        analyze_input(
211            input,
212            AnalysisContext {
213                limits,
214                parser_options,
215                cache,
216                cancel,
217            },
218        )
219    };
220    if inputs.len() < PARALLEL_FILE_THRESHOLD {
221        return inputs.into_iter().map(analyze).collect();
222    }
223    match worker_pool() {
224        Some(pool) => pool.install(|| inputs.into_par_iter().map(analyze).collect()),
225        None => inputs.into_iter().map(analyze).collect(),
226    }
227}

230struct AnalysisBatch {
231    files_considered: usize,
232    files_covered: usize,
233    coverage_gaps: Vec<CoverageGap>,
234    before_diagnostics: Vec<EvaluatedDiagnostic>,
235    after_diagnostics: Vec<EvaluatedDiagnostic>,
236    cancelled: bool,
237}
238
239impl AnalysisBatch {
240    fn push(&mut self, result: AnalyzedInput) {
241        self.files_considered = self.files_considered.saturating_add(1);
242        match result {
243            AnalyzedInput::Ready { before, after } => self.push_ready(before, after),
244            AnalyzedInput::Unsupported {
245                path,
246                language,
247                reason,
248            } => self.coverage_gaps.push(CoverageGap {
249                path,
250                status: CoverageStatus::Unsupported,
251                language,
252                reason: Some(reason),
253            }),
254            AnalyzedInput::Cancelled { path, language } => {
255                self.cancelled = true;
256                self.coverage_gaps.push(CoverageGap {
257                    path,
258                    status: CoverageStatus::Incomplete,
259                    language,
260                    reason: Some("evaluation cancelled".into()),
261                });
262            }
263        }
264    }
265
266    fn push_ready(&mut self, before: Option<Box<AnalyzedFile>>, after: Option<Box<AnalyzedFile>>) {
267        self.files_covered = self.files_covered.saturating_add(1);
268        if let Some(item) = before {
269            self.before_diagnostics
270                .extend(materialize(&item.file, &item.facts));
271        }
272        if let Some(item) = after {
273            self.after_diagnostics
274                .extend(materialize(&item.file, &item.facts));
275        }
276    }
277}

279fn collect_analysis(
280    analyzed: Vec<Result<AnalyzedInput, anyhow::Error>>,
281) -> anyhow::Result<AnalysisBatch> {
282    let mut batch = AnalysisBatch::default();
283    for result in analyzed {
284        batch.push(result?);
285    }
286    Ok(batch)

299fn finalize_assessment(mut final_state: FinalAssessment) -> Assessment {
300    final_state
301        .batch
302        .before_diagnostics
303        .sort_by(diagnostic_order);
304    final_state
305        .batch
306        .after_diagnostics
307        .sort_by(diagnostic_order);
308    let after_diagnostics = std::mem::take(&mut final_state.batch.after_diagnostics);
309    let mut diagnostics = compared_diagnostics(
310        final_state.request.comparison,
311        &final_state.batch.before_diagnostics,
312        after_diagnostics,
313        &final_state.path_continuity,
314        final_state.request.public_fingerprint_comparison,
315    );
316    let diagnostics_truncated = diagnostics.len() > MAX_DIAGNOSTICS;
317    diagnostics.truncate(MAX_DIAGNOSTICS);
318    if diagnostics_truncated {
319        final_state.batch.coverage_gaps.push(CoverageGap {
320            path: RepoPath::request_marker(),
321            status: CoverageStatus::Incomplete,
322            language: None,
323            reason: Some(format!(
324                "diagnostics exceeded the {MAX_DIAGNOSTICS}-finding output limit"
325            )),
326        });
327    }
328    final_state
329        .batch
330        .coverage_gaps
331        .sort_by(|left, right| left.path.cmp(&right.path));
332    let cache = metadata_delta(&final_state.cache_before, final_state.cache_after);
333    let status = assessment_status(
334        final_state.batch.cancelled,
335        diagnostics_truncated,
336        &diagnostics,
337        &final_state.batch.coverage_gaps,
338    );
339    Assessment {
340        status,
341        diagnostics,
342        coverage: Coverage {
343            files_considered: final_state.batch.files_considered,
344            files_covered: final_state.batch.files_covered,
345            gaps: final_state.batch.coverage_gaps,
346        },
347        valid_as_of: final_state.request.valid_as_of,
348        provider: ProviderMetadata::default(),
349        timing: Timing {
350            duration_ms: u64::try_from(final_state.started.elapsed().as_millis())
351                .unwrap_or(u64::MAX),
352            files_read: final_state.files_read,
353            files_parsed: cache.misses,
354        },
355        cache,
356    }
357}

359fn compared_diagnostics(
360    comparison: Comparison,
361    before: &[EvaluatedDiagnostic],
362    after: Vec<EvaluatedDiagnostic>,
363    continuity: &BTreeMap<RepoPath, RepoPath>,
364    public_fingerprint_comparison: bool,
365) -> Vec<Diagnostic> {
366    match comparison {
367        Comparison::All => after.into_iter().map(|item| item.diagnostic).collect(),
368        Comparison::Introduced if public_fingerprint_comparison => {
369            public_fingerprint_difference(after, before)
370        }
371        Comparison::Introduced => semantic_difference(after, before, continuity),
372    }
373}
374
375fn public_fingerprint_difference(
376    candidates: Vec<EvaluatedDiagnostic>,
377    subtract: &[EvaluatedDiagnostic],
378) -> Vec<Diagnostic> {
379    let prior = subtract
380        .iter()
381        .map(|item| item.diagnostic.fingerprint.as_str())
382        .collect::<BTreeSet<_>>();
383    candidates
384        .into_iter()
385        .filter(|item| !prior.contains(item.diagnostic.fingerprint.as_str()))
386        .map(|item| item.diagnostic)
387        .collect()
388}

390fn assessment_status(
391    cancelled: bool,
392    diagnostics_truncated: bool,
393    diagnostics: &[Diagnostic],
394    coverage_gaps: &[CoverageGap],
395) -> AssessmentStatus {
396    if cancelled {
397        return AssessmentStatus::Cancelled;
398    }
399    if diagnostics_truncated || has_coverage(coverage_gaps, CoverageStatus::Incomplete) {
400        return AssessmentStatus::Incomplete;
401    }
402    if !diagnostics.is_empty() {
403        return AssessmentStatus::Findings;
404    }
405    AssessmentStatus::Clean
406}
407
408fn has_coverage(gaps: &[CoverageGap], status: CoverageStatus) -> bool {
409    gaps.iter().any(|item| item.status == status)
410}

431fn exact_content_path_continuity(inputs: &[EvaluationInput]) -> BTreeMap<RepoPath, RepoPath> {
432    type ContinuityKey = (crate::identity::ContentId, crate::model::Language, String);
433    let mut removed = BTreeMap::<ContinuityKey, Vec<RepoPath>>::new();
434    let mut added = BTreeMap::<ContinuityKey, Vec<RepoPath>>::new();
435    for input in inputs {
436        match (&input.before, &input.after) {
437            (Some(before), None) => removed
438                .entry((
439                    before.content_id,
440                    before.language,
441                    before.language_mode.clone(),
442                ))
443                .or_default()
444                .push(input.path.clone()),
445            (None, Some(after)) => added
446                .entry((
447                    after.content_id,
448                    after.language,
449                    after.language_mode.clone(),
450                ))
451                .or_default()
452                .push(input.path.clone()),
453            _ => {}
454        }
455    }
456    let mut continuity = BTreeMap::new();
457    for (key, mut before_paths) in removed {
458        let Some(after_paths) = added.get_mut(&key) else {
459            continue;
460        };
461        before_paths.sort();
462        after_paths.sort();
463        for (before, after) in before_paths.into_iter().zip(after_paths.iter().cloned()) {
464            continuity.insert(before, after);
465        }
466    }
467    continuity
468}

499fn analyze_input(
500    input: EvaluationInput,
501    context: AnalysisContext<'_>,
502) -> Result<AnalyzedInput, anyhow::Error> {
503    let language = input
504        .after
505        .as_ref()
506        .or(input.before.as_ref())
507        .map(|file| file.language);
508    if input.before.is_none() && input.after.is_none() {
509        return Ok(AnalyzedInput::Unsupported {
510            path: input.path,
511            language: None,
512            reason: "path is missing or its language is outside the supported envelope".into(),
513        });
514    }
515    match analyze_pair(input.before, input.after, &context) {
516        Ok((before, after)) => Ok(AnalyzedInput::Ready {
517            before: before.map(Box::new),
518            after: after.map(Box::new),
519        }),
520        Err(AnalysisError::Unsupported(reason)) => Ok(AnalyzedInput::Unsupported {
521            path: input.path,
522            language,
523            reason,
524        }),
525        Err(AnalysisError::Cancelled) => Ok(AnalyzedInput::Cancelled {
526            path: input.path,
527            language,
528        }),
529        Err(error) => Err(error.into()),
530    }
531}

533fn analyze_pair(
534    before: Option<Arc<SourceFile>>,
535    after: Option<Arc<SourceFile>>,
536    context: &AnalysisContext<'_>,
537) -> Result<(Option<AnalyzedFile>, Option<AnalyzedFile>), AnalysisError> {
538    let before = before.map(|file| analyze_one(file, context)).transpose()?;
539    let after = after.map(|file| analyze_one(file, context)).transpose()?;

562fn materialize(file: &SourceFile, facts: &FileFacts) -> Vec<EvaluatedDiagnostic> {
563    facts
564        .diagnostics
565        .iter()
566        .map(|raw| {
567            let comparison_key = hex::encode(hash_domain(
568                "diagnostic-comparison/v1",
569                &[
570                    raw.rule_id.as_bytes(),
571                    file.language.family().as_bytes(),
572                    raw.entity_key.as_bytes(),
573                    raw.cause_key.as_bytes(),
574                ],
575            ));
576            let fingerprint = hex::encode(hash_domain(
577                "diagnostic-fingerprint/v1",
578                &[file.path.as_bytes(), comparison_key.as_bytes()],
579            ));
580            EvaluatedDiagnostic {
581                diagnostic: Diagnostic {
582                    rule_id: raw.rule_id.clone(),
583                    severity: raw.severity,
584                    message: raw.message.clone(),
585                    path: file.path.clone(),
586                    range: raw.range,
587                    fingerprint,
588                    language: file.language,
589                    evidence: raw.evidence.clone(),
590                },
591                comparison_key,
592            }
593        })
594        .collect()
595}

597fn diagnostic_order(left: &EvaluatedDiagnostic, right: &EvaluatedDiagnostic) -> std::cmp::Ordering {
598    left.diagnostic
599        .path
600        .cmp(&right.diagnostic.path)
601        .then_with(|| left.diagnostic.rule_id.cmp(&right.diagnostic.rule_id))
602        .then_with(|| {
603            left.diagnostic
604                .fingerprint
605                .cmp(&right.diagnostic.fingerprint)
606        })
607}

609fn semantic_difference(
610    candidates: Vec<EvaluatedDiagnostic>,
611    subtract: &[EvaluatedDiagnostic],
612    continuity: &BTreeMap<RepoPath, RepoPath>,
613) -> Vec<Diagnostic> {
614    let mut exact_counts = BTreeMap::<(RepoPath, String), usize>::new();
615    let mut alias_counts = BTreeMap::<(RepoPath, String), usize>::new();
616    for item in subtract {
617        *exact_counts
618            .entry((item.diagnostic.path.clone(), item.comparison_key.clone()))
619            .or_default() += 1;
620        if let Some(after) = continuity.get(&item.diagnostic.path) {
621            *alias_counts
622                .entry((after.clone(), item.comparison_key.clone()))
623                .or_default() += 1;
624        }
625    }
626    let mut pending = Vec::new();
627    for item in candidates {
628        let exact_key = (item.diagnostic.path.clone(), item.comparison_key.clone());
629        if !decrement(&mut exact_counts, &exact_key) {
630            pending.push(item);
631        }
632    }
633    pending
634        .into_iter()
635        .filter_map(|item| {
636            let alias_key = (item.diagnostic.path.clone(), item.comparison_key.clone());
637            if decrement(&mut alias_counts, &alias_key) {
638                None
639            } else {
640                Some(item.diagnostic)
641            }
642        })
643        .collect()
644}
645
646fn decrement<K: Ord>(counts: &mut BTreeMap<K, usize>, key: &K) -> bool {
647    counts.get_mut(key).is_some_and(|count| {
648        if *count == 0 {
649            false
650        } else {
651            *count -= 1;
652            true
653        }
654    })
655}
