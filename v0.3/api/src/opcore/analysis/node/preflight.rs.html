5#[derive(Clone, Copy, Eq, PartialEq)]
6enum ScanState {
7    Code,
8    SingleQuote,
9    DoubleQuote,
10    Template,
11    LineComment,
12    BlockComment,
13}

15pub(super) fn preflight_structure(source: &str, cancel: &CancelToken) -> Result<(), AnalysisError> {
16    let bytes = source.as_bytes();
17    let mut index = 0usize;
18    let mut scanner = StructuralScanner::default();
19    while index < bytes.len() {
20        if index.is_multiple_of(1_024) {
21            check_cancel(cancel)?;
22        }
23        let byte = bytes[index];
24        let next = bytes.get(index.saturating_add(1)).copied();
25        index = index.saturating_add(scanner.consume(byte, next)?);
26        index = index.saturating_add(1);
27    }
28    Ok(())
29}

31struct StructuralScanner {
32    state: ScanState,
33    escaped: bool,
34    depth: usize,
35    unary: usize,
36    template_expression_depths: Vec<usize>,
37}
38
39impl Default for StructuralScanner {
40    fn default() -> Self {
41        Self {
42            state: ScanState::Code,
43            escaped: false,
44            depth: 0,
45            unary: 0,
46            template_expression_depths: Vec::new(),
47        }
48    }
49}
50
51impl StructuralScanner {
52    fn consume(&mut self, byte: u8, next: Option<u8>) -> Result<usize, AnalysisError> {
53        let skipped = match self.state {
54            ScanState::SingleQuote | ScanState::DoubleQuote => self.consume_quote(byte),
55            ScanState::Template => self.consume_template(byte, next)?,
56            ScanState::LineComment => self.consume_line_comment(byte),
57            ScanState::BlockComment => self.consume_block_comment(byte, next),
58            ScanState::Code => self.consume_code(byte, next),
59        };
60        reject_excessive_structure(self.depth, self.unary)?;
61        Ok(skipped)
62    }
63
64    fn consume_quote(&mut self, byte: u8) -> usize {
65        if self.take_escape(byte) {
66            return 0;
67        }
68        let closes = (self.state == ScanState::SingleQuote && byte == b'\'')
69            || (self.state == ScanState::DoubleQuote && byte == b'"');
70        if closes {
71            self.leave_literal();
72        }
73        0
74    }
75
76    fn consume_template(&mut self, byte: u8, next: Option<u8>) -> Result<usize, AnalysisError> {
77        if self.take_escape(byte) {
78            return Ok(0);
79        }
80        if byte == b'`' {
81            self.leave_literal();
82            return Ok(0);
83        }
84        if byte != b'$' || next != Some(b'{') {
85            return Ok(0);
86        }
87        self.depth = self.depth.saturating_add(1);
88        self.template_expression_depths.push(self.depth);
89        reject_excessive_structure(self.depth, self.unary)?;
90        self.state = ScanState::Code;
91        Ok(1)
92    }
93
94    fn consume_line_comment(&mut self, byte: u8) -> usize {
95        if matches!(byte, b'\n' | b'\r') {
96            self.state = ScanState::Code;
97        }
98        0
99    }
100
101    fn consume_block_comment(&mut self, byte: u8, next: Option<u8>) -> usize {
102        if byte == b'*' && next == Some(b'/') {
103            self.state = ScanState::Code;
104            return 1;
105        }
106        0
107    }
108
109    fn consume_code(&mut self, byte: u8, next: Option<u8>) -> usize {
110        if let Some(state) = comment_start(byte, next) {
111            self.state = state;
112            return 1;
113        }
114        if let Some(state) = literal_start(byte) {
115            self.state = state;
116            return 0;
117        }
118        self.consume_code_operator(byte);
119        0
120    }
121
122    fn consume_code_operator(&mut self, byte: u8) {
123        if matches!(byte, b'(' | b'[' | b'{') {
124            self.depth = self.depth.saturating_add(1);
125            self.unary = 0;
126        } else if matches!(byte, b')' | b']') {
127            self.depth = self.depth.saturating_sub(1);
128            self.unary = 0;
129        } else if byte == b'}' {
130            self.close_brace();
131        } else if matches!(byte, b'!' | b'~' | b'+' | b'-') {
132            self.unary = self.unary.saturating_add(1);
133        } else if !byte.is_ascii_whitespace() {
134            self.unary = 0;
135        }
136    }
137
138    fn close_brace(&mut self) {
139        if self.template_expression_depths.last() == Some(&self.depth) {
140            self.template_expression_depths.pop();
141            self.state = ScanState::Template;
142        }
143        self.depth = self.depth.saturating_sub(1);
144        self.unary = 0;
145    }
146
147    fn take_escape(&mut self, byte: u8) -> bool {
148        if self.escaped {
149            self.escaped = false;
150            return true;
151        }
152        if byte == b'\\' {
153            self.escaped = true;
154            return true;
155        }
156        false
157    }
158
159    fn leave_literal(&mut self) {
160        self.state = ScanState::Code;
161        self.unary = 0;
162    }
163}
164
165fn comment_start(byte: u8, next: Option<u8>) -> Option<ScanState> {
166    match (byte, next) {
167        (b'/', Some(b'/')) => Some(ScanState::LineComment),
168        (b'/', Some(b'*')) => Some(ScanState::BlockComment),
169        _ => None,
170    }
171}
172
173const fn literal_start(byte: u8) -> Option<ScanState> {
174    match byte {
175        b'\'' => Some(ScanState::SingleQuote),
176        b'"' => Some(ScanState::DoubleQuote),
177        b'`' => Some(ScanState::Template),
178        _ => None,
179    }
180}
181
182fn reject_excessive_structure(depth: usize, unary: usize) -> Result<(), AnalysisError> {
183    if depth > MAX_STRUCTURAL_DEPTH || unary > MAX_STRUCTURAL_DEPTH {
184        return Err(AnalysisError::Unsupported(format!(
185            "source structural depth exceeds the {MAX_STRUCTURAL_DEPTH}-level safety limit"
186        )));
187    }
188    Ok(())
189}
