17const CONDITIONAL: &str = "conditional_attribute";
18const PATH_OVERRIDE: &str = "path_attribute";
19const UNSUPPORTED_PATH: &str = "unsupported_path";
20
21pub(super) fn extract(
22    root: &SyntaxNode,
23    cancel: &CancelToken,
24) -> Result<DependencyFacts, AnalysisError> {
25    let mut collector = Collector::new(cancel);
26    for node in root.descendants() {
27        if let Some(module) = ast::Module::cast(node.clone()) {
28            collector.observe_module(&module)?;
29        } else if let Some(usage) = ast::Use::cast(node) {
30            collector.observe_use(&usage)?;
31        }
32    }
33    Ok(collector.finish())
34}
35
36struct Collector<'a> {
37    cancel: &'a CancelToken,
38    references: BTreeSet<DependencyReference>,
39    truncated: bool,
40    visited: usize,
41}
42
43impl<'a> Collector<'a> {
44    fn new(cancel: &'a CancelToken) -> Self {
45        Self {
46            cancel,
47            references: BTreeSet::new(),
48            truncated: false,
49            visited: 0,
50        }
51    }
52
53    fn observe_module(&mut self, module: &ast::Module) -> Result<(), AnalysisError> {
54        self.check_cancel()?;
55        if module.item_list().is_some() {
56            return Ok(());
57        }
58        let Some(name) = module.name() else {
59            self.push_unsupported(UNSUPPORTED_PATH);
60            return Ok(());
61        };
62        if has_attribute(module, "path") {
63            self.push_unsupported(PATH_OVERRIDE);
64            return Ok(());
65        }
66        if is_conditional(module.syntax()) {
67            self.push_unsupported(CONDITIONAL);
68            return Ok(());
69        }
70        let mut segments = inline_scope(module.syntax());
71        segments.push(normalize_identifier(name.text().as_str()));
72        self.push(DependencyReferenceKind::RustModule, segments.join("::"), 0);
73        Ok(())
74    }
75
76    fn observe_use(&mut self, usage: &ast::Use) -> Result<(), AnalysisError> {
77        self.check_cancel()?;
78        if is_conditional(usage.syntax()) {
79            self.push_unsupported(CONDITIONAL);
80            return Ok(());
81        }
82        let Some(tree) = usage.use_tree() else {
83            self.push_unsupported(UNSUPPORTED_PATH);
84            return Ok(());
85        };
86        let mut paths = Vec::new();
87        if !flatten_use_tree(&tree, &[], &mut paths) {
88            self.push_unsupported(UNSUPPORTED_PATH);
89            return Ok(());
90        }
91        let scope = inline_scope(usage.syntax());
92        for segments in paths {
93            let Some(specifier) = normalize_use_path(segments, &scope) else {
94                continue;
95            };
96            self.push(DependencyReferenceKind::RustUse, specifier, 0);
97        }
98        Ok(())
99    }
100
101    fn push_unsupported(&mut self, reason: &str) {
102        self.push(DependencyReferenceKind::RustUnsupported, reason.into(), 0);
103    }
104
105    fn push(&mut self, kind: DependencyReferenceKind, specifier: String, level: u32) {
106        if self.references.len() >= MAX_DEPENDENCY_FACTS_PER_FILE {
107            self.truncated = true;
108            return;
109        }
110        self.references.insert(DependencyReference {
111            kind,
112            specifier,
113            level,
114        });
115    }
116
117    fn check_cancel(&mut self) -> Result<(), AnalysisError> {
118        self.visited = self.visited.saturating_add(1);
119        if self
120            .visited
121            .is_multiple_of(crate::cancel::CANCELLATION_POLL_INTERVAL)
122            && self.cancel.is_cancelled()
123        {
124            return Err(AnalysisError::Cancelled);
125        }
126        Ok(())
127    }
128
129    fn finish(self) -> DependencyFacts {
130        DependencyFacts {
131            status: if self.truncated {
132                DependencyExtractionStatus::Truncated
133            } else {
134                DependencyExtractionStatus::Complete
135            },
136            references: self.references.into_iter().collect(),
137        }
138    }
139}

141fn flatten_use_tree(tree: &ast::UseTree, prefix: &[String], output: &mut Vec<Vec<String>>) -> bool {
142    let mut path = prefix.to_vec();
143    if let Some(tree_path) = tree.path() {
144        let Some(mut segments) = path_segments(&tree_path) else {
145            return false;
146        };
147        if prefix.is_empty()
148            && tree
149                .syntax()
150                .text()
151                .to_string()
152                .trim_start()
153                .starts_with("::")
154        {
155            segments.insert(0, "::".into());
156        }
157        path.extend(segments);
158    }
159    if let Some(list) = tree.use_tree_list() {
160        return list
161            .use_trees()
162            .all(|child| flatten_use_tree(&child, &path, output));
163    }
164    if path.last().is_some_and(|segment| segment == "self") && path.len() > 1 {
165        path.pop();
166    }
167    if !path.is_empty() {
168        output.push(path);
169    }
170    true
171}

173fn path_segments(path: &ast::Path) -> Option<Vec<String>> {
174    path.segments()
175        .map(|segment| match segment.kind()? {
176            PathSegmentKind::Name(name) => Some(normalize_identifier(name.text().as_str())),
177            PathSegmentKind::SelfKw => Some("self".into()),
178            PathSegmentKind::SuperKw => Some("super".into()),
179            PathSegmentKind::CrateKw => Some("crate".into()),
180            PathSegmentKind::SelfTypeKw | PathSegmentKind::Type { .. } => None,
181        })
182        .collect()
183}

185fn normalize_use_path(mut path: Vec<String>, inline: &[String]) -> Option<String> {
186    let first = path.first()?.as_str();
187    if first == "crate" {
188        return Some(path.join("::"));
189    }
190    if first == "::" || matches!(first, "std" | "core" | "alloc" | "proc_macro") {
191        return None;
192    }
193    if first == "self" {
194        path.remove(0);
195        let mut normalized = Vec::with_capacity(1 + inline.len() + path.len());
196        normalized.push("self".into());
197        normalized.extend(inline.iter().cloned());
198        normalized.extend(path);
199        return Some(normalized.join("::"));
200    }
201    if first == "super" {
202        let mut supers = 0usize;
203        while path.first().is_some_and(|segment| segment == "super") {
204            path.remove(0);
205            supers = supers.saturating_add(1);
206        }
207        let retained = inline.len().saturating_sub(supers);
208        let external_supers = supers.saturating_sub(inline.len());
209        let mut normalized = Vec::with_capacity(external_supers + retained + path.len() + 1);
210        if external_supers == 0 {
211            normalized.push("self".into());
212            normalized.extend(inline[..retained].iter().cloned());
213        } else {
214            normalized.extend(std::iter::repeat_n("super".into(), external_supers));
215        }
216        normalized.extend(path);
217        return Some(normalized.join("::"));
218    }
219    let mut normalized = inline.to_vec();
220    normalized.extend(path);
221    Some(normalized.join("::"))
222}

224fn inline_scope(node: &SyntaxNode) -> Vec<String> {
225    let mut modules = node
226        .ancestors()
227        .skip(1)
228        .filter_map(ast::Module::cast)
229        .filter(|module| module.item_list().is_some())
230        .filter_map(|module| module.name())
231        .map(|name| normalize_identifier(name.text().as_str()))
232        .collect::<Vec<_>>();
233    modules.reverse();
234    modules
235}

237fn is_conditional(node: &SyntaxNode) -> bool {
238    node.ancestors().any(|ancestor| {
239        ast::Module::cast(ancestor.clone()).is_some_and(|module| {
240            has_attribute_including_inner(&module, "cfg")
241                || has_attribute_including_inner(&module, "cfg_attr")
242        }) || ast::Use::cast(ancestor.clone())
243            .is_some_and(|usage| has_attribute(&usage, "cfg") || has_attribute(&usage, "cfg_attr"))
244            || ast::SourceFile::cast(ancestor).is_some_and(|source| {
245                has_attribute_including_inner(&source, "cfg")
246                    || has_attribute_including_inner(&source, "cfg_attr")
247            })
248    })
249}

251fn has_attribute(owner: &impl HasAttrs, name: &str) -> bool {
252    attributes_named(owner.attrs(), name)
253}
254
255fn has_attribute_including_inner(owner: &impl HasAttrs, name: &str) -> bool {
256    attributes_named(ast::attrs_including_inner(owner), name)
257}
258
259fn attributes_named(attributes: impl Iterator<Item = ast::Attr>, name: &str) -> bool {
260    attributes.into_iter().any(|attribute| {
261        let text = attribute.syntax().text().to_string();
262        let trimmed = text
263            .trim_start_matches('#')
264            .trim_start_matches('!')
265            .trim_start();
266        let Some(body) = trimmed.strip_prefix('[').map(str::trim_start) else {
267            return false;
268        };
269        body.strip_prefix(name).is_some_and(|rest| {
270            rest.starts_with(|character: char| {
271                character.is_ascii_whitespace() || matches!(character, '(' | '=' | ']')
272            })
273        })
274    })
275}

277fn normalize_identifier(identifier: &str) -> String {
278    identifier.strip_prefix("r#").unwrap_or(identifier).into()
