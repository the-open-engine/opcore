1use crate::model::{Position, SourceRange};
2
3#[derive(Debug)]
4pub(super) struct LineIndex {
5    starts: Vec<usize>,
6    source_len: usize,
7}
8
9impl LineIndex {
10    pub(super) fn new(source: &[u8]) -> Self {
11        let mut starts = vec![0];
12        starts.extend(
13            source
14                .iter()
15                .enumerate()
16                .filter(|(_, byte)| **byte == b'\n')
17                .map(|(index, _)| index + 1),
18        );
19        Self {
20            starts,
21            source_len: source.len(),
22        }
23    }
24
25    pub(super) fn position(&self, offset: usize) -> Position {
26        let offset = offset.min(self.source_len);
27        let line_index = self
28            .starts
29            .partition_point(|start| *start <= offset)
30            .saturating_sub(1);
31        Position {
32            line: u32::try_from(line_index + 1).unwrap_or(u32::MAX),
33            column: u32::try_from(offset.saturating_sub(self.starts[line_index]) + 1)
34                .unwrap_or(u32::MAX),
35            byte: u32::try_from(offset).unwrap_or(u32::MAX),
36        }
37    }
38
39    pub(super) fn range(&self, start: usize, end: usize) -> SourceRange {
40        SourceRange {
41            start: self.position(start),
42            end: self.position(end),
43        }
44    }
45}
46
47pub(super) fn byte_range(source: &[u8], start: usize, end: usize) -> SourceRange {
48    LineIndex::new(source).range(start, end)
