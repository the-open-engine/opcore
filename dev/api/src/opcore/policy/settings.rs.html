12#[derive(Clone, Debug, Default, Deserialize, Eq, PartialEq, Serialize)]
13#[serde(default, deny_unknown_fields, rename_all = "camelCase")]
14pub struct TargetPolicy {
15    pub exclude: Vec<RepoPath>,
16}
17
18impl TargetPolicy {
19    #[must_use]
20    pub fn includes(&self, path: &RepoPath) -> bool {
21        !self.exclude.iter().any(|excluded| {
22            path == excluded
23                || path
24                    .as_bytes()
25                    .strip_prefix(excluded.as_bytes())
26                    .is_some_and(|suffix| suffix.starts_with(b"/"))
27        })
28    }
29
30    pub(super) fn validate(&self) -> Result<()> {
31        ensure!(
32            self.exclude.len() <= MAX_TARGET_PATHS,
33            "targets.exclude exceeds {MAX_TARGET_PATHS} paths"
34        );
35        let unique = self.exclude.iter().collect::<BTreeSet<_>>();
36        ensure!(
37            unique.len() == self.exclude.len(),
38            "targets.exclude contains a duplicate path"
39        );
40        Ok(())
41    }
42}

66#[derive(Clone, Debug, Deserialize, Eq, PartialEq, Serialize)]
67#[serde(default, deny_unknown_fields, rename_all = "camelCase")]
68pub struct ProviderOptions {
69    pub roots: Vec<String>,
70}
71
72impl Default for ProviderOptions {
73    fn default() -> Self {
74        Self {
75            roots: vec![".".into()],
76        }
77    }
78}
79
80impl ProviderOptions {
81    fn validate(&self, name: &str) -> Result<()> {
82        ensure!(
83            self.roots.len() <= MAX_TARGET_PATHS,
84            "providers.{name}.roots exceeds {MAX_TARGET_PATHS} paths"
85        );
86        let mut previous = Vec::<&str>::new();
87        for root in &self.roots {
88            if root != "." {
89                RepoPath::from_protocol(root)
90                    .map_err(|error| anyhow::anyhow!("providers.{name}.roots: {error}"))?;
91            }
92            ensure!(
93                !previous.iter().any(|other| roots_overlap(root, other)),
94                "providers.{name}.roots must contain distinct non-overlapping project paths"
95            );
96            previous.push(root);
97        }
98        Ok(())
99    }
100}
101
102fn roots_overlap(left: &str, right: &str) -> bool {
103    left == "."
104        || right == "."
105        || left == right
106        || left
107            .strip_prefix(right)
108            .is_some_and(|suffix| suffix.starts_with('/'))
109        || right
110            .strip_prefix(left)
111            .is_some_and(|suffix| suffix.starts_with('/'))
112}

128        match provider {
129            NativeProvider::RustNative => &self.rust_native,

131            NativeProvider::PythonNative => &self.python_native,
132        }

135    pub(super) fn validate(&self) -> Result<()> {

137            NativeProvider::RustNative,
138            NativeProvider::NodeNative,

140        ] {
141            self.options(provider).validate(provider.as_str())?;
142        }
143        Ok(())

160macro_rules! override_fields {
161    ($(#[$meta:meta])* $name:ident, $target:ty, {$($field:ident : $type:ty),+ $(,)?}) => {
162        $(#[$meta])*
163        #[derive(Clone, Debug, Default, Deserialize, Eq, PartialEq, Serialize)]
164        #[serde(default, deny_unknown_fields, rename_all = "camelCase")]
165        pub struct $name { $(#[serde(skip_serializing_if = "Option::is_none")] pub $field: Option<$type>,)+ }
166        impl $name {
167            pub(super) fn apply(&self, target: &mut $target) {
168                $(if let Some(value) = &self.$field { target.$field.clone_from(value); })+
169            }
170        }
171    };
172}
