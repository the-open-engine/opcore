3use crate::{
4    analysis::AnalysisError,
5    cancel::{CANCELLATION_POLL_INTERVAL, CancelToken},
6};

8#[derive(Clone, Copy)]
9pub(super) enum WalkEvent<'tree> {
10    Enter(Node<'tree>),
11    Leave(Node<'tree>),
12}
13
14pub(super) fn walk<'tree>(
15    root: Node<'tree>,
16    cancel: &CancelToken,
17    mut visitor: impl FnMut(WalkEvent<'tree>) -> Result<bool, AnalysisError>,
18) -> Result<(), AnalysisError> {
19    let mut cursor = root.walk();
20    let mut visited = 0usize;
21    loop {
22        visited = visited.saturating_add(1);
23        poll_cancel(visited, cancel)?;
24        let node = cursor.node();
25        let descend = visitor(WalkEvent::Enter(node))?;
26        if descend_to_child(descend, &mut cursor) {
27            continue;
28        }
29        let _ = visitor(WalkEvent::Leave(node))?;
30        if cursor.goto_next_sibling() {
31            continue;
32        }
33        if !ascend(&mut cursor, &mut visitor)? {
34            return check_cancel(cancel);
35        }
36    }
37}
38
39fn descend_to_child(descend: bool, cursor: &mut TreeCursor<'_>) -> bool {
40    descend && cursor.goto_first_child()
41}
42
43fn poll_cancel(visited: usize, cancel: &CancelToken) -> Result<(), AnalysisError> {
44    if visited.is_multiple_of(CANCELLATION_POLL_INTERVAL) {
45        check_cancel(cancel)
46    } else {
47        Ok(())
48    }
49}
50
51fn ascend<'tree>(
52    cursor: &mut TreeCursor<'tree>,
53    visitor: &mut impl FnMut(WalkEvent<'tree>) -> Result<bool, AnalysisError>,
54) -> Result<bool, AnalysisError> {
55    loop {
56        if !cursor.goto_parent() {
57            return Ok(false);
58        }
59        let _ = visitor(WalkEvent::Leave(cursor.node()))?;
60        if cursor.goto_next_sibling() {
61            return Ok(true);
62        }
63    }
64}

66pub(super) fn walk_without_cancel(root: Node<'_>, mut visitor: impl FnMut(Node<'_>) -> bool) {
67    let mut cursor = root.walk();
68    loop {
69        let descend = visitor(cursor.node());
70        if descend && cursor.goto_first_child() {
71            continue;
72        }
73        if cursor.goto_next_sibling() {
74            continue;
75        }
76        if !ascend_without_events(&mut cursor) {
77            return;
78        }
79    }
80}
81
82fn ascend_without_events(cursor: &mut TreeCursor<'_>) -> bool {
83    loop {
84        if !cursor.goto_parent() {
85            return false;
86        }
87        if cursor.goto_next_sibling() {
88            return true;
89        }
90    }
91}
92
93pub(super) fn check_cancel(cancel: &CancelToken) -> Result<(), AnalysisError> {
94    if cancel.is_cancelled() {
95        Err(AnalysisError::Cancelled)
96    } else {
97        Ok(())
98    }
