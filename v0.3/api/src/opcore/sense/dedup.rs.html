44pub(super) fn introduced_duplicates(inputs: DedupInputs<'_>) -> DedupResult {
45    if inputs.before.id() == inputs.after.id() {
46        return identical_snapshot_result(inputs);
47    }
48    let mut issues = Vec::new();
49    let changed_paths = canonical_changed_paths(inputs.changed_paths, inputs.continuity);
50    let (mut findings, suppressed_paths, mut findings_truncated) =
51        identical_file_findings(IdenticalFileInputs {
52            before: inputs.before,
53            after: inputs.after,
54            changed_paths: &changed_paths,
55            continuity: inputs.continuity,
56            minimum_bytes: inputs.minimum_identical_bytes,
57            finding_limit: inputs.finding_limit,
58        });
59    let empty_continuity = BTreeMap::new();
60    let before = callable_view(
61        inputs.before,
62        inputs.file_facts,
63        inputs.fact_keys,
64        inputs.continuity,
65    );
66    let after = callable_view(
67        inputs.after,
68        inputs.file_facts,
69        inputs.fact_keys,
70        &empty_continuity,
71    );
72    let callable_limit = inputs.finding_limit.saturating_sub(findings.len());
73    let (callable_findings, callable_truncated) = callable_findings(
74        &before.groups,
75        &after.groups,
76        &changed_paths,
77        &suppressed_paths,
78        callable_limit,
79    );
80    findings.extend(callable_findings);
81    findings_truncated |= callable_truncated;
82    let region_limit = inputs.finding_limit.saturating_sub(findings.len());
83    let region_result = regions::region_findings(regions::RegionInputs {
84        before: inputs.before,
85        after: inputs.after,
86        file_facts: inputs.file_facts,
87        fact_keys: inputs.fact_keys,
88        changed_paths: &changed_paths,
89        continuity: inputs.continuity,
90        suppressed_paths: &suppressed_paths,
91        callable_findings: &findings,
92        finding_limit: region_limit,
93    });
94    issues.extend(snapshot_limit_issues(
95        inputs.before,
96        inputs.file_facts,
97        inputs.fact_keys,
98        &[SenseView::Before],
99        &region_result.malformed_fact_keys,
100    ));
101    issues.extend(snapshot_limit_issues(
102        inputs.after,
103        inputs.file_facts,
104        inputs.fact_keys,
105        &[SenseView::After],
106        &region_result.malformed_fact_keys,
107    ));
108    findings.extend(region_result.findings);
109    findings_truncated |= region_result.findings_truncated;
110    let mut before_coverage = before.coverage;
111    let mut after_coverage = after.coverage;
112    region_result.before.apply(&mut before_coverage);
113    region_result.after.apply(&mut after_coverage);
114    issues.extend(region_result.comparison_issues);
115    DedupResult {
116        findings,
117        findings_truncated,
118        coverage: DedupCoverage {
119            before: before_coverage,
120            after: after_coverage,
121        },
122        issues,
123    }
124}

126fn identical_snapshot_result(inputs: DedupInputs<'_>) -> DedupResult {
127    let mut coverage = fingerprint_coverage(inputs.before, inputs.file_facts, inputs.fact_keys);
128    let (region_coverage, region_issues, malformed_fact_keys) =
129        regions::validate_identical_snapshot(inputs.before, inputs.file_facts, inputs.fact_keys);
130    let mut issues = snapshot_limit_issues(
131        inputs.before,
132        inputs.file_facts,
133        inputs.fact_keys,
134        &[SenseView::Before, SenseView::After],
135        &malformed_fact_keys,
136    );
137    region_coverage.apply(&mut coverage);
138    issues.extend(region_issues);
139    DedupResult {
140        findings: Vec::new(),
141        findings_truncated: false,
142        coverage: DedupCoverage {
143            before: coverage.clone(),
144            after: coverage,
145        },
146        issues,
147    }
148}

173fn snapshot_limit_issues(
174    snapshot: &SourceSnapshot,
175    file_facts: &facts::FactMap,
176    fact_keys: &FactKeyLookup,
177    views: &[SenseView],
178    malformed_fact_keys: &BTreeSet<FileFactKey>,
179) -> Vec<SenseIssue> {
180    let mut state = SnapshotLimitState::default();
181    for file in snapshot.files() {
182        let Some((fact_key, file_facts)) = lookup_file_fact(file, file_facts, fact_keys) else {
183            continue;
184        };
185        observe_callable_limits(&mut state, &file.path, file_facts);
186        observe_region_limits(
187            &mut state,
188            &file.path,
189            file_facts,
190            malformed_fact_keys.contains(&fact_key),
191        );
192    }
193    render_snapshot_limit_issues(state, views)
194}

196fn observe_callable_limits(
197    state: &mut SnapshotLimitState,
198    path: &RepoPath,
199    file_facts: &FileFacts,
200) {
201    let count = file_facts.callable_fingerprints.callables.len();
202    let bounded = count <= MAX_CALLABLE_FINGERPRINTS_PER_FILE;
203    if file_facts.callable_fingerprints.status == FingerprintExtractionStatus::Truncated || !bounded
204    {
205        state.callable_files.push(path);
206    }
207    if !bounded || file_facts.callable_fingerprints.status != FingerprintExtractionStatus::Complete
208    {
209        return;
210    }
211    let remaining = MAX_CALLABLE_FINGERPRINTS_PER_REQUEST.saturating_sub(state.callables);
212    if count > remaining {
213        state.callable_request.push(path);
214    }
215    state.callables = state.callables.saturating_add(count.min(remaining));
216}

218fn observe_region_limits(
219    state: &mut SnapshotLimitState,
220    path: &RepoPath,
221    file_facts: &FileFacts,
222    malformed: bool,
223) {
224    let count = file_facts.region_fingerprints.anchors.len();
225    let bounded = count <= MAX_REGION_FINGERPRINTS_PER_FILE;
226    if file_facts.region_fingerprints.status == FingerprintExtractionStatus::Truncated || !bounded {
227        state.region_files.push(path);
228    }
229    if bounded
230        && !malformed
231        && file_facts.region_fingerprints.status == FingerprintExtractionStatus::Complete
232        && !admit_region_anchors(&mut state.regions, count)
233    {
234        state.region_request.push(path);
235    }
236}

238fn render_snapshot_limit_issues(state: SnapshotLimitState, views: &[SenseView]) -> Vec<SenseIssue> {
239    let mut issues = Vec::new();
240    push_limit_issue(
241        &mut issues,
242        state.callable_files,
243        LimitIssueSpec {
244            code: "dedup_callable_file_limit",
245            stage: DedupLimitStage::CallableFileExtraction,
246            views,
247            limit: MAX_CALLABLE_FINGERPRINTS_PER_FILE,
248            processed: MAX_CALLABLE_FINGERPRINTS_PER_FILE,
249            message: "callable-body duplicate extraction is incomplete",
250        },
251    );
252    push_limit_issue(
253        &mut issues,
254        state.region_files,
255        LimitIssueSpec {
256            code: "dedup_region_file_limit",
257            stage: DedupLimitStage::TokenRegionFileExtraction,
258            views,
259            limit: MAX_REGION_FINGERPRINTS_PER_FILE,
260            processed: MAX_REGION_FINGERPRINTS_PER_FILE,
261            message: "token-region duplicate extraction is incomplete",
262        },
263    );
264    push_limit_issue(
265        &mut issues,
266        state.callable_request,
267        LimitIssueSpec {
268            code: "dedup_callable_request_limit",
269            stage: DedupLimitStage::CallableRequestCollection,
270            views,
271            limit: MAX_CALLABLE_FINGERPRINTS_PER_REQUEST,
272            processed: state.callables,
273            message: "request-wide callable duplicate collection is incomplete",
274        },
275    );
276    push_limit_issue(
277        &mut issues,
278        state.region_request,
279        LimitIssueSpec {
280            code: "dedup_region_request_limit",
281            stage: DedupLimitStage::TokenRegionRequestCollection,
282            views,
283            limit: MAX_REGION_FINGERPRINTS_PER_REQUEST,
284            processed: state.regions,
285            message: "request-wide token-region duplicate collection is incomplete",
286        },
287    );
288    issues
289}

291fn admit_region_anchors(retained: &mut usize, count: usize) -> bool {
292    let Some(total) = prospective_region_anchor_total(*retained, count) else {
293        return false;
294    };
295    *retained = total;
296    true
297}
298
299fn prospective_region_anchor_total(retained: usize, count: usize) -> Option<usize> {
300    (count <= MAX_REGION_FINGERPRINTS_PER_REQUEST.saturating_sub(retained))
301        .then(|| retained.saturating_add(count))
302}

314fn push_limit_issue(
315    issues: &mut Vec<SenseIssue>,
316    affected: AffectedPaths,
317    spec: LimitIssueSpec<'_>,
318) {
319    if affected.count == 0 {
320        return;
321    }
322    let mut issue = SenseIssue::new(
323        spec.code,
324        format!(
325            "{}: the fixed limit of {} was reached; at least {} path(s) were affected",
326            spec.message, spec.limit, affected.count
327        ),
328    );
329    issue.stage = Some(spec.stage);
330    issue.views = spec.views.to_vec();
331    issue.limit = Some(spec.limit);
332    issue.processed = Some(spec.processed);
333    issue.paths = affected.paths;
334    issue.paths_truncated = affected.count > issue.paths.len();
335    issue.next_step = Some(
336        concat!(
337            "No runtime option raises this fixed safety limit. Split unusually dense source files ",
338            "or, for generated/vendor trees, use literal targets.exclude entries such as ",
339            "{\"schemaVersion\":1,\"targets\":{\"exclude\":[\"generated\",\"vendor\"]}}. See ",
340            "https://the-open-engine.github.io/opcore/v",
341            env!("CARGO_PKG_VERSION_MAJOR"),
342            ".",
343            env!("CARGO_PKG_VERSION_MINOR"),
344            "/docs/configuration.html#select-targets."
345        )
346        .into(),
347    );
348    issues.push(issue);
349}

381fn identical_file_findings(
382    inputs: IdenticalFileInputs<'_>,
383) -> (Vec<IntroducedDuplicate>, BTreeSet<RepoPath>, bool) {
384    let empty_continuity = BTreeMap::new();
385    let before_groups = file_groups(inputs.before, inputs.minimum_bytes, inputs.continuity);
386    let after_groups = file_groups(inputs.after, inputs.minimum_bytes, &empty_continuity);
387    let mut findings = Vec::new();
388    let mut suppressed_paths = BTreeSet::new();
389    let mut truncated = false;
390    for (content_id, after_group) in after_groups {
391        let empty = BTreeSet::new();
392        let before_paths = before_groups
393            .get(&content_id)
394            .map_or(&empty, |group| &group.paths);
395        if after_group.paths.len() < 2 || after_group.paths.len() <= before_paths.len() {
396            continue;
397        }
398        let introduced_paths = after_group
399            .paths
400            .difference(before_paths)
401            .cloned()
402            .collect::<BTreeSet<_>>();
403        if introduced_paths.is_disjoint(inputs.changed_paths) {
404            continue;
405        }
406        suppressed_paths.extend(introduced_paths);
407        if findings.len() >= inputs.finding_limit {
408            truncated = true;
409            continue;
410        }
411        let mut occurrences = after_group
412            .paths
413            .iter()
414            .map(|path| DuplicateOccurrence {
415                path: path.clone(),
416                range: None,
417                callable_kind: None,
418                changed: inputs.changed_paths.contains(path),
419            })
420            .collect::<Vec<_>>();
421        occurrences
422            .sort_by(|left, right| (!left.changed, &left.path).cmp(&(!right.changed, &right.path)));
423        let (occurrences, occurrences_truncated) = bound_occurrences(occurrences);
424        findings.push(IntroducedDuplicate {
425            class_id: format!("file-{}", short_id(&content_id.hex())),
426            kind: DuplicateKind::IdenticalFile,
427            language_family: None,
428            before_count: before_paths.len(),
429            after_count: after_group.paths.len(),
430            introduced_count: after_group.paths.len().saturating_sub(before_paths.len()),
431            bytes: Some(after_group.bytes),
432            token_count: None,
433            occurrences,
434            occurrences_truncated,
435        });
436    }
437    (findings, suppressed_paths, truncated)
438}

440fn file_groups(
441    snapshot: &SourceSnapshot,
442    minimum_bytes: usize,
443    continuity: &BTreeMap<RepoPath, RepoPath>,
444) -> BTreeMap<ContentId, FileGroup> {
445    let mut groups = BTreeMap::<ContentId, FileGroup>::new();
446    for file in snapshot
447        .files()
448        .filter(|file| file.bytes.len() >= minimum_bytes)
449    {
450        let path = continuity
451            .get(&file.path)
452            .cloned()
453            .unwrap_or_else(|| file.path.clone());
454        let group = groups.entry(file.content_id).or_insert_with(|| FileGroup {
455            bytes: file.bytes.len(),
456            paths: BTreeSet::new(),
457        });
458        group.paths.insert(path);
459    }
460    groups
461}

478fn fingerprint_coverage(
479    snapshot: &SourceSnapshot,
480    file_facts: &facts::FactMap,
481    fact_keys: &FactKeyLookup,
482) -> FingerprintCoverage {
483    let mut coverage = FingerprintCoverage::default();
484    for file in snapshot.files() {
485        coverage.files = coverage.files.saturating_add(1);
486        let Some(file_facts) = lookup_file_facts(file, file_facts, fact_keys) else {
487            coverage.unsupported_files = coverage.unsupported_files.saturating_add(1);
488            continue;
489        };
490        record_non_region_coverage(&mut coverage, file_facts);
491    }
492    coverage
493}

495fn callable_view(
496    snapshot: &SourceSnapshot,
497    file_facts: &facts::FactMap,
498    fact_keys: &FactKeyLookup,
499    continuity: &BTreeMap<RepoPath, RepoPath>,
500) -> CallableView {
501    let mut groups = CallableGroups::new();
502    let mut coverage = FingerprintCoverage::default();
503    for file in snapshot.files() {
504        coverage.files = coverage.files.saturating_add(1);
505        let Some(file_facts) = lookup_file_facts(file, file_facts, fact_keys) else {
506            coverage.unsupported_files = coverage.unsupported_files.saturating_add(1);
507            continue;
508        };
509        record_fingerprint_status(&mut coverage, file_facts);
510        if file_facts.callable_fingerprints.status != FingerprintExtractionStatus::Complete
511            || file_facts.callable_fingerprints.callables.is_empty()
512            || file_facts.callable_fingerprints.callables.len() > MAX_CALLABLE_FINGERPRINTS_PER_FILE
513        {
514            continue;
515        }
516        let path = continuity
517            .get(&file.path)
518            .cloned()
519            .unwrap_or_else(|| file.path.clone());
520        for callable in &file_facts.callable_fingerprints.callables {
521            if coverage.callable_occurrences >= MAX_CALLABLE_FINGERPRINTS_PER_REQUEST {
522                coverage.request_truncated = true;
523                continue;
524            }
525            coverage.callable_occurrences = coverage.callable_occurrences.saturating_add(1);
526            groups
527                .entry(file.language.family())
528                .or_default()
529                .entry(callable.digest.clone())
530                .or_default()
531                .push(CallableOccurrence {
532                    path: path.clone(),
533                    range: callable.range,
534                    kind: callable.kind,
535                    token_count: usize::try_from(callable.token_count).unwrap_or(usize::MAX),
536                });
537        }
538    }
539    for occurrences in groups.values_mut().flat_map(BTreeMap::values_mut) {
540        occurrences.sort_by(|left, right| {
541            (&left.path, left.range, left.kind).cmp(&(&right.path, right.range, right.kind))
542        });
543    }
544    CallableView { groups, coverage }
545}

555fn lookup_file_fact<'a>(
556    file: &crate::model::SourceFile,
557    file_facts: &'a facts::FactMap,
558    fact_keys: &FactKeyLookup,
559) -> Option<(FileFactKey, &'a FileFacts)> {

563    file_facts.get(key).map(|facts| (*key, facts.as_ref()))
564}

566fn record_non_region_coverage(coverage: &mut FingerprintCoverage, file_facts: &FileFacts) {
567    record_fingerprint_status(coverage, file_facts);
568    if file_facts.callable_fingerprints.status != FingerprintExtractionStatus::Complete
569        || file_facts.callable_fingerprints.callables.len() > MAX_CALLABLE_FINGERPRINTS_PER_FILE
570    {
571        return;
572    }
573    let remaining =
574        MAX_CALLABLE_FINGERPRINTS_PER_REQUEST.saturating_sub(coverage.callable_occurrences);
575    coverage.callable_occurrences = coverage.callable_occurrences.saturating_add(
576        file_facts
577            .callable_fingerprints
578            .callables
579            .len()
580            .min(remaining),
581    );
582    coverage.request_truncated |= file_facts.callable_fingerprints.callables.len() > remaining;
583}

585fn record_fingerprint_status(coverage: &mut FingerprintCoverage, file_facts: &FileFacts) {
586    let mut status = file_facts
587        .callable_fingerprints
588        .status
589        .max(file_facts.region_fingerprints.status);
590    if file_facts.callable_fingerprints.callables.len() > MAX_CALLABLE_FINGERPRINTS_PER_FILE
591        || file_facts.region_fingerprints.anchors.len() > MAX_REGION_FINGERPRINTS_PER_FILE
592    {
593        status = FingerprintExtractionStatus::Truncated;
594    }
595    match status {
596        FingerprintExtractionStatus::Complete => {
597            coverage.complete_files = coverage.complete_files.saturating_add(1);
598        }
599        FingerprintExtractionStatus::Unsupported => {
600            coverage.unsupported_files = coverage.unsupported_files.saturating_add(1);
601        }
602        FingerprintExtractionStatus::ParserFailed => {
603            coverage.parser_failed_files = coverage.parser_failed_files.saturating_add(1);
604        }
605        FingerprintExtractionStatus::Truncated => {
606            coverage.truncated_files = coverage.truncated_files.saturating_add(1);
607        }
608    }
609}

611fn callable_findings(
612    before: &CallableGroups,
613    after: &CallableGroups,
614    changed_paths: &BTreeSet<RepoPath>,
615    suppressed_paths: &BTreeSet<RepoPath>,
616    finding_limit: usize,
617) -> (Vec<IntroducedDuplicate>, bool) {
618    let mut findings = Vec::new();
619    let mut truncated = false;
620    for (language_family, groups) in after {
621        for (digest, after_occurrences) in groups {
622            let before_occurrences = before
623                .get(language_family)
624                .and_then(|group| group.get(digest));
625            let before_count = before_occurrences.map_or(0, Vec::len);
626            if after_occurrences.len() < 2 || after_occurrences.len() <= before_count {
627                continue;
628            }
629            let before_by_path =
630                path_counts(before_occurrences.into_iter().flatten(), |occurrence| {
631                    &occurrence.path
632                });
633            let after_by_path =
634                path_counts(after_occurrences.iter(), |occurrence| &occurrence.path);
635            let grew_paths = paths_with_growth(&before_by_path, &after_by_path);
636            if grew_paths.is_disjoint(changed_paths)
637                || (!grew_paths.is_empty() && grew_paths.is_subset(suppressed_paths))
638            {
639                continue;
640            }
641            if findings.len() >= finding_limit {
642                truncated = true;
643                continue;
644            }
645            let mut occurrences = after_occurrences
646                .iter()
647                .map(|occurrence| DuplicateOccurrence {
648                    path: occurrence.path.clone(),
649                    range: Some(occurrence.range),
650                    callable_kind: Some(occurrence.kind),
651                    changed: changed_paths.contains(&occurrence.path),
652                })
653                .collect::<Vec<_>>();
654            occurrences.sort_by(|left, right| {
655                (!left.changed, &left.path, left.range, left.callable_kind).cmp(&(
656                    !right.changed,
657                    &right.path,
658                    right.range,
659                    right.callable_kind,
660                ))
661            });
662            let (occurrences, occurrences_truncated) = bound_occurrences(occurrences);
663            findings.push(IntroducedDuplicate {
664                class_id: format!("dup-{}", short_id(digest)),
665                kind: DuplicateKind::CallableBody,
666                language_family: Some((*language_family).into()),
667                before_count,
668                after_count: after_occurrences.len(),
669                introduced_count: after_occurrences.len().saturating_sub(before_count),
670                bytes: None,
671                token_count: after_occurrences
672                    .first()
673                    .map(|occurrence| occurrence.token_count),
674                occurrences,
675                occurrences_truncated,
676            });
677        }
678    }
679    (findings, truncated)
680}

682pub(super) fn path_counts<'a, T: 'a>(
683    occurrences: impl IntoIterator<Item = &'a T>,
684    path: impl Fn(&T) -> &RepoPath,
685) -> BTreeMap<RepoPath, usize> {
686    let mut counts = BTreeMap::new();
687    for occurrence in occurrences {
688        let count = counts.entry(path(occurrence).clone()).or_insert(0usize);
689        *count = count.saturating_add(1);
690    }
691    counts
692}

694pub(super) fn paths_with_growth(
695    before: &BTreeMap<RepoPath, usize>,
696    after: &BTreeMap<RepoPath, usize>,
697) -> BTreeSet<RepoPath> {
698    after
699        .iter()
700        .filter(|(path, count)| before.get(*path).copied().unwrap_or(0) < **count)
701        .map(|(path, _)| path.clone())
702        .collect()
703}

705fn bound_occurrences(
706    mut occurrences: Vec<DuplicateOccurrence>,
707) -> (Vec<DuplicateOccurrence>, bool) {
708    let truncated = occurrences.len() > MAX_OBSERVATION_PATHS;
709    occurrences.truncate(MAX_OBSERVATION_PATHS);
710    (occurrences, truncated)
711}

885    fn swap_first_encoded_token_records(facts: &mut RegionFingerprintFacts) {
886        let bytes = base64::engine::general_purpose::STANDARD
887            .decode(&facts.encoded_tokens)
888            .unwrap();
889        let record_end = |start: usize| {
890            let prefix = u16::from_be_bytes(bytes[start + 9..start + 11].try_into().unwrap());
891            start + 11 + usize::from(prefix)
892        };
893        let first_end = record_end(0);
894        let second_end = record_end(first_end);
895        let mut reordered = Vec::with_capacity(bytes.len());
896        reordered.extend_from_slice(&bytes[first_end..second_end]);
897        reordered.extend_from_slice(&bytes[..first_end]);
898        reordered.extend_from_slice(&bytes[second_end..]);
899        facts.encoded_tokens = base64::engine::general_purpose::STANDARD.encode(reordered);
900    }
