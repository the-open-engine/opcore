37#[derive(Default)]
38struct DecodedEvidence(BTreeMap<FileFactKey, EvidenceState>);
39
40impl DecodedEvidence {
41    fn state(
42        &mut self,
43        fact_key: FileFactKey,
44        source: &SourceFile,
45        facts: &RegionFingerprintFacts,
46        retain: bool,
47    ) -> EvidenceState {
48        let state = self.0.entry(fact_key).or_insert_with(|| {
49            if retain {
50                DecodedTokens::decode(source, facts)
51                    .map(Arc::new)
52                    .map_or(EvidenceState::Malformed, EvidenceState::Decoded)
53            } else if DecodedTokens::validate(source, facts) {
54                EvidenceState::Validated
55            } else {
56                EvidenceState::Malformed
57            }
58        });
59        if retain && matches!(state, EvidenceState::Validated) {
60            *state = DecodedTokens::decode(source, facts)
61                .map(Arc::new)
62                .map_or(EvidenceState::Malformed, EvidenceState::Decoded);
63        }
64        state.clone()
65    }
66
67    fn malformed_fact_keys(&self) -> BTreeSet<FileFactKey> {
68        self.0
69            .iter()
70            .filter_map(|(key, state)| matches!(state, EvidenceState::Malformed).then_some(*key))
71            .collect()
72    }
73}

123pub(super) fn region_findings(inputs: RegionInputs<'_>) -> RegionResult {
124    let mut decoded_evidence = DecodedEvidence::default();
125    let before = RegionView::new(RegionViewInputs {
126        view: SenseView::Before,
127        snapshot: inputs.before,
128        file_facts: inputs.file_facts,
129        fact_keys: inputs.fact_keys,
130        continuity: inputs.continuity,
131        decoded_evidence: &mut decoded_evidence,
132    });
133    let after = RegionView::new(RegionViewInputs {
134        view: SenseView::After,
135        snapshot: inputs.after,
136        file_facts: inputs.file_facts,
137        fact_keys: inputs.fact_keys,
138        continuity: &BTreeMap::new(),
139        decoded_evidence: &mut decoded_evidence,
140    });
141    let context = EvaluationContext {
142        inputs,
143        before: &before,
144        after: &after,
145    };
146    let mut state = EvaluationState::default();
147    state.comparison.observe_malformed_view(&before);
148    state.comparison.observe_malformed_view(&after);
149    if !state.comparison.incomplete {
150        'families: for (family, groups) in &context.after.groups {
151            for (anchor, after_occurrences) in groups {
152                state.evaluate_group(context, family, *anchor, after_occurrences);
153                if state.comparison.incomplete {
154                    break 'families;
155                }
156            }
157        }
158    }
159
160    let comparison_issues = state.comparison.issues();
161    let malformed_fact_keys = decoded_evidence.malformed_fact_keys();
162    RegionResult {
163        findings: state.findings,
164        findings_truncated: state.comparison.findings_truncated,
165        before: before.coverage(&state.comparison),
166        after: after.coverage(&state.comparison),
167        comparison_issues,
168        malformed_fact_keys,
169    }
170}

173    snapshot: &SourceSnapshot,
174    file_facts: &facts::FactMap,
175    fact_keys: &FactKeyLookup,
176) -> (RegionCoverage, Vec<SenseIssue>, BTreeSet<FileFactKey>) {

195        decoded_evidence.malformed_fact_keys(),
196    )

206#[derive(Default)]
207struct EvaluationState {
208    comparison: ComparisonState,
209    findings: Vec<IntroducedDuplicate>,
210    reported_classes: BTreeSet<(String, String)>,
211    covered: BTreeMap<usize, Vec<Range<usize>>>,
212}
213
214impl EvaluationState {
215    fn evaluate_group(
216        &mut self,
217        context: EvaluationContext<'_, '_>,
218        family: &'static str,
219        anchor: AnchorKey,
220        after_occurrences: &[AnchorOccurrence],
221    ) {
222        if !anchor_class_may_be_introduced(AnchorClassInputs {
223            after: after_occurrences,
224            after_view: context.after,
225            changed_paths: context.inputs.changed_paths,
226            suppressed_paths: context.inputs.suppressed_paths,
227        }) {
228            return;
229        }
230        for seed in after_occurrences {
231            self.evaluate_seed(context, family, anchor, after_occurrences, *seed);
232            if self.comparison.incomplete {
233                break;
234            }
235        }
236    }
237
238    fn evaluate_seed(
239        &mut self,
240        context: EvaluationContext<'_, '_>,
241        family: &'static str,
242        anchor: AnchorKey,
243        occurrences: &[AnchorOccurrence],
244        seed: AnchorOccurrence,
245    ) {
246        let seed_path = &context.after.files[seed.file].path;
247        if !context.inputs.changed_paths.contains(seed_path)
248            || context.inputs.suppressed_paths.contains(seed_path)
249            || interval_contains(&self.covered, seed.file, seed.token_index)
250        {
251            return;
252        }
253        let candidates = candidate_regions(
254            context.after,
255            anchor,
256            seed,
257            occurrences,
258            &mut self.comparison,
259        );
260        let Some(class) = self.best_class(context, family, candidates) else {
261            return;
262        };
263        mark_covered(&mut self.covered, &class.after_occurrences);
264        if !self
265            .reported_classes
266            .insert((family.to_owned(), class.digest.clone()))
267        {
268            return;
269        }
270        if self.findings.len() >= context.inputs.finding_limit {
271            self.comparison.findings_truncated = true;
272            return;
273        }
274        self.findings.push(class.into_finding());
275    }
276
277    fn best_class(
278        &mut self,
279        context: EvaluationContext<'_, '_>,
280        family: &'static str,
281        candidates: Vec<Candidate>,
282    ) -> Option<ExactClass> {
283        let mut best = None;
284        for candidate in candidates {
285            let Some(class) = exact_class(
286                ExactClassInputs {
287                    before: context.before,
288                    after: context.after,
289                    family,
290                    changed_paths: context.inputs.changed_paths,
291                    suppressed_paths: context.inputs.suppressed_paths,
292                    callable_findings: context.inputs.callable_findings,
293                },
294                candidate,
295                &mut self.comparison,
296            ) else {
297                if self.comparison.incomplete {
298                    break;
299                }
300                continue;
301            };
302            if class_precedes(best.as_ref(), &class) {
303                best = Some(class);
304            }
305        }
306        best
307    }
308}
309
310fn class_precedes(current: Option<&ExactClass>, candidate: &ExactClass) -> bool {
311    current.is_none_or(|current| {
312        (candidate.after_occurrences.len(), candidate.token_count)
313            > (current.after_occurrences.len(), current.token_count)
314    })
315}

325fn anchor_class_may_be_introduced(inputs: AnchorClassInputs<'_, '_>) -> bool {
326    inputs.after.len() >= 2
327        && has_distinct_anchors(inputs.after)
328        && inputs.after.iter().any(|occurrence| {
329            let path = &inputs.after_view.files[occurrence.file].path;
330            inputs.changed_paths.contains(path) && !inputs.suppressed_paths.contains(path)
331        })
332}
333
334fn has_distinct_anchors(occurrences: &[AnchorOccurrence]) -> bool {
335    let Some(first) = occurrences.first() else {
336        return false;
337    };
338    occurrences[1..]
339        .iter()
340        .any(|candidate| distinct_anchor(*first, *candidate))
341}

412fn candidate_regions(
413    view: &RegionView<'_>,
414    anchor: AnchorKey,
415    seed: AnchorOccurrence,
416    occurrences: &[AnchorOccurrence],
417    comparison: &mut ComparisonState,
418) -> Vec<Candidate> {
419    let mut candidates = Vec::new();
420    let mut candidate_ranges = BTreeSet::new();
421    for peer in occurrences
422        .iter()
423        .copied()
424        .filter(|peer| distinct_anchor(seed, *peer))
425    {
426        let Some((start, end)) = maximal_exact_region(view, seed, peer, comparison) else {
427            if comparison.incomplete {
428                break;
429            }
430            continue;
431        };
432        if end.saturating_sub(start) < MIN_DEDUP_REGION_TOKENS {
433            continue;
434        }
435        let candidate = Candidate {
436            file: seed.file,
437            start,
438            end,
439            anchor,
440            anchor_offset: usize::try_from(seed.token_index)
441                .unwrap_or(usize::MAX)
442                .saturating_sub(start),
443        };
444        if candidate_ranges.insert((start, end)) {
445            candidates.push(candidate);
446        }
447    }
448    candidates.sort_by(|left, right| (right.len(), left.start).cmp(&(left.len(), right.start)));
449    candidates
450}

452fn distinct_anchor(left: AnchorOccurrence, right: AnchorOccurrence) -> bool {
453    left.file != right.file
454        || usize::try_from(left.token_index.abs_diff(right.token_index)).unwrap_or(usize::MAX)
455            >= MIN_DEDUP_REGION_TOKENS
456}

458fn maximal_exact_region(
459    view: &RegionView<'_>,
460    left: AnchorOccurrence,
461    right: AnchorOccurrence,
462    comparison: &mut ComparisonState,
463) -> Option<(usize, usize)> {
464    let left_anchor = usize::try_from(left.token_index).ok()?;
465    let right_anchor = usize::try_from(right.token_index).ok()?;
466    if !anchor_tokens_match(
467        view,
468        TokenLocation {
469            file: left.file,
470            token: left_anchor,
471        },
472        TokenLocation {
473            file: right.file,
474            token: right_anchor,
475        },
476        comparison,
477    )? {
478        return None;
479    }
480    let left_start = extend_left(
481        view,
482        TokenLocation {
483            file: left.file,
484            token: left_anchor,
485        },
486        TokenLocation {
487            file: right.file,
488            token: right_anchor,
489        },
490        comparison,
491    )?;
492    let left_end = extend_right(
493        view,
494        TokenLocation {
495            file: left.file,
496            token: left_anchor.saturating_add(DEDUP_REGION_KGRAM_TOKENS),
497        },
498        TokenLocation {
499            file: right.file,
500            token: right_anchor.saturating_add(DEDUP_REGION_KGRAM_TOKENS),
501        },
502        comparison,
503    )?;
504    Some((left_start, left_end))
505}

507fn anchor_tokens_match(
508    view: &RegionView<'_>,
509    left: TokenLocation,
510    right: TokenLocation,
511    comparison: &mut ComparisonState,
512) -> Option<bool> {
513    for offset in 0..DEDUP_REGION_KGRAM_TOKENS {
514        if !comparison.tokens_equal(TokenPair::same_view(
515            view,
516            left.offset(offset),
517            right.offset(offset),
518        ))? {
519            return Some(false);
520        }
521    }
522    Some(true)
523}
524
525fn extend_left(
526    view: &RegionView<'_>,
527    mut left: TokenLocation,
528    mut right: TokenLocation,
529    comparison: &mut ComparisonState,
530) -> Option<usize> {
531    while left.token > 0 && right.token > 0 {
532        let prior_left = left.prior();
533        let prior_right = right.prior();
534        if !comparison.tokens_equal(TokenPair::same_view(view, prior_left, prior_right))? {
535            break;
536        }
537        left = prior_left;
538        right = prior_right;
539    }
540    Some(left.token)
541}
542
543fn extend_right(
544    view: &RegionView<'_>,
545    mut left: TokenLocation,
546    mut right: TokenLocation,
547    comparison: &mut ComparisonState,
548) -> Option<usize> {
549    while comparison.token_exists(view, left.file, left.token)
550        && comparison.token_exists(view, right.file, right.token)
551    {
552        if !comparison.tokens_equal(TokenPair::same_view(view, left, right))? {
553            break;
554        }
555        left.token = left.token.saturating_add(1);
556        right.token = right.token.saturating_add(1);
557    }
558    Some(left.token)
559}

561struct ExactClass {
562    digest: String,
563    family: &'static str,
564    before_count: usize,
565    after_occurrences: Vec<ExactOccurrence>,
566    introduced_count: usize,
567    token_count: usize,
568}

570impl ExactClass {
571    fn into_finding(self) -> IntroducedDuplicate {
572        let mut occurrences = self
573            .after_occurrences
574            .into_iter()
575            .map(|occurrence| DuplicateOccurrence {
576                path: occurrence.path,
577                range: Some(occurrence.range),
578                callable_kind: None,
579                changed: occurrence.changed,
580            })
581            .collect::<Vec<_>>();
582        occurrences.sort_by(|left, right| {
583            (!left.changed, &left.path, left.range).cmp(&(!right.changed, &right.path, right.range))
584        });
585        let after_count = occurrences.len();
586        let (occurrences, occurrences_truncated) = bound_occurrences(occurrences);
587        IntroducedDuplicate {
588            class_id: format!("region-{}", short_id(&self.digest)),
589            kind: DuplicateKind::TokenRegion,
590            language_family: Some(self.family.into()),
591            before_count: self.before_count,
592            after_count,
593            introduced_count: self.introduced_count,
594            bytes: None,
595            token_count: Some(self.token_count),
596            occurrences,
597            occurrences_truncated,
598        }
599    }
600}

611
612fn exact_class(
613    inputs: ExactClassInputs<'_>,
614    candidate: Candidate,
615    comparison: &mut ComparisonState,
616) -> Option<ExactClass> {
617    let before_occurrences = exact_occurrences(
618        OccurrenceInputs {
619            candidate_view: inputs.after,
620            candidate,
621            target: inputs.before,
622            family: inputs.family,
623            changed_paths: inputs.changed_paths,
624        },
625        comparison,
626    )?;
627    let after_occurrences = exact_occurrences(
628        OccurrenceInputs {
629            candidate_view: inputs.after,
630            candidate,
631            target: inputs.after,
632            family: inputs.family,
633            changed_paths: inputs.changed_paths,
634        },
635        comparison,
636    )?;
637    if after_occurrences.len() < 2 || after_occurrences.len() <= before_occurrences.len() {
638        return None;
639    }
640    let before_by_path = path_counts(&before_occurrences, |occurrence| &occurrence.path);
641    let after_by_path = path_counts(&after_occurrences, |occurrence| &occurrence.path);
642    let grew_paths = paths_with_growth(&before_by_path, &after_by_path);
643    if grew_paths.is_disjoint(inputs.changed_paths)
644        || (!grew_paths.is_empty() && grew_paths.is_subset(inputs.suppressed_paths))
645        || shadowed_by_callable(&after_occurrences, &grew_paths, inputs.callable_findings)
646    {
647        return None;
648    }
649    let Some(digest) = inputs
650        .after
651        .region_digest(candidate.file, candidate.start..candidate.end)
652    else {
653        comparison.mark_malformed_location(inputs.after, candidate.file);
654        return None;
655    };
656    Some(ExactClass {
657        digest,
658        family: inputs.family,
659        before_count: before_occurrences.len(),
660        introduced_count: after_occurrences
661            .len()
662            .saturating_sub(before_occurrences.len()),
663        after_occurrences,
664        token_count: candidate.len(),
665    })
666}

668fn shadowed_by_callable(
669    occurrences: &[ExactOccurrence],
670    grew_paths: &BTreeSet<RepoPath>,
671    callable_findings: &[IntroducedDuplicate],
672) -> bool {
673    !grew_paths.is_empty()
674        && grew_paths.iter().all(|path| {
675            occurrences
676                .iter()
677                .filter(|occurrence| &occurrence.path == path)
678                .all(|region| {
679                    callable_findings.iter().any(|finding| {
680                        finding.kind == DuplicateKind::CallableBody
681                            && finding.occurrences.iter().any(|callable| {
682                                callable.path == region.path
683                                    && callable.range.is_some_and(|range| {
684                                        range.start.byte < region.range.end.byte
685                                            && range.end.byte > region.range.start.byte
686                                    })
687                            })
688                    })
689                })
690        })
691}

711
712fn exact_occurrences(
713    inputs: OccurrenceInputs<'_, '_>,
714    comparison: &mut ComparisonState,
715) -> Option<Vec<ExactOccurrence>> {
716    let Some(group) = inputs
717        .target
718        .groups
719        .get(inputs.family)
720        .and_then(|groups| groups.get(&inputs.candidate.anchor))
721    else {
722        return Some(Vec::new());
723    };
724    let mut occurrences = BTreeMap::<(RepoPath, usize), ExactOccurrence>::new();
725    for anchor in group {
726        let Ok(anchor_index) = usize::try_from(anchor.token_index) else {
727            comparison.mark_malformed_location(inputs.target, anchor.file);
728            return None;
729        };
730        let Some(start) = anchor_index.checked_sub(inputs.candidate.anchor_offset) else {
731            continue;
732        };
733        let end = start.saturating_add(inputs.candidate.len());
734        if !comparison.regions_equal(&RegionPair {
735            left_view: inputs.candidate_view,
736            left_file: inputs.candidate.file,
737            left: inputs.candidate.start..inputs.candidate.end,
738            right_view: inputs.target,
739            right_file: anchor.file,
740            right: start..end,
741        })? {
742            continue;
743        }
744        let file = &inputs.target.files[anchor.file];
745        let Some(range) = file.range(start..end) else {
746            comparison.mark_malformed_location(inputs.target, anchor.file);
747            return None;
748        };
749        occurrences
750            .entry((file.path.clone(), start))
751            .or_insert_with(|| ExactOccurrence {
752                file: anchor.file,
753                path: file.path.clone(),
754                start,
755                end,
756                range,
757                changed: inputs.changed_paths.contains(&file.path),
758            });
759    }
760    let mut occurrences = occurrences.into_values().collect::<Vec<_>>();
761    occurrences.sort_by(|left, right| {
762        (&left.path, left.start, left.end).cmp(&(&right.path, right.start, right.end))
763    });
764    let mut non_overlapping = Vec::<ExactOccurrence>::new();
765    for occurrence in occurrences {
766        if non_overlapping
767            .last()
768            .is_none_or(|prior| prior.path != occurrence.path || prior.end <= occurrence.start)
769        {
770            non_overlapping.push(occurrence);
771        }
772    }
773    Some(non_overlapping)
774}

806struct RegionView<'a> {
807    view: SenseView,
808    files: Vec<RegionFile<'a>>,
809    groups: BTreeMap<&'static str, BTreeMap<AnchorKey, Vec<AnchorOccurrence>>>,
810    retained_anchors: usize,
811    retained_tokens: usize,
812    request_truncated: bool,
813    malformed_paths: BTreeSet<RepoPath>,
814    malformed_paths_truncated: bool,
815}
816
817impl<'a> RegionView<'a> {
818    fn new(inputs: RegionViewInputs<'a, '_, '_>) -> Self {
819        let RegionViewInputs {
820            view: sense_view,
821            snapshot,
822            file_facts,
823            fact_keys,
824            continuity,
825            decoded_evidence,
826        } = inputs;
827        let mut view = Self {
828            view: sense_view,
829            files: Vec::new(),
830            groups: BTreeMap::new(),
831            retained_anchors: 0,
832            retained_tokens: 0,
833            request_truncated: false,
834            malformed_paths: BTreeSet::new(),
835            malformed_paths_truncated: false,
836        };
837        for source in snapshot.files() {
838            let Some((fact_key, facts)) = lookup_file_fact(source, file_facts, fact_keys) else {
839                continue;
840            };
841            if facts.region_fingerprints.status != FingerprintExtractionStatus::Complete {
842                continue;
843            }
844            let path = continuity
845                .get(&source.path)
846                .cloned()
847                .unwrap_or_else(|| source.path.clone());
848            let file_bounded =
849                facts.region_fingerprints.anchors.len() <= MAX_REGION_FINGERPRINTS_PER_FILE;
850            let retained_anchors = file_bounded
851                .then(|| {
852                    prospective_region_anchor_total(
853                        view.retained_anchors,
854                        facts.region_fingerprints.anchors.len(),
855                    )
856                })
857                .flatten();
858            let state = decoded_evidence.state(
859                fact_key,
860                source,
861                &facts.region_fingerprints,
862                retained_anchors.is_some(),
863            );
864            let EvidenceState::Decoded(decoded) = state else {
865                if matches!(state, EvidenceState::Malformed) {
866                    record_bounded_path(
867                        &mut view.malformed_paths,
868                        &mut view.malformed_paths_truncated,
869                        path,
870                    );
871                } else if file_bounded {
872                    view.request_truncated = true;
873                }
874                continue;
875            };
876            let Some(retained_anchors) = retained_anchors else {
877                view.request_truncated = true;
878                continue;
879            };
880            view.retained_anchors = retained_anchors;
881            let file = view.files.len();
882            view.retained_tokens = view.retained_tokens.saturating_add(
883                usize::try_from(facts.region_fingerprints.token_count).unwrap_or(usize::MAX),
884            );
885            view.files.push(RegionFile {
886                source,
887                path,
888                decoded,
889            });
890            for anchor in &facts.region_fingerprints.anchors {
891                view.groups
892                    .entry(source.language.family())
893                    .or_default()
894                    .entry((anchor.hash_one, anchor.hash_two))
895                    .or_default()
896                    .push(AnchorOccurrence {
897                        file,
898                        token_index: anchor.token_index,
899                    });
900            }
901        }
902        for occurrences in view.groups.values_mut().flat_map(BTreeMap::values_mut) {
903            occurrences.sort_by(|left, right| {
904                (&view.files[left.file].path, left.token_index)
905                    .cmp(&(&view.files[right.file].path, right.token_index))
906            });
907        }
908        view
909    }
910
911    fn coverage(&self, comparison: &ComparisonState) -> RegionCoverage {
912        RegionCoverage {
913            anchors: self.retained_anchors,
914            tokens: self.retained_tokens,
915            request_truncated: self.request_truncated
916                || (comparison.limit_reached && comparison.views.contains(&self.view)),
917            malformed_evidence: !self.malformed_paths.is_empty()
918                || self.malformed_paths_truncated
919                || (comparison.malformed_evidence && comparison.views.contains(&self.view)),
920        }
921    }
922
923    fn exact_token(&self, location: TokenLocation) -> Option<ExactToken<'_>> {
924        let file = self.files.get(location.file)?;
925        let decoded = file.tokens();
926        let token = decoded.tokens.get(location.token)?;
927        Some(ExactToken {
928            tag: token.tag,
929            prefix: decoded.bytes.get(token.prefix.clone())?,
930            text: file.source.bytes.get(token.start..token.end)?,
931        })
932    }
933
934    fn region_digest(&self, file: usize, range: Range<usize>) -> Option<String> {
935        let file = self.files.get(file)?;
936        let decoded = file.tokens();
937        let mut encoded = Vec::new();
938        for token in decoded.tokens.get(range)? {
939            let prefix = decoded.bytes.get(token.prefix.clone())?;
940            encoded.extend_from_slice(
941                &u32::try_from(prefix.len())
942                    .unwrap_or(u32::MAX)
943                    .to_be_bytes(),
944            );
945            encoded.extend_from_slice(prefix);
946            encoded.push(token.tag);
947            let text = file.source.bytes.get(token.start..token.end)?;
948            encoded.extend_from_slice(&u32::try_from(text.len()).unwrap_or(u32::MAX).to_be_bytes());
949            encoded.extend_from_slice(text);
950        }
951        Some(hex::encode(hash_domain(
952            "dedup-token-region/v1",
953            &[file.source.language.family().as_bytes(), &encoded],
954        )))
955    }
956}

958fn record_bounded_path(paths: &mut BTreeSet<RepoPath>, paths_truncated: &mut bool, path: RepoPath) {
959    if paths.contains(&path) {
960        return;
961    }
962    if paths.len() >= MAX_OBSERVATION_PATHS {
963        *paths_truncated = true;
964        return;
965    }
966    paths.insert(path);
967}

998#[expect(
999    clippy::struct_excessive_bools,
1000    reason = "comparison completeness, report truncation, limit identity, and sample truncation are independent states"
1001)]
1002struct ComparisonState {
1003    remaining: usize,
1004    incomplete: bool,
1005    findings_truncated: bool,
1006    limit_reached: bool,
1007    malformed_evidence: bool,
1008    views: BTreeSet<SenseView>,
1009    limit_paths: BTreeSet<RepoPath>,
1010    malformed_paths: BTreeSet<RepoPath>,
1011    malformed_paths_truncated: bool,
1012}
1013
1014impl Default for ComparisonState {
1015    fn default() -> Self {
1016        Self {
1017            remaining: MAX_DEDUP_TOKEN_COMPARISONS,
1018            incomplete: false,
1019            findings_truncated: false,
1020            limit_reached: false,
1021            malformed_evidence: false,
1022            views: BTreeSet::new(),
1023            limit_paths: BTreeSet::new(),
1024            malformed_paths: BTreeSet::new(),
1025            malformed_paths_truncated: false,
1026        }
1027    }
1028}
1029
1030impl ComparisonState {
1031    fn observe_malformed_view(&mut self, view: &RegionView<'_>) {
1032        if view.malformed_paths.is_empty() && !view.malformed_paths_truncated {
1033            return;
1034        }
1035        self.incomplete = true;
1036        self.malformed_evidence = true;
1037        self.views.insert(view.view);
1038        for path in &view.malformed_paths {
1039            record_bounded_path(
1040                &mut self.malformed_paths,
1041                &mut self.malformed_paths_truncated,
1042                path.clone(),
1043            );
1044        }
1045        self.malformed_paths_truncated |= view.malformed_paths_truncated;
1046    }
1047
1048    fn record_limit_location(&mut self, view: &RegionView<'_>, file: usize) {
1049        self.views.insert(view.view);
1050        let Some(path) = view.files.get(file).map(|file| &file.path) else {
1051            return;
1052        };
1053        if self.limit_paths.len() < MAX_OBSERVATION_PATHS {
1054            self.limit_paths.insert(path.clone());
1055        }
1056    }
1057
1058    fn record_limit_pair(&mut self, pair: &TokenPair<'_, '_>) {
1059        self.record_limit_location(pair.left_view, pair.left.file);
1060        self.record_limit_location(pair.right_view, pair.right.file);
1061    }
1062
1063    fn mark_malformed_location(&mut self, view: &RegionView<'_>, file: usize) {
1064        self.incomplete = true;
1065        self.malformed_evidence = true;
1066        self.views.insert(view.view);
1067        let Some(path) = view.files.get(file).map(|file| &file.path) else {
1068            return;
1069        };
1070        record_bounded_path(
1071            &mut self.malformed_paths,
1072            &mut self.malformed_paths_truncated,
1073            path.clone(),
1074        );
1075    }
1076
1077    fn issues(&self) -> Vec<SenseIssue> {
1078        let mut issues = Vec::new();
1079        if let Some(issue) = self.limit_issue() {
1080            issues.push(issue);
1081        }
1082        if self.malformed_evidence {
1083            let mut issue = SenseIssue::new(
1084                "dedup_exact_evidence_malformed",
1085                "exact token evidence was malformed; duplicate comparison could not complete",
1086            );
1087            issue.views = self.views.iter().copied().collect();
1088            issue.paths = self.malformed_paths.iter().cloned().collect();
1089            issue.paths_truncated = self.malformed_paths_truncated;
1090            issue.next_step = Some(
1091                concat!(
1092                    "Rerun with a fresh local cache. If the error repeats, report the affected ",
1093                    "files; --allow-partial cannot make malformed evidence complete."
1094                )
1095                .into(),
1096            );
1097            issues.push(issue);
1098        }
1099        issues
1100    }
1101
1102    fn limit_issue(&self) -> Option<SenseIssue> {
1103        if !self.limit_reached {
1104            return None;
1105        }
1106        let mut issue = SenseIssue::new(
1107            "dedup_exact_comparison_limit",
1108            format!(
1109                "exact token comparison reached the fixed limit of {MAX_DEDUP_TOKEN_COMPARISONS}; candidate duplicate regions remain unchecked"
1110            ),
1111        );
1112        issue.stage = Some(DedupLimitStage::ExactTokenComparison);
1113        issue.views = self.views.iter().copied().collect();
1114        issue.limit = Some(MAX_DEDUP_TOKEN_COMPARISONS);
1115        issue.processed = Some(MAX_DEDUP_TOKEN_COMPARISONS);
1116        issue.paths = self.limit_paths.iter().cloned().collect();
1117        // Evaluation stops at the exhausted comparison. Paths from unchecked candidates are
1118        // intentionally unknown, so this remains a bounded sample rather than complete evidence.
1119        issue.paths_truncated = true;
1120        issue.next_step = Some(concat!(
1121            "No runtime option raises this fixed safety limit. Split unusually repetitive source ",
1122            "or, for generated/vendor trees, use literal targets.exclude entries such as ",
1123            "{\"schemaVersion\":1,\"targets\":{\"exclude\":[\"generated\",\"vendor\"]}}. See ",
1124            "https://the-open-engine.github.io/opcore/v",
1125            env!("CARGO_PKG_VERSION_MAJOR"),
1126            ".",
1127            env!("CARGO_PKG_VERSION_MINOR"),
1128            "/docs/configuration.html#select-targets."
1129        ).into());
1130        Some(issue)
1131    }
1132
1133    fn token_exists(&mut self, view: &RegionView<'_>, file_index: usize, token: usize) -> bool {
1134        let Some(file) = view.files.get(file_index) else {
1135            self.mark_malformed_location(view, file_index);
1136            return false;
1137        };
1138        let tokens = file.tokens();
1139        token < tokens.tokens.len()
1140    }
1141
1142    fn tokens_equal(&mut self, pair: TokenPair<'_, '_>) -> Option<bool> {
1143        if self.remaining == 0 {
1144            self.incomplete = true;
1145            self.limit_reached = true;
1146            self.record_limit_pair(&pair);
1147            return None;
1148        }
1149        self.remaining = self.remaining.saturating_sub(1);
1150        let Some(left) = pair.left_view.exact_token(pair.left) else {
1151            self.mark_malformed_location(pair.left_view, pair.left.file);
1152            return None;
1153        };
1154        let Some(right) = pair.right_view.exact_token(pair.right) else {
1155            self.mark_malformed_location(pair.right_view, pair.right.file);
1156            return None;
1157        };
1158        Some(left == right)
1159    }
1160
1161    fn regions_equal(&mut self, pair: &RegionPair<'_, '_>) -> Option<bool> {
1162        if pair.left.len() != pair.right.len() {
1163            return Some(false);
1164        }
1165        if !self.token_exists(
1166            pair.left_view,
1167            pair.left_file,
1168            pair.left.end.saturating_sub(1),
1169        ) || !self.token_exists(
1170            pair.right_view,
1171            pair.right_file,
1172            pair.right.end.saturating_sub(1),
1173        ) {
1174            return if self.incomplete { None } else { Some(false) };
1175        }
1176        for offset in 0..pair.left.len() {
1177            if !self.tokens_equal(TokenPair {
1178                left_view: pair.left_view,
1179                left: TokenLocation {
1180                    file: pair.left_file,
1181                    token: pair.left.start.saturating_add(offset),
1182                },
1183                right_view: pair.right_view,
1184                right: TokenLocation {
1185                    file: pair.right_file,
1186                    token: pair.right.start.saturating_add(offset),
1187                },
1188            })? {
1189                return Some(false);
1190            }
1191        }
1192        Some(true)
1193    }
1194}
