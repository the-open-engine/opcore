20/// Renders one bundled provider manifest bound to the exact installed executable bytes.
21///
22/// # Errors
23///
24/// Returns an error when the path is not an absolute regular file or cannot be canonicalized,
25/// opened, read, or hashed.
26pub fn render_for(executable: &Path, profile: ProviderProfile) -> Result<Value> {
27    ensure!(
28        executable.is_absolute(),
29        "installed executable path must be absolute"
30    );
31    let executable = executable
32        .canonicalize()
33        .context("canonicalize installed executable")?;
34    let digest = artifact::digest_path(&executable).context("hash installed executable")?;
35    let sha256 = digest
36        .strip_prefix("sha256:")
37        .context("installed executable digest has an invalid domain")?
38        .to_owned();
39    let settings = manifest_profile(profile);
40    Ok(json!({
41        "manifestVersion": MANIFEST_VERSION,
42        "server": {
43            "id": profile.provider_id(),
44            "name": profile.provider_name(),
45            "version": env!("CARGO_PKG_VERSION")
46        },
47        "protocolVersions": [PROTOCOL_VERSION],
48        "roles": ["judge"],
49        "capabilities": ["check"],
50        "capabilityProfiles": [profile.capability_profile()],
51        "entrypoint": {
52            "transport": "stdio",
53            "bin": executable,
54            "args": settings.arguments
55        },
56        "artifact": {
57            "fingerprint": digest,
58            "checksums": [{ "path": executable, "sha256": sha256 }]
59        },
60        "provenance": {
61            "publisher": "the-open-engine",
62            "source": "https://github.com/the-open-engine/opcore",
63            "license": "Apache-2.0"
64        },
65        "accessExpectations": {
66            "filesystem": { "read": settings.filesystem_read, "write": settings.filesystem_write },
67            "network": { "outbound": false, "allowlist": [] },
68            "secrets": { "names": [] },
69            "environment": { "inherit": false, "variables": settings.environment },
70            "dataClasses": settings.data_classes
71        }
72    }))
73}

83fn manifest_profile(profile: ProviderProfile) -> ManifestProfile {
84    match profile {
85        ProviderProfile::Fast => ManifestProfile {
86            arguments: json!(["serve", "--stdio"]),
87            environment: Vec::new(),
88            filesystem_read: Vec::new(),
89            filesystem_write: Vec::new(),
90            data_classes: vec!["source-code", "diff-metadata"],
91        },
92        ProviderProfile::RustNative => {
93            let mut environment = super::native::child_environment(profile).to_vec();
94            environment.push("OPCORE_CARGO");
95            ManifestProfile {
96                arguments: json!(["serve", "--stdio", "--profile", "rust-native"]),
97                environment,
98                filesystem_read: vec!["host-provided-rust-toolchain", "host-provided-cargo-cache"],
99                filesystem_write: vec![
100                    "provider-private-scratch",
101                    "host-provided-cargo-cache",
102                    "host-provided-rust-toolchain",
103                ],
104                data_classes: vec!["source-code", "repository-content", "diff-metadata"],
105            }
106        }
107        ProviderProfile::NodeNative => native_manifest_profile(
108            profile,
109            "OPCORE_NPM",
110            "host-provided-node-toolchain",
111            "host-provided-npm-cache",
112            true,
113        ),
114        ProviderProfile::PythonNative => native_manifest_profile(
115            profile,
116            "OPCORE_PYRIGHT",
117            "host-provided-python-type-checker",
118            "host-provided-python-environment",
119            false,
120        ),
121    }
122}

124fn native_manifest_profile(
125    profile: ProviderProfile,
126    configured_tool: &'static str,
127    toolchain: &'static str,
128    cache: &'static str,
129    writable_cache: bool,
130) -> ManifestProfile {
131    let mut environment = super::native::child_environment(profile).to_vec();
132    environment.push(configured_tool);
133    if profile == ProviderProfile::PythonNative {
134        environment.push("OPCORE_MYPY");
135    }
136    let mut filesystem_write = vec!["provider-private-scratch"];
137    if writable_cache {
138        filesystem_write.push(cache);
139    }
140    ManifestProfile {
141        arguments: json!(["serve", "--stdio", "--profile", profile.cli_name()]),
142        environment,
143        filesystem_read: vec![toolchain, cache],
144        filesystem_write,
145        data_classes: vec!["source-code", "repository-content", "diff-metadata"],
146    }
147}
