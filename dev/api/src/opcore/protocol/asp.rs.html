275impl ScopeWire {
276    fn normalize(self) -> Result<Scope, RpcFailure> {
277        match self {
278            Self::Named(NamedScope::Changeset) => Ok(Scope::Changeset),
279            Self::Named(NamedScope::Workspace) => Ok(Scope::Workspace),
280            Self::Paths(PathScope { paths }) if paths.is_empty() => Err(RpcFailure::input(
281                "path scope must contain at least one path",
282            )),
283            Self::Paths(PathScope { paths }) => {
284                let unique: BTreeSet<_> = paths.iter().collect();
285                if unique.len() != paths.len() {
286                    return Err(RpcFailure::input("path scope contains duplicates"));
287                }
288                Ok(Scope::Paths { paths })
289            }
290        }
291    }

319#[derive(Clone, Copy)]
320struct RequestDeadline(Instant);
321
322impl RequestDeadline {
323    fn new(wallclock_ms: u64) -> Result<Self, RpcFailure> {
324        Instant::now()
325            .checked_add(Duration::from_millis(wallclock_ms))
326            .map(Self)
327            .ok_or_else(|| RpcFailure::policy("evaluation wallclock grant overflowed"))
328    }
329
330    fn remaining(self) -> Result<Duration, RpcFailure> {
331        self.0
332            .checked_duration_since(Instant::now())
333            .filter(|remaining| !remaining.is_zero())
334            .ok_or_else(|| RpcFailure::unavailable("evaluation exceeded its wallclock grant"))
335    }
336
337    fn remaining_ms(self) -> Result<u64, RpcFailure> {
338        let remaining = self.remaining()?;
339        let millis = u64::try_from(remaining.as_millis()).unwrap_or(u64::MAX);
340        Ok(millis.max(1))
341    }
342
343    fn work_timeout(self, reserve: Duration) -> Result<Duration, RpcFailure> {
344        self.remaining()?
345            .checked_sub(reserve)
346            .filter(|remaining| !remaining.is_zero())
347            .ok_or_else(|| {
348                RpcFailure::unavailable("insufficient wallclock remains for native cleanup")
349            })
350    }
351
352    fn instant(self) -> Instant {
353        self.0
354    }
355}

413fn initialize_object(params: &Value) -> Result<&serde_json::Map<String, Value>, RpcFailure> {
414    let object = params
415        .as_object()
416        .ok_or_else(|| RpcFailure::input("initialize params must be an object"))?;
417    reject_keys(
418        object,
419        &[
420            "protocolVersion",
421            "host",
422            "hostCapabilities",
423            "workspace",
424            "assuranceMode",
425            "authorityMode",
426            "trustTier",
427        ],
428    )?;
429    if object.get("protocolVersion").and_then(Value::as_str) != Some(PROTOCOL_VERSION) {
430        return Err(RpcFailure {
431            code: -32014,
432            message: "unsupported-version",
433            fail_class: "contract",
434            retryable: false,
435            detail: format!("only {PROTOCOL_VERSION} is supported"),
436        });
437    }
438    if object
439        .get("hostCapabilities")
440        .is_some_and(|value| !value.is_object())
441    {
442        return Err(RpcFailure::input("hostCapabilities must be an object"));
443    }
444    for field in ["assuranceMode", "authorityMode"] {
445        if let Some(value) = object.get(field) {
446            validate_enum_value(
447                value,
448                &["advisory", "gated", "mediated-write", "isolated"],
449                field,
450            )?;
451        }
452    }
453    if let Some(value) = object.get("trustTier") {
454        validate_enum_value(
455            value,
456            &["first-party", "certified", "untrusted"],
457            "trustTier",
458        )?;
459    }
460    Ok(object)
461}

463fn validate_enum_value(value: &Value, choices: &[&str], field: &str) -> Result<(), RpcFailure> {
464    if value.as_str().is_some_and(|value| choices.contains(&value)) {
465        Ok(())
466    } else {
467        Err(RpcFailure::input(format!("{field} is invalid")))
468    }
469}

471fn validate_initialize_host(object: &serde_json::Map<String, Value>) -> Result<(), RpcFailure> {
472    let host = object
473        .get("host")
474        .and_then(Value::as_object)
475        .ok_or_else(|| RpcFailure::input("initialize.host is required"))?;
476    if host.get("name").and_then(Value::as_str).is_none()
477        || host.get("version").and_then(Value::as_str).is_none()
478    {
479        return Err(RpcFailure::input(
480            "initialize.host requires name and version",
481        ));
482    }
483    Ok(())
484}

486fn initialize_baseline(object: &serde_json::Map<String, Value>) -> Result<Value, RpcFailure> {
487    let workspace = object
488        .get("workspace")
489        .and_then(Value::as_object)
490        .ok_or_else(|| RpcFailure::input("initialize.workspace is required"))?;
491    if workspace.get("root").and_then(Value::as_str).is_none() {
492        return Err(RpcFailure::input("initialize.workspace.root is required"));
493    }
494    let baseline: Baseline = serde_json::from_value(
495        workspace
496            .get("baseline")
497            .cloned()
498            .ok_or_else(|| RpcFailure::input("initialize.workspace.baseline is required"))?,
499    )
500    .map_err(|error| RpcFailure::input(format!("invalid initialize baseline: {error}")))?;
501    baseline.validate_timestamp()?;
502    let baseline_json = serde_json::to_value(&baseline)
503        .map_err(|error| RpcFailure::input(format!("cannot serialize baseline: {error}")))?;
504    Ok(baseline_json)
505}

537pub fn initialized(params: &Value, expected_baseline: &Value) -> Result<AspSession, RpcFailure> {
538    let object = params
539        .as_object()
540        .ok_or_else(|| RpcFailure::input("initialized params must be an object"))?;
541    reject_keys(object, &["grantedPermissions", "baseline"])?;
542    let baseline = initialized_baseline(object, expected_baseline)?;
543    let permissions = object
544        .get("grantedPermissions")
545        .and_then(Value::as_object)
546        .ok_or_else(|| RpcFailure::input("initialized.grantedPermissions is required"))?;
547    validate_permissions(permissions)?;
548    let wallclock_ms = granted_wallclock(permissions)?;
549    Ok(AspSession {
550        baseline: baseline.clone(),
551        wallclock_ms,
552        initialized_grant: Value::Object(permissions.clone()),
553    })
554}

556fn initialized_baseline<'a>(
557    object: &'a serde_json::Map<String, Value>,
558    expected: &Value,
559) -> Result<&'a Value, RpcFailure> {
560    let baseline = object
561        .get("baseline")
562        .ok_or_else(|| RpcFailure::input("initialized.baseline is required"))?;
563    let parsed: Baseline = serde_json::from_value(baseline.clone())
564        .map_err(|error| RpcFailure::input(format!("invalid initialized baseline: {error}")))?;
565    parsed.validate_timestamp()?;
566    if canonical_json(baseline)? != canonical_json(expected)? {
567        return Err(RpcFailure::health(
568            "initialized baseline differs from initialize baseline",
569        ));
570    }
571    Ok(baseline)
572}

574fn validate_permissions(permissions: &serde_json::Map<String, Value>) -> Result<(), RpcFailure> {
575    reject_keys(
576        permissions,
577        &[
578            "read",
579            "write",
580            "network",
581            "networkAllowlist",
582            "resourceLimits",
583        ],
584    )?;
585    validate_denied_permission(permissions, "write")?;
586    validate_denied_permission(permissions, "network")?;
587    validate_network_allowlist(permissions.get("networkAllowlist"))?;
588    match permissions.get("read") {
589        Some(Value::Array(values)) if values.iter().all(Value::is_string) => {}
590        Some(Value::Array(_)) => {
591            return Err(RpcFailure::input("granted read entries must be strings"));
592        }
593        None => {}
594        _ => {
595            return Err(RpcFailure::input(
596                "granted read permission must be an array",
597            ));
598        }
599    }
600    Ok(())
601}
602
603fn validate_network_allowlist(value: Option<&Value>) -> Result<(), RpcFailure> {
604    let Some(value) = value else {
605        return Ok(());
606    };
607    let entries = value
608        .as_array()
609        .ok_or_else(|| RpcFailure::input("networkAllowlist must be an array"))?;
610    for entry in entries {
611        let entry = entry
612            .as_object()
613            .ok_or_else(|| RpcFailure::input("networkAllowlist entries must be objects"))?;
614        reject_keys(entry, &["host", "port", "proto"])?;
615        if entry.get("host").and_then(Value::as_str).is_none() {
616            return Err(RpcFailure::input(
617                "networkAllowlist entry requires a string host",
618            ));
619        }
620        if entry.get("port").is_some_and(|port| !port.is_i64()) {
621            return Err(RpcFailure::input(
622                "networkAllowlist port must be an integer",
623            ));
624        }
625        if let Some(proto) = entry.get("proto") {
626            validate_enum_value(proto, &["https", "http", "tcp"], "networkAllowlist proto")?;
627        }
628    }
629    Ok(())
630}

632fn validate_denied_permission(
633    permissions: &serde_json::Map<String, Value>,
634    name: &str,
635) -> Result<(), RpcFailure> {
636    match permissions.get(name) {
637        None | Some(Value::Bool(false)) => Ok(()),
638        Some(Value::Bool(true)) => Err(RpcFailure::policy(format!(
639            "provider cannot accept {name} permission"
640        ))),
641        Some(_) => Err(RpcFailure::input(format!(
642            "granted {name} permission must be a boolean"
643        ))),
644    }
645}

647fn granted_wallclock(permissions: &serde_json::Map<String, Value>) -> Result<u64, RpcFailure> {
648    let limits = permissions
649        .get("resourceLimits")
650        .and_then(Value::as_object)
651        .ok_or_else(|| RpcFailure::input("initialized resourceLimits are required"))?;
652    reject_keys(limits, &["cpuPct", "memoryMb", "wallclockMs", "fd"])?;
653    if limits.values().any(|value| !value.is_u64()) {
654        return Err(RpcFailure::input(
655            "resource limits must be non-negative integers",
656        ));
657    }
658    let wallclock_ms = limits
659        .get("wallclockMs")
660        .and_then(Value::as_u64)
661        .unwrap_or(30_000)
662        .min(300_000);
663    if wallclock_ms == 0 {
664        return Err(RpcFailure::policy(
665            "initialized wallclockMs grant must permit at least one millisecond",
666        ));
667    }
668    Ok(wallclock_ms)
669}

715pub(crate) async fn evaluate_for(
716    params: Value,
717    session: AspSession,
718    profile: ProviderProfile,
719    runtime: EvaluationRuntime<'_>,
720) -> Result<Value, RpcFailure> {
721    let deadline = RequestDeadline::new(session.wallclock_ms)?;
722    let normalized = match normalize_params_for(params, &session, profile)? {
723        NormalizedRequest::Supported(params) => params,
724        NormalizedRequest::Unsupported(params) => {
725            return Ok(unsupported_comparison_assessment(
726                &params, &session, profile,
727            ));
728        }
729    };
730    ensure_not_cancelled(&runtime.cancel)?;
731    if matches!(normalized.scope, Scope::Workspace) && !has_full_workspace_read(&session) {
732        return Ok(unsupported_normalized_assessment(
733            &normalized,
734            &session,
735            profile,
736            "workspace scope requires the exact full-workspace read grant",
737        ));
738    }
739    if profile.is_native() {
740        return native_provider::evaluate(native_provider::NativeEvaluation {
741            params: normalized,
742            session,
743            profile,
744            host: runtime.host,
745            cancel: runtime.cancel,
746            deadline,
747            project_root: runtime.project_root,
748        })
749        .await;
750    }
751    evaluate_fast(normalized, &session, runtime, deadline).await
752}
753
754async fn evaluate_fast(
755    normalized: NormalizedParams,
756    session: &AspSession,
757    runtime: EvaluationRuntime<'_>,
758    deadline: RequestDeadline,
759) -> Result<Value, RpcFailure> {
760    let prepared = match prepare_views(
761        &normalized,
762        session,
763        runtime.host,
764        &runtime.cancel,
765        deadline,
766    )
767    .await
768    {
769        Ok(prepared) => prepared,
770        Err(WorkspaceFailure::Rpc(failure)) => return Err(failure),
771        Err(WorkspaceFailure::Coverage(gap)) => {
772            return Ok(workspace_gap_assessment(
773                &normalized,
774                session,
775                ProviderProfile::Fast,
776                &gap,
777            ));
778        }
779    };
780    ensure_not_cancelled(&runtime.cancel)?;
781    let assessment = runtime
782        .engine
783        .evaluate(
784            EvaluationRequest {
785                before: Some(prepared.before),
786                after: prepared.after,
787                scope: normalized.scope.clone(),
788                comparison: normalized.comparison,
789                paths: prepared.paths,
790                limits: normalized.configuration.verify,
791                valid_as_of: normalized.changeset_digest.clone(),
792                public_fingerprint_comparison: true,
793            },
794            runtime.cancel.clone(),
795        )
796        .await
797        .map_err(|error| RpcFailure::unavailable(error.to_string()))?;
798    if runtime.cancel.is_cancelled() || assessment.status == AssessmentStatus::Cancelled {
799        return Err(RpcFailure::cancelled());
800    }
801    Ok(to_asp_assessment(
802        assessment,
803        &normalized,
804        &prepared.read_ids,
805        &session.baseline,
806        &session.initialized_grant,
807    ))
808}

811    if cancel.is_cancelled() {
812        Err(RpcFailure::cancelled())
813    } else {
814        Ok(())
815    }
816}

940fn validate_request_names(values: &[String], label: &str) -> Result<(), RpcFailure> {
941    if values.len() > MAX_FILES {
942        return Err(RpcFailure::input(format!(
943            "{label} exceeds {MAX_FILES} entries"
944        )));
945    }
946    let unique = values.iter().collect::<BTreeSet<_>>();
947    if unique.len() != values.len() {
948        return Err(RpcFailure::input(format!("{label} contains duplicates")));
949    }
950    if values.iter().any(String::is_empty) {
951        return Err(RpcFailure::input(format!(
952            "{label} entries must not be empty"
953        )));
954    }
955    Ok(())
956}

1171fn validate_changeset(changeset: &ChangeSet) -> Result<(), RpcFailure> {
1172    changeset.baseline.validate_timestamp()?;
1173    if changeset.baseline.rev.is_empty() {
1174        return Err(RpcFailure::input("changeset baseline rev is empty"));
1175    }
1176    if changeset.changes.len() > MAX_FILES {
1177        return Err(RpcFailure::input(format!(
1178            "changeset exceeds {MAX_FILES} changes"
1179        )));
1180    }
1181    let mut destinations = BTreeSet::new();
1182    let mut sources = BTreeSet::new();
1183    for change in &changeset.changes {
1184        if !destinations.insert(change.path.clone()) {
1185            return Err(RpcFailure::input(format!(
1186                "duplicate change destination {}",
1187                change.path
1188            )));
1189        }
1190        validate_change(change, &mut sources)?;
1191    }
1192    if destinations.iter().any(|path| sources.contains(path)) {
1193        return Err(RpcFailure::input(
1194            "a path cannot be both a change destination and rename source",
1195        ));
1196    }
1197    Ok(())
1198}

1200pub(crate) fn is_rfc3339_timestamp(value: &str) -> bool {
1201    let bytes = value.as_bytes();
1202    if bytes.len() < 20 || !timestamp_separators(bytes) {
1203        return false;
1204    }
1205    let Some((year, month, day)) = timestamp_date(bytes) else {
1206        return false;
1207    };
1208    let Some((hour, minute, second)) = timestamp_time(bytes) else {
1209        return false;
1210    };
1211    valid_date(year, month, day)
1212        && valid_time(hour, minute, second)
1213        && valid_timestamp_suffix(bytes)
1214}
1215
1216fn timestamp_separators(bytes: &[u8]) -> bool {
1217    bytes.get(4) == Some(&b'-')
1218        && bytes.get(7) == Some(&b'-')
1219        && matches!(bytes.get(10), Some(b'T' | b't'))
1220        && bytes.get(13) == Some(&b':')
1221        && bytes.get(16) == Some(&b':')
1222}
1223
1224fn timestamp_date(bytes: &[u8]) -> Option<(u32, u32, u32)> {
1225    Some((
1226        decimal(&bytes[0..4])?,
1227        decimal(&bytes[5..7])?,
1228        decimal(&bytes[8..10])?,
1229    ))
1230}
1231
1232fn timestamp_time(bytes: &[u8]) -> Option<(u32, u32, u32)> {
1233    Some((
1234        decimal(&bytes[11..13])?,
1235        decimal(&bytes[14..16])?,
1236        decimal(&bytes[17..19])?,
1237    ))
1238}
1239
1240fn valid_date(year: u32, month: u32, day: u32) -> bool {
1241    (1..=12).contains(&month) && day > 0 && day <= days_in_month(year, month)
1242}
1243
1244fn valid_time(hour: u32, minute: u32, second: u32) -> bool {
1245    hour <= 23 && minute <= 59 && second <= 60
1246}
1247
1248fn valid_timestamp_suffix(bytes: &[u8]) -> bool {
1249    let mut cursor = 19;
1250    if bytes.get(cursor) == Some(&b'.') {
1251        cursor += 1;
1252        let fraction_start = cursor;
1253        while bytes.get(cursor).is_some_and(u8::is_ascii_digit) {
1254            cursor += 1;
1255        }
1256        if cursor == fraction_start {
1257            return false;
1258        }
1259    }
1260    match bytes.get(cursor) {
1261        Some(b'Z' | b'z') => cursor + 1 == bytes.len(),
1262        Some(b'+' | b'-') if cursor + 6 == bytes.len() => {
1263            bytes.get(cursor + 3) == Some(&b':')
1264                && decimal(&bytes[cursor + 1..cursor + 3]).is_some_and(|value| value <= 23)
1265                && decimal(&bytes[cursor + 4..cursor + 6]).is_some_and(|value| value <= 59)
1266        }
1267        _ => false,
1268    }
1269}
1270
1271fn decimal(bytes: &[u8]) -> Option<u32> {
1272    bytes.iter().try_fold(0_u32, |value, digit| {
1273        digit
1274            .is_ascii_digit()
1275            .then(|| value * 10 + u32::from(*digit - b'0'))
1276    })
1277}
1278
1279const fn days_in_month(year: u32, month: u32) -> u32 {
1280    match month {
1281        2 if year.is_multiple_of(400) || (year.is_multiple_of(4) && !year.is_multiple_of(100)) => {
1282            29
1283        }
1284        2 => 28,
1285        4 | 6 | 9 | 11 => 30,
1286        _ => 31,
1287    }
1288}

1290fn validate_change(change: &Change, sources: &mut BTreeSet<RepoPath>) -> Result<(), RpcFailure> {
1291    let shape = (
1292        change.before.is_some(),
1293        change.after.is_some(),
1294        change.from.is_some(),
1295    );
1296    match (change.kind, shape) {
1297        (ChangeKind::Create, (false, true, false))
1298        | (ChangeKind::Modify, (true, true, false))
1299        | (ChangeKind::Delete, (true, false, false)) => {}
1300        (ChangeKind::Rename, (true, _, true)) => validate_rename(change, sources)?,
1301        _ => return Err(illegal_change(change)),
1302    }
1303    validate_optional_blob(change.before.as_deref())?;
1304    validate_optional_blob(change.after.as_deref())
1305}
1306
1307fn validate_rename(change: &Change, sources: &mut BTreeSet<RepoPath>) -> Result<(), RpcFailure> {
1308    let Some(from) = change.from.as_ref() else {
1309        return Err(illegal_change(change));
1310    };
1311    if from == &change.path {
1312        return Err(RpcFailure::input(
1313            "rename source and destination are identical",
1314        ));
1315    }
1316    if !sources.insert(from.clone()) {
1317        return Err(RpcFailure::input(format!("duplicate rename source {from}")));
1318    }
1319    Ok(())
1320}

1322fn illegal_change(change: &Change) -> RpcFailure {
1323    RpcFailure::input(format!(
1324        "illegal fields for {:?} change at {}",
1325        change.kind, change.path
1326    ))
1327}

1329fn validate_optional_blob(blob: Option<&str>) -> Result<(), RpcFailure> {
1330    blob.map_or(Ok(()), validate_blob_ref)
1331}
1332
1333fn validate_blob_ref(value: &str) -> Result<(), RpcFailure> {
1334    let Some(rest) = value.strip_prefix("blob:") else {
1335        return Err(RpcFailure::input(format!("invalid blob ref {value}")));
1336    };
1337    let Some((algorithm, digest)) = rest.split_once(':') else {
1338        return Err(RpcFailure::input(format!("invalid blob ref {value}")));
1339    };
1340    if algorithm.is_empty()
1341        || !algorithm
1342            .bytes()
1343            .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit())
1344        || digest.is_empty()
1345        || !digest
1346            .bytes()
1347            .all(|byte| byte.is_ascii_hexdigit() && !byte.is_ascii_uppercase())
1348    {
1349        return Err(RpcFailure::input(format!("invalid blob ref {value}")));
1350    }
1351    Ok(())
1352}
1353
1354fn verify_blob(reference: &str, bytes: &[u8]) -> Result<(), RpcFailure> {
1355    let Some(rest) = reference.strip_prefix("blob:") else {
1356        return Err(RpcFailure::input(format!("invalid blob ref {reference}")));
1357    };
1358    let Some((algorithm, expected)) = rest.split_once(':') else {
1359        return Err(RpcFailure::input(format!("invalid blob ref {reference}")));
1360    };
1361    if algorithm != "sha256" {
1362        return Err(RpcFailure::unavailable(format!(
1363            "unsupported blob digest algorithm {algorithm}"
1364        )));
1365    }
1366    let actual = hex::encode(Sha256::digest(bytes));
1367    if actual != expected {
1368        return Err(RpcFailure::unavailable(format!(
1369            "blob hash mismatch for {reference}"
1370        )));
1371    }
1372    Ok(())
1373}
