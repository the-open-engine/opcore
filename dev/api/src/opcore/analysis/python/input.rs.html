15pub(super) fn preflight_structure(source: &str, cancel: &CancelToken) -> Result<(), AnalysisError> {
16    let mut structure = PythonStructure::default();
17    let mut tokens = lexer::lex(source, Mode::Module);
18    let mut index = 0usize;
19    loop {
20        if index.is_multiple_of(CANCELLATION_POLL_INTERVAL) {
21            cancellation_boundary(cancel)?;
22        }
23        let token = tokens.next_token();
24        if token.is_eof() {
25            break;
26        }
27        index = index.saturating_add(1);
28        if structure.ignore_or_reset(token) {
29            continue;
30        }
31        if structure.observe_layout(token) {
32            continue;
33        }
34        structure.observe_line_start(token);
35        structure.observe_expression(token);
36        structure.observe_unary(token);
37        structure.validate()?;
38    }
39    drop(tokens.finish());
40    Ok(())
41}

50pub(super) fn fingerprint_tokens(tokens: &Tokens) -> Vec<PythonFingerprintToken> {
51    tokens
52        .iter()
53        .filter_map(|token| {
54            python_fingerprint_tag(token.kind()).map(|tag| PythonFingerprintToken {
55                start: token.start().to_usize(),
56                end: token.end().to_usize(),
57                tag,
58            })
59        })
60        .collect()
61}

63const fn python_fingerprint_tag(token: TokenKind) -> Option<u8> {
64    match token {
65        TokenKind::Comment | TokenKind::NonLogicalNewline | TokenKind::EndOfFile => None,
66        TokenKind::Indent => Some(1),
67        TokenKind::Dedent => Some(2),
68        TokenKind::Newline => Some(3),
69        _ => Some(0),
70    }
71}

73struct PythonStructure {
74    delimiters: usize,
75    indents: usize,
76    unary: usize,
77    expression_nodes: usize,
78    logical_line_start: bool,
79    elif_chain_by_indent: Vec<usize>,
80}
81
82impl Default for PythonStructure {
83    fn default() -> Self {
84        Self {
85            delimiters: 0,
86            indents: 0,
87            unary: 0,
88            expression_nodes: 0,
89            logical_line_start: true,
90            elif_chain_by_indent: vec![0],
91        }
92    }
93}
94
95impl PythonStructure {
96    fn ignore_or_reset(&mut self, token: TokenKind) -> bool {
97        if matches!(token, TokenKind::Comment | TokenKind::NonLogicalNewline) {
98            return true;
99        }
100        if !matches!(token, TokenKind::Newline) {
101            return false;
102        }
103        self.unary = 0;
104        self.expression_nodes = 0;
105        self.logical_line_start = true;
106        true
107    }
108
109    fn observe_layout(&mut self, token: TokenKind) -> bool {
110        match token {
111            TokenKind::Lpar | TokenKind::Lsqb | TokenKind::Lbrace => {
112                self.delimiters = self.delimiters.saturating_add(1);
113            }
114            TokenKind::Rpar | TokenKind::Rsqb | TokenKind::Rbrace => {
115                self.delimiters = self.delimiters.saturating_sub(1);
116            }
117            TokenKind::Indent => {
118                self.indents = self.indents.saturating_add(1);
119                self.elif_chain_by_indent
120                    .resize(self.indents.saturating_add(1), 0);
121                return true;
122            }
123            TokenKind::Dedent => {
124                self.indents = self.indents.saturating_sub(1);
125                return true;
126            }
127            _ => {}
128        }
129        false
130    }
131
132    fn observe_line_start(&mut self, token: TokenKind) {
133        if !self.logical_line_start {
134            return;
135        }
136        let chain = &mut self.elif_chain_by_indent[self.indents];
137        match token {
138            TokenKind::If => *chain = 1,
139            TokenKind::Elif => *chain = chain.saturating_add(1),
140            TokenKind::Else => {}
141            _ => *chain = 0,
142        }
143        self.logical_line_start = false;
144    }
145
146    fn observe_expression(&mut self, token: TokenKind) {
147        if matches!(
148            token,
149            TokenKind::Plus
150                | TokenKind::Minus
151                | TokenKind::Star
152                | TokenKind::Slash
153                | TokenKind::Vbar
154                | TokenKind::Amper
155                | TokenKind::Less
156                | TokenKind::Greater
157                | TokenKind::Dot
158                | TokenKind::Percent
159                | TokenKind::EqEqual
160                | TokenKind::NotEqual
161                | TokenKind::LessEqual
162                | TokenKind::GreaterEqual
163                | TokenKind::CircumFlex
164                | TokenKind::LeftShift
165                | TokenKind::RightShift
166                | TokenKind::DoubleStar
167                | TokenKind::DoubleSlash
168                | TokenKind::And
169                | TokenKind::Or
170                | TokenKind::In
171                | TokenKind::Is
172                | TokenKind::If
173                | TokenKind::Else
174                | TokenKind::Lambda
175                | TokenKind::Lpar
176                | TokenKind::Lsqb
177        ) {
178            self.expression_nodes = self.expression_nodes.saturating_add(1);
179        } else if matches!(token, TokenKind::Comma | TokenKind::Semi) {
180            self.expression_nodes = 0;
181        }
182    }
183
184    fn observe_unary(&mut self, token: TokenKind) {
185        if matches!(
186            token,
187            TokenKind::Not
188                | TokenKind::Plus
189                | TokenKind::Minus
190                | TokenKind::Tilde
191                | TokenKind::Await
192        ) {
193            self.unary = self.unary.saturating_add(1);
194        } else {
195            self.unary = 0;
196        }
197    }
198
199    fn validate(&self) -> Result<(), AnalysisError> {
200        if self.delimiters > MAX_STRUCTURAL_DEPTH
201            || self.indents > MAX_STRUCTURAL_DEPTH
202            || self.unary > MAX_STRUCTURAL_DEPTH
203            || self.expression_nodes > MAX_STRUCTURAL_DEPTH
204            || self.elif_chain_by_indent[self.indents] > MAX_STRUCTURAL_DEPTH
205        {
206            return Err(AnalysisError::Unsupported(format!(
207                "source structural depth exceeds the {MAX_STRUCTURAL_DEPTH}-level safety limit"
208            )));
209        }
210        Ok(())
211    }
212}

214pub(super) struct DecodedSource<'a> {
215    pub(super) text: &'a str,
216    pub(super) byte_base: usize,
217}
218
219pub(super) fn decode_source(bytes: &[u8]) -> Result<DecodedSource<'_>, AnalysisError> {
220    let (body, byte_base) = bytes
221        .strip_prefix(UTF8_BOM)
222        .map_or((bytes, 0), |without_bom| (without_bom, UTF8_BOM.len()));
223
224    if let Some(encoding) = declared_encoding(body)
225        && !is_utf8_encoding(&encoding)
226    {
227        return Err(AnalysisError::Unsupported(format!(
228            "declared Python source encoding '{encoding}' is unsupported; UTF-8 is required"
229        )));
230    }
231
232    let text = std::str::from_utf8(body)
233        .map_err(|_| AnalysisError::Unsupported("Python source is not valid UTF-8".into()))?;
234    Ok(DecodedSource { text, byte_base })
235}

237fn declared_encoding(bytes: &[u8]) -> Option<String> {
238    let mut lines = bytes.splitn(3, |byte| *byte == b'\n');
239    let first_with_ending = lines.next().unwrap_or_default();
240    let first = first_with_ending
241        .strip_suffix(b"\r")
242        .unwrap_or(first_with_ending);
243    if let Some(encoding) = encoding_cookie(first) {
244        return Some(encoding);
245    }
246    if !comment_or_blank(first) {
247        return None;
248    }
249    lines
250        .next()
251        .and_then(|line| encoding_cookie(line.strip_suffix(b"\r").unwrap_or(line)))
252}

254fn comment_or_blank(line: &[u8]) -> bool {
255    line.iter()
256        .copied()
257        .find(|byte| !matches!(byte, b' ' | b'\t' | 0x0c))
258        .is_none_or(|byte| byte == b'#')
259}
260
261fn encoding_cookie(line: &[u8]) -> Option<String> {
262    let comment_start = line
263        .iter()
264        .position(|byte| !matches!(byte, b' ' | b'\t' | 0x0c))?;
265    if line[comment_start] != b'#' {
266        return None;
267    }
268    let comment = &line[comment_start + 1..];
269    let lower = comment
270        .iter()
271        .map(u8::to_ascii_lowercase)
272        .collect::<Vec<_>>();
273    for (index, window) in lower.windows(b"coding".len()).enumerate() {
274        if window != b"coding" {
275            continue;
276        }
277        let mut cursor = index + b"coding".len();
278        while lower.get(cursor).is_some_and(u8::is_ascii_whitespace) {
279            cursor += 1;
280        }
281        if !matches!(lower.get(cursor), Some(b':' | b'=')) {
282            continue;
283        }
284        cursor += 1;
285        while lower.get(cursor).is_some_and(u8::is_ascii_whitespace) {
286            cursor += 1;
287        }
288        let start = cursor;
289        while lower
290            .get(cursor)
291            .is_some_and(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.'))
292        {
293            cursor += 1;
294        }
295        if cursor > start {
296            return Some(String::from_utf8_lossy(&comment[start..cursor]).into_owned());
297        }
298    }
299    None
300}

302fn is_utf8_encoding(encoding: &str) -> bool {
303    matches!(
304        encoding
305            .bytes()
306            .filter(|byte| !matches!(byte, b'-' | b'_' | b'.'))
307            .map(|byte| char::from(byte).to_ascii_lowercase())
308            .collect::<String>()
309            .as_str(),
310        "utf" | "u8" | "utf8" | "utf8sig" | "cp65001"
311    )
