8impl<'a> RegionView<'a> {
9    pub(super) fn new(inputs: &RegionViewInputs<'a, '_>) -> Self {
10        let mut view = Self {
11            view: inputs.view,
12            files: Vec::new(),
13            groups: BTreeMap::new(),
14            retained_anchors: 0,
15            retained_tokens: 0,
16            decoded_evidence: inputs.decoded_evidence.clone(),
17            #[cfg(test)]
18            indexed: 0,
19            malformed_paths: BTreeSet::new(),
20            malformed_paths_truncated: false,
21        };
22        for source in inputs.snapshot.files() {
23            view.add_file(source, inputs);
24        }
25        view
26    }
27
28    fn add_file(&mut self, source: &'a SourceFile, inputs: &RegionViewInputs<'a, '_>) {
29        let Some((key, facts)) = lookup_file_fact(source, inputs.file_facts, inputs.fact_keys)
30        else {
31            return;
32        };
33        let region = &facts.region_fingerprints;
34        if region.status != FingerprintExtractionStatus::Complete {
35            return;
36        }
37        let path = inputs
38            .continuity
39            .get(&source.path)
40            .cloned()
41            .unwrap_or_else(|| source.path.clone());
42        let state = self
43            .decoded_evidence
44            .borrow_mut()
45            .state(key, source, region, false);
46        if matches!(state, EvidenceState::Malformed) {
47            record_bounded_path(
48                &mut self.malformed_paths,
49                &mut self.malformed_paths_truncated,
50                path,
51            );
52            return;
53        }
54        if region.anchors.len() > MAX_REGION_FINGERPRINTS_PER_FILE {
55            return;
56        }
57        let Some(anchor_order) = self.decoded_evidence.borrow_mut().anchor_order(key, region)
58        else {
59            record_bounded_path(
60                &mut self.malformed_paths,
61                &mut self.malformed_paths_truncated,
62                path,
63            );
64            return;
65        };
66        self.retained_anchors = self.retained_anchors.saturating_add(region.anchors.len());
67        self.retained_tokens = self
68            .retained_tokens
69            .saturating_add(usize::try_from(region.token_count).unwrap_or(usize::MAX));
70        self.files.push(RegionFile {
71            source,
72            path,
73            fact_key: key,
74            facts: Arc::clone(&inputs.file_facts[&key]),
75            anchor_order,
76        });
77    }

79    pub(super) fn index_batch(&mut self, selection: &batches::Selection) {
80        self.groups.clear();
81        for (file_index, (file, range)) in self.files.iter().zip(&selection.ranges).enumerate() {
82            for index in &file.anchor_order[range.clone()] {
83                let anchor = &file.facts.region_fingerprints.anchors[usize::from(*index)];
84                self.groups
85                    .entry(file.source.language.family())
86                    .or_default()
87                    .entry((anchor.hash_one, anchor.hash_two))
88                    .or_default()
89                    .push(AnchorOccurrence {
90                        file: file_index,
91                        token_index: anchor.token_index,
92                    });
93                #[cfg(test)]
94                {
95                    self.indexed += 1;
96                }
97            }
98        }
99        for occurrences in self.groups.values_mut().flat_map(BTreeMap::values_mut) {
100            occurrences.sort_by(|left, right| {
101                (&self.files[left.file].path, left.token_index)
102                    .cmp(&(&self.files[right.file].path, right.token_index))
103            });
104        }
105    }

107    pub(super) fn coverage(&self, comparison: &ComparisonState) -> RegionCoverage {
108        RegionCoverage {
109            anchors: self.retained_anchors,
110            tokens: self.retained_tokens,
111            request_truncated: (comparison.limit_reached || comparison.decode_processed.is_some())
112                && comparison.views.contains(&self.view),
113            malformed_evidence: !self.malformed_paths.is_empty()
114                || self.malformed_paths_truncated
115                || (comparison.malformed_evidence && comparison.views.contains(&self.view)),
116        }
117    }

119    pub(super) fn tokens(&self, file: usize) -> Option<Arc<DecodedTokens>> {
120        let file = self.files.get(file)?;
121        self.decoded_evidence
122            .borrow_mut()
123            .state(
124                file.fact_key,
125                file.source,
126                &file.facts.region_fingerprints,
127                true,
128            )
129            .decoded()
130    }

132    pub(super) fn exact_token<'token>(
133        &'token self,
134        location: TokenLocation,
135        decoded: &'token DecodedTokens,
136    ) -> Option<ExactToken<'token>> {
137        let file = self.files.get(location.file)?;
138        let token = decoded.tokens.get(location.token)?;
139        Some(ExactToken {
140            tag: token.tag,
141            prefix: decoded.bytes.get(token.prefix.clone())?,
142            text: file.source.bytes.get(token.start..token.end)?,
143        })
144    }
145
146    pub(super) fn region_range(&self, file: usize, range: Range<usize>) -> Option<SourceRange> {
147        self.tokens(file)?.source_range(range)
148    }

150    pub(super) fn region_digest(&self, file: usize, range: Range<usize>) -> Option<String> {
151        let decoded = self.tokens(file)?;
152        let file = self.files.get(file)?;
153        let mut encoded = Vec::new();
154        for token in decoded.tokens.get(range)? {
155            let prefix = decoded.bytes.get(token.prefix.clone())?;
156            encoded.extend_from_slice(
157                &u32::try_from(prefix.len())
158                    .unwrap_or(u32::MAX)
159                    .to_be_bytes(),
160            );
161            encoded.extend_from_slice(prefix);
162            encoded.push(token.tag);
163            let text = file.source.bytes.get(token.start..token.end)?;
164            encoded.extend_from_slice(&u32::try_from(text.len()).unwrap_or(u32::MAX).to_be_bytes());
165            encoded.extend_from_slice(text);
166        }
167        Some(hex::encode(hash_domain(
168            "dedup-token-region/v1",
169            &[file.source.language.family().as_bytes(), &encoded],
170        )))
171    }
