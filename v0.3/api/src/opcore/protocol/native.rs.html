127    tool_digest: String,
128    environment_digest: String,
129    runtime: Option<RuntimeIdentity>,

132#[derive(Clone)]
133struct RuntimeIdentity {
134    kind: &'static str,
135    path: PathBuf,
136    digest: String,
137    source: &'static str,
138    marker: Option<(PathBuf, String)>,
139}
140
141impl NativeExecutionIdentity {
142    pub(crate) fn capture(
143        profile: crate::protocol::asp::ProviderProfile,
144        tool_program: &Path,
145        cancel: &CancelToken,
146        deadline: Instant,
147    ) -> Result<Self, String> {
148        let tool_digest = digest_tool(tool_program, cancel, deadline)?;
149        let environment = child_environment(profile)
150            .iter()
151            .filter_map(|key| env::var_os(key).map(|value| (OsString::from(key), value)))
152            .collect::<Vec<_>>();
153        let runtime = capture_runtime_identity(profile, cancel, deadline)?;
154        let environment_digest = digest_environment(profile, &environment, runtime.as_ref());
155        Ok(Self {
156            environment,
157            tool_digest,
158            environment_digest,
159            runtime,
160        })
161    }
162
163    fn verify_tool(
164        &self,
165        tool_program: &Path,
166        cancel: &CancelToken,
167        deadline: Instant,
168    ) -> Result<(), String> {
169        if digest_tool(tool_program, cancel, deadline)? == self.tool_digest {
170            Ok(())
171        } else {
172            Err("the native tool executable changed during evaluation".into())
173        }
174    }
175
176    fn verify_runtime(&self, cancel: &CancelToken, deadline: Instant) -> Result<(), String> {
177        let Some(runtime) = &self.runtime else {
178            return Err("the native runtime was not selected".into());
179        };
180        if digest_tool(&runtime.path, cancel, deadline)? != runtime.digest {
181            return Err("the native runtime executable changed during evaluation".into());
182        }
183        if let Some((path, expected)) = &runtime.marker
184            && digest_tool(path, cancel, deadline)? != *expected
185        {
186            return Err("the native runtime environment marker changed during evaluation".into());
187        }
188        Ok(())
189    }
190
191    fn verify(
192        &self,
193        tool_program: &Path,
194        cancel: &CancelToken,
195        deadline: Instant,
196    ) -> Result<(), String> {
197        self.verify_tool(tool_program, cancel, deadline)?;
198        self.verify_runtime(cancel, deadline)
199    }
200
201    fn runtime_path(&self) -> Option<&Path> {
202        self.runtime.as_ref().map(|runtime| runtime.path.as_path())
203    }
204
205    fn runtime_source(&self) -> Option<&'static str> {
206        self.runtime.as_ref().map(|runtime| runtime.source)
207    }
208}
209
210fn capture_runtime_identity(
211    profile: crate::protocol::asp::ProviderProfile,
212    cancel: &CancelToken,
213    deadline: Instant,
214) -> Result<Option<RuntimeIdentity>, String> {
215    let Some((kind, path, source)) = runtime_selection(profile)? else {
216        return Ok(None);
217    };
218    let digest = digest_tool(&path, cancel, deadline)?;
219    let marker = (profile == crate::protocol::asp::ProviderProfile::PythonNative)
220        .then(|| python_environment_marker(&path, cancel, deadline))
221        .transpose()?
222        .flatten();
223    Ok(Some(RuntimeIdentity {
224        kind,
225        path,
226        digest,
227        source,
228        marker,
229    }))
230}
231
232fn runtime_selection(
233    profile: crate::protocol::asp::ProviderProfile,
234) -> Result<Option<(&'static str, PathBuf, &'static str)>, String> {
235    match profile {
236        crate::protocol::asp::ProviderProfile::Fast => Ok(None),
237        crate::protocol::asp::ProviderProfile::RustNative => {
238            Ok(runtime_on_path("rustc").map(|path| ("rustc-launcher", path, "path")))
239        }
240        crate::protocol::asp::ProviderProfile::NodeNative => {
241            Ok(runtime_on_path("node").map(|path| ("node", path, "path")))
242        }
243        crate::protocol::asp::ProviderProfile::PythonNative => python_runtime_selection()
244            .map(|selection| selection.map(|(path, source)| ("python-interpreter", path, source))),
245    }
246}

248pub(super) fn python_runtime_selection() -> Result<Option<(PathBuf, &'static str)>, String> {
249    let configured = env::var_os("OPCORE_PYTHON").map(PathBuf::from);
250    let selection = if let Some(path) = configured {
251        Some((path, python_environment_source()?))
252    } else {
253        runtime_on_path("python3")
254            .or_else(|| runtime_on_path("python"))
255            .map(|path| (path, "ambient"))
256    };
257    if selection
258        .as_ref()
259        .is_some_and(|(path, _)| !path.is_absolute() || !path.is_file())
260    {
261        return Err("the selected Python interpreter is not an absolute file".into());
262    }
263    Ok(selection)
264}
265
266fn python_environment_source() -> Result<&'static str, String> {
267    match env::var("OPCORE_PYTHON_SOURCE").as_deref() {
268        Ok("project-dot-venv") => Ok("project-dot-venv"),
269        Ok("active-project-venv") => Ok("active-project-venv"),
270        Ok("explicit") | Err(env::VarError::NotPresent) => Ok("explicit"),
271        Ok(_) => Err("OPCORE_PYTHON_SOURCE is unsupported".into()),
272        Err(env::VarError::NotUnicode(_)) => {
273            Err("OPCORE_PYTHON_SOURCE is not valid Unicode".into())
274        }
275    }
276}

278fn python_environment_marker(
279    interpreter: &Path,
280    cancel: &CancelToken,
281    deadline: Instant,
282) -> Result<Option<(PathBuf, String)>, String> {
283    let Some(root) = interpreter.parent().and_then(Path::parent) else {
284        return Ok(None);
285    };
286    let marker = root.join("pyvenv.cfg");
287    if !marker.is_file() {
288        return Ok(None);
289    }
290    let digest = digest_tool(&marker, cancel, deadline)?;
291    Ok(Some((marker, digest)))
292}
293
294fn runtime_on_path(executable: &str) -> Option<PathBuf> {
295    let path = env::var_os("PATH")?;
296    env::split_paths(&path).find_map(|directory| {
297        let candidate = directory.join(if cfg!(windows) {
298            format!("{executable}.exe")
299        } else {
300            executable.to_owned()
301        });
302        (candidate.is_absolute() && candidate.is_file()).then_some(candidate)
303    })
304}

318/// A host may lower these limits to honor its request deadline; hard ceilings stay fixed.
319#[derive(Clone, Copy, Debug, Eq, PartialEq)]
320pub(crate) struct NativeLimits {
321    timeout: Duration,
322    max_output_bytes: usize,
323}
324
325impl NativeLimits {
326    pub(crate) fn bounded(timeout: Duration, max_output_bytes: usize) -> Option<Self> {
327        let valid = !timeout.is_zero()
328            && timeout <= MAX_TIMEOUT
329            && max_output_bytes > 0
330            && max_output_bytes <= MAX_OUTPUT_BYTES;
331        valid.then_some(Self {
332            timeout,
333            max_output_bytes,
334        })
335    }
336}
337
338impl Default for NativeLimits {
339    fn default() -> Self {
340        Self {
341            timeout: DEFAULT_TIMEOUT,
342            max_output_bytes: DEFAULT_OUTPUT_BYTES,
343        }
344    }
345}

439/// Runs the profile's intentionally fixed native command. Policy and decisions remain host-owned.
440pub(crate) fn evaluate(request: &NativeCheckRequest<'_>) -> NativeCheckResult {
441    match request.profile {
442        crate::protocol::asp::ProviderProfile::RustNative => evaluate_rust(request),
443        crate::protocol::asp::ProviderProfile::NodeNative => node::evaluate(request),
444        crate::protocol::asp::ProviderProfile::PythonNative => python::evaluate(request),
445        crate::protocol::asp::ProviderProfile::Fast => unavailable_result(
446            request,
447            "fast checks do not use the native execution backend",
448        ),
449    }
450}

453    profile: crate::protocol::asp::ProviderProfile,
454    files: &BTreeMap<RepoPath, Arc<[u8]>>,

461        crate::protocol::asp::ProviderProfile::PythonNative => python::applicable(files),
462        crate::protocol::asp::ProviderProfile::Fast => false,
463    }

466pub(crate) fn not_applicable(profile: crate::protocol::asp::ProviderProfile) -> NativeCheckResult {
467    let detail = match profile {
468        crate::protocol::asp::ProviderProfile::RustNative => "the candidate has no root Cargo.toml",
469        crate::protocol::asp::ProviderProfile::NodeNative => {
470            "the candidate has no root locked npm TypeScript project"
471        }
472        crate::protocol::asp::ProviderProfile::PythonNative => "the candidate has no Python source",
473        crate::protocol::asp::ProviderProfile::Fast => "fast is not a native provider",
474    };
475    NativeCheckResult {

477        diagnostics: Vec::new(),

481            "elapsedMs": 0,
482            "isolation": "not-executed"
483        }),
484        comparison_safe: true,
485    }

488fn evaluate_rust(request: &NativeCheckRequest<'_>) -> NativeCheckResult {
489    let started = Instant::now();
490    let deadline = started
491        .checked_add(request.limits.timeout)
492        .unwrap_or(started);
493    let tool = request.tool_program.to_string_lossy().into_owned();
494    if request.cancel.is_cancelled() {
495        return NativeEvidence::new(request, tool, Vec::new(), false).finish(
496            NativeStatus::Cancelled,
497            Vec::new(),
498            "Rust-native evaluation was cancelled",
499            started,
500        );
501    }
502    if request.identity.runtime_path().is_none() {
503        return NativeEvidence::new(request, tool, Vec::new(), false).finish(
504            NativeStatus::Unavailable,
505            Vec::new(),
506            "Rust-native could not bind the PATH-selected rustc launcher",
507            started,
508        );
509    }
510    let applicability = match process::preflight(request, deadline) {
511        Ok(applicability) => applicability,
512        Err(failure) => {
513            return NativeEvidence::new(request, tool, Vec::new(), false).finish(
514                failure.status,
515                Vec::new(),
516                failure.detail,
517                started,
518            );
519        }
520    };
521    let process::Applicability::Applicable {
522        working_root,
523        locked,
524    } = applicability
525    else {
526        return NativeEvidence::new(
527            request,
528            tool,
529            process::cargo_argv(false, request.project_root.is_some()),
530            false,
531        )
532        .finish(
533            NativeStatus::NotApplicable,
534            Vec::new(),
535            "the candidate has no root Cargo.toml",
536            started,
537        );
538    };
539
540    let argv = process::cargo_argv(locked, request.project_root.is_some());
541    let mut evidence = NativeEvidence::new(request, tool, argv.clone(), locked);
542    let execution = match process::run(request, &working_root, &argv, deadline) {
543        Ok(execution) => execution,
544        Err(failure) => {
545            evidence.exit_code = failure.exit_code;
546            evidence.captured_output_bytes = failure.captured_output_bytes;
547            return evidence.finish(failure.status, Vec::new(), failure.detail, started);
548        }
549    };
550    evidence.exit_code = execution.exit_status.code();
551    evidence.captured_output_bytes = execution.captured_output_bytes();
552    evidence.stderr_excerpt = output::stderr_excerpt(
553        &execution.stderr,
554        &execution.workspace_root,
555        execution.target(),
556    );
557    if request
558        .identity
559        .verify(request.tool_program, request.cancel, deadline)
560        .is_err()
561    {
562        let (status, detail) = if request.cancel.is_cancelled() {
563            (
564                NativeStatus::Cancelled,
565                "Rust-native evaluation was cancelled",
566            )
567        } else {
568            (
569                NativeStatus::Incomplete,
570                "the Cargo or rustc launcher changed or could not be verified during native evaluation",
571            )
572        };
573        return evidence.finish(status, Vec::new(), detail, started);
574    }
575    assess_execution(&execution, request.source_files, evidence, started)
576}
577
578fn assess_execution(
579    execution: &process::Execution,
580    source_files: &BTreeMap<RepoPath, Arc<[u8]>>,
581    evidence: NativeEvidence,
582    started: Instant,
583) -> NativeCheckResult {
584    if !execution.exit_status.success() && output::unavailable(&execution.stderr, &execution.stdout)
585    {
586        let detail = detail_with_excerpt(
587            "Cargo or an offline dependency/toolchain input was unavailable",
588            evidence.stderr_excerpt.as_deref(),
589        );
590        return evidence.finish(NativeStatus::Unavailable, Vec::new(), detail, started);
591    }
592    let Ok(parsed) = output::parse(
593        &execution.stdout,
594        &execution.workspace_root,
595        execution.target(),
596        source_files,
597    ) else {
598        return evidence.finish(
599            NativeStatus::Incomplete,
600            Vec::new(),
601            "Cargo returned malformed or unsupported message-format JSON",
602            started,
603        );
604    };
605    if parsed.build_finished != Some(true) && execution.exit_status.success() {
606        return evidence.finish(
607            NativeStatus::Incomplete,
608            Vec::new(),
609            "Cargo did not report a successful completed build",
610            started,
611        );
612    }
613    if !execution.exit_status.success() && parsed.error_count == 0 {
614        let detail = detail_with_excerpt(
615            "Cargo failed without a structured compiler error",
616            evidence.stderr_excerpt.as_deref(),
617        );
618        return evidence.finish(NativeStatus::Incomplete, Vec::new(), detail, started);
619    }
620    let status = if parsed.diagnostics.is_empty() {
621        NativeStatus::Clean
622    } else {
623        NativeStatus::Findings
624    };
625    evidence.finish(status, parsed.diagnostics, "", started)
626}

630        request: &NativeCheckRequest<'_>,
631        tool: String,
632        argv: Vec<String>,
633        locked: bool,
634    ) -> Self {
635        Self {

639            runtime: request

641                .runtime
642                .as_ref()
643                .map(|runtime| runtime.kind),
644            runtime_digest: request
645                .identity
646                .runtime
647                .as_ref()
648                .map(|runtime| runtime.digest.clone()),
649            argv,
650            elapsed_ms: 0,
651            exit_code: None,
652            locked,
653            project_root: request.project_root.map(ToString::to_string),
654            captured_output_bytes: 0,
655            timeout_ms: millis(request.limits.timeout),
656            max_output_bytes: request.limits.max_output_bytes,
657            cargo_offline: true,
658            target_cache: "ephemeral",
659            isolation: "host-sandbox-required",
660            file_mode_policy: "regular-owner-read-write",
661            generated_lock_digest: None,
662            stderr_excerpt: None,
663        }
664    }
665
666    fn finish(
667        mut self,
668        status: NativeStatus,

678        NativeCheckResult {
679            status,
680            diagnostics,
681            detail: (!detail.is_empty()).then_some(detail),
682            evidence,
683            comparison_safe,
684        }
685    }

846pub(super) fn diagnostic_fingerprint(input: NativeFingerprintInput<'_>) -> String {
847    let line = input
848        .range
849        .and_then(|range| usize::try_from(range.start.line).ok())
850        .and_then(|line| {
851            input
852                .files
853                .get(input.path)?
854                .split(|byte| *byte == b'\n')
855                .nth(line)
856        })
857        .map(normalized_bytes)
858        .unwrap_or_default();
859    hex::encode(hash_domain(
860        input.domain,
861        &[input.code.as_bytes(), input.message.as_bytes(), &line],
862    ))
863}
864
865fn normalized_bytes(bytes: &[u8]) -> Vec<u8> {
866    let mut normalized = Vec::with_capacity(bytes.len());
867    let mut whitespace = false;
868    for byte in bytes {
869        if byte.is_ascii_whitespace() {
870            whitespace = !normalized.is_empty();
871        } else {
872            if whitespace {
873                normalized.push(b' ');
874                whitespace = false;
875            }
876            normalized.push(*byte);
877        }
878    }
879    normalized
880}

883    diagnostics.sort_by(|left, right| {
884        (&left.location, &left.code, &left.message, &left.fingerprint).cmp(&(
885            &right.location,
886            &right.code,
887            &right.message,
888            &right.fingerprint,
889        ))
890    });
891    diagnostics.dedup();
892}

894pub(super) fn output_detail(summary: &str, stdout: &[u8], stderr: &[u8]) -> String {
895    let stderr = String::from_utf8_lossy(stderr);
896    let stdout = String::from_utf8_lossy(stdout);
897    let lines = || stderr.lines().chain(stdout.lines());
898    let mut excerpt = lines()
899        .filter(|line| failure_line(line) && !bare_failure_marker(line))
900        .take(3)
901        .map(str::trim)
902        .collect::<Vec<_>>()
903        .join(" | ");
904    if excerpt.is_empty()
905        && let Some(line) = lines().find(|line| !line.trim().is_empty())
906    {
907        excerpt.push_str(line.trim());
908    }
909    let excerpt = (!excerpt.is_empty()).then(|| excerpt.chars().take(512).collect::<String>());
910    excerpt.map_or_else(
911        || summary.to_owned(),
912        |excerpt| format!("{summary}: {excerpt}"),
913    )
914}
915
916fn bare_failure_marker(line: &str) -> bool {
917    let marker = line.trim().strip_prefix("npm ").unwrap_or(line.trim());
918    ["error", "fatal", "failed"]
919        .iter()
920        .any(|candidate| marker.eq_ignore_ascii_case(candidate))
921}
922
923fn failure_line(line: &str) -> bool {
924    [b"error".as_slice(), b"fatal", b"failed"]
925        .iter()
926        .any(|needle| {
927            line.as_bytes()
928                .windows(needle.len())
929                .any(|window| window.eq_ignore_ascii_case(needle))
930        })
931}
932
933fn detail_with_excerpt(summary: &str, excerpt: Option<&str>) -> String {
934    excerpt.map_or_else(
935        || summary.to_owned(),
936        |excerpt| format!("{summary}: {excerpt}"),
937    )
938}

940fn digest_tool(path: &Path, cancel: &CancelToken, deadline: Instant) -> Result<String, String> {
941    require_tool_identity_active(cancel, deadline)?;
942    let before = bounded_tool_metadata(path)?;
943    let mut file = File::open(path).map_err(|error| format!("open native tool: {error}"))?;
944    let digest = read_tool_digest(&mut file, before.len(), cancel, deadline)?;
945    let after = file
946        .metadata()
947        .map_err(|error| format!("reinspect native tool: {error}"))?;
948    ensure_unchanged_tool(&before, &after)?;
949    Ok(format!("sha256:{}", hex::encode(digest.finalize())))
950}
951
952fn bounded_tool_metadata(path: &Path) -> Result<Metadata, String> {
953    let metadata = path
954        .metadata()
955        .map_err(|error| format!("inspect native tool: {error}"))?;
956    if !metadata.is_file() || metadata.len() > MAX_TOOL_BYTES {
957        return Err("the native tool is not a bounded regular file".into());
958    }
959    Ok(metadata)
960}
961
962fn read_tool_digest(
963    file: &mut File,
964    expected_bytes: u64,
965    cancel: &CancelToken,
966    deadline: Instant,
967) -> Result<Sha256, String> {
968    let mut digest = Sha256::new();
969    let mut buffer = vec![0_u8; 64 * 1024].into_boxed_slice();
970    let maximum = expected_bytes
971        .checked_add(1)
972        .ok_or_else(|| "native tool byte bound overflowed".to_owned())?;
973    let mut total = 0_u64;
974    loop {
975        require_tool_identity_active(cancel, deadline)?;
976        let remaining = maximum.saturating_sub(total);
977        if remaining == 0 {
978            return Err("the native tool grew while it was read".into());
979        }
980        let limit = usize::try_from(remaining)
981            .unwrap_or(usize::MAX)
982            .min(buffer.len());
983        let read = file
984            .read(&mut buffer[..limit])
985            .map_err(|error| format!("read native tool: {error}"))?;
986        if read == 0 {
987            break;
988        }
989        total = total
990            .checked_add(u64::try_from(read).unwrap_or(u64::MAX))
991            .ok_or_else(|| "native tool byte count overflowed".to_owned())?;
992        digest.update(&buffer[..read]);
993    }
994    if total != expected_bytes {
995        return Err("the native tool changed length while it was read".into());
996    }
997    Ok(digest)
998}
999
1000fn require_tool_identity_active(cancel: &CancelToken, deadline: Instant) -> Result<(), String> {
1001    if cancel.is_cancelled() {
1002        return Err("native tool identity capture was cancelled".into());
1003    }
1004    if Instant::now() >= deadline {
1005        return Err("native tool identity capture exceeded its wallclock grant".into());
1006    }
1007    Ok(())
1008}

1010fn ensure_unchanged_tool(before: &Metadata, after: &Metadata) -> Result<(), String> {
1011    if !same_tool_identity(before, after) {
1012        return Err("the native tool changed while it was read".into());
1013    }
1014    Ok(())
1015}

1017#[cfg(unix)]
1018fn same_tool_identity(before: &Metadata, after: &Metadata) -> bool {
1019    use std::os::unix::fs::MetadataExt as _;
1020
1021    before.dev() == after.dev()
1022        && before.ino() == after.ino()
1023        && before.len() == after.len()
1024        && before.ctime() == after.ctime()
1025        && before.ctime_nsec() == after.ctime_nsec()
1026        && before.modified().ok() == after.modified().ok()
1027}
1028
1029#[cfg(not(unix))]
1030fn same_tool_identity(before: &Metadata, after: &Metadata) -> bool {
1031    before.len() == after.len() && before.modified().ok() == after.modified().ok()
1032}

1034fn digest_environment(
1035    profile: crate::protocol::asp::ProviderProfile,
1036    environment: &[(OsString, OsString)],
1037    runtime: Option<&RuntimeIdentity>,
1038) -> String {
1039    let mut digest = Sha256::new();
1040    update_bytes_digest(&mut digest, profile.cli_name().as_bytes());
1041    for (key, value) in environment {
1042        update_os_digest(&mut digest, key);
1043        update_os_digest(&mut digest, value);
1044    }
1045    for (key, value) in fixed_environment(profile) {
1046        update_bytes_digest(&mut digest, key.as_bytes());
1047        update_bytes_digest(&mut digest, value.as_bytes());
1048    }
1049    if let Some(runtime) = runtime {
1050        update_bytes_digest(&mut digest, runtime.kind.as_bytes());
1051        update_bytes_digest(&mut digest, runtime.digest.as_bytes());
1052        update_bytes_digest(&mut digest, runtime.source.as_bytes());
1053        if let Some((_, marker_digest)) = &runtime.marker {
1054            update_bytes_digest(&mut digest, marker_digest.as_bytes());
1055        }
1056    }
1057    format!("sha256:{}", hex::encode(digest.finalize()))
1058}
1059
1060fn update_os_digest(digest: &mut Sha256, value: &OsStr) {
1061    #[cfg(unix)]
1062    {
1063        use std::os::unix::ffi::OsStrExt as _;
1064        update_bytes_digest(digest, value.as_bytes());
1065    }
1066    #[cfg(not(unix))]
1067    update_bytes_digest(digest, value.to_string_lossy().as_bytes());
1068}
1069
1070fn update_bytes_digest(digest: &mut Sha256, value: &[u8]) {
1071    digest.update(u64::try_from(value.len()).unwrap_or(u64::MAX).to_be_bytes());
1072    digest.update(value);
1073}
