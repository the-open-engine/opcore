33pub(super) fn applicable(files: &BTreeMap<RepoPath, Arc<[u8]>>) -> bool {
34    files.keys().any(|path| path.as_bytes() == b"package.json")
35        && has_npm_lock(files)
36        && !type_script_configs(files).is_empty()
37}

39struct NodeEvaluator<'request, 'input> {
40    request: &'request NativeCheckRequest<'input>,
41    started: Instant,
42    deadline: Instant,
43    configs: Vec<String>,
44    captured_output_bytes: usize,
45    compiler: Option<String>,
46    compiler_digest: Option<String>,
47    covered_files: BTreeSet<RepoPath>,
48}
49
50impl<'request, 'input> NodeEvaluator<'request, 'input> {
51    fn new(request: &'request NativeCheckRequest<'input>) -> Self {
52        let (started, deadline) = evaluation_window(request.limits.timeout);
53        Self {
54            request,
55            started,
56            deadline,
57            configs: type_script_configs(request.source_files),
58            captured_output_bytes: 0,
59            compiler: None,
60            compiler_digest: None,
61            covered_files: BTreeSet::new(),
62        }
63    }
64
65    fn run(mut self) -> NativeCheckResult {
66        if let Some(result) = self.preflight() {
67            return result;
68        }
69        let environment = native_environment(self.request.workspace_root);
70        let outcome = match self.install(&environment) {
71            Ok(()) => self.check_projects(&environment),
72            Err(result) => result,
73        };
74        if cleanup_node_modules(self.request.workspace_root, self.request.source_files).is_err() {
75            return self.finish(
76                NativeStatus::Incomplete,
77                Vec::new(),
78                "Node-native could not remove generated node_modules content",
79                outcome_exit_code(&outcome),
80                false,
81            );
82        }
83        if self
84            .request
85            .identity
86            .verify(
87                self.request.tool_program,
88                self.request.cancel,
89                self.deadline,
90            )
91            .is_err()
92        {
93            self.identity_failure(
94                "the npm executable or Node runtime changed or could not be verified during evaluation"
95                    .into(),
96            )
97        } else {
98            outcome
99        }
100    }
101
102    fn preflight(&self) -> Option<NativeCheckResult> {
103        let has_package = self
104            .request
105            .source_files
106            .keys()
107            .any(|path| path.as_bytes() == b"package.json");
108        if !has_package || self.configs.is_empty() {
109            return Some(self.finish(
110                NativeStatus::NotApplicable,
111                Vec::new(),
112                "the candidate has no root package.json and TypeScript project",
113                None,
114                true,
115            ));
116        }
117        if self.request.identity.runtime_path().is_none() {
118            return Some(self.finish(
119                NativeStatus::Unavailable,
120                Vec::new(),
121                "Node-native could not bind the PATH-selected Node runtime",
122                None,
123                false,
124            ));
125        }
126        if self.configs.len() > MAX_CONFIGS {
127            return Some(self.finish(
128                NativeStatus::Incomplete,
129                Vec::new(),
130                format!("TypeScript project count exceeds the {MAX_CONFIGS}-config limit"),
131                None,
132                false,
133            ));
134        }
135        if contains_tracked_node_modules(self.request.source_files) {
136            return Some(self.finish(
137                NativeStatus::Incomplete,
138                Vec::new(),
139                "tracked node_modules content is outside the Node-native materialization profile",
140                None,
141                false,
142            ));
143        }
144        if !has_npm_lock(self.request.source_files) {
145            return Some(self.finish(
146                NativeStatus::NotApplicable,
147                Vec::new(),
148                "Node-native requires a root npm package-lock.json or npm-shrinkwrap.json",
149                None,
150                true,
151            ));
152        }
153        self.request.cancel.is_cancelled().then(|| {
154            self.finish(
155                NativeStatus::Cancelled,
156                Vec::new(),
157                "Node-native evaluation was cancelled",
158                None,
159                false,
160            )
161        })
162    }
163
164    fn install(&mut self, environment: &[(OsString, OsString)]) -> Result<(), NativeCheckResult> {
165        let execution = process::run_tool(
166            self.request,
167            self.request.workspace_root,
168            &npm_argv(),
169            environment,
170            self.deadline,
171        )
172        .map_err(|failure| {
173            self.captured_output_bytes = failure.captured_output_bytes;
174            self.finish(
175                failure.status,
176                Vec::new(),
177                failure.detail,
178                failure.exit_code,
179                false,
180            )
181        })?;
182        self.captured_output_bytes = execution.captured_output_bytes();
183        if execution.exit_status.success() {
184            return Ok(());
185        }
186        let unavailable = npm_unavailable(&execution.stdout, &execution.stderr);
187        Err(self.finish(
188            if unavailable {
189                NativeStatus::Unavailable
190            } else {
191                NativeStatus::Incomplete
192            },
193            Vec::new(),
194            output_detail(
195                if unavailable {
196                    "npm could not satisfy the locked dependency graph from its offline cache"
197                } else {
198                    "npm ci failed before TypeScript checking"
199                },
200                &execution.stdout,
201                &execution.stderr,
202            ),
203            execution.exit_status.code(),
204            false,
205        ))
206    }
207
208    fn check_projects(&mut self, environment: &[(OsString, OsString)]) -> NativeCheckResult {
209        let (compiler, identity) = match self.compiler_identity() {
210            Ok(value) => value,
211            Err(result) => return result,
212        };
213        let diagnostics = match self.collect_projects(&compiler, &identity, environment) {
214            Ok(diagnostics) => diagnostics,
215            Err(result) => return result,
216        };
217        if identity
218            .verify_tool(&compiler, self.request.cancel, self.deadline)
219            .is_err()
220        {
221            return self.identity_failure(
222                "the TypeScript compiler changed or could not be verified during evaluation".into(),
223            );
224        }
225        self.finish_projects(diagnostics)
226    }
227
228    fn compiler_identity(
229        &mut self,
230    ) -> Result<(PathBuf, NativeExecutionIdentity), NativeCheckResult> {
231        let Some(compiler) = local_tsc(self.request.workspace_root) else {
232            return Err(self.finish(
233                NativeStatus::Unavailable,
234                Vec::new(),
235                "the locked npm installation does not provide a root project-local tsc",
236                Some(0),
237                false,
238            ));
239        };
240        let identity = match NativeExecutionIdentity::capture(
241            self.request.profile,
242            &compiler,
243            self.request.cancel,
244            self.deadline,
245        ) {
246            Ok(identity) => identity,
247            Err(detail) => return Err(self.identity_failure(detail)),
248        };
249        self.compiler = Some("node_modules/.bin/tsc".into());
250        self.compiler_digest = Some(identity.tool_digest.clone());
251        Ok((compiler, identity))
252    }
253
254    fn collect_projects(
255        &mut self,
256        compiler: &Path,
257        identity: &NativeExecutionIdentity,
258        environment: &[(OsString, OsString)],
259    ) -> Result<Vec<NativeDiagnostic>, NativeCheckResult> {
260        let mut diagnostics = Vec::new();
261        for config in self.configs.clone() {
262            let mut project = self.check_project(&config, compiler, identity, environment)?;
263            diagnostics.append(&mut project.diagnostics);
264            self.covered_files.append(&mut project.files);
265            if diagnostics.len() > MAX_DIAGNOSTICS {
266                return Err(self.finish(
267                    NativeStatus::Incomplete,
268                    Vec::new(),
269                    format!("TypeScript diagnostics exceed the {MAX_DIAGNOSTICS}-finding limit"),
270                    Some(1),
271                    false,
272                ));
273            }
274        }
275        Ok(diagnostics)
276    }
277
278    fn finish_projects(&self, mut diagnostics: Vec<NativeDiagnostic>) -> NativeCheckResult {
279        sort_diagnostics(&mut diagnostics);
280        let clean = diagnostics.is_empty();
281        self.finish(
282            if clean {
283                NativeStatus::Clean
284            } else {
285                NativeStatus::Findings
286            },
287            diagnostics,
288            "",
289            Some(i32::from(!clean)),
290            true,
291        )
292    }

294    fn check_project(
295        &mut self,
296        config: &str,
297        compiler: &Path,
298        identity: &NativeExecutionIdentity,
299        environment: &[(OsString, OsString)],
300    ) -> Result<ProjectCheck, NativeCheckResult> {
301        let argv = tsc_argv(config);
302        let compiler_request = NativeCheckRequest {
303            profile: self.request.profile,
304            workspace_root: self.request.workspace_root,
305            project_root: self.request.project_root,
306            source_files: self.request.source_files,
307            tool_program: compiler,
308            identity,
309            cancel: self.request.cancel,
310            limits: self.request.limits,
311        };
312        let execution = process::run_tool(
313            &compiler_request,
314            self.request.workspace_root,
315            &argv,
316            environment,
317            self.deadline,
318        )
319        .map_err(|failure| {
320            self.captured_output_bytes = self
321                .captured_output_bytes
322                .saturating_add(failure.captured_output_bytes);
323            self.finish(
324                failure.status,
325                Vec::new(),
326                failure.detail,
327                failure.exit_code,
328                false,
329            )
330        })?;
331        self.captured_output_bytes = self
332            .captured_output_bytes
333            .saturating_add(execution.captured_output_bytes());
334        if self.captured_output_bytes > self.request.limits.max_output_bytes {
335            return Err(self.finish(
336                NativeStatus::Incomplete,
337                Vec::new(),
338                "Node-native cumulative output exceeded the configured byte limit",
339                execution.exit_status.code(),
340                false,
341            ));
342        }
343        let diagnostics = parse_tsc_output(
344            &execution.stdout,
345            &execution.stderr,
346            self.request.workspace_root,
347            self.request.source_files,
348            config,
349        );
350        if execution.exit_status.success()
351            || (execution.exit_status.code().is_some() && !diagnostics.is_empty())
352        {
353            Ok(ProjectCheck {
354                diagnostics,
355                files: listed_project_files(
356                    &execution.stdout,
357                    self.request.workspace_root,
358                    self.request.source_files,
359                ),
360            })
361        } else {
362            Err(self.finish(
363                NativeStatus::Incomplete,
364                Vec::new(),
365                output_detail(
366                    "tsc failed without a supported diagnostic",
367                    &execution.stdout,
368                    &execution.stderr,
369                ),
370                execution.exit_status.code(),
371                false,
372            ))
373        }
374    }
375
376    fn identity_failure(&self, detail: String) -> NativeCheckResult {
377        self.finish(
378            if self.request.cancel.is_cancelled() {
379                NativeStatus::Cancelled
380            } else {
381                NativeStatus::Incomplete
382            },
383            Vec::new(),
384            detail,
385            None,
386            false,
387        )
388    }
389
390    fn finish(
391        &self,
392        status: NativeStatus,
393        diagnostics: Vec<NativeDiagnostic>,
394        detail: impl Into<String>,
395        exit_code: Option<i32>,
396        comparison_safe: bool,
397    ) -> NativeCheckResult {
398        let install_argv = npm_argv();
399        let mut evidence = common_evidence(
400            self.request,
401            self.started,
402            self.captured_output_bytes,
403            exit_code,
404        );
405        evidence.insert("installArgv".into(), json!(install_argv));
406        evidence.insert("compiler".into(), json!(self.compiler));
407        evidence.insert("compilerDigest".into(), json!(self.compiler_digest));
408        evidence.insert("configs".into(), json!(self.configs));
409        evidence.insert("coveredSourceFiles".into(), json!(self.covered_files.len()));
410        evidence.insert(
411            "unconfiguredTypeScriptSourceFiles".into(),
412            json!(
413                type_script_source_files(self.request.source_files)
414                    .difference(&self.covered_files)
415                    .count()
416            ),
417        );
418        evidence.insert(
419            "unconfiguredJavaScriptSourceFiles".into(),
420            json!(
421                node_source_files(self.request.source_files)
422                    .difference(&self.covered_files)
423                    .filter(|path| !is_type_script(path))
424                    .count()
425            ),
426        );
427        evidence.insert(
428            "selectedSourceFiles".into(),
429            json!(node_source_files(self.request.source_files).len()),
430        );
431        evidence.insert(
432            "locked".into(),
433            json!(has_npm_lock(self.request.source_files)),
434        );
435        evidence.insert("npmOffline".into(), json!(true));
436        evidence.insert("lifecycleScripts".into(), json!(false));
437        evidence.insert("dependencyTree".into(), json!("ephemeral"));
438        complete_result(
439            status,
440            diagnostics,
441            detail.into(),
442            evidence,
443            comparison_safe,
444        )
445    }
446}

501fn type_script_configs(files: &BTreeMap<RepoPath, Arc<[u8]>>) -> Vec<String> {
502    files
503        .keys()
504        .filter_map(|path| {
505            let value = path.as_utf8()?;
506            value.rsplit('/').next().and_then(|name| {
507                (name == "tsconfig.json"
508                    || (name.starts_with("tsconfig.")
509                        && Path::new(name)
510                            .extension()
511                            .is_some_and(|extension| extension.eq_ignore_ascii_case("json"))))
512                .then(|| value.to_owned())
513            })
514        })
515        .collect()
516}

523fn node_source_files(files: &BTreeMap<RepoPath, Arc<[u8]>>) -> BTreeSet<RepoPath> {
524    files
525        .keys()
526        .filter(|path| {
527            let bytes = path.as_bytes();
528            [
529                b".js".as_slice(),
530                b".jsx",
531                b".mjs",
532                b".cjs",
533                b".ts",
534                b".tsx",
535                b".mts",
536                b".cts",
537            ]
538            .iter()
539            .any(|suffix| has_extension(bytes, suffix))
540        })
541        .cloned()
542        .collect()
543}

597fn cleanup_node_modules(
598    workspace_root: &Path,
599    files: &BTreeMap<RepoPath, Arc<[u8]>>,
600) -> std::io::Result<()> {
601    let paths = files
602        .keys()
603        .filter(|path| path.as_bytes().ends_with(b"package.json"))
604        .filter_map(|path| path.to_path_buf().parent().map(Path::to_owned))
605        .map(|parent| workspace_root.join(parent).join("node_modules"))
606        .chain(std::iter::once(workspace_root.join("node_modules")))
607        .collect::<BTreeSet<_>>();
608    let mut paths = paths.into_iter().collect::<Vec<_>>();
609    paths.sort_by_key(|path| std::cmp::Reverse(path.components().count()));
610    for path in paths {
611        match fs::remove_dir_all(&path) {
612            Ok(()) => {}
613            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
614            Err(error) => return Err(error),
615        }
616    }
617    Ok(())
618}

620fn npm_unavailable(stdout: &[u8], stderr: &[u8]) -> bool {
621    let output = format!(
622        "{}\n{}",
623        String::from_utf8_lossy(stdout),
624        String::from_utf8_lossy(stderr)
625    )
626    .to_ascii_lowercase();
627    [
628        "enotcached",
629        "cache mode is 'only-if-cached'",
630        "could not resolve dependency",
631        "no cached response available",
632    ]
633    .iter()
634    .any(|needle| output.contains(needle))
635}

637fn parse_tsc_output(
638    stdout: &[u8],
639    stderr: &[u8],
640    workspace_root: &Path,
641    source_files: &BTreeMap<RepoPath, Arc<[u8]>>,
642    config: &str,
643) -> Vec<NativeDiagnostic> {
644    let stdout = String::from_utf8_lossy(stdout);
645    let stderr = String::from_utf8_lossy(stderr);
646    stdout
647        .lines()
648        .chain(stderr.lines())
649        .filter_map(|line| parse_tsc_line(line, workspace_root, source_files, config))
650        .collect()
651}
652
653fn parse_tsc_line(
654    line: &str,
655    workspace_root: &Path,
656    source_files: &BTreeMap<RepoPath, Arc<[u8]>>,
657    config: &str,
658) -> Option<NativeDiagnostic> {
659    let line = line.trim();
660    let (raw_path, range, diagnostic) =
661        if let Some((prefix, diagnostic)) = line.split_once(": error TS") {
662            let (raw_path, range) = tsc_location(prefix);
663            (raw_path, range, diagnostic)
664        } else {
665            (config, None, line.strip_prefix("error TS")?)
666        };
667    let (raw_code, message) = diagnostic.split_once(':')?;
668    if raw_code.is_empty()
669        || !raw_code.bytes().all(|byte| byte.is_ascii_digit())
670        || message.len() > MAX_MESSAGE_BYTES
671    {
672        return None;
673    }
674    let fallback = RepoPath::from_protocol(config).ok()?;
675    let path = resolve_diagnostic_path(raw_path, workspace_root, source_files).unwrap_or(fallback);
676    let message = message.trim().to_owned();
677    let code = format!("{NODE_NATIVE_RULE}/TS{raw_code}");
678    let fingerprint = diagnostic_fingerprint(NativeFingerprintInput {
679        domain: "node-native-diagnostic/v1",
680        code: &code,
681        message: &message,
682        path: &path,
683        range,
684        files: source_files,
685    });
686    Some(NativeDiagnostic {
687        code,
688        severity: NativeSeverity::Error,
689        source: NODE_NATIVE_PROVIDER_ID,
690        message,
691        help: None,
692        location: NativeLocation {
693            path: path.to_string(),
694            range,
695        },
696        fingerprint,
697    })
698}

700fn tsc_location(prefix: &str) -> (&str, Option<NativeRange>) {
701    let Some(open) = prefix.rfind('(') else {
702        return (prefix.trim(), None);
703    };
704    let Some(coordinates) = prefix
705        .get(open + 1..)
706        .and_then(|value| value.strip_suffix(')'))
707    else {
708        return (prefix.trim(), None);
709    };
710    let Some((line, character)) = coordinates.split_once(',') else {
711        return (prefix.trim(), None);
712    };
713    let Some(line) = line
714        .parse::<u32>()
715        .ok()
716        .and_then(|value| value.checked_sub(1))
717    else {
718        return (prefix.trim(), None);
719    };
720    let Some(character) = character
721        .parse::<u32>()
722        .ok()
723        .and_then(|value| value.checked_sub(1))
724    else {
725        return (prefix.trim(), None);
726    };
727    let position = NativePosition {
728        line,
729        char: character,
730    };
731    (
732        prefix.get(..open).unwrap_or(prefix).trim(),
733        Some(NativeRange {
734            start: position,
735            end: position,
736        }),
737    )
738}

740fn outcome_exit_code(result: &NativeCheckResult) -> Option<i32> {
741    result
742        .evidence
743        .get("exitCode")
744        .and_then(serde_json::Value::as_i64)
745        .and_then(|code| i32::try_from(code).ok())
746}

773    #[test]
774    fn parses_plain_tsc_diagnostics_with_zero_based_locations() {
775        let path = RepoPath::from_protocol("src/index.ts").expect("fixture path");
776        let files = BTreeMap::from([(
777            path,
778            Arc::<[u8]>::from(b"const value: string = 1;\n".as_slice()),
779        )]);
780        let diagnostic = parse_tsc_line(
781            "src/index.ts(1,7): error TS2322: Type 'number' is not assignable to type 'string'.",
782            Path::new("/tmp/workspace"),
783            &files,
784            "tsconfig.json",
785        )
786        .expect("diagnostic");
787        assert_eq!(diagnostic.location.path, "src/index.ts");
788        assert_eq!(
789            diagnostic.location.range.map(|range| range.start),
790            Some(NativePosition { line: 0, char: 6 })
791        );
792        assert!(diagnostic.code.ends_with("/TS2322"));
793
794        let config = RepoPath::from_protocol("tsconfig.json").expect("config path");
795        let files = BTreeMap::from([(config, Arc::<[u8]>::from(b"{}\n".as_slice()))]);
796        let diagnostic = parse_tsc_line(
797            "error TS18003: No inputs were found in config file.",
798            Path::new("/tmp/workspace"),
799            &files,
800            "tsconfig.json",
801        )
802        .expect("global diagnostic");
803        assert_eq!(diagnostic.location.path, "tsconfig.json");
804        assert!(diagnostic.location.range.is_none());
805    }

809        let result = run_node(
810            PROJECT,
811            r#"#!/bin/sh
812mkdir -p node_modules/.bin
813cat > node_modules/.bin/tsc <<'EOF'
814#!/bin/sh
815echo "src/index.ts(1,30): error TS2322: simulated type failure"
816echo "error TS18003: simulated project failure"
817echo "$PWD/src/index.ts"
818exit 1
819EOF
820chmod 700 node_modules/.bin/tsc
821exit 0
822"#,
823            false,
824        );
825        assert_eq!(result.status, NativeStatus::Findings);
826        assert_eq!(result.diagnostics.len(), 2);
827        assert!(
828            result
829                .diagnostics
830                .iter()
831                .any(|diagnostic| diagnostic.location.path == "src/index.ts")
832        );
833        assert_eq!(result.evidence["dependencyTree"], "ephemeral");
834        assert_eq!(result.evidence["compiler"], "node_modules/.bin/tsc");
835
836        let clean = run_node(
837            PROJECT,
838            r#"#!/bin/sh
839mkdir -p node_modules/.bin
840cat > node_modules/.bin/tsc <<'EOF'
841#!/bin/sh
842echo "$PWD/src/index.ts"
843exit 0
844EOF
845chmod 700 node_modules/.bin/tsc
846exit 0
847"#,
848            false,
849        );
850        assert_eq!(clean.status, NativeStatus::Clean);
851        assert!(clean.diagnostics.is_empty());
852        assert!(clean.comparison_safe);
853    }

855    #[test]
856    fn node_evaluator_preserves_install_and_preflight_failures() {
857        let unavailable = run_node(
858            PROJECT,
859            "#!/bin/sh\necho 'npm error code ENOTCACHED' >&2\nexit 1\n",
860            false,
861        );
862        assert_eq!(unavailable.status, NativeStatus::Unavailable);
863        assert!(
864            unavailable
865                .detail
866                .as_deref()
867                .is_some_and(|detail| detail.contains("offline cache"))
868        );
869        assert!(!unavailable.comparison_safe);
870
871        let no_lock = run_node(
872            &[
873                ("package.json", "{}\n"),
874                ("tsconfig.json", "{}\n"),
875                ("src/index.ts", "export {};\n"),
876            ],
877            "#!/bin/sh\nexit 0\n",
878            false,
879        );
880        assert_eq!(no_lock.status, NativeStatus::NotApplicable);
881        assert!(no_lock.comparison_safe);
882
883        let tracked_modules = run_node(
884            &[
885                ("package.json", "{}\n"),
886                ("package-lock.json", "{}\n"),
887                ("tsconfig.json", "{}\n"),
888                ("node_modules/tracked.js", "export {};\n"),
889            ],
890            "#!/bin/sh\nexit 0\n",
891            false,
892        );
893        assert_eq!(tracked_modules.status, NativeStatus::Incomplete);
894
895        let cancelled = run_node(PROJECT, "#!/bin/sh\nexit 0\n", true);
896        assert_eq!(cancelled.status, NativeStatus::Cancelled);
897    }
