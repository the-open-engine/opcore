20pub(crate) fn inspect_native_projects(
21    repository: &GitRepository,
22    policy: &PolicySnapshot,
23) -> Vec<(NativeProvider, Result<String>)> {
24    policy
25        .policy
26        .native
27        .iter()
28        .map(|provider| {
29            let profile = match provider {
30                NativeProvider::RustNative => ProviderProfile::RustNative,
31                NativeProvider::NodeNative => ProviderProfile::NodeNative,
32                NativeProvider::PythonNative => ProviderProfile::PythonNative,
33            };
34            (*provider, inspect_project(repository, policy, profile))
35        })
36        .collect()
37}

39fn inspect_project(
40    repository: &GitRepository,
41    policy: &PolicySnapshot,
42    profile: ProviderProfile,
43) -> Result<String> {
44    let initial = HostSelection {
45        profiles: vec![profile],
46        roots: selected_roots(&[])?,
47        excluded_roots: Vec::new(),
48        skip_inapplicable: false,
49        comparison: CheckComparison::All,
50    };
51    let selected = initial.for_profile(profile, &CheckArgs::default(), policy)?;
52    ensure!(
53        !selected.roots.is_empty(),
54        "all configured project roots are excluded"
55    );
56    let plan = native_capture_plan(repository, &policy.view, &selected, &policy.policy.targets)?;
57    let paths = repository.native_candidate_paths_from(&policy.view, &plan.roots, plan.filter)?;
58    ensure!(
59        paths.len() <= MAX_FILES,
60        "native project exceeds {MAX_FILES} captured input paths"
61    );
62    let mut controls = BTreeSet::new();
63    let mut interpreters = Vec::new();
64    for root in &selected.roots {
65        controls.extend(required_controls(&paths, root, profile)?);
66        if profile == ProviderProfile::PythonNative {
67            let context = project_context_from_paths(&paths, root, profile);
68            interpreters.push(inspect_python_interpreter(repository.root(), &context)?);
69        }
70    }
71    inspect_controls(repository, &policy.view, &controls)?;
72    ensure!(
73        policy.is_current()?,
74        "configuration changed during project inspection; rerun doctor"
75    );
76    Ok(project_summary(&selected.roots, &interpreters))
77}

79fn project_summary(roots: &[WorkspaceRoot], interpreters: &[String]) -> String {
80    let roots = roots
81        .iter()
82        .map(|root| root.label.as_str())
83        .collect::<Vec<_>>()
84        .join(", ");
85    let environment = if interpreters.is_empty() {
86        String::new()
87    } else {
88        format!("; Python interpreters: {}", interpreters.join(", "))
89    };
90    format!(
91        "project inputs present for {roots}{environment}; setup inspection only, run the workflow to verify dependencies and checker configuration"
92    )
93}

95fn inspect_python_interpreter(repository_root: &Path, context: &WorkspaceRoot) -> Result<String> {
96    let selected = super::project_python_override(repository_root, context.prefix.as_ref());
97    let selected = match selected {
98        Some(selected) => Some(selected),
99        None => crate::protocol::native::python_runtime_selection().map_err(anyhow::Error::msg)?,
100    };
101    let (path, source) = selected.with_context(|| {
102        format!(
103            "{} requires a Python interpreter; select OPCORE_PYTHON, a project .venv, or Python on PATH",
104            context.label
105        )
106    })?;
107    let metadata = path
108        .metadata()
109        .with_context(|| format!("inspect Python interpreter {}", path.display()))?;
110    ensure!(
111        path.is_absolute() && metadata.is_file(),
112        "selected Python interpreter must be an absolute regular file: {}",
113        path.display()
114    );
115    #[cfg(unix)]
116    {
117        use std::os::unix::fs::PermissionsExt as _;
118        ensure!(
119            metadata.permissions().mode() & 0o111 != 0,
120            "selected Python interpreter is not executable: {}",
121            path.display()
122        );
123    }
124    Ok(format!("{}={} ({source})", context.label, path.display()))
125}

127fn required_controls(
128    paths: &BTreeSet<RepoPath>,
129    root: &WorkspaceRoot,
130    profile: ProviderProfile,
131) -> Result<BTreeSet<RepoPath>> {
132    let context = project_context_from_paths(paths, root, profile);
133    let controls = match profile {
134        ProviderProfile::RustNative => {
135            ensure!(
136                has_context_file(paths, root, b"Cargo.toml"),
137                "{} requires Cargo.toml",
138                root.label
139            );
140            paths
141                .iter()
142                .filter(|path| context_filename(path, root) == Some(b"Cargo.toml"))
143                .cloned()
144                .collect()
145        }
146        ProviderProfile::NodeNative => node_controls(paths, &context)?,
147        ProviderProfile::PythonNative => {
148            ensure!(
149                paths.iter().any(|path| {
150                    is_python(path) && super::selection::path_in_workspace_root(path, &context)
151                }),
152                "{} contains no Python source",
153                root.label
154            );
155            paths
156                .iter()
157                .filter(|path| {
158                    context_filename(path, &context)
159                        .is_some_and(|name| super::selection::provider_context_file(profile, name))
160                })
161                .cloned()
162                .collect()
163        }
164        ProviderProfile::Fast => anyhow::bail!("Fast has no native project prerequisites"),
165    };
166    Ok(controls)
167}

169fn node_controls(
170    paths: &BTreeSet<RepoPath>,
171    context: &WorkspaceRoot,
172) -> Result<BTreeSet<RepoPath>> {
173    for (present, required) in [
174        (
175            has_context_file(paths, context, b"package.json"),
176            "package.json",
177        ),
178        (
179            has_context_file(paths, context, b"package-lock.json")
180                || has_context_file(paths, context, b"npm-shrinkwrap.json"),
181            "an npm lockfile",
182        ),
183        (
184            paths
185                .iter()
186                .any(|path| context_filename(path, context).is_some_and(is_tsconfig_name)),
187            "tsconfig.json or tsconfig.*.json",
188        ),
189    ] {
190        ensure!(
191            present,
192            "{} requires {required} for node-native",
193            context.label
194        );
195    }
196    Ok(paths
197        .iter()
198        .filter(|path| {
199            context_filename(path, context).is_some_and(|name| {
200                super::selection::provider_context_file(ProviderProfile::NodeNative, name)
201            })
202        })
203        .cloned()
204        .collect())
205}

207fn inspect_controls(
208    repository: &GitRepository,
209    view: &ConfigView,
210    paths: &BTreeSet<RepoPath>,
211) -> Result<()> {
212    if paths.is_empty() {
213        return Ok(());
214    }
215    let capture = match view {
216        ConfigView::Worktree => repository.capture_auxiliary(paths)?,
217        ConfigView::Index => repository.capture_staged_auxiliary(paths)?,
218        ConfigView::Tree(reference) => repository.capture_tree_auxiliary(paths, reference, None)?,
219    };
220    for path in paths {
221        capture
222            .paths
223            .get(path)
224            .and_then(|state| state.after.as_ref())
225            .with_context(|| {
226                format!("required project control file {path} is missing or is not a regular file")
227            })?;
228    }
229    ensure!(
230        repository.is_auxiliary_current(&capture)?,
231        "project controls changed during inspection; rerun doctor"
232    );
233    Ok(())
234}
