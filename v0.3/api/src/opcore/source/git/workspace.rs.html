52impl GitRepository {
53    pub fn capture_workspace(&self) -> Result<GitWorkspaceCapture, SourceError> {
54        let mut last_error = None;
55        for _ in 0..CAPTURE_ATTEMPTS {
56            let before_sentinel = self.workspace_sentinel(None)?;
57            match self.capture_workspace_once(&before_sentinel) {
58                Ok(capture) => {
59                    let after_sentinel = self.workspace_sentinel(Some(&before_sentinel))?;
60                    if before_sentinel == after_sentinel {
61                        return Ok(capture);
62                    }
63                    last_error = Some(
64                        "HEAD, index, paths, or working tree changed while files were read"
65                            .to_owned(),
66                    );
67                }
68                Err(error @ SourceError::Changed(_)) => last_error = Some(error.to_string()),
69                Err(error) => return Err(error),
70            }
71        }
72        Err(SourceError::Changed(last_error.unwrap_or_else(|| {
73            "workspace capture did not stabilize".into()
74        })))
75    }
76
77    pub fn is_workspace_current(&self, capture: &GitWorkspaceCapture) -> Result<bool, SourceError> {
78        match &capture.sentinel {
79            SenseSentinel::Worktree(sentinel) => {
80                Ok(*sentinel == self.workspace_sentinel(Some(sentinel))?)
81            }
82            SenseSentinel::Staged(sentinel) => Ok(*sentinel == self.staged_workspace_sentinel()?),
83            SenseSentinel::Tree {
84                sentinel,
85                tree_ref,
86                base_ref,
87            } => Ok(*sentinel
88                == self.sentinel_with(
89                    GitScope::Tree,
90                    &[],
91                    is_dependency_capture_candidate,
92                    base_ref.as_deref(),
93                    Some(tree_ref),
94                )?),
95        }
96    }
97
98    pub fn capture_staged_workspace(&self) -> Result<GitWorkspaceCapture, SourceError> {
99        let mut last_error = None;
100        for _ in 0..CAPTURE_ATTEMPTS {
101            let before_sentinel = self.staged_workspace_sentinel()?;
102            match self.capture_staged_workspace_once(&before_sentinel) {
103                Ok(capture) => {
104                    if before_sentinel == self.staged_workspace_sentinel()? {
105                        return Ok(capture);
106                    }
107                    last_error =
108                        Some("HEAD or worktree-local index changed while files were read".into());
109                }
110                Err(error @ SourceError::Changed(_)) => last_error = Some(error.to_string()),
111                Err(error) => return Err(error),
112            }
113        }
114        Err(SourceError::Changed(last_error.unwrap_or_else(|| {
115            "staged workspace capture did not stabilize".into()
116        })))
117    }

119    /// Captures the complete selected dependency views from immutable Git trees.
120    pub fn capture_tree_workspace(
121        &self,
122        tree_ref: &str,
123        base_ref: Option<&str>,
124    ) -> Result<GitWorkspaceCapture, SourceError> {
125        super::stable_capture(
126            || {
127                self.sentinel_with(
128                    GitScope::Tree,
129                    &[],
130                    is_dependency_capture_candidate,
131                    base_ref,
132                    Some(tree_ref),
133                )
134            },
135            |sentinel| {
136                let mut comparison = sentinel.clone();
137                if base_ref.is_none() {
138                    comparison.base_tree = comparison
139                        .target_tree
140                        .clone()
141                        .ok_or_else(|| SourceError::Invalid("missing target tree".into()))?;
142                }
143                let mut capture = self.capture_staged_workspace_once(&comparison)?;
144                capture.sentinel = SenseSentinel::Tree {
145                    sentinel: sentinel.clone(),
146                    tree_ref: tree_ref.to_owned(),
147                    base_ref: base_ref.map(str::to_owned),
148                };
149                Ok(capture)
150            },
151            |_| {
152                self.sentinel_with(
153                    GitScope::Tree,
154                    &[],
155                    is_dependency_capture_candidate,
156                    base_ref,
157                    Some(tree_ref),
158                )
159            },
160            "Git refs changed while dependency files were captured",
161            "dependency tree capture did not stabilize",
162        )
163    }

165    fn capture_workspace_once(
166        &self,
167        sentinel: &WorkspaceSentinel,
168    ) -> Result<GitWorkspaceCapture, SourceError> {
169        reject_unmerged(&self.root)?;
170        let mut entries = super::objects::all_tree_entries_filtered(
171            &self.root,
172            &sentinel.git.base_tree,
173            &|path| self.dependency_candidate(path),
174        )?;
175        entries.retain(|path, _| is_dependency_capture_candidate(path));
176        let before_paths = entries.keys().cloned().collect::<BTreeSet<_>>();
177        let paths = before_paths
178            .union(&sentinel.paths)
179            .cloned()
180            .collect::<BTreeSet<_>>();
181        require_path_bound(&paths)?;
182
183        let after_paths = workspace_after_paths(&self.root, &before_paths, &sentinel.paths)?;
184        let before_unsupported_files = unsupported_count(&before_paths);
185        let after_unsupported_files = unsupported_count(&after_paths);
186        let unsupported_changed_files = sentinel
187            .git
188            .worktree_paths
189            .iter()
190            .filter(|path| is_dependency_source_candidate(path) && detect_language(path).is_none())
191            .count();
192        entries.retain(|path, _| detect_language(path).is_some());
193        let before_files = files_from_git_objects(&self.root, &entries)?;
194        let overlay_files = self.read_worktree_files(&sentinel.paths)?;
195        let (before_files, after_files) =
196            apply_workspace_overlay(before_files, &sentinel.paths, overlay_files);
197        let changed_paths = changed_source_paths(&before_files, &after_files);
198        Ok(build_workspace_capture(
199            WorkspaceCaptureParts {
200                paths,
201                changed_paths,
202                before_unsupported_files,
203                after_unsupported_files,
204                unsupported_changed_files,
205                before_files,
206                after_files,
207            },
208            sentinel,
209        ))
210    }

212    fn capture_staged_workspace_once(
213        &self,
214        sentinel: &Sentinel,
215    ) -> Result<GitWorkspaceCapture, SourceError> {
216        if sentinel.target_tree.is_none() {
217            reject_unmerged(&self.root)?;
218        }
219        let mut before_entries =
220            super::objects::all_tree_entries_filtered(&self.root, &sentinel.base_tree, &|path| {
221                self.dependency_candidate(path)
222            })?;
223        let mut after_entries = if let Some(target) = &sentinel.target_tree {
224            super::objects::all_tree_entries_filtered(&self.root, target, &|path| {
225                self.dependency_candidate(path)
226            })?
227        } else {
228            super::objects::all_index_entries_filtered(&self.root, &|path| {
229                self.dependency_candidate(path)
230            })?
231        };
232        before_entries.retain(|path, _| is_dependency_capture_candidate(path));
233        after_entries.retain(|path, _| is_dependency_capture_candidate(path));
234        let before_paths = before_entries.keys().cloned().collect::<BTreeSet<_>>();
235        let after_paths = after_entries.keys().cloned().collect::<BTreeSet<_>>();
236        let paths = before_paths.union(&after_paths).cloned().collect();
237        require_path_bound(&paths)?;
238
239        let before_unsupported_files = unsupported_count(&before_paths);
240        let after_unsupported_files = unsupported_count(&after_paths);
241        let unsupported_changed_files = paths
242            .iter()
243            .filter(|path| {
244                is_dependency_source_candidate(path)
245                    && detect_language(path).is_none()
246                    && before_entries.get(*path) != after_entries.get(*path)
247            })
248            .count();
249        let after_source_paths = after_entries
250            .keys()
251            .filter(|path| detect_language(path).is_some())
252            .cloned()
253            .collect::<BTreeSet<_>>();
254        let changed_after_entries = after_entries
255            .iter()
256            .filter(|(path, entry)| {
257                detect_language(path).is_some() && before_entries.get(*path) != Some(*entry)
258            })
259            .map(|(path, entry)| (path.clone(), entry.clone()))
260            .collect();
261        before_entries.retain(|path, _| detect_language(path).is_some());
262        let before_files = files_from_git_objects(&self.root, &before_entries)?;
263        let changed_after_files = files_from_git_objects(&self.root, &changed_after_entries)?;
264        let after_files =
265            staged_after_files(&before_files, &after_source_paths, changed_after_files);
266        let changed_paths = changed_source_paths(&before_files, &after_files);
267        Ok(build_staged_capture(
268            WorkspaceCaptureParts {
269                paths,
270                changed_paths,
271                before_unsupported_files,
272                after_unsupported_files,
273                unsupported_changed_files,
274                before_files,
275                after_files,
276            },
277            sentinel,
278        ))
279    }

282        self.sentinel_with(
283            GitScope::Staged,
284            &[],
285            is_dependency_capture_candidate,
286            None,
287            None,

291    fn workspace_sentinel(
292        &self,
293        prior: Option<&WorkspaceSentinel>,
294    ) -> Result<WorkspaceSentinel, SourceError> {
295        let git = self.dependency_sentinel()?;
296        let special_paths = match prior {
297            Some(prior) if prior.git.index_hash == git.index_hash => prior.special_paths.clone(),
298            _ => special_index_paths(&self.root, &|path| self.dependency_candidate(path))?,
299        };
300        let mut paths = git
301            .worktree_paths
302            .iter()
303            .filter(|path| self.dependency_candidate(path))
304            .cloned()
305            .collect::<BTreeSet<_>>();
306        paths.extend(
307            special_paths
308                .iter()
309                .filter(|path| self.dependency_candidate(path))
310                .cloned(),
311        );
312        let generations = workspace_generation_state(&self.root, &paths)?;
313        Ok(WorkspaceSentinel {
314            git,
315            paths,
316            special_paths,
317            generations,
318        })
319    }

323    pub fn apply_hypothetical(
324        mut self,
325        overlays: &BTreeMap<RepoPath, Option<Vec<u8>>>,
326        request_digest: &str,
327    ) -> Result<Self, SourceError> {
328        let before = Arc::clone(&self.after);
329        let mut files = before
330            .files()
331            .cloned()
332            .map(|file| (file.path.clone(), file))
333            .collect::<BTreeMap<_, _>>();
334        let mut after_paths = self.paths.clone();
335        for (path, content) in overlays {
336            if !is_dependency_capture_candidate(path) {
337                continue;
338            }
339            if let Some(content) = content {
340                after_paths.insert(path.clone());
341                if detect_language(path).is_some() {
342                    if content.len() > MAX_SOURCE_BYTES {
343                        return Err(SourceError::Incomplete(format!(
344                            "{path} exceeds {MAX_SOURCE_BYTES} bytes"
345                        )));
346                    }
347                    if let Some(file) = source_file(path.clone(), content.clone()) {
348                        files.insert(path.clone(), file);
349                    }
350                }
351            } else {
352                after_paths.remove(path);
353                files.remove(path);
354            }
355        }
356        require_path_bound(&after_paths)?;
357        let total_bytes = files
358            .values()
359            .fold(0usize, |total, file| total.saturating_add(file.bytes.len()));
360        if total_bytes > MAX_TOTAL_SOURCE_BYTES {
361            return Err(SourceError::Incomplete(format!(
362                "hypothetical source view exceeds {MAX_TOTAL_SOURCE_BYTES} total bytes"
363            )));
364        }
365        let after = Arc::new(SourceSnapshot::new(files.into_values()));
366        let changed_paths = changed_source_paths(before.files(), after.files());
367        let unsupported_changed_files = overlays
368            .keys()
369            .filter(|path| is_dependency_source_candidate(path) && detect_language(path).is_none())
370            .count();
371        self.before = before;
372        self.after = after;
373        self.paths = self.paths.union(&after_paths).cloned().collect();
374        self.changed_paths = changed_paths;
375        self.before_unsupported_files = self.after_unsupported_files;
376        self.after_unsupported_files = unsupported_count(&after_paths);
377        self.unsupported_changed_files = unsupported_changed_files;
378        self.valid_as_of = serde_json::json!({
379            "kind": "git-hypothetical",
380            "base": self.valid_as_of,
381            "request": request_digest,
382            "beforeView": self.before.id(),
383            "afterView": self.after.id(),
384        })
385        .to_string();
386        Ok(self)
387    }
388}

397fn workspace_after_paths(
398    root: &Path,
399    before: &BTreeSet<RepoPath>,
400    overlay: &BTreeSet<RepoPath>,
401) -> Result<BTreeSet<RepoPath>, SourceError> {
402    let mut after = before.clone();
403    let mut missing = BTreeSet::new();
404    for path in overlay {
405        after.remove(path);
406        match fs::symlink_metadata(root.join(path.to_path_buf())) {
407            Ok(_) => {
408                after.insert(path.clone());
409            }
410            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
411                missing.insert(path.clone());
412            }
413            Err(error) => return Err(SourceError::Io(error.to_string())),
414        }
415    }
416    after.extend(sparse_paths(root, &missing)?);
417    Ok(after)
418}
419
420fn require_path_bound(paths: &BTreeSet<RepoPath>) -> Result<(), SourceError> {
421    if paths.len() > MAX_FILES {
422        return Err(SourceError::Incomplete(format!(
423            "{} paths exceed limit {MAX_FILES}",
424            paths.len()
425        )));
426    }
427    Ok(())
428}

430fn apply_workspace_overlay(
431    before_files: Vec<SourceFile>,
432    overlay_paths: &BTreeSet<RepoPath>,
433    overlay_files: Vec<SourceFile>,
434) -> (Vec<SourceFile>, Vec<SourceFile>) {
435    let mut after_by_path = before_files
436        .iter()
437        .cloned()
438        .map(|file| (file.path.clone(), file))
439        .collect::<BTreeMap<_, _>>();
440    for path in overlay_paths {
441        after_by_path.remove(path);
442    }
443    for file in overlay_files {
444        after_by_path.insert(file.path.clone(), file);
445    }
446    (before_files, after_by_path.into_values().collect())
447}

449fn changed_source_paths<'a>(
450    before_files: impl IntoIterator<Item = &'a SourceFile>,
451    after_files: impl IntoIterator<Item = &'a SourceFile>,
452) -> BTreeSet<RepoPath> {
453    let mut changed = BTreeSet::new();
454    let mut before = before_files.into_iter().peekable();
455    let mut after = after_files.into_iter().peekable();
456    while let (Some(before_file), Some(after_file)) = (before.peek(), after.peek()) {
457        match before_file.path.cmp(&after_file.path) {
458            std::cmp::Ordering::Less => {
459                if let Some(file) = before.next() {
460                    changed.insert(file.path.clone());
461                }
462            }
463            std::cmp::Ordering::Greater => {
464                if let Some(file) = after.next() {
465                    changed.insert(file.path.clone());
466                }
467            }
468            std::cmp::Ordering::Equal => {
469                if before_file.content_id != after_file.content_id {
470                    changed.insert(before_file.path.clone());
471                }
472                before.next();
473                after.next();
474            }
475        }
476    }
477    changed.extend(before.map(|file| file.path.clone()));
478    changed.extend(after.map(|file| file.path.clone()));
479    changed
480}

482fn staged_after_files(
483    before_files: &[SourceFile],
484    after_paths: &BTreeSet<RepoPath>,
485    changed_files: Vec<SourceFile>,
486) -> Vec<SourceFile> {
487    let mut after = before_files
488        .iter()
489        .filter(|file| after_paths.contains(&file.path))
490        .cloned()
491        .map(|file| (file.path.clone(), file))
492        .collect::<BTreeMap<_, _>>();
493    after.extend(
494        changed_files
495            .into_iter()
496            .map(|file| (file.path.clone(), file)),
497    );
498    after.into_values().collect()
499}

511fn build_workspace_capture(
512    parts: WorkspaceCaptureParts,
513    sentinel: &WorkspaceSentinel,
514) -> GitWorkspaceCapture {
515    let before = Arc::new(SourceSnapshot::new(parts.before_files));
516    let after = Arc::new(SourceSnapshot::new(parts.after_files));
517    let valid_as_of = serde_json::json!({
518        "kind": "git-workspace",
519        "headTree": sentinel.git.base_tree,
520        "index": hex::encode(sentinel.git.index_hash),
521        "scopeState": hex::encode(sentinel.git.scope_state),
522        "generations": hex::encode(sentinel.generations),
523        "beforeView": before.id(),
524        "afterView": after.id(),
525    })
526    .to_string();
527    GitWorkspaceCapture {
528        before,
529        after,
530        paths: parts.paths,
531        changed_paths: parts.changed_paths,
532        before_unsupported_files: parts.before_unsupported_files,
533        after_unsupported_files: parts.after_unsupported_files,
534        unsupported_changed_files: parts.unsupported_changed_files,
535        valid_as_of,
536        sentinel: SenseSentinel::Worktree(sentinel.clone()),
537    }
538}

557pub(super) fn workspace_generation_state(
558    root: &Path,
559    paths: &BTreeSet<RepoPath>,
560) -> Result<[u8; 32], SourceError> {
561    let mut generations = BTreeMap::new();
562    for path in paths {
563        let absolute = root.join(path.to_path_buf());
564        let state = match fs::symlink_metadata(absolute) {
565            Ok(metadata) => metadata_state(&metadata),
566            Err(error) if error.kind() == std::io::ErrorKind::NotFound => [0; 56],
567            Err(error) => return Err(SourceError::Io(error.to_string())),
568        };
569        generations.insert(path.clone(), state);
570    }
571    Ok(generation_state(generations))
572}
