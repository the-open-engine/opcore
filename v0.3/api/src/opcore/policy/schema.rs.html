1//! Editor schema derived from the same typed defaults used by configuration resolution.
2
3use serde_json::{Value, json};
4
5use super::{Policy, settings::MAX_TARGET_PATHS};
6
7pub fn schema() -> Value {
8    let defaults = serde_json::to_value(Policy::default()).unwrap_or(Value::Null);
9    let mut properties = serde_json::Map::new();
10    properties.insert("$schema".into(), json!({"type":"string"}));
11    properties.insert("schemaVersion".into(), json!({"const":1,"default":1}));
12    for name in ["verify", "sense", "coverage"] {
13        properties.insert(name.into(), scalar_section(&defaults[name]));
14    }
15    properties.insert(
16        "targets".into(),
17        object(json!({"exclude":path_list(false)})),
18    );
19    let provider = object(json!({"roots":path_list(true)}));
20    let providers =
21        object(json!({"rust-native":provider,"node-native":provider,"python-native":provider}));
22    properties.insert("providers".into(), providers);
23    properties.insert(
24        "native".into(),
25        json!({"type":"array","uniqueItems":true,"maxItems":3,
26        "default":[],"items":{"enum":["rust-native","node-native","python-native"]}}),
27    );
28    properties.insert("documentation".into(), object(json!({"bindings": {
29        "type":"array", "maxItems":crate::documentation::MAX_DOCUMENTATION_BINDINGS,"default":[],
30        "items": {"type":"object", "additionalProperties":false,
31            "required":["source","document"], "properties":{"source":path(),"document":path()}}
32    }})));
33    let mut workflow = object(
34        properties
35            .iter()
36            .filter(|(key, _)| {
37                matches!(
38                    key.as_str(),
39                    "verify" | "sense" | "coverage" | "targets" | "providers" | "native"
40                )
41            })
42            .map(|(key, value)| (key.clone(), value.clone()))
43            .collect::<serde_json::Map<_, _>>()
44            .into(),
45    );
46    remove_override_defaults(&mut workflow);
47    let mut post_edit = workflow.clone();
48    post_edit["properties"]["native"]["maxItems"] = json!(0);
49    properties.insert(
50        "workflows".into(),
51        object(json!({"post-edit":post_edit,"pre-commit":workflow,"ci":workflow})),
52    );
53    json!({"$schema":"https://json-schema.org/draft/2020-12/schema",
54        "title":"Opcore repository configuration", "type":"object",
55        "additionalProperties":false,"required":["schemaVersion"],"properties":properties})
56}

58fn remove_override_defaults(value: &mut Value) {
59    match value {
60        Value::Object(fields) => {
61            fields.remove("default");
62            for field in fields.values_mut() {
63                remove_override_defaults(field);
64            }
65        }
66        Value::Array(values) => {
67            for value in values {
68                remove_override_defaults(value);
69            }
70        }
71        _ => {}
72    }
73}
74
75fn object(properties: Value) -> Value {
76    let mut object = json!({"type":"object","additionalProperties":false});
77    object["properties"] = properties;
78    object
79}
80
81fn scalar_section(defaults: &Value) -> Value {
82    let properties = defaults
83        .as_object()
84        .into_iter()
85        .flatten()
86        .map(|(name, default)| {
87            let mut property = if default.is_boolean() {
88                json!({"type":"boolean"})
89            } else {
90                json!({"type":"integer","minimum":0})
91            };
92            if let Some((minimum, maximum)) = numeric_bounds(name) {
93                property["minimum"] = json!(minimum);
94                property["maximum"] = json!(maximum);
95            }
96            property["default"] = default.clone();
97            (name.clone(), property)
98        })
99        .collect::<serde_json::Map<_, _>>();
100    object(properties.into())
101}
102
103fn numeric_bounds(name: &str) -> Option<(usize, usize)> {
104    use crate::limits;
105    Some(match name {
106        "maxFileLines" | "maxFunctionLines" | "maxCyclomaticComplexity" => {
107            (0, limits::MAX_SOURCE_BYTES)
108        }
109        "maxLineBytes" => (0, limits::MAX_LINE_BYTES),
110        "maxParameters" => (0, limits::MAX_DEPENDENCY_FACTS_PER_FILE),
111        "maxNesting" => (0, limits::MAX_STRUCTURAL_DEPTH),
112        "importantFanIn" => (1, limits::MAX_FILES),
113        "minimumIdenticalBytes" => (1, limits::MAX_SOURCE_BYTES),
114        "maxDependencyTargets" => (0, limits::MAX_GRAPH_EDGES),
115        "maxEdgeSelectors" | "maxModuleExports" | "maxShapeMembers" => {
116            (0, limits::MAX_INTERFACE_FACTS_PER_FILE)
117        }
118        _ => return None,
119    })
120}

122fn path() -> Value {
123    json!({"type":"string","minLength":1,"maxLength":crate::path::MAX_REPOSITORY_PATH_BYTES,
124        "description":"Literal repository-relative file or subtree path; no glob syntax or dot/parent components.",
125        "pattern":"^(?![A-Za-z]:)(?!/)(?!.*\\\\)(?!.*(?:^|/)(?:\\.|\\.\\.)(?:/|$))(?!.*//)(?!.*\\u0000)[^/]+(?:/[^/]+)*$"})
126}
127
128fn path_list(root: bool) -> Value {
129    let item = if root {
130        json!({"anyOf":[{"const":"."},path()]})
131    } else {
132        path()
133    };
134    json!({"type":"array","uniqueItems":true,"maxItems":MAX_TARGET_PATHS,"items":item,
135        "default":if root { json!(["."]) } else { json!([]) }})
136}
