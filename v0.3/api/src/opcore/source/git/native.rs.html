45/// A bounded full-file Git view used only by the ASP host/native provider boundary.
46///
47/// Unlike `SourceSnapshot`, this view deliberately retains manifests, lockfiles, build scripts,
48/// and non-source resources. It is never fed to the fast parser engine or persisted in its cache.
49#[derive(Clone, Debug)]
50pub(crate) struct GitNativeCapture {
51    pub(crate) baseline_rev: String,
52    pub(crate) paths: BTreeMap<RepoPath, NativePathState>,
53    pub(crate) changed_paths: BTreeSet<RepoPath>,
54    pub(crate) valid_as_of: String,
55    roots: Vec<RepoPath>,
56    filter: super::NativeCaptureFilter,
57    sentinel: NativeSentinel,
58    base_ref: Option<String>,
59}

61#[derive(Clone, Debug, Eq, PartialEq)]
62enum NativeSentinel {
63    Worktree {
64        head_tree: String,
65        observed_head_tree: String,
66        index_state: [u8; 32],
67        paths: BTreeSet<RepoPath>,
68        content_state: [u8; 32],
69        generation_state: [u8; 32],
70    },
71    Staged {
72        head_tree: String,
73        observed_head_tree: String,
74        index_state: [u8; 32],
75        before_paths: BTreeSet<RepoPath>,
76        after_paths: BTreeSet<RepoPath>,
77    },
78    Tree {
79        head_tree: String,
80        target_tree: String,
81        tree_ref: String,
82        base_ref: Option<String>,
83        before_paths: BTreeSet<RepoPath>,
84        after_paths: BTreeSet<RepoPath>,
85    },
86}

91    pub(crate) fn capture_native_workspace(
92        &self,
93        staged: bool,
94        roots: &[RepoPath],
95        filter: super::NativeCaptureFilter,
96    ) -> Result<GitNativeCapture, SourceError> {
97        stable_capture(
98            || self.native_sentinel(staged, roots, filter),
99            |sentinel| self.capture_native_once(sentinel, roots, filter),
100            |_| self.native_sentinel(staged, roots, filter),
101            "HEAD, index, native path set, or worktree bytes changed during capture",
102            "native workspace capture did not stabilize",
103        )
104    }

106    pub(crate) fn is_native_current(
107        &self,
108        capture: &GitNativeCapture,
109    ) -> Result<bool, SourceError> {
110        if let NativeSentinel::Tree {
111            tree_ref, base_ref, ..
112        } = &capture.sentinel
113        {
114            Ok(capture.sentinel
115                == self.native_tree_sentinel(
116                    tree_ref,
117                    base_ref.as_deref(),
118                    &capture.roots,
119                    capture.filter,
120                )?)
121        } else {
122            let staged = matches!(capture.sentinel, NativeSentinel::Staged { .. });
123            Ok(capture.sentinel
124                == self.native_sentinel_from(
125                    staged,
126                    capture.base_ref.as_deref(),
127                    &capture.roots,
128                    capture.filter,
129                )?)
130        }
131    }

147    pub(crate) fn capture_native_from(
148        &self,
149        view: &crate::policy::ConfigView,
150        base_ref: Option<&str>,
151        roots: &[RepoPath],
152        filter: super::NativeCaptureFilter,
153    ) -> Result<GitNativeCapture, SourceError> {
154        let crate::policy::ConfigView::Tree(tree_ref) = view else {
155            let staged = matches!(view, crate::policy::ConfigView::Index);
156            let mut capture = stable_capture(
157                || self.native_sentinel_from(staged, base_ref, roots, filter),
158                |sentinel| self.capture_native_once(sentinel, roots, filter),
159                |_| self.native_sentinel_from(staged, base_ref, roots, filter),
160                "Git refs, index or native inputs changed during capture",
161                "native capture did not stabilize",
162            )?;
163            capture.base_ref = base_ref.map(str::to_owned);
164            return Ok(capture);
165        };
166        stable_capture(
167            || self.native_tree_sentinel(tree_ref, base_ref, roots, filter),
168            |sentinel| self.capture_native_once(sentinel, roots, filter),
169            |_| self.native_tree_sentinel(tree_ref, base_ref, roots, filter),
170            "Git refs changed while native workspace was captured",
171            "native tree capture did not stabilize",
172        )
173    }

175    pub(crate) fn native_candidate_paths_from(
176        &self,
177        view: &crate::policy::ConfigView,
178        roots: &[RepoPath],
179        filter: super::NativeCaptureFilter,
180    ) -> Result<BTreeSet<RepoPath>, SourceError> {
181        match view {
182            crate::policy::ConfigView::Tree(reference) => {

184                native_tree_paths(&self.root, &tree, roots, filter, &self.targets)

194    fn native_tree_sentinel(
195        &self,
196        tree_ref: &str,
197        base_ref: Option<&str>,
198        roots: &[RepoPath],
199        filter: super::NativeCaptureFilter,
200    ) -> Result<NativeSentinel, SourceError> {
201        let target_tree = super::resolve_tree(&self.root, tree_ref)?;
202        let head_tree = base_ref.map_or_else(
203            || Ok(target_tree.clone()),
204            |reference| super::resolve_tree(&self.root, reference),
205        )?;
206        let before_paths = native_tree_paths(&self.root, &head_tree, roots, filter, &self.targets)?;
207        let after_paths =
208            native_tree_paths(&self.root, &target_tree, roots, filter, &self.targets)?;
209        Ok(NativeSentinel::Tree {
210            head_tree,
211            target_tree,
212            tree_ref: tree_ref.to_owned(),
213            base_ref: base_ref.map(str::to_owned),
214            before_paths,
215            after_paths,
216        })
217    }

229    fn native_sentinel_from(
230        &self,
231        staged: bool,
232        base_ref: Option<&str>,
233        roots: &[RepoPath],
234        filter: super::NativeCaptureFilter,
235    ) -> Result<NativeSentinel, SourceError> {
236        let observed_head_tree = head_tree_or_empty(&self.root)?;
237        let head_tree = base_ref.map_or_else(
238            || Ok(observed_head_tree.clone()),
239            |reference| super::resolve_tree(&self.root, reference),
240        )?;
241        let index_state = sentinel_index_hash(GitScope::Changed, &self.index_path)?;
242        let before_paths = native_tree_paths(&self.root, &head_tree, roots, filter, &self.targets)?;
243        if staged {
244            return self.staged_native_sentinel(
245                (head_tree, observed_head_tree),
246                index_state,
247                before_paths,
248                roots,
249                filter,
250            );
251        }
252        self.worktree_native_sentinel(head_tree, observed_head_tree, index_state, roots, filter)
253    }

274        &self,
275        head_tree: String,

293    fn capture_native_once(
294        &self,
295        sentinel: &NativeSentinel,
296        roots: &[RepoPath],
297        filter: super::NativeCaptureFilter,
298    ) -> Result<GitNativeCapture, SourceError> {
299        if !matches!(sentinel, NativeSentinel::Tree { .. }) {
300            reject_unmerged(&self.root)?;
301        }
302        match sentinel {
303            NativeSentinel::Worktree {
304                head_tree, paths, ..
305            } => self.capture_native_worktree(head_tree, paths, sentinel, roots, filter),
306            NativeSentinel::Staged { .. } => self.capture_native_index(sentinel, roots, filter),
307            NativeSentinel::Tree { .. } => self.capture_native_tree(sentinel, roots, filter),
308        }
309    }

311    fn capture_native_tree(
312        &self,
313        sentinel: &NativeSentinel,
314        roots: &[RepoPath],
315        filter: super::NativeCaptureFilter,
316    ) -> Result<GitNativeCapture, SourceError> {
317        let NativeSentinel::Tree {
318            head_tree,
319            target_tree,
320            before_paths,
321            after_paths,
322            ..
323        } = sentinel
324        else {
325            return Err(SourceError::Invalid(
326                "native tree capture requires a tree sentinel".into(),
327            ));
328        };
329        validate_path_sets(before_paths, after_paths)?;
330        let before = raw_git_blobs(
331            &self.root,
332            &tree_entries(&self.root, head_tree, before_paths)?,
333        )?;
334        let after = raw_git_blobs(
335            &self.root,
336            &tree_entries(&self.root, target_tree, after_paths)?,
337        )?;
338        Ok(build_capture(
339            before,
340            after,
341            sentinel.clone(),
342            roots.to_vec(),
343            filter,
344        ))
345    }

347    fn capture_native_worktree(
348        &self,
349        head_tree: &str,
350        after_paths: &BTreeSet<RepoPath>,
351        sentinel: &NativeSentinel,
352        roots: &[RepoPath],
353        filter: super::NativeCaptureFilter,
354    ) -> Result<GitNativeCapture, SourceError> {
355        let before_paths = native_tree_paths(&self.root, head_tree, roots, filter, &self.targets)?;
356        validate_path_sets(&before_paths, after_paths)?;
357        let before_entries = tree_entries(&self.root, head_tree, &before_paths)?;
358        let before = raw_git_blobs(&self.root, &before_entries)?;
359        let after = raw_worktree_blobs(&self.root, after_paths)?;
360        Ok(build_capture(
361            before,
362            after,
363            sentinel.clone(),
364            roots.to_vec(),
365            filter,
366        ))
367    }

369    fn capture_native_index(
370        &self,
371        sentinel: &NativeSentinel,
372        roots: &[RepoPath],
373        filter: super::NativeCaptureFilter,
374    ) -> Result<GitNativeCapture, SourceError> {
375        let NativeSentinel::Staged {
376            head_tree,
377            before_paths,
378            after_paths,
379            ..
380        } = sentinel
381        else {
382            return Err(SourceError::Invalid(
383                "worktree sentinel used for staged native capture".into(),
384            ));
385        };
386        validate_path_sets(before_paths, after_paths)?;
387        let before_entries = tree_entries(&self.root, head_tree, before_paths)?;
388        let after_entries = index_entries(&self.root, after_paths)?;
389        let before = raw_git_blobs(&self.root, &before_entries)?;
390        let after = raw_git_blobs(&self.root, &after_entries)?;
391        Ok(build_capture(
392            before,
393            after,
394            sentinel.clone(),
395            roots.to_vec(),
396            filter,
397        ))
398    }

401fn validate_path_sets(
402    before_paths: &BTreeSet<RepoPath>,
403    after_paths: &BTreeSet<RepoPath>,
404) -> Result<(), SourceError> {
405    require_path_bound(before_paths)?;
406    require_path_bound(after_paths)?;
407    require_path_bound(&before_paths.union(after_paths).cloned().collect())?;
408    Ok(())
409}
410
411fn require_path_bound(paths: &BTreeSet<RepoPath>) -> Result<(), SourceError> {
412    if paths.len() > MAX_FILES {
413        return Err(SourceError::Incomplete(format!(
414            "native workspace exceeds {MAX_FILES} files"
415        )));
416    }
417    Ok(())
418}

420fn raw_git_blobs(
421    root: &std::path::Path,
422    entries: &BTreeMap<RepoPath, TreeEntry>,
423) -> Result<BTreeMap<RepoPath, NativeBlob>, SourceError> {
424    git_blobs_from_objects_bounded(root, entries.iter().collect(), MAX_TOTAL_SOURCE_BYTES).map(
425        |blobs| {
426            blobs
427                .into_iter()
428                .map(|blob| (blob.path, NativeBlob::new(blob.bytes)))
429                .collect()
430        },
431    )
432}

434fn raw_worktree_blobs(
435    root: &std::path::Path,
436    paths: &BTreeSet<RepoPath>,
437) -> Result<BTreeMap<RepoPath, NativeBlob>, SourceError> {
438    let mut blobs = BTreeMap::new();
439    let mut missing = BTreeSet::new();
440    let mut total = 0usize;
441    for path in paths {
442        if let Some(blob) = raw_worktree_blob(root, path, &mut total)? {
443            blobs.insert(path.clone(), blob);
444        } else {
445            missing.insert(path.clone());
446        }
447    }
448    let sparse = sparse_paths(root, &missing)?;
449    let entries = index_entries(root, &sparse)?;
450    let remaining = MAX_TOTAL_SOURCE_BYTES.saturating_sub(total);
451    for (path, blob) in raw_git_blobs_bounded(root, &entries, remaining)? {
452        blobs.insert(path, blob);
453    }
454    Ok(blobs)
455}
456
457fn raw_worktree_blob(
458    root: &std::path::Path,
459    path: &RepoPath,
460    total: &mut usize,
461) -> Result<Option<NativeBlob>, SourceError> {
462    match read_optional_worktree_blob(root, path)? {
463        Some(bytes) => {
464            add_bounded_blob_length(total, bytes.len(), path, MAX_TOTAL_SOURCE_BYTES)?;
465            Ok(Some(NativeBlob::new(bytes)))
466        }
467        None => Ok(None),
468    }
469}

471fn raw_git_blobs_bounded(
472    root: &std::path::Path,
473    entries: &BTreeMap<RepoPath, TreeEntry>,
474    total_limit: usize,

477        blobs
478            .into_iter()

484fn build_capture(
485    mut before: BTreeMap<RepoPath, NativeBlob>,
486    mut after: BTreeMap<RepoPath, NativeBlob>,
487    sentinel: NativeSentinel,
488    roots: Vec<RepoPath>,
489    filter: super::NativeCaptureFilter,
490) -> GitNativeCapture {
491    let mut all_paths = before.keys().cloned().collect::<BTreeSet<_>>();
492    all_paths.extend(after.keys().cloned());
493    let mut changed_paths = BTreeSet::new();
494    let mut states = BTreeMap::new();
495    for path in all_paths {
496        let before_blob = before.remove(&path);
497        let mut after_blob = after.remove(&path);
498        if let (Some(left), Some(right)) = (&before_blob, &mut after_blob)
499            && left.blob_id == right.blob_id
500        {
501            right.bytes = Arc::clone(&left.bytes);
502        }
503        if before_blob.as_ref().map(|blob| &blob.blob_id)
504            != after_blob.as_ref().map(|blob| &blob.blob_id)
505        {
506            changed_paths.insert(path.clone());
507        }
508        states.insert(
509            path,
510            NativePathState {
511                before: before_blob,
512                after: after_blob,
513            },
514        );
515    }
516    let before_view = content_view_digest("git-native-view/v1", &states, |state| {
517        state.before.as_ref().map(|blob| blob.blob_id.as_bytes())
518    });
519    let after_view = content_view_digest("git-native-view/v1", &states, |state| {
520        state.after.as_ref().map(|blob| blob.blob_id.as_bytes())
521    });
522    let baseline_tree = match &sentinel {
523        NativeSentinel::Worktree { head_tree, .. }
524        | NativeSentinel::Staged { head_tree, .. }
525        | NativeSentinel::Tree { head_tree, .. } => head_tree,
526    };
527    let kind = if matches!(sentinel, NativeSentinel::Staged { .. }) {
528        "git-native-index"
529    } else if matches!(sentinel, NativeSentinel::Tree { .. }) {
530        "git-native-tree"
531    } else {
532        "git-native-worktree"
533    };
534    GitNativeCapture {
535        baseline_rev: format!("git:tree:{baseline_tree}"),
536        paths: states,
537        changed_paths,
538        valid_as_of: serde_json::json!({
539            "kind": kind,
540            "headTree": baseline_tree,
541            "beforeView": before_view,
542            "afterView": after_view,
543        })
544        .to_string(),
545        roots,
546        filter,
547        sentinel,
548        base_ref: None,
549    }
550}
