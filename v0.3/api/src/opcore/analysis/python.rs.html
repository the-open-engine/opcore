36const PARSER_NAME: &str = "ruff-python-parser";
37pub fn analyze(
38    file: &SourceFile,
39    limits: &RuleLimits,
40    cancel: &CancelToken,
41) -> Result<FileFacts, AnalysisError> {
42    cancellation_boundary(cancel)?;
43    let decoded = decode_source(&file.bytes)?;
44    cancellation_boundary(cancel)?;
45    preflight_structure(decoded.text, cancel)?;
46
47    let lines = LineMap::new(decoded.text, decoded.byte_base);
48    let parsed = match parse_module(decoded.text) {
49        Ok(parsed) => parsed,
50        Err(error) => {
51            let diagnostic = parse_diagnostic(
52                error.location.start(),
53                parse_error_kind(&error.error),
54                parse_error_message(&error.error),
55                &lines,
56            );
57            return Ok(parser_failure_facts(
58                PARSER_NAME,
59                PYTHON_PARSER_VERSION,
60                vec![diagnostic],
61            ));
62        }
63    };
64    cancellation_boundary(cancel)?;
65
66    let fingerprint_tokens = fingerprint_tokens(parsed.tokens());
67    let region_fingerprints = python_region_fingerprints(&fingerprint_tokens, decoded.text, &lines);
68    let interfaces = python_interface_exports(&parsed.syntax().body, decoded.text, parsed.tokens());
69    let mut walker = MetricsWalker::new(limits, cancel, &lines, &fingerprint_tokens, interfaces);
70    walker.visit_body(&parsed.syntax().body);
71    if walker.cancelled {
72        return Err(AnalysisError::Cancelled);
73    }
74    cancellation_boundary(cancel)?;
75    walker.diagnostics.sort_by(|left, right| {
76        left.range
77            .map(|range| (range.start.byte, range.end.byte))
78            .cmp(&right.range.map(|range| (range.start.byte, range.end.byte)))
79            .then_with(|| left.rule_id.cmp(&right.rule_id))
80            .then_with(|| left.entity_key.cmp(&right.entity_key))
81    });
82
83    let dependencies = walker.take_dependency_facts();
84    let callable_fingerprints = walker.take_callable_fingerprints();
85    let interfaces = walker.take_interface_facts();
86    Ok(FileFacts {
87        diagnostics: walker.diagnostics,
88        parser: PARSER_NAME.into(),
89        parser_version: PYTHON_PARSER_VERSION.into(),
90        dependencies,
91        callable_fingerprints,
92        region_fingerprints,
93        interfaces,
94    })
95}

97fn parse_diagnostic(
98    offset: TextSize,
99    error_kind: &'static str,
100    message: &'static str,
101    lines: &LineMap<'_>,
102) -> RawDiagnostic {
103    let start = offset.to_usize().min(lines.source.len());
104    let width = lines.source[start..]
105        .chars()
106        .next()
107        .map_or(0, char::len_utf8);
108    let mut evidence = parser_evidence();
109    evidence.insert("errorKind".into(), serde_json::json!(error_kind));
110    RawDiagnostic {
111        rule_id: "python.syntax".into(),
112        severity: Severity::Error,
113        message: message.into(),
114        range: Some(lines.range(start, start.saturating_add(width))),
115        entity_key: format!(
116            "module-syntax:{error_kind}:{}",
117            crate::analysis::line_anchor(lines.source.as_bytes(), start)
118        ),
119        cause_key: error_kind.into(),
120        evidence,
121    }
122}

124const fn parse_error_kind(error: &ParseErrorType) -> &'static str {
125    match error {
126        ParseErrorType::UnexpectedIndentation => "indentation",
127        ParseErrorType::FStringError(_) => "f-string",
128        ParseErrorType::TStringError(_) => "t-string",
129        ParseErrorType::RecursionLimitExceeded => "recursion-limit",
130        ParseErrorType::ExpectedToken { found, .. } if found.is_eof() => "unexpected-eof",
131        ParseErrorType::Lexical(error) => lexical_error_kind(error),
132        _ => "syntax",
133    }
134}

136const fn lexical_error_kind(error: &LexicalErrorType) -> &'static str {
137    match error {
138        LexicalErrorType::StringError | LexicalErrorType::UnclosedStringError => "string",
139        LexicalErrorType::UnicodeError
140        | LexicalErrorType::MissingUnicodeLbrace
141        | LexicalErrorType::MissingUnicodeRbrace => "unicode",
142        _ => lexical_misc_error_kind(error),
143    }
144}
145
146const fn lexical_misc_error_kind(error: &LexicalErrorType) -> &'static str {
147    match error {
148        LexicalErrorType::IndentationError => "indentation",
149        LexicalErrorType::UnrecognizedToken { .. } => "unrecognized-token",
150        LexicalErrorType::FStringError(_) => "f-string",
151        LexicalErrorType::TStringError(_) => "t-string",
152        LexicalErrorType::Eof => "unexpected-eof",
153        LexicalErrorType::LineContinuationError => "line-continuation",
154        LexicalErrorType::InvalidByteLiteral => "byte-literal",
155        _ => "lexer",
156    }
157}

159const fn parse_error_message(error: &ParseErrorType) -> &'static str {
160    match error {
161        ParseErrorType::ExpectedToken { found, .. } if found.is_eof() => {
162            "unexpected end of Python source"
163        }
164        ParseErrorType::UnexpectedIndentation => "unexpected Python indentation",
165        ParseErrorType::Lexical(error) => lexical_error_message(error),
166        ParseErrorType::FStringError(_) => "invalid Python f-string",
167        ParseErrorType::TStringError(_) => "invalid Python t-string",
168        ParseErrorType::RecursionLimitExceeded => "Python syntax exceeds the parser depth limit",
169        _ => "invalid Python syntax",
170    }
171}

176        LexicalErrorType::FStringError(_) => "invalid Python f-string",
177        LexicalErrorType::TStringError(_) => "invalid Python t-string",
178        _ => "invalid Python syntax",

186struct LineMap<'a> {
187    source: &'a str,
188    starts: Vec<usize>,
189    byte_base: usize,
190}
191
192impl<'a> LineMap<'a> {
193    fn new(source: &'a str, byte_base: usize) -> Self {
194        let bytes = source.as_bytes();
195        let mut starts = vec![0];
196        let mut index = 0;
197        while index < bytes.len() {
198            match bytes[index] {
199                b'\r' => {
200                    index += 1;
201                    if bytes.get(index) == Some(&b'\n') {
202                        index += 1;
203                    }
204                    starts.push(index);
205                }
206                b'\n' => {
207                    index += 1;
208                    starts.push(index);
209                }
210                _ => index += 1,
211            }
212        }
213        Self {
214            source,
215            starts,
216            byte_base,
217        }
218    }
219
220    fn position_usize(&self, offset: usize) -> Position {
221        let offset = offset.min(self.source.len());
222        let line_index = self
223            .starts
224            .partition_point(|start| *start <= offset)
225            .saturating_sub(1);
226        let line_start = self.starts[line_index];
227        Position {
228            line: u32::try_from(line_index.saturating_add(1)).unwrap_or(u32::MAX),
229            column: u32::try_from(offset.saturating_sub(line_start).saturating_add(1))
230                .unwrap_or(u32::MAX),
231            byte: u32::try_from(offset.saturating_add(self.byte_base)).unwrap_or(u32::MAX),
232        }
233    }
234
235    fn range(&self, start: usize, end: usize) -> SourceRange {
236        SourceRange {
237            start: self.position_usize(start),
238            end: self.position_usize(end),
239        }
240    }
241
242    fn text_range(&self, range: TextRange) -> SourceRange {
243        self.range(range.start().to_usize(), range.end().to_usize())
244    }
245
246    fn line_span(&self, range: TextRange) -> u32 {
247        let start = range.start().to_usize();
248        let end = range.end().to_usize();
249        let start_line = self.position_usize(start).line;
250        let end_line = self.position_usize(end.saturating_sub(1).max(start)).line;
251        end_line.saturating_sub(start_line).saturating_add(1)
252    }
253}

282struct MetricsWalker<'a> {
283    limits: &'a RuleLimits,
284    cancel: &'a CancelToken,
285    lines: &'a LineMap<'a>,
286    frames: Vec<Option<FunctionMetrics>>,
287    scope: Vec<String>,
288    entity_occurrences: BTreeMap<String, u32>,
289    diagnostics: Vec<RawDiagnostic>,
290    dependencies: Vec<DependencyReference>,
291    dependencies_truncated: bool,
292    fingerprint_tokens: &'a [PythonFingerprintToken],
293    fingerprints: FingerprintCollector,
294    interfaces: InterfaceFacts,
295    interfaces_truncated: bool,
296    cancelled: bool,
297}

299impl<'a> MetricsWalker<'a> {
300    fn new(
301        limits: &'a RuleLimits,
302        cancel: &'a CancelToken,
303        lines: &'a LineMap<'a>,
304        fingerprint_tokens: &'a [PythonFingerprintToken],
305        interfaces: InterfaceFacts,
306    ) -> Self {
307        Self {
308            limits,
309            cancel,
310            lines,
311            frames: Vec::new(),
312            scope: Vec::new(),
313            entity_occurrences: BTreeMap::new(),
314            diagnostics: Vec::new(),
315            dependencies: Vec::new(),
316            dependencies_truncated: false,
317            fingerprint_tokens,
318            fingerprints: FingerprintCollector::default(),
319            interfaces,
320            interfaces_truncated: false,
321            cancelled: false,
322        }
323    }
324
325    fn current(&mut self) -> Option<&mut FunctionMetrics> {
326        self.frames.last_mut().and_then(Option::as_mut)
327    }
328
329    fn stop_requested(&mut self) -> bool {
330        self.cancelled |= self.cancel.is_cancelled();
331        self.cancelled
332    }
333
334    fn enter_control(&mut self, complexity: u32) -> Option<u32> {
335        let current = self.current()?;
336        current.complexity = current.complexity.saturating_add(complexity);
337        let previous = current.control_depth;
338        current.control_depth = current.control_depth.saturating_add(1);
339        current.max_nesting = current.max_nesting.max(current.control_depth);
340        Some(previous)
341    }
342
343    fn leave_control(&mut self, previous: Option<u32>) {
344        if let (Some(previous), Some(current)) = (previous, self.current()) {
345            current.control_depth = previous;
346        }
347    }
348
349    fn add_complexity(&mut self, amount: usize) {
350        if let Some(current) = self.current() {
351            current.complexity = current
352                .complexity
353                .saturating_add(u32::try_from(amount).unwrap_or(u32::MAX));
354        }
355    }
356
357    fn enter_callable(
358        &mut self,
359        kind: CallableKind,
360        name: String,
361        range: TextRange,
362        parameters: u32,
363    ) {
364        self.scope.push(name);
365        let base_entity = format!("callable:{}", self.scope.join("."));
366        let occurrence = self
367            .entity_occurrences
368            .entry(base_entity.clone())
369            .or_default();
370        let entity = format!("{base_entity}:{}", *occurrence);
371        *occurrence = occurrence.saturating_add(1);
372        self.frames.push(Some(FunctionMetrics {
373            kind,
374            entity,
375            range,
376            parameters,
377            complexity: 1,
378            control_depth: 0,
379            max_nesting: 0,
380        }));
381    }
382
383    fn observe_callable_fingerprint(
384        &mut self,
385        kind: FingerprintCallableKind,
386        range: Option<TextRange>,
387    ) {
388        if self.frames.iter().any(Option::is_some) {
389            return;
390        }
391        let Some(range) = range else {
392            return;
393        };
394        let start = range.start().to_usize();
395        let end = range.end().to_usize();
396        let source = self.lines.source.as_bytes();
397        let output_range = self.lines.text_range(range);
398        self.fingerprints.push(fingerprint(
399            "python",
400            kind,
401            output_range,
402            self.fingerprint_tokens
403                .iter()
404                .filter(|token| token.start >= start && token.end <= end)
405                .map(|token| FingerprintToken {
406                    tag: token.tag,
407                    text: source.get(token.start..token.end).unwrap_or_default(),
408                    line: self.lines.position_usize(token.start).line,
409                    counted: token.tag == 0,
410                    start_byte: self.lines.position_usize(token.start).byte,
411                    end_byte: self.lines.position_usize(token.end).byte,
412                }),
413        ));
414    }
415
416    fn leave_callable(&mut self) {
417        let metrics = self.frames.pop().flatten();
418        self.scope.pop();
419        if let Some(metrics) = metrics {
420            self.emit_metrics(&metrics);
421        }
422    }
423
424    fn emit_metrics(&mut self, metrics: &FunctionMetrics) {
425        let function_lines = self.lines.line_span(metrics.range);
426        self.emit_metric(
427            metrics,
428            "complexity.max-function-lines",
429            "functionLines",
430            function_lines,
431            self.limits.max_function_lines,
432        );
433        self.emit_metric(
434            metrics,
435            "complexity.max-parameters",
436            "parameters",
437            metrics.parameters,
438            self.limits.max_parameters,
439        );
440        self.emit_metric(
441            metrics,
442            "complexity.max-nesting",
443            "nesting",
444            metrics.max_nesting,
445            self.limits.max_nesting,
446        );
447        self.emit_metric(
448            metrics,
449            "complexity.max-cyclomatic-complexity",
450            "cyclomaticComplexity",
451            metrics.complexity,
452            self.limits.max_cyclomatic_complexity,
453        );
454    }
455
456    fn emit_metric(
457        &mut self,
458        metrics: &FunctionMetrics,
459        rule_id: &str,
460        metric_name: &str,
461        actual: u32,
462        limit: u32,
463    ) {
464        if actual <= limit {
465            return;
466        }
467        let mut evidence = parser_evidence();
468        evidence.insert("metric".into(), serde_json::json!(metric_name));
469        evidence.insert("actual".into(), serde_json::json!(actual));
470        evidence.insert("limit".into(), serde_json::json!(limit));
471        self.diagnostics.push(RawDiagnostic {
472            rule_id: rule_id.into(),
473            severity: Severity::Warning,
474            message: format!(
475                "{} has {actual} {metric_name}; configured maximum is {limit}",
476                metrics.kind.label()
477            ),
478            range: Some(self.lines.text_range(metrics.range)),
479            entity_key: metrics.entity.clone(),
480            cause_key: format!("limit:{limit}"),
481            evidence,
482        });
483    }
484
485    fn enter_comprehension(&mut self, generators: &[ast::Comprehension]) -> Option<u32> {
486        let decisions = generators
487            .len()
488            .saturating_add(generators.iter().map(|generator| generator.ifs.len()).sum());
489        let current = self.current()?;
490        current.complexity = current
491            .complexity
492            .saturating_add(u32::try_from(decisions).unwrap_or(u32::MAX));
493        let previous = current.control_depth;
494        current.control_depth = current
495            .control_depth
496            .saturating_add(u32::try_from(generators.len()).unwrap_or(u32::MAX));
497        current.max_nesting = current.max_nesting.max(current.control_depth);
498        Some(previous)
499    }
500
501    fn enter_expression_control(&mut self, expression: &ast::Expr) -> Option<u32> {
502        match expression {
503            ast::Expr::ListComp(node) => self.enter_comprehension(&node.generators),
504            ast::Expr::SetComp(node) => self.enter_comprehension(&node.generators),
505            ast::Expr::DictComp(node) => self.enter_comprehension(&node.generators),
506            ast::Expr::Generator(node) => self.enter_comprehension(&node.generators),
507            _ => None,
508        }
509    }

511    fn push_dependency(
512        &mut self,
513        kind: DependencyReferenceKind,
514        specifier: impl Into<String>,
515        level: u32,
516    ) {
517        if self.dependencies.len() >= MAX_DEPENDENCY_FACTS_PER_FILE {
518            self.dependencies_truncated = true;
519            return;
520        }
521        self.dependencies.push(DependencyReference {
522            kind,
523            specifier: specifier.into(),
524            level,
525        });
526    }
527
528    fn observe_dependency(&mut self, statement: &ast::Stmt) {
529        match statement {
530            ast::Stmt::Import(import) => {
531                for alias in &import.names {
532                    self.push_dependency(
533                        DependencyReferenceKind::PythonAbsolute,
534                        alias.name.to_string(),
535                        0,
536                    );
537                }
538            }
539            ast::Stmt::ImportFrom(import) => {
540                let level = import.level;
541                if level == 0 {
542                    self.push_dependency(
543                        DependencyReferenceKind::PythonAbsolute,
544                        import.module.as_ref().map_or("", |module| module.as_str()),
545                        0,
546                    );
547                } else if let Some(module) = &import.module {
548                    self.push_dependency(
549                        DependencyReferenceKind::PythonRelative,
550                        module.to_string(),
551                        level,
552                    );
553                    for alias in &import.names {
554                        let selector = if alias.name.as_str() == "*" {
555                            self.interfaces.gaps.namespace_imports =
556                                self.interfaces.gaps.namespace_imports.saturating_add(1);
557                            InterfaceSelector {
558                                kind: InterfaceSelectorKind::Namespace,
559                                name: String::new(),
560                            }
561                        } else {
562                            InterfaceSelector {
563                                kind: InterfaceSelectorKind::Named,
564                                name: alias.name.to_string(),
565                            }
566                        };
567                        self.push_interface_import(InterfaceImportFact {
568                            specifier: module.to_string(),
569                            level,
570                            role: InterfaceImportRole::Import,
571                            namespace: InterfaceNamespace::Runtime,
572                            selector,
573                        });
574                    }
575                } else {
576                    self.interfaces.gaps.unsupported_patterns = self
577                        .interfaces
578                        .gaps
579                        .unsupported_patterns
580                        .saturating_add(import.names.len());
581                    for alias in &import.names {
582                        self.push_dependency(
583                            DependencyReferenceKind::PythonUnsupportedRelative,
584                            alias.name.to_string(),
585                            level,
586                        );
587                    }
588                }
589            }
590            _ => {}
591        }
592    }

594    fn take_dependency_facts(&mut self) -> DependencyFacts {
595        self.dependencies.sort();
596        DependencyFacts {
597            status: if self.dependencies_truncated {
598                DependencyExtractionStatus::Truncated
599            } else {
600                DependencyExtractionStatus::Complete
601            },
602            references: std::mem::take(&mut self.dependencies),
603        }
604    }
605
606    fn take_callable_fingerprints(&mut self) -> CallableFingerprintFacts {
607        std::mem::take(&mut self.fingerprints).finish()
608    }
609
610    fn push_interface_import(&mut self, fact: InterfaceImportFact) {
611        let fact_count = self
612            .interfaces
613            .imports
614            .len()
615            .saturating_add(self.interfaces.exports.len())
616            .saturating_add(self.interfaces.shapes.len());
617        if fact_count >= MAX_INTERFACE_FACTS_PER_FILE {
618            self.interfaces_truncated = true;
619        } else {
620            self.interfaces.imports.push(fact);
621        }
622    }
623
624    fn take_interface_facts(&mut self) -> InterfaceFacts {
625        self.interfaces.imports.sort();
626        self.interfaces.imports.dedup();
627        self.interfaces.exports.sort();
628        self.interfaces.exports.dedup();
629        self.interfaces.status = if self.interfaces_truncated {
630            InterfaceExtractionStatus::Truncated
631        } else if self.interfaces.status == InterfaceExtractionStatus::Partial
632            || self.interfaces.gaps.namespace_imports > 0
633            || self.interfaces.gaps.unsupported_patterns > 0
634        {
635            InterfaceExtractionStatus::Partial
636        } else {
637            InterfaceExtractionStatus::Complete
638        };
639        std::mem::take(&mut self.interfaces)
640    }

664fn python_interface_exports(suite: &[ast::Stmt], source: &str, tokens: &Tokens) -> InterfaceFacts {
665    let occurrences = tokens
666        .iter()
667        .filter(|token| {
668            token.kind() == TokenKind::Name
669                && source.get(token.range().to_std_range()) == Some("__all__")
670        })
671        .count();
672    let assignment = suite.iter().find_map(static_all_assignment);
673    if occurrences == 0 {
674        return InterfaceFacts {
675            status: InterfaceExtractionStatus::Complete,
676            public_surface_status: InterfaceSurfaceStatus::Unsupported,
677            ..InterfaceFacts::default()
678        };
679    }
680    let Some(names) = assignment.filter(|_| occurrences == 1) else {
681        return InterfaceFacts {
682            status: InterfaceExtractionStatus::Partial,
683            public_surface_status: InterfaceSurfaceStatus::Partial,
684            gaps: crate::model::InterfaceGapCounts {
685                unsupported_patterns: 1,
686                ..crate::model::InterfaceGapCounts::default()
687            },
688            ..InterfaceFacts::default()
689        };
690    };
691    let mut facts = InterfaceFacts {
692        status: InterfaceExtractionStatus::Complete,
693        public_surface_status: InterfaceSurfaceStatus::Complete,
694        ..InterfaceFacts::default()
695    };
696    let name_count = names.len();
697    for name in names.into_iter().take(MAX_INTERFACE_FACTS_PER_FILE) {
698        facts.exports.push(InterfaceExportFact {
699            exported_name: name,
700            namespace: InterfaceExportNamespace::Value,
701            origin: InterfaceExportOrigin::Local,
702            declaration_kind: InterfaceDeclarationKind::Unknown,
703        });
704    }
705    if facts.exports.len() < name_count {
706        facts.status = InterfaceExtractionStatus::Truncated;
707        facts.public_surface_status = InterfaceSurfaceStatus::Truncated;
708    }
709    facts.exports.sort();
710    facts.exports.dedup();
711    facts
712}

714fn static_all_assignment(statement: &ast::Stmt) -> Option<Vec<String>> {
715    let value = match statement {
716        ast::Stmt::Assign(assign)
717            if assign.targets.len() == 1 && is_all_name(&assign.targets[0]) =>
718        {
719            Some(assign.value.as_ref())
720        }
721        ast::Stmt::AnnAssign(assign) if is_all_name(&assign.target) => assign.value.as_deref(),
722        _ => None,
723    }?;
724    static_string_sequence(value)
725}
726
727fn is_all_name(expression: &ast::Expr) -> bool {
728    matches!(expression, ast::Expr::Name(name) if name.id.as_str() == "__all__")
729}
730
731fn static_string_sequence(expression: &ast::Expr) -> Option<Vec<String>> {
732    let elements = match expression {
733        ast::Expr::List(list) => &list.elts,
734        ast::Expr::Tuple(tuple) => &tuple.elts,
735        _ => return None,
736    };
737    elements
738        .iter()
739        .map(|element| match element {
740            ast::Expr::StringLiteral(literal) => Some(literal.value.to_str().to_owned()),
741            _ => None,
742        })
743        .collect()
744}
745
746impl<'ast> Visitor<'ast> for MetricsWalker<'_> {
747    fn visit_stmt(&mut self, node: &'ast ast::Stmt) {
748        if self.stop_requested() {
749            return;
750        }
751        self.observe_dependency(node);
752        match node {
753            ast::Stmt::FunctionDef(function) => {
754                self.visit_function(function);
755                return;
756            }
757            ast::Stmt::ClassDef(class) => {
758                self.scope.push(class.name.to_string());
759                self.frames.push(None);
760                walk_stmt(self, node);
761                self.frames.pop();
762                self.scope.pop();
763                return;
764            }
765            _ => {}
766        }
767        let control = match node {
768            ast::Stmt::If(_) | ast::Stmt::For(_) | ast::Stmt::While(_) => self.enter_control(1),
769            ast::Stmt::With(_) => self.enter_control(0),
770            ast::Stmt::Try(node) => {
771                self.enter_control(u32::try_from(node.handlers.len()).unwrap_or(u32::MAX))
772            }
773            ast::Stmt::Match(node) => {
774                self.enter_control(u32::try_from(node.cases.len()).unwrap_or(u32::MAX))
775            }
776            _ => None,
777        };
778        walk_stmt(self, node);
779        self.leave_control(control);
780    }
781
782    fn visit_elif_else_clause(&mut self, clause: &'ast ast::ElifElseClause) {
783        if clause.test.is_some() {
784            self.add_complexity(1);
785        }
786        walk_elif_else_clause(self, clause);
787    }
788
789    fn visit_expr(&mut self, node: &'ast ast::Expr) {
790        if self.stop_requested() {
791            return;
792        }
793        match node {
794            ast::Expr::Lambda(lambda) => {
795                self.visit_lambda(lambda);
796                return;
797            }
798            ast::Expr::BoolOp(node) => self.add_complexity(node.values.len().saturating_sub(1)),
799            ast::Expr::If(_) => self.add_complexity(1),
800            _ => {}
801        }
802        let control = self.enter_expression_control(node);
803        walk_expr(self, node);
804        self.leave_control(control);
805    }
806}
807
808impl MetricsWalker<'_> {
809    fn visit_function(&mut self, node: &ast::StmtFunctionDef) {
810        for decorator in &node.decorator_list {
811            self.visit_decorator(decorator);
812        }
813        if let Some(type_params) = &node.type_params {
814            self.visit_type_params(type_params);
815        }
816        self.visit_parameters(&node.parameters);
817        if let Some(returns) = &node.returns {
818            self.visit_annotation(returns);
819        }
820        self.observe_callable_fingerprint(
821            FingerprintCallableKind::Function,
822            suite_range(&node.body),
823        );
824        let metric_start = node
825            .decorator_list
826            .iter()
827            .map(Ranged::start)
828            .min()
829            .unwrap_or_else(|| node.start());
830        self.enter_callable(
831            if node.is_async {
832                CallableKind::AsyncFunction
833            } else {
834                CallableKind::Function
835            },
836            node.name.to_string(),
837            TextRange::new(metric_start, node.end()),
838            parameter_count(&node.parameters),
839        );
840        self.visit_body(&node.body);
841        self.leave_callable();
842    }
843
844    fn visit_lambda(&mut self, node: &ast::ExprLambda) {
845        self.observe_callable_fingerprint(FingerprintCallableKind::Lambda, Some(node.body.range()));
846        let anchor =
847            crate::analysis::line_anchor(self.lines.source.as_bytes(), node.start().to_usize());
848        self.enter_callable(
849            CallableKind::Lambda,
850            format!("<lambda:{anchor}>"),
851            node.range,
852            node.parameters.as_deref().map_or(0, parameter_count),
853        );
854        if let Some(parameters) = &node.parameters {
855            self.visit_parameters(parameters);
856        }
857        self.visit_expr(&node.body);
858        self.leave_callable();
859    }
860}
861
862fn suite_range(body: &[ast::Stmt]) -> Option<TextRange> {
863    Some(TextRange::new(
864        body.first().map(Ranged::start)?,
865        body.last().map(Ranged::end)?,
866    ))
867}
868
869fn parameter_count(parameters: &ast::Parameters) -> u32 {
870    let count = parameters
871        .posonlyargs
872        .len()
873        .saturating_add(parameters.args.len())
874        .saturating_add(parameters.kwonlyargs.len())
875        .saturating_add(usize::from(parameters.vararg.is_some()))
876        .saturating_add(usize::from(parameters.kwarg.is_some()));
877    u32::try_from(count).unwrap_or(u32::MAX)
878}

881    if cancel.is_cancelled() {
882        Err(AnalysisError::Cancelled)
883    } else {
884        Ok(())
885    }
