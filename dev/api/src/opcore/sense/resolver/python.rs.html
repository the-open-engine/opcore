2    model::{DependencyReference, DependencyReferenceKind},

8pub(super) fn resolve(
9    importer: &RepoPath,
10    reference: &DependencyReference,
11    index: &ResolutionIndex,
12) -> Resolution {
13    let absolute = reference.kind == DependencyReferenceKind::PythonAbsolute;
14    let Some(base) = normalized(importer, &reference.specifier, reference.level) else {
15        return if absolute {
16            Resolution::Ambiguous
17        } else {
18            Resolution::Unresolved
19        };
20    };
21    match alias_resolution(index.python_aliases.get(&base)) {
22        Resolution::Unresolved if absolute => Resolution::Ambiguous,
23        resolution => resolution,
24    }
25}

27fn normalized(importer: &RepoPath, module: &str, level: u32) -> Option<String> {
28    if module.is_empty() || module.contains(['/', '\\']) {
29        return None;
30    }
31    if level == 0 {
32        if module.contains('.') {
33            return None;
34        }
35        return normalized_relative(importer, module, 0);
36    }
37    let parents = usize::try_from(level.saturating_sub(1)).ok()?;
38    let module_path = module.replace('.', "/");
39    normalized_relative(importer, &module_path, parents)
40}
