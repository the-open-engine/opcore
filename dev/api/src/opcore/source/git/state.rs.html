31pub(super) fn read_optional_worktree_blob(
32    root: &Path,
33    path: &RepoPath,
34) -> Result<Option<Vec<u8>>, SourceError> {
35    match fs::symlink_metadata(root.join(path.to_path_buf())) {
36        Ok(_) => read_regular_no_follow(root, path).map(Some),
37        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(None),
38        Err(error) => Err(io_error(error)),
39    }

42fn current_worktree_state_with(
43    root: &Path,
44    paths: &BTreeSet<RepoPath>,
45    total_limit: usize,
46    include: fn(&RepoPath) -> bool,
47) -> Result<([u8; 32], [u8; 32]), SourceError> {
48    let mut manifest = collect_worktree_manifest(root, paths, total_limit, include)?;
49    let remaining = total_limit.saturating_sub(manifest.total_bytes);
50    merge_sparse_contents(root, &manifest.missing, &mut manifest.contents, remaining)?;
51    Ok((
52        worktree_state(paths, &manifest.contents, include),
53        generation_state(manifest.generations),
54    ))
55}

57#[derive(Default)]
58struct WorktreeManifest {
59    contents: BTreeMap<RepoPath, ContentId>,
60    generations: BTreeMap<RepoPath, [u8; 56]>,
61    missing: BTreeSet<RepoPath>,
62    total_bytes: usize,
63}
64
65fn collect_worktree_manifest(
66    root: &Path,
67    paths: &BTreeSet<RepoPath>,
68    total_limit: usize,
69    include: fn(&RepoPath) -> bool,
70) -> Result<WorktreeManifest, SourceError> {
71    let mut manifest = WorktreeManifest::default();
72    for path in paths {
73        if !include(path) {
74            continue;
75        }
76        scan_worktree_manifest_path(root, path, &mut manifest, total_limit)?;
77    }
78    Ok(manifest)
79}
80
81fn scan_worktree_manifest_path(
82    root: &Path,
83    path: &RepoPath,
84    manifest: &mut WorktreeManifest,
85    total_limit: usize,
86) -> Result<(), SourceError> {
87    let absolute = root.join(path.to_path_buf());
88    match fs::symlink_metadata(&absolute) {
89        Ok(_) => {
90            let bytes = read_regular_no_follow(root, path)?;
91            add_bounded_blob_length(&mut manifest.total_bytes, bytes.len(), path, total_limit)?;
92            manifest
93                .contents
94                .insert(path.clone(), ContentId::of(&bytes));
95            let metadata = fs::metadata(&absolute).map_err(io_error)?;
96            manifest
97                .generations
98                .insert(path.clone(), metadata_state(&metadata));
99        }
100        Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
101            manifest.missing.insert(path.clone());
102            manifest.generations.insert(path.clone(), [0; 56]);
103        }
104        Err(error) => return Err(io_error(error)),
105    }
106    Ok(())
107}

109fn merge_sparse_contents(
110    root: &Path,
111    missing: &BTreeSet<RepoPath>,
112    contents: &mut BTreeMap<RepoPath, ContentId>,
113    total_limit: usize,
114) -> Result<(), SourceError> {
115    let sparse = sparse_paths(root, missing)?;
116    let entries = index_entries(root, &sparse)?;
117    let selected = entries.iter().collect::<Vec<_>>();
118    for blob in git_blobs_from_objects_bounded(root, selected, total_limit)? {
119        contents.insert(blob.path, ContentId::of(&blob.bytes));
120    }
121    Ok(())
122}

124pub(super) fn generation_state(generations: BTreeMap<RepoPath, [u8; 56]>) -> [u8; 32] {
125    let mut generation_manifest = Vec::new();
126    for (path, state) in generations {
127        generation_manifest.extend_from_slice(&(path.as_bytes().len() as u64).to_be_bytes());
128        generation_manifest.extend_from_slice(path.as_bytes());
129        generation_manifest.extend_from_slice(&state);
130    }
131    hash_domain("git-worktree-generation/v1", &[&generation_manifest])
132}

145pub(super) fn worktree_state(
146    paths: &BTreeSet<RepoPath>,
147    contents: &BTreeMap<RepoPath, ContentId>,
148    include: fn(&RepoPath) -> bool,
149) -> [u8; 32] {
150    let mut manifest = Vec::new();
151    for path in paths {
152        manifest.extend_from_slice(&(path.as_bytes().len() as u64).to_be_bytes());
153        manifest.extend_from_slice(path.as_bytes());
154        if !include(path) {
155            manifest.push(0);
156        } else if let Some(content) = contents.get(path) {
157            manifest.push(1);
158            manifest.extend_from_slice(content.as_bytes());
159        } else {
160            manifest.push(2);
161        }
162    }
163    hash_domain("git-worktree-content/v1", &[&manifest])
164}

166pub(super) fn file_state(path: &Path) -> std::io::Result<[u8; 32]> {
167    let bytes = fs::read(path)?;
168    let metadata = fs::metadata(path)?;
169    Ok(hash_domain(
170        "git-file-state/v1",
171        &[&bytes, &metadata_state(&metadata)],
172    ))
173}

175pub(super) fn metadata_state(metadata: &fs::Metadata) -> [u8; 56] {
176    let mut state = [0; 56];
177    state[0..8].copy_from_slice(&metadata.dev().to_be_bytes());
178    state[8..16].copy_from_slice(&metadata.ino().to_be_bytes());
179    state[16..24].copy_from_slice(&metadata.len().to_be_bytes());
180    state[24..32].copy_from_slice(&metadata.mtime().to_be_bytes());
181    state[32..40].copy_from_slice(&metadata.mtime_nsec().to_be_bytes());
182    state[40..48].copy_from_slice(&metadata.ctime().to_be_bytes());
183    state[48..56].copy_from_slice(&metadata.ctime_nsec().to_be_bytes());
184    state
185}

187pub(super) fn read_regular_no_follow(root: &Path, path: &RepoPath) -> Result<Vec<u8>, SourceError> {
188    let relative = path.to_path_buf();
189    validate_parent_components(root, &relative, path)?;
190    let absolute = root.join(relative);
191    let before = fs::symlink_metadata(&absolute).map_err(io_error)?;
192    validate_regular_metadata(&before, path)?;
193    let file = open_no_follow(&absolute)?;
194    let opened = file.metadata().map_err(io_error)?;
195    ensure_same_file(&before, &opened, path)?;
196    let bytes = read_bounded_file(file, &opened, path)?;
197    let after = fs::metadata(&absolute).map_err(io_error)?;
198    ensure_read_stable(&opened, &after, path)?;
199    Ok(bytes)
200}

202fn validate_parent_components(
203    root: &Path,
204    relative: &Path,
205    path: &RepoPath,
206) -> Result<(), SourceError> {
207    let mut parent = root.to_path_buf();
208    if let Some(components) = relative.parent() {
209        for component in components.components() {
210            parent.push(component);
211            let metadata = fs::symlink_metadata(&parent).map_err(io_error)?;
212            if metadata.file_type().is_symlink() || !metadata.is_dir() {
213                return Err(SourceError::Denied(format!(
214                    "symlinked or non-directory parent for {path}"
215                )));
216            }
217        }
218    }
219    Ok(())
220}
221
222fn validate_regular_metadata(metadata: &fs::Metadata, path: &RepoPath) -> Result<(), SourceError> {
223    if metadata.file_type().is_symlink() || !metadata.is_file() {
224        return Err(SourceError::Denied(format!("{path} is not a regular file")));
225    }
226    if metadata.len() > MAX_SOURCE_BYTES as u64 {
227        return Err(SourceError::Incomplete(format!(
228            "{path} exceeds {MAX_SOURCE_BYTES} bytes"
229        )));
230    }
231    Ok(())
232}

234fn open_no_follow(path: &Path) -> Result<fs::File, SourceError> {
235    OpenOptions::new()
236        .read(true)
237        .custom_flags(libc::O_NOFOLLOW)
238        .open(path)
239        .map_err(io_error)
240}

242fn ensure_same_file(
243    before: &fs::Metadata,
244    opened: &fs::Metadata,
245    path: &RepoPath,
246) -> Result<(), SourceError> {
247    if (before.dev(), before.ino()) != (opened.dev(), opened.ino()) {
248        return Err(SourceError::Changed(format!("{path} changed before open")));
249    }
250    Ok(())
251}

253fn read_bounded_file(
254    file: fs::File,
255    metadata: &fs::Metadata,
256    path: &RepoPath,
257) -> Result<Vec<u8>, SourceError> {
258    let capacity = usize::try_from(metadata.len()).map_err(|_| {
259        SourceError::Incomplete(format!("{path} size is not representable on this platform"))
260    })?;
261    let mut bytes = Vec::with_capacity(capacity);
262    file.take((MAX_SOURCE_BYTES + 1) as u64)
263        .read_to_end(&mut bytes)
264        .map_err(io_error)?;
265    if bytes.len() > MAX_SOURCE_BYTES {
266        return Err(SourceError::Incomplete(format!(
267            "{path} exceeds {MAX_SOURCE_BYTES} bytes"
268        )));
269    }
270    Ok(bytes)
271}
272
273fn ensure_read_stable(
274    opened: &fs::Metadata,
275    after: &fs::Metadata,
276    path: &RepoPath,
277) -> Result<(), SourceError> {
278    if read_state(opened) != read_state(after) {
279        return Err(SourceError::Changed(format!("{path} changed while read")));
280    }
281    Ok(())
282}

284fn read_state(metadata: &fs::Metadata) -> (u64, u64, u64, i64, i64) {
285    (
286        metadata.dev(),
287        metadata.ino(),
288        metadata.len(),
289        metadata.mtime(),
290        metadata.mtime_nsec(),
291    )
