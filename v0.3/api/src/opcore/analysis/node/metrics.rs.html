8pub(super) fn metric_diagnostics(
9    tokens: &[LexToken],
10    delimiters: &Delimiters,
11    functions: &[FunctionFact],
12    limits: &RuleLimits,
13    cancel: &CancelToken,
14) -> Result<Vec<RawDiagnostic>, AnalysisError> {
15    let owners = function_owners(tokens.len(), functions);
16    let complexities = function_complexities(tokens, functions, &owners);
17    let nestings = function_nestings(tokens, delimiters, functions, &owners);
18    let mut diagnostics = Vec::new();
19    let mut entity_occurrences = BTreeMap::<String, usize>::new();
20    for (function_index, function) in functions.iter().enumerate() {
21        if function_index.is_multiple_of(CANCELLATION_POLL_INTERVAL) {
22            check_cancel(cancel)?;
23        }
24        let range = byte_range_for_tokens(tokens, function.start, function.end);
25        let lines = range
26            .end
27            .line
28            .saturating_sub(range.start.line)
29            .saturating_add(1);
30        let params = function.parameter_count;
31        let complexity = complexities[function_index];
32        let nesting = nestings[function_index];
33        let base_entity = format!("function:{}", function.name);
34        let occurrence = entity_occurrences.entry(base_entity.clone()).or_default();
35        let entity = format!("{base_entity}:{}", *occurrence);
36        *occurrence = occurrence.saturating_add(1);
37
38        maybe_metric(
39            &mut diagnostics,
40            MetricDiagnostic {
41                rule_id: "complexity.max-function-lines",
42                label: "function lines",
43                actual: lines,
44                limit: limits.max_function_lines,
45                range,
46                entity: &entity,
47            },
48        );
49        maybe_metric(
50            &mut diagnostics,
51            MetricDiagnostic {
52                rule_id: "complexity.max-parameters",
53                label: "function parameters",
54                actual: params,
55                limit: limits.max_parameters,
56                range,
57                entity: &entity,
58            },
59        );
60        maybe_metric(
61            &mut diagnostics,
62            MetricDiagnostic {
63                rule_id: "complexity.max-nesting",
64                label: "function nesting",
65                actual: nesting,
66                limit: limits.max_nesting,
67                range,
68                entity: &entity,
69            },
70        );
71        maybe_metric(
72            &mut diagnostics,
73            MetricDiagnostic {
74                rule_id: "complexity.max-cyclomatic-complexity",
75                label: "cyclomatic complexity",
76                actual: complexity,
77                limit: limits.max_cyclomatic_complexity,
78                range,
79                entity: &entity,
80            },
81        );
82    }
83    Ok(diagnostics)
84}

86fn function_owners(token_count: usize, functions: &[FunctionFact]) -> Vec<Option<usize>> {
87    let mut starts = vec![Vec::new(); token_count];
88    for (index, function) in functions.iter().enumerate() {
89        if function.start < token_count {
90            starts[function.start].push(index);
91        }
92    }
93    for indexes in &mut starts {
94        indexes.sort_by_key(|index| std::cmp::Reverse(functions[*index].end));
95    }
96    let mut owners = vec![None; token_count];
97    let mut stack = Vec::<usize>::new();
98    for token in 0..token_count {
99        while stack
100            .last()
101            .is_some_and(|index| functions[*index].end < token)
102        {
103            let _ = stack.pop();
104        }
105        stack.extend(starts[token].iter().copied());
106        owners[token] = stack.last().copied();
107    }
108    owners
109}

111fn function_complexities(
112    tokens: &[LexToken],
113    functions: &[FunctionFact],
114    owners: &[Option<usize>],
115) -> Vec<u32> {
116    let mut complexities = vec![1u32; functions.len()];
117    let type_only = type_only_token_mask(tokens, functions, owners);
118    for (index, token) in tokens.iter().enumerate() {
119        let Some(owner) = owners[index] else {
120            continue;
121        };
122        if !type_only[index]
123            && in_function_body(functions[owner].body, index)
124            && matches!(
125                token.kind,
126                Kind::If
127                    | Kind::For
128                    | Kind::While
129                    | Kind::Case
130                    | Kind::Catch
131                    | Kind::Question
132                    | Kind::Amp2
133                    | Kind::Pipe2
134                    | Kind::Question2
135            )
136        {
137            complexities[owner] = complexities[owner].saturating_add(1);
138        }
139    }
140    complexities
141}

143fn type_only_token_mask(
144    tokens: &[LexToken],
145    functions: &[FunctionFact],
146    owners: &[Option<usize>],
147) -> Vec<bool> {
148    let mut mask = vec![false; tokens.len()];
149    for (start, token) in tokens.iter().enumerate() {
150        let Some(owner) = owners[start] else {
151            continue;
152        };
153        if token.kind != Kind::Type
154            || !in_function_body(functions[owner].body, start)
155            || !tokens
156                .get(start + 1)
157                .is_some_and(|next| next.kind == Kind::Ident)
158        {
159            continue;
160        }
161        for index in start..tokens.len() {
162            if owners[index] != Some(owner) {
163                break;
164            }
165            mask[index] = true;
166            if tokens[index].kind == Kind::Semicolon {
167                break;
168            }
169        }
170    }
171    mask
172}

174fn function_nestings(
175    tokens: &[LexToken],
176    delimiters: &Delimiters,
177    functions: &[FunctionFact],
178    owners: &[Option<usize>],
179) -> Vec<u32> {
180    let mut intervals = nesting_intervals(tokens, delimiters, functions, owners);
181    intervals
182        .iter_mut()
183        .map(|function_intervals| maximum_nesting(function_intervals))
184        .collect()
185}
186
187fn nesting_intervals(
188    tokens: &[LexToken],
189    delimiters: &Delimiters,
190    functions: &[FunctionFact],
191    owners: &[Option<usize>],
192) -> Vec<Vec<(usize, usize)>> {
193    let mut intervals = vec![Vec::new(); functions.len()];
194    for (index, token) in tokens.iter().enumerate() {
195        let Some(owner) = owners[index] else {
196            continue;
197        };
198        if !in_function_body(functions[owner].body, index)
199            || !is_nesting_keyword(token.kind)
200            || (token.kind == Kind::Else
201                && tokens
202                    .get(index + 1)
203                    .is_some_and(|next| next.kind == Kind::If))
204        {
205            continue;
206        }
207        let function_end = functions[owner].body.map_or(index, |(_, end)| end);
208        if let Some(start) = control_statement_start(tokens, delimiters, index, function_end)
209            && owners[start] == Some(owner)
210            && let Some(interval) = nesting_interval(tokens, delimiters, index, start, function_end)
211        {
212            intervals[owner].push(interval);
213        }
214    }
215    intervals
216}

218fn nesting_interval(
219    tokens: &[LexToken],
220    delimiters: &Delimiters,
221    keyword: usize,
222    start: usize,
223    function_end: usize,
224) -> Option<(usize, usize)> {
225    if tokens[start].kind == Kind::LCurly {
226        return delimiters.brace_open[start].map(|close| (start, close));
227    }
228    Some((
229        keyword,
230        unbraced_statement_end(tokens, delimiters, start, function_end),
231    ))
232}

234fn maximum_nesting(intervals: &mut [(usize, usize)]) -> u32 {
235    intervals.sort_unstable();
236    let mut active = Vec::<usize>::new();
237    let mut maximum = 0u32;
238    for (open, close) in intervals {
239        while active
240            .last()
241            .is_some_and(|active_close| *active_close <= *open)
242        {
243            let _ = active.pop();
244        }
245        active.push(*close);
246        maximum = maximum.max(u32::try_from(active.len()).unwrap_or(u32::MAX));
247    }
248    maximum
249}

251fn in_function_body(body: Option<(usize, usize)>, token: usize) -> bool {
252    body.is_some_and(|(open, close)| open < token && token < close)
253}

255const fn is_nesting_keyword(kind: Kind) -> bool {
256    matches!(
257        kind,
258        Kind::If
259            | Kind::Else
260            | Kind::For
261            | Kind::While
262            | Kind::Switch
263            | Kind::Catch
264            | Kind::Try
265            | Kind::Finally
266            | Kind::Do
267            | Kind::With
268    )
269}

271fn control_statement_start(
272    tokens: &[LexToken],
273    delimiters: &Delimiters,
274    keyword: usize,
275    function_end: usize,
276) -> Option<usize> {
277    let mut index = keyword + 1;
278    if matches!(
279        tokens[keyword].kind,
280        Kind::If | Kind::For | Kind::While | Kind::Switch | Kind::Catch | Kind::With
281    ) && tokens
282        .get(index)
283        .is_some_and(|token| token.kind == Kind::LParen)
284    {
285        index = delimiters.paren_open[index]?.saturating_add(1);
286    }
287    (index < function_end).then_some(index)
288}

290fn unbraced_statement_end(
291    tokens: &[LexToken],
292    delimiters: &Delimiters,
293    start: usize,
294    function_end: usize,
295) -> usize {
296    let mut index = start;
297    while index < function_end {
298        match tokens[index].kind {
299            Kind::LParen => index = delimiters.paren_open[index].unwrap_or(index),
300            Kind::LBrack => index = delimiters.bracket_open[index].unwrap_or(index),
301            Kind::LCurly => index = delimiters.brace_open[index].unwrap_or(index),
302            Kind::Semicolon => return index,
303            _ => {}
304        }
305        index = index.saturating_add(1);
306    }
307    function_end
308}

310#[derive(Clone, Copy)]
311struct MetricDiagnostic<'a> {
312    rule_id: &'a str,
313    label: &'a str,
314    actual: u32,
315    limit: u32,
316    range: SourceRange,
317    entity: &'a str,
318}

320fn maybe_metric(diagnostics: &mut Vec<RawDiagnostic>, metric: MetricDiagnostic<'_>) {
321    if metric.actual <= metric.limit {
322        return;
323    }
324    let evidence = metric_evidence(parser_evidence(), metric.actual, metric.limit);
325    diagnostics.push(RawDiagnostic {
326        rule_id: metric.rule_id.into(),
327        severity: Severity::Warning,
328        message: format!(
329            "{}: {}; configured maximum is {}",
330            metric.label, metric.actual, metric.limit
331        ),
332        range: Some(metric.range),
333        entity_key: metric.entity.into(),
334        cause_key: format!("limit:{}", metric.limit),
335        evidence,
336    });
337}

339fn byte_range_for_tokens(tokens: &[LexToken], start: usize, end: usize) -> SourceRange {
340    let start_token = &tokens[start];
341    let end_token = &tokens[end];
342    SourceRange {
343        start: start_token.start_position,
344        end: end_token.end_position,
345    }
