81#[must_use]
82pub fn detect_language(path: &RepoPath) -> Option<(Language, String)> {
83    path.as_utf8()?;
84    language_rule(path.as_bytes()).map(|(language, mode)| {
85        let mode = match language {
86            Language::Shell => format!("{mode}:{}", std::env::consts::OS),
87            Language::Go => format!("{mode}:{}:{}", go_target_os(), go_target_arch()),
88            _ => mode.into(),
89        };
90        (language, mode)
91    })
92}
93
94#[must_use]
95pub fn resolve_language(path: &RepoPath, bytes: &[u8]) -> Option<(Language, String)> {
96    let (language, mode) = detect_language(path)?;
97    if language != Language::Shell {
98        return Some((language, mode));
99    }
100    let dialect = shell_shebang_dialect(bytes).unwrap_or_else(|| {
101        if has_extension(path.as_bytes(), b".bash") {
102            "bash"
103        } else {
104            "sh"
105        }
106    });
107    Some((
108        language,
109        format!("shell:{dialect}:{}", std::env::consts::OS),
110    ))
111}

113#[must_use]
114pub fn is_source_candidate(path: &RepoPath) -> bool {
115    if language_rule(path.as_bytes()).is_some() {
116        return true;
117    }
118    UNSUPPORTED_SOURCE_EXTENSIONS
119        .iter()
120        .any(|extension| has_extension(path.as_bytes(), extension))
121}

141fn language_rule(path: &[u8]) -> Option<(Language, &'static str)> {
142    let filename = path.rsplit(|byte| *byte == b'/').next().unwrap_or(path);
143    if filename.eq_ignore_ascii_case(b".terraform.lock.hcl") {
144        return Some((Language::Hcl, "hcl:terraform-lock:native"));
145    }
146    for (name, mode) in [
147        (b"terragrunt.hcl".as_slice(), "hcl:terragrunt:native"),
148        (b"terragrunt.hcl.json".as_slice(), "hcl:terragrunt:json"),
149        (
150            b"terragrunt.stack.hcl".as_slice(),
151            "hcl:terragrunt-stack:native",
152        ),
153        (
154            b"terragrunt.values.hcl".as_slice(),
155            "hcl:terragrunt-values:native",
156        ),
157    ] {
158        if filename.eq_ignore_ascii_case(name) {
159            return Some((Language::Hcl, mode));
160        }
161    }
162    if has_tfvars_suffix(filename) {
163        let mode = if has_extension(filename, b".json") {
164            "hcl:terraform-vars:json"
165        } else {
166            "hcl:terraform-vars:native"
167        };
168        return Some((Language::Hcl, mode));
169    }
170    language_rules()
171        .iter()
172        .find(|(suffix, _, _)| has_extension(path, suffix.as_bytes()))
173        .map(|(_, language, mode)| (*language, *mode))
174}

176pub(crate) fn strip_language_suffix(path: &str, language: Language) -> Option<&str> {
177    language_rules()
178        .iter()
179        .find(|(suffix, candidate, _)| {
180            *candidate == language && has_extension(path.as_bytes(), suffix.as_bytes())
181        })
182        .map(|(suffix, _, _)| &path[..path.len() - suffix.len()])
183}

185pub(crate) fn has_extension(path: &[u8], extension: &[u8]) -> bool {
186    path.len() >= extension.len()
187        && path[path.len() - extension.len()..].eq_ignore_ascii_case(extension)
188}
189
190fn has_tfvars_suffix(filename: &[u8]) -> bool {
191    filename
192        .windows(b".tfvars".len())
193        .position(|window| window.eq_ignore_ascii_case(b".tfvars"))
194        .is_some_and(|index| {
195            let remainder = &filename[index + b".tfvars".len()..];
196            remainder.is_empty() || remainder.starts_with(b".")
197        })
198}

200fn shell_shebang_dialect(bytes: &[u8]) -> Option<&'static str> {
201    let line_end = bytes
202        .iter()
203        .position(|byte| matches!(byte, b'\r' | b'\n'))
204        .unwrap_or(bytes.len());
205    let command = bytes.get(..line_end)?.strip_prefix(b"#!")?;
206    let mut words = command
207        .split(u8::is_ascii_whitespace)
208        .filter(|word| !word.is_empty());
209    let interpreter = shell_command_name(words.next()?)?;
210    if interpreter == b"env" {
211        env_shell_dialect(&mut words)
212    } else {
213        known_shell_dialect(interpreter)
214    }
215}
216
217fn env_shell_dialect<'a>(words: &mut impl Iterator<Item = &'a [u8]>) -> Option<&'static str> {
218    while let Some(word) = words.next() {
219        if env_option_takes_value(word) {
220            words.next()?;
221            continue;
222        }
223        if env_option_without_value(word) || env_assignment(word) {
224            continue;
225        }
226        if word.starts_with(b"-") {
227            return None;
228        }
229        return known_shell_dialect(shell_command_name(word)?);
230    }
231    None
232}
233
234fn env_option_takes_value(word: &[u8]) -> bool {
235    matches!(word, b"-u" | b"--unset")
236}
237
238fn env_option_without_value(word: &[u8]) -> bool {
239    matches!(
240        word,
241        b"-S" | b"--split-string" | b"-i" | b"--ignore-environment" | b"--"
242    ) || word.starts_with(b"--unset=")
243}
244
245fn env_assignment(word: &[u8]) -> bool {
246    word.contains(&b'=')
247}
248
249fn known_shell_dialect(interpreter: &[u8]) -> Option<&'static str> {
250    match interpreter {
251        b"bash" => Some("bash"),
252        b"sh" | b"dash" => Some("sh"),
253        _ => None,
254    }
255}
256
257fn shell_command_name(word: &[u8]) -> Option<&[u8]> {
258    let name = word.rsplit(|byte| matches!(byte, b'/' | b'\\')).next()?;
259    (!name.is_empty()).then_some(name)
260}

303fn go_target_os() -> &'static str {
304    match std::env::consts::OS {
305        "macos" => "darwin",
306        value => value,
307    }
308}
309
310fn go_target_arch() -> &'static str {
311    match std::env::consts::ARCH {
312        "x86_64" => "amd64",
313        "aarch64" => "arm64",
314        "x86" => "386",
315        value => value,
316    }

324    fn classifies_supported_unsupported_and_non_source_suffixes() {
325        for path in [
326            "a.js",
327            "a.jsx",
328            "a.mjs",
329            "a.cjs",
330            "a.ts",
331            "a.tsx",
332            "a.mts",
333            "a.cts",
334            "a.d.ts",
335            "a.d.mts",
336            "a.d.cts",
337            "a.rs",
338            "a.go",
339            "a_test.go",
340            "a.py",
341            "a.pyi",
342            "main.tf",
343            "main.tofu",
344            "vars.tfvars",
345            "module.tftest.hcl",
346            "terragrunt.hcl",
347            "generic.hcl",
348            "script.sh",
349            "script.bash",
350            "schema.proto",
351        ] {
352            assert!(
353                detect_language(&RepoPath::from_protocol(path).unwrap()).is_some(),
354                "{path}"
355            );
356        }
357        let go = RepoPath::from_protocol("a.go").unwrap();
358        assert_eq!(detect_language(&go).unwrap().0, Language::Go);
359        assert!(is_source_candidate(&go));
360        assert!(is_dependency_source_candidate(
361            &RepoPath::from_protocol("script.sh").unwrap()
362        ));
363        assert!(is_source_candidate(
364            &RepoPath::from_protocol("script.sh").unwrap()
365        ));
366        let protobuf = RepoPath::from_protocol("schema.proto").unwrap();
367        assert_eq!(detect_language(&protobuf).unwrap().0, Language::Protobuf);
368        for path in ["src/Main.java", "src/main.zig"] {
369            let path = RepoPath::from_protocol(path).unwrap();
370            assert_eq!(detect_language(&path), None);
371            assert!(is_source_candidate(&path));
372            assert!(is_dependency_source_candidate(&path));
373        }
374        assert!(!is_source_candidate(
375            &RepoPath::from_protocol("README.md").unwrap()
376        ));
377        assert!(!is_dependency_source_candidate(
378            &RepoPath::from_protocol("README.md").unwrap()
379        ));
380        let go_module = RepoPath::from_protocol("nested/go.mod").unwrap();
381        assert!(is_dependency_capture_candidate(&go_module));
382        assert!(!is_dependency_source_candidate(&go_module));
383        assert!(is_go_module_file(&go_module));
384
385        let mixed_case = RepoPath::from_protocol("SRC/COMPONENT.D.TS").unwrap();
386        assert_eq!(
387            detect_language(&mixed_case),
388            Some((Language::TypeScript, "typescript:declaration".into()))
389        );
390        let non_utf8 = RepoPath::new(b"src/\xffmodule.RS".to_vec()).unwrap();
391        assert_eq!(detect_language(&non_utf8), None);
392        assert!(is_source_candidate(&non_utf8));
393
394        assert_eq!(
395            strip_language_suffix("src/COMPONENT.D.TS", Language::TypeScript),
396            Some("src/COMPONENT")
397        );
398    }
