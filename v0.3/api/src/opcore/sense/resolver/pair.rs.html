11pub(crate) struct PairInput<'a> {
12    pub(crate) before: &'a SourceSnapshot,
13    pub(crate) after: &'a SourceSnapshot,
14    pub(crate) file_facts: &'a facts::FactMap,
15    pub(crate) parser_options: &'a [u8],
16    pub(crate) interface_paths: &'a BTreeSet<RepoPath>,
17    pub(crate) go_modules: &'a super::super::engine::GoModuleViews,
18}

20pub(crate) fn project_pair(input: PairInput<'_>) -> (Projection, Projection, BTreeSet<RepoPath>) {
21    let before_index = ResolutionIndex::new(
22        input.before,
23        input.file_facts,
24        input.parser_options,
25        &input.go_modules.before,
26    );
27    let metadata_changed = input.go_modules.before != input.go_modules.after;
28    let before_interface_paths =
29        interface_paths(&before_index, input.interface_paths, metadata_changed);
30    let before_projection = project_with_index(
31        input.before,
32        &context(&input, &before_index, &before_interface_paths),
33    );
34    if views_match(&input) {
35        return (
36            before_projection.clone(),
37            before_projection,
38            before_interface_paths,
39        );
40    }
41    project_changed(
42        input,
43        &before_index,
44        &before_interface_paths,
45        before_projection,
46        metadata_changed,
47    )
48}
49
50fn project_changed(
51    input: PairInput<'_>,
52    before_index: &ResolutionIndex,
53    before_interface_paths: &BTreeSet<RepoPath>,
54    before_projection: Projection,
55    metadata_changed: bool,
56) -> (Projection, Projection, BTreeSet<RepoPath>) {
57    let incremental_safe = incremental_safe(&input, before_index);
58    let after_index = (!incremental_safe).then(|| {
59        ResolutionIndex::new(
60            input.after,
61            input.file_facts,
62            input.parser_options,
63            &input.go_modules.after,
64        )
65    });
66    let comparison_index = after_index.as_ref().unwrap_or(before_index);
67    let after_interface_paths =
68        interface_paths(comparison_index, input.interface_paths, metadata_changed);
69    let logical_changed_paths = before_interface_paths
70        .union(&after_interface_paths)
71        .cloned()
72        .collect();
73    let resolution_context = context(&input, comparison_index, &after_interface_paths);
74    let after_projection = if full_projection_required(incremental_safe, &before_projection) {
75        project_with_index(input.after, &resolution_context)
76    } else {
77        update_changed_importers(
78            input.before,
79            input.after,
80            before_projection.clone(),
81            &resolution_context,
82            &after_interface_paths,
83        )
84    };
85    (before_projection, after_projection, logical_changed_paths)
86}

88fn context<'a>(
89    input: &PairInput<'a>,
90    index: &'a ResolutionIndex,
91    interface_paths: &'a BTreeSet<RepoPath>,
92) -> ResolutionContext<'a> {
93    ResolutionContext {
94        file_facts: input.file_facts,
95        parser_options: input.parser_options,
96        index,
97        interface_paths,
98    }
99}

101fn views_match(input: &PairInput<'_>) -> bool {
102    input.before.id() == input.after.id() && input.go_modules.before == input.go_modules.after
103}
104
105fn incremental_safe(input: &PairInput<'_>, before_index: &ResolutionIndex) -> bool {
106    can_incrementally_project(input.before, input.after, input.interface_paths)
107        && rust::index_is_stable(input.before, input.after, input.interface_paths)
108        && !go_sensitive(input, before_index)
109}
110
111fn go_sensitive(input: &PairInput<'_>, before_index: &ResolutionIndex) -> bool {
112    before_index.go.has_go()
113        || input
114            .after
115            .files()
116            .any(|file| file.language == Language::Go)
117        || !input.go_modules.before.is_empty()
118        || !input.go_modules.after.is_empty()
119}
120
121fn full_projection_required(incremental_safe: bool, before: &Projection) -> bool {
122    !incremental_safe
123        || before.coverage.edges_truncated
124        || !before.resolution_gaps.is_empty()
125        || before.resolution_gaps_truncated
126        || before.interface_coverage.selectors_truncated
127}
128
129fn interface_paths(
130    index: &ResolutionIndex,
131    paths: &BTreeSet<RepoPath>,
132    metadata_changed: bool,
133) -> BTreeSet<RepoPath> {
134    let mut logical = paths
135        .iter()
136        .map(|path| index.go.logical_path_for(path))
137        .collect::<BTreeSet<_>>();
138    if metadata_changed {
139        logical.extend(index.go.logical_paths().cloned());
140    }
141    logical
142}

144pub(super) fn can_incrementally_project(
145    before: &SourceSnapshot,
146    after: &SourceSnapshot,
147    changed_paths: &BTreeSet<RepoPath>,
148) -> bool {
149    before.len() == after.len()
150        && before.files().zip(after.files()).all(|(before, after)| {
151            before.path == after.path
152                && before.language == after.language
153                && (before.content_id == after.content_id || changed_paths.contains(&before.path))
154        })
155}
