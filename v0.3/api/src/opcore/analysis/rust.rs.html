38/// Analyzes one Rust source file without executing repository code.
39///
40/// # Errors
41///
42/// Returns an error when analysis is cancelled, the edition is unsupported, or a hard
43/// parser-safety bound is exceeded.
44pub fn analyze(
45    file: &SourceFile,
46    limits: &RuleLimits,
47    cancel: &CancelToken,
48) -> Result<FileFacts, AnalysisError> {
49    check_cancel(cancel)?;
50    validate_rust_language(file)?;
51    let editions = parse_editions(&file.language_mode)?;
52    let default_edition = editions[0];
53    let (source_bytes, byte_base) = source_without_bom(&file.bytes);
54    let lines = LineIndex::new(source_bytes, byte_base);
55    let text = match std::str::from_utf8(source_bytes) {
56        Ok(text) => text,
57        Err(error) => {
58            return Ok(invalid_utf8_facts(
59                &lines,
60                source_bytes,
61                error,
62                default_edition,
63            ));
64        }
65    };
66    let fingerprint_tokens = preflight_structure(text, cancel)?;
67    analyze_valid_rust(
68        RustAnalysisInput {
69            text,
70            lines: &lines,
71            limits,
72            cancel,
73            fingerprint_tokens: &fingerprint_tokens,
74        },
75        editions,
76    )
77}

88fn analyze_valid_rust(
89    input: RustAnalysisInput<'_>,
90    editions: Vec<Edition>,
91) -> Result<FileFacts, AnalysisError> {
92    let (parse, edition, mut syntax_errors) = select_parse(input.text, editions, input.cancel)?;
93    check_cancel(input.cancel)?;
94    sort_syntax_errors(&mut syntax_errors);
95    if let Some(facts) = syntax_facts(
96        &syntax_errors,
97        input.lines,
98        input.text,
99        edition,
100        input.cancel,
101    )? {
102        return Ok(facts);
103    }
104    let tree = parse.tree();
105    let root = tree.syntax();
106    reject_nightly_features(root)?;
107    let dependencies = dependencies::extract(root, input.cancel)?;
108    let (diagnostics, callable_fingerprints) = metric_diagnostics(
109        root,
110        RustMetricContext {
111            text: input.text,
112            lines: input.lines,
113            edition,
114            limits: input.limits,
115            cancel: input.cancel,
116        },
117        input.fingerprint_tokens,
118    )?;
119    let region_fingerprints =
120        rust_region_fingerprints(input.fingerprint_tokens, input.text, input.lines);
121    check_cancel(input.cancel)?;
122    Ok(parser_facts(
123        diagnostics,
124        callable_fingerprints,
125        region_fingerprints,
126        dependencies,
127    ))
128}

130fn sort_syntax_errors(errors: &mut [ra_ap_syntax::SyntaxError]) {
131    errors.sort_by(|left, right| {
132        text_range_key(left.range())
133            .cmp(&text_range_key(right.range()))
134            .then_with(|| left.to_string().cmp(&right.to_string()))
135    });
136}

138fn syntax_facts(
139    errors: &[ra_ap_syntax::SyntaxError],
140    lines: &LineIndex,
141    text: &str,
142    edition: Edition,
143    cancel: &CancelToken,
144) -> Result<Option<FileFacts>, AnalysisError> {
145    if errors.is_empty() {
146        return Ok(None);
147    }
148    let diagnostics = syntax_diagnostics(errors, lines, text.as_bytes(), edition, cancel)?;
149    Ok(Some(parser_facts_with_status(
150        diagnostics,
151        DependencyExtractionStatus::ParserFailed,
152        parser_failed(),
153        region_parser_failed(),
154    )))
155}

235fn select_parse(
236    text: &str,
237    editions: Vec<Edition>,
238    cancel: &CancelToken,
239) -> Result<
240    (
241        ra_ap_syntax::Parse<RaSourceFile>,
242        Edition,
243        Vec<ra_ap_syntax::SyntaxError>,
244    ),
245    AnalysisError,
246> {
247    check_cancel(cancel)?;
248    let mut editions = editions.into_iter();
249    let Some(mut edition) = editions.next() else {
250        return Err(AnalysisError::Unsupported(
251            "Rust language mode resolved no supported edition".into(),
252        ));
253    };
254    let mut parse = RaSourceFile::parse(text, edition);
255    let mut errors = syntax_errors_for_edition(&parse, edition);
256    for candidate in editions {
257        if errors.is_empty() {
258            break;
259        }
260        let candidate_parse = RaSourceFile::parse(text, candidate);
261        let candidate_errors = syntax_errors_for_edition(&candidate_parse, candidate);
262        if candidate_errors.is_empty() {
263            edition = candidate;
264            parse = candidate_parse;
265            errors = candidate_errors;
266            break;
267        }
268    }
269    Ok((parse, edition, errors))
270}

272fn uses_nightly_feature(root: &SyntaxNode) -> bool {
273    root.descendants()
274        .filter_map(ast::Attr::cast)
275        .any(|attribute| {
276            attribute.kind().is_inner() && attribute.simple_name().as_deref() == Some("feature")
277        })
278}

280fn syntax_errors_for_edition(
281    parse: &ra_ap_syntax::Parse<RaSourceFile>,
282    edition: Edition,
283) -> Vec<ra_ap_syntax::SyntaxError> {
284    parse
285        .errors()
286        .into_iter()
287        .filter(|error| {
288            edition != Edition::Edition2018 || !error.to_string().contains("unknown literal prefix")
289        })
290        .collect()
291}

293fn preflight_structure(
294    source: &str,
295    cancel: &CancelToken,
296) -> Result<Vec<RustFingerprintToken>, AnalysisError> {
297    let mut structure = RustStructure::default();
298    let mut fingerprint_tokens = Vec::new();
299    let mut offset = 0usize;
300    for (index, token) in tokenize(source, FrontmatterAllowed::No).enumerate() {
301        if index.is_multiple_of(CANCELLATION_POLL_INTERVAL) {
302            check_cancel(cancel)?;
303        }
304        let start = offset;
305        offset = offset.saturating_add(usize::try_from(token.len).unwrap_or(usize::MAX));
306        if !matches!(
307            token.kind,
308            TokenKind::Whitespace | TokenKind::LineComment { .. } | TokenKind::BlockComment { .. }
309        ) {
310            fingerprint_tokens.push(RustFingerprintToken { start, end: offset });
311        }
312        if structure.consume(token.kind) {
313            structure.validate()?;
314        }
315    }
316    Ok(fingerprint_tokens)
317}

346#[derive(Default)]
347struct RustStructure {
348    delimiters: usize,
349    angles: usize,
350    unary: usize,
351    expression_nodes: usize,
352    brace_expressions: Vec<usize>,
353}
354
355impl RustStructure {
356    fn consume(&mut self, kind: TokenKind) -> bool {
357        match kind {
358            TokenKind::OpenParen | TokenKind::OpenBrace | TokenKind::OpenBracket => {
359                self.open(kind);
360            }
361            TokenKind::CloseParen | TokenKind::CloseBrace | TokenKind::CloseBracket => {
362                self.close(kind);
363            }
364            TokenKind::Lt => {
365                self.angles = self.angles.saturating_add(1);
366                self.expression_nodes = self.expression_nodes.saturating_add(1);
367            }
368            TokenKind::Gt => {
369                self.angles = self.angles.saturating_sub(1);
370                self.expression_nodes = self.expression_nodes.saturating_add(1);
371            }
372            TokenKind::Plus
373            | TokenKind::Minus
374            | TokenKind::Star
375            | TokenKind::Slash
376            | TokenKind::And
377            | TokenKind::Or
378            | TokenKind::Caret
379            | TokenKind::Percent
380            | TokenKind::Eq
381            | TokenKind::Dot
382            | TokenKind::Question => {
383                self.operator(kind);
384            }
385            TokenKind::Semi | TokenKind::Comma => {
386                self.angles = 0;
387                self.unary = 0;
388                self.expression_nodes = 0;
389            }
390            TokenKind::Bang | TokenKind::Tilde => self.unary = self.unary.saturating_add(1),
391            TokenKind::Whitespace
392            | TokenKind::LineComment { .. }
393            | TokenKind::BlockComment { .. } => return false,
394            _ => self.unary = 0,
395        }
396        true
397    }
398
399    fn open(&mut self, kind: TokenKind) {
400        self.delimiters = self.delimiters.saturating_add(1);
401        if matches!(kind, TokenKind::OpenBrace) {
402            self.brace_expressions.push(self.expression_nodes);
403            self.angles = 0;
404            self.expression_nodes = 0;
405        } else {
406            self.expression_nodes = self.expression_nodes.saturating_add(1);
407        }
408        self.unary = 0;
409    }
410
411    fn close(&mut self, kind: TokenKind) {
412        self.delimiters = self.delimiters.saturating_sub(1);
413        if matches!(kind, TokenKind::CloseBrace) {
414            self.angles = 0;
415            self.expression_nodes = self.brace_expressions.pop().unwrap_or(0);
416        }
417        self.unary = 0;
418    }
419
420    fn operator(&mut self, kind: TokenKind) {
421        self.expression_nodes = self.expression_nodes.saturating_add(1);
422        if matches!(kind, TokenKind::Minus | TokenKind::Star | TokenKind::And) {
423            self.unary = self.unary.saturating_add(1);
424        } else {
425            self.unary = 0;
426        }
427    }
428
429    fn validate(&self) -> Result<(), AnalysisError> {
430        if self.delimiters > MAX_STRUCTURAL_DEPTH
431            || self.angles > MAX_STRUCTURAL_DEPTH
432            || self.unary > MAX_STRUCTURAL_DEPTH
433            || self.expression_nodes > MAX_STRUCTURAL_DEPTH
434        {
435            return Err(AnalysisError::Unsupported(format!(
436                "source structural depth exceeds the {MAX_STRUCTURAL_DEPTH}-level safety limit"
437            )));
438        }
439        Ok(())
440    }
441}

443fn parse_editions(mode: &str) -> Result<Vec<Edition>, AnalysisError> {
444    match mode.trim().to_ascii_lowercase().as_str() {
445        "2018" | "rust2018" | "rust-2018" | "rust:2018" | "edition2018" => {
446            Ok(vec![Edition::Edition2018])
447        }
448        "2021" | "rust2021" | "rust-2021" | "rust:2021" | "edition2021" => {
449            Ok(vec![Edition::Edition2021])
450        }
451        "2024" | "rust2024" | "rust-2024" | "rust:2024" | "edition2024" => {
452            Ok(vec![Edition::Edition2024])
453        }
454        "stable" | "rust:stable" => Ok(vec![
455            Edition::Edition2024,
456            Edition::Edition2021,
457            Edition::Edition2018,
458        ]),
459        _ => Err(AnalysisError::Unsupported(format!(
460            "unsupported Rust language mode '{mode}'; expected stable edition 2018, 2021, or 2024"
461        ))),
462    }
463}

465fn syntax_diagnostics(
466    errors: &[ra_ap_syntax::SyntaxError],
467    lines: &LineIndex,
468    source: &[u8],
469    edition: Edition,
470    cancel: &CancelToken,
471) -> Result<Vec<RawDiagnostic>, AnalysisError> {
472    let truncated = errors.len() > MAX_DIAGNOSTICS;
473    let retained = if truncated {
474        MAX_DIAGNOSTICS.saturating_sub(1)
475    } else {
476        errors.len()
477    };
478    let mut diagnostics = Vec::with_capacity(retained.saturating_add(usize::from(truncated)));
479    let mut message_occurrences = BTreeMap::<String, usize>::new();
480
481    for (index, error) in errors.iter().take(retained).enumerate() {
482        if index.is_multiple_of(CANCELLATION_POLL_INTERVAL) {
483            check_cancel(cancel)?;
484        }
485        let message = stable_message(&error.to_string());
486        let anchor = crate::analysis::line_anchor(source, usize::from(error.range().start()));
487        let occurrence_key = format!("{message}:{anchor}");
488        let occurrence = message_occurrences.entry(occurrence_key).or_default();
489        let entity_key = format!("syntax:{message}:{anchor}:{}", *occurrence);
490        *occurrence = occurrence.saturating_add(1);
491        let mut evidence = parser_evidence(edition);
492        evidence.insert("errorKind".into(), serde_json::json!("parser"));
493        diagnostics.push(RawDiagnostic {
494            rule_id: "rust.syntax".into(),
495            severity: Severity::Error,
496            message: format!("Rust syntax error: {message}"),
497            range: Some(lines.byte_range(
498                usize::from(error.range().start()),
499                usize::from(error.range().end()),
500            )),
501            entity_key,
502            cause_key: format!("parser:{RUST_PARSER_VERSION}:{message}"),
503            evidence,
504        });
505    }
506
507    if truncated {
508        let mut evidence = parser_evidence(edition);
509        evidence.insert("actual".into(), serde_json::json!(errors.len()));
510        evidence.insert("retained".into(), serde_json::json!(retained));
511        evidence.insert("truncated".into(), serde_json::json!(true));
512        diagnostics.push(RawDiagnostic {
513            rule_id: "rust.syntax".into(),
514            severity: Severity::Error,
515            message: format!(
516                "Rust parser produced {} errors; only the first {retained} are reported",
517                errors.len()
518            ),
519            range: None,
520            entity_key: "syntax:truncated".into(),
521            cause_key: format!("parser:{RUST_PARSER_VERSION}:truncated"),
522            evidence,
523        });
524    }
525    check_cancel(cancel)?;
526    Ok(diagnostics)
527}
