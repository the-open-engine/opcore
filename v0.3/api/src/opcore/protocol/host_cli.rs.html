139impl HostEvaluation {
140    pub(crate) fn is_current(&self, repository: &GitRepository) -> Result<bool> {
141        let selected = repository.with_targets(&self.targets);
142        for capture in &self.captures {
143            if !selected.is_native_current(&capture.source)? {
144                return Ok(false);
145            }
146            if !capture.plan.filter.is_fast_only()
147                && capture.plan
148                    != native_capture_plan(
149                        &selected,
150                        &self.view,
151                        &capture.selection,
152                        &self.targets,
153                    )?
154            {
155                return Ok(false);
156            }
157        }
158        Ok(true)
159    }
160}

162pub(super) async fn run(args: CheckArgs, repository: GitRepository) -> Result<()> {
163    let evaluation = match stable_report(&args, &repository).await {
164        Ok(evaluation) => evaluation,
165        Err(error) => {
166            emit_host_error(args.json, "evaluation_failed", &error)?;
167            return Err(error);
168        }
169    };
170    if args.json {
171        println!("{}", serde_json::to_string(&evaluation.report)?);
172    } else {
173        print!("{}", evaluation.human);
174    }
175    let decision = match evaluation.report["decision"].as_str() {
176        Some("allow") => Decision::Allow,
177        Some("deny") => Decision::Deny,
178        _ => Decision::Indeterminate,
179    };
180    enforce(decision, args.advisory)
181}

183fn emit_host_error(json_output: bool, code: &str, error: &anyhow::Error) -> Result<()> {
184    if !json_output {
185        return Ok(());
186    }
187    let rendered = serde_json::to_string(&json!({
188        "schema": "opcore.host-check-error.v1",
189        "decision": "indeterminate",
190        "code": code,
191        "message": format!("{error:#}"),
192    }))?;
193    anyhow::ensure!(
194        rendered.len() <= MAX_ASSESSMENT_BYTES,
195        "host error report exceeds {MAX_ASSESSMENT_BYTES} bytes"
196    );
197    println!("{rendered}");
198    Ok(())
199}

201async fn stable_report(args: &CheckArgs, repository: &GitRepository) -> Result<HostEvaluation> {
202    for _ in 0..2 {
203        let policy = PolicySnapshot::capture(repository, args.config_view(), args.workflow)
204            .context("load repository configuration for provider check")?;
205        let evaluation = evaluate(args, repository, &policy).await?;
206        if evaluation.is_current(repository)? && policy.is_current()? {
207            return Ok(evaluation);
208        }
209    }
210    anyhow::bail!("Git source or configuration changed before the provider result was returned")
211}

219/// Evaluates selected providers silently with a caller-owned resolved configuration.
220pub(crate) async fn evaluate(
221    args: &CheckArgs,
222    repository: &GitRepository,
223    policy: &PolicySnapshot,
224) -> Result<HostEvaluation> {
225    let selection = HostSelection::from_args(args)?;
226    let mut reports = Vec::new();
227    let mut captures = Vec::new();
228    for profile in &selection.profiles {
229        let selected = selection.for_profile(*profile, args, policy)?;
230        if selected.roots.is_empty() {
231            reports.push(excluded_report(policy, &selected, *profile));
232            continue;
233        }
234        let source = repository.with_targets(&policy.policy.targets);
235        let plan = native_capture_plan(&source, &policy.view, &selected, &policy.policy.targets)?;
236        let capture = source
237            .capture_native_from(&policy.view, args.base.as_deref(), &plan.roots, plan.filter)
238            .context("capture immutable provider Git view")?;
239        reports.push(evaluate_capture(&capture, policy, repository.root(), &selected).await);
240        captures.push(HostCapture {
241            source: capture,
242            selection: selected,
243            plan,
244        });
245    }
246    let report = merge_reports(reports, &selection, &policy.policy.targets);
247    let mut value = report_value(&report);
248    value["configuration"] = json!({
249        "view": policy.view,
250        "workflow": policy.workflow,
251        "effective": policy.policy,
252    });
253    anyhow::ensure!(
254        serde_json::to_vec(&value)?.len() <= MAX_ASSESSMENT_BYTES,
255        "combined provider report exceeds {MAX_ASSESSMENT_BYTES} bytes"
256    );
257    Ok(HostEvaluation {
258        passed: report.decision == Decision::Allow,
259        human: render_human(&report),
260        report: value,
261        captures,
262        targets: policy.policy.targets.clone(),
263        view: policy.view.clone(),
264    })
265}

267fn merge_reports(
268    reports: Vec<HostReport>,
269    selection: &HostSelection,
270    targets: &TargetPolicy,
271) -> HostReport {
272    let mut combined = HostReport {
273        decision: Decision::Indeterminate,
274        changeset_digest: String::new(),
275        root_digests: Vec::new(),
276        capture_valid_as_of: String::new(),
277        policy_digest: String::new(),
278        comparison: selection.comparison,
279        runs: Vec::new(),
280    };
281    let mut freshness = Vec::new();
282    for report in reports {
283        combined.policy_digest = report.policy_digest;
284        combined.root_digests.extend(report.root_digests);
285        combined.runs.extend(report.runs);
286        freshness.push(report.capture_valid_as_of);
287    }
288    combined.root_digests.sort_by_key(|(root, _, provider, _)| {
289        (root.clone(), provider_position(selection, provider))
290    });
291    combined.runs.sort_by_key(|run| {
292        (
293            run.root.clone(),
294            provider_position(selection, run.profile.provider_id()),
295        )
296    });
297    combined.capture_valid_as_of =
298        json!({ "providers": freshness, "targets": targets }).to_string();
299    combined.changeset_digest = combined_digest(&combined.root_digests);
300    combined.decision = compose_decision(&combined.runs);
301    combined
302}

312fn native_capture_plan(
313    repository: &GitRepository,
314    view: &ConfigView,
315    selection: &HostSelection,
316    targets: &TargetPolicy,
317) -> Result<NativeCapturePlan> {
318    let roots = capture_roots(selection);
319    let base_filter = capture_filter(selection);
320    let candidate_paths = repository
321        .native_candidate_paths_from(view, &roots, base_filter)
322        .context("discover provider Git workspace paths")?;
323    if !selection.skip_inapplicable || selection.profiles.contains(&ProviderProfile::Fast) {
324        validate_selected_roots(&candidate_paths, selection)?;
325    }
326    validate_native_exclusions(&candidate_paths, selection, targets)?;
327    let rust_applicable = selection.profiles.contains(&ProviderProfile::RustNative)
328        && rust_applicable_to_any_root(&candidate_paths, &selection.roots);
329    let plan = if rust_applicable {
330        NativeCapturePlan {
331            roots: Vec::new(),
332            filter: base_filter,
333        }
334    } else {
335        NativeCapturePlan {
336            roots: project_capture_roots(&candidate_paths, selection),
337            filter: base_filter,
338        }
339    };
340    Ok(plan)
341}

351fn capture_filter(selection: &HostSelection) -> NativeCaptureFilter {
352    selection
353        .profiles
354        .iter()
355        .fold(NativeCaptureFilter::empty(), |filter, profile| {
356            let kind = match profile {
357                ProviderProfile::Fast => NativeCaptureKind::Fast,
358                ProviderProfile::RustNative => NativeCaptureKind::Rust,
359                ProviderProfile::NodeNative => NativeCaptureKind::Node,
360                ProviderProfile::PythonNative => NativeCaptureKind::Python,
361            };
362            filter.with(kind)
363        })
364}

366fn rust_applicable_to_any_root(paths: &BTreeSet<RepoPath>, roots: &[WorkspaceRoot]) -> bool {
367    roots.iter().any(|root| {
368        paths.iter().any(|path| {
369            if let Some(prefix) = &root.prefix {
370                path.as_bytes() == [prefix.as_bytes(), b"/Cargo.toml"].concat()
371            } else {
372                path.as_bytes() == b"Cargo.toml"
373            }
374        })
375    })
376}

378impl HostSelection {
379    fn from_args(args: &CheckArgs) -> Result<Self> {
380        let (profiles, skip_inapplicable) = selected_profiles(args)?;
381        let native_selected = profiles
382            .iter()
383            .any(|profile| *profile != ProviderProfile::Fast);
384        anyhow::ensure!(
385            !native_selected || args.allow_unsandboxed_native,
386            concat!(
387                "native checks may execute repository-controlled code with access to host files, ",
388                "credentials, and network; rerun with --allow-unsandboxed-native only for a trusted repository"
389            )
390        );
391        anyhow::ensure!(
392            native_selected || !args.allow_unsandboxed_native,
393            "--allow-unsandboxed-native requires a native provider"
394        );
395        Ok(Self {
396            profiles,
397            skip_inapplicable,
398            roots: selected_roots(&args.roots)?,
399            excluded_roots: Vec::new(),
400            comparison: args.comparison.unwrap_or_else(|| {
401                if args.all || args.committed || (args.tree.is_some() && args.base.is_none()) {
402                    CheckComparison::All
403                } else {
404                    CheckComparison::Introduced
405                }
406            }),
407        })
408    }
409}

411fn validate_selected_roots(
412    candidate_paths: &BTreeSet<RepoPath>,
413    selection: &HostSelection,
414) -> Result<()> {
415    for root in &selection.roots {
416        let Some(prefix) = &root.prefix else {
417            continue;
418        };
419        anyhow::ensure!(
420            candidate_paths
421                .iter()
422                .any(|path| path_has_prefix(path.as_bytes(), prefix.as_bytes())),
423            "selected root {} contains no files supported by the selected providers",
424            root.label
425        );
426    }
427    Ok(())
428}

430async fn evaluate_capture(
431    capture: &GitNativeCapture,
432    policy: &PolicySnapshot,
433    candidate_root: &std::path::Path,
434    selection: &HostSelection,
435) -> HostReport {
436    let mut runs = selection
437        .excluded_roots
438        .iter()
439        .flat_map(|root| {
440            selection.profiles.iter().map(|profile| {
441                skipped_run(
442                    root.label.clone(),
443                    root.label.clone(),
444                    *profile,
445                    "excluded by targets.exclude",
446                )
447            })
448        })
449        .collect::<Vec<_>>();
450    let mut root_digests = Vec::new();
451    for root in &selection.roots {
452        for profile in &selection.profiles {
453            let prepared = match host_input(capture, root, *profile, selection.comparison, policy) {
454                Ok(prepared) => prepared,
455                Err(error) => {
456                    runs.push(failed_run(
457                        root.label.clone(),
458                        root.label.clone(),
459                        *profile,
460                        error.to_string(),
461                    ));
462                    continue;
463                }
464            };
465            root_digests.push((
466                prepared.label.clone(),
467                prepared.context.clone(),
468                profile.provider_id().to_owned(),
469                prepared.changeset_digest.clone(),
470            ));
471            run_prepared(&mut runs, prepared, candidate_root, selection, *profile).await;
472        }
473    }
474    let changeset_digest = combined_digest(&root_digests);
475    HostReport {
476        decision: compose_decision(&runs),
477        changeset_digest,
478        root_digests,
479        capture_valid_as_of: capture.valid_as_of.clone(),
480        policy_digest: policy.digest.clone(),
481        comparison: selection.comparison,
482        runs,
483    }
484}

486async fn run_prepared(
487    runs: &mut Vec<ProviderRun>,
488    prepared: PreparedRoot,
489    candidate_root: &std::path::Path,
490    selection: &HostSelection,
491    profile: ProviderProfile,
492) {
493    if selection.skip_inapplicable
494        && let Some(reason) = inapplicable_reason(
495            profile,
496            &prepared.candidate_paths,
497            prepared.project_root.as_ref(),
498        )
499    {
500        runs.push(skipped_run(
501            prepared.label,
502            prepared.context,
503            profile,
504            reason,
505        ));
506        return;
507    }
508    runs.push(
509        run_provider(
510            profile,
511            ProviderRunRequest {
512                root: prepared.label,
513                context: prepared.context,
514                workspace: prepared.workspace,
515                params: prepared.params,
516                candidate_root: candidate_root.to_path_buf(),
517                context_root: prepared.context_root,
518                project_root: prepared.project_root,
519            },
520        )
521        .await,
522    );
523}

525fn host_input(
526    capture: &GitNativeCapture,
527    root: &WorkspaceRoot,
528    profile: ProviderProfile,
529    comparison: CheckComparison,
530    policy: &PolicySnapshot,
531) -> Result<PreparedRoot> {
532    let context = project_context(capture, root, profile);
533    let project_root = (profile == ProviderProfile::RustNative)
534        .then(|| relative_root(root, &context))
535        .flatten();
536    let baseline = json!({ "rev": capture.baseline_rev });
537    let mut workspace = HostWorkspace::new(baseline.clone())?;
538    let mapping = workspace_mapping(capture, root, &context, profile)?;
539    let candidate_paths = populate_workspace(capture, &mapping, &mut workspace)?;
540    let changes = root_changes(capture, &mapping, &mut workspace)?;
541    let changeset = json!({ "baseline": baseline, "changes": changes });
542    let changeset_digest = json_digest(&changeset)?;
543    let params = json!({
544        "changeset": changeset,
545        "scope": "workspace",
546        "comparison": comparison.as_str(),
547        "configuration": if profile == ProviderProfile::Fast {
548            json!({"verify": policy.policy.verify})
549        } else {
550            json!({})
551        }
552    });
553    Ok(PreparedRoot {
554        label: root.label.clone(),
555        context: context.label,
556        workspace,
557        params,
558        changeset_digest,
559        candidate_paths,
560        context_root: context.prefix,
561        project_root,
562    })
563}

565fn populate_workspace(
566    capture: &GitNativeCapture,
567    mapping: &std::collections::BTreeMap<RepoPath, RepoPath>,
568    workspace: &mut HostWorkspace,
569) -> Result<BTreeSet<RepoPath>> {
570    let mut candidate_paths = BTreeSet::new();
571    for (original, path) in mapping {
572        let state = capture
573            .paths
574            .get(original)
575            .with_context(|| format!("missing captured state for {original}"))?;
576        if let Some(blob) = &state.before {
577            let id = workspace.add_baseline_file(path.clone(), Arc::clone(&blob.bytes))?;
578            anyhow::ensure!(
579                id == blob.blob_id,
580                "captured baseline blob identity changed"
581            );
582        }
583        if state.after.is_some() {
584            candidate_paths.insert(path.clone());
585        }
586    }
587    Ok(candidate_paths)
588}

590fn root_changes(
591    capture: &GitNativeCapture,
592    mapping: &std::collections::BTreeMap<RepoPath, RepoPath>,
593    workspace: &mut HostWorkspace,
594) -> Result<Vec<Value>> {
595    let mut changes = Vec::new();
596    for path in &capture.changed_paths {
597        let Some(rebased) = mapping.get(path) else {
598            continue;
599        };
600        let state = capture
601            .paths
602            .get(path)
603            .with_context(|| format!("missing captured state for {path}"))?;
604        changes.push(change_value(rebased, state, workspace)?);
605    }
606    Ok(changes)
607}
608
609fn change_value(
610    path: &RepoPath,
611    state: &NativePathState,
612    workspace: &mut HostWorkspace,
613) -> Result<Value> {
614    match (&state.before, &state.after) {
615        (None, Some(after)) => {
616            let id = workspace.add_candidate_blob(Arc::clone(&after.bytes))?;
617            Ok(json!({ "path": path, "kind": "create", "after": id }))
618        }
619        (Some(before), None) => {
620            Ok(json!({ "path": path, "kind": "delete", "before": before.blob_id }))
621        }
622        (Some(before), Some(after)) => {
623            let id = workspace.add_candidate_blob(Arc::clone(&after.bytes))?;
624            Ok(json!({
625                "path": path,
626                "kind": "modify",
627                "before": before.blob_id,
628                "after": id
629            }))
630        }
631        (None, None) => anyhow::bail!("changed path {path} has neither before nor after content"),
632    }
633}

635async fn run_provider(profile: ProviderProfile, request: ProviderRunRequest) -> ProviderRun {
636    let ProviderRunRequest {
637        root,
638        context,
639        workspace,
640        params,
641        candidate_root,
642        context_root,
643        project_root,
644    } = request;
645    let result = tokio::task::spawn_blocking(move || {
646        let command = provider_command(
647            profile,
648            project_root.as_ref(),
649            &candidate_root,
650            context_root.as_ref(),
651        )?;
652        let request = HostRequest::new(workspace, params)?
653            .with_wallclock_ms(provider_wallclock(profile))?
654            .with_candidate_root(&candidate_root)?;
655        host::evaluate(&command, &request)
656    })
657    .await;
658    match result {
659        Ok(Ok(assessment)) => ProviderRun {
660            root,
661            context,
662            profile,
663            assessment: Some(assessment),
664            failure: None,
665            skipped: None,
666        },
667        Ok(Err(error)) => failed_run(
668            root,
669            context,
670            profile,
671            format!("provider failed: {error:#}"),
672        ),
673        Err(error) => failed_run(
674            root,
675            context,
676            profile,
677            format!("provider worker failed: {error}"),
678        ),
679    }
680}

682fn provider_command(
683    profile: ProviderProfile,
684    project_root: Option<&RepoPath>,
685    candidate_root: &Path,
686    context_root: Option<&RepoPath>,
687) -> Result<ProviderCommand> {
688    let command = ProviderCommand::current_scoped(profile, project_root)?;
689    if profile == ProviderProfile::PythonNative
690        && let Some((interpreter, source)) = project_python_override(candidate_root, context_root)
691    {
692        return Ok(command
693            .with_environment("OPCORE_PYTHON", interpreter.into_os_string())
694            .with_environment("OPCORE_PYTHON_SOURCE", source));
695    }
696    Ok(command)
697}

699fn project_python_override(
700    candidate_root: &Path,
701    context_root: Option<&RepoPath>,
702) -> Option<(PathBuf, &'static str)> {
703    if let Some(interpreter) = env::var_os("OPCORE_PYTHON") {
704        return Some((PathBuf::from(interpreter), "explicit"));
705    }
706    let project_root = context_root.map_or_else(
707        || candidate_root.to_path_buf(),
708        |context| candidate_root.join(context.to_path_buf()),
709    );
710    let dot_venv = python_venv_interpreter(&project_root.join(".venv"));
711    if dot_venv.is_file() {
712        return Some((dot_venv, "project-dot-venv"));
713    }
714    active_project_python(&project_root).map(|active| (active, "active-project-venv"))
715}
716
717fn active_project_python(project_root: &Path) -> Option<PathBuf> {
718    let environment = PathBuf::from(env::var_os("VIRTUAL_ENV")?);
719    if !environment.is_absolute() {
720        return None;
721    }
722    let project_root = project_root.canonicalize().ok()?;
723    let environment = environment.canonicalize().ok()?;
724    if !environment.starts_with(&project_root) {
725        return None;
726    }
727    let interpreter = python_venv_interpreter(&environment);
728    interpreter.is_file().then_some(interpreter)
729}

731fn python_venv_interpreter(environment: &Path) -> PathBuf {
732    if cfg!(windows) {
733        environment.join("Scripts/python.exe")
734    } else {
735        environment.join("bin/python")
736    }

739fn inapplicable_reason(
740    profile: ProviderProfile,
741    candidate_paths: &BTreeSet<RepoPath>,
742    project_root: Option<&RepoPath>,
743) -> Option<&'static str> {
744    match profile {
745        ProviderProfile::RustNative
746            if !candidate_paths.iter().any(|path| {
747                project_root.map_or_else(
748                    || path.as_bytes() == b"Cargo.toml",
749                    |root| path.as_bytes() == [root.as_bytes(), b"/Cargo.toml"].concat().as_slice(),
750                )
751            }) =>
752        {
753            Some("no root Cargo.toml")
754        }
755        ProviderProfile::NodeNative if !node_applicable(candidate_paths) => {
756            Some("no root locked npm TypeScript project")
757        }
758        ProviderProfile::PythonNative if !candidate_paths.iter().any(is_python) => {
759            Some("no Python source")
760        }
761        _ => None,
762    }
763}

765fn node_applicable(paths: &BTreeSet<RepoPath>) -> bool {
766    has_exact(paths, b"package.json")
767        && (has_exact(paths, b"package-lock.json") || has_exact(paths, b"npm-shrinkwrap.json"))
768        && paths.iter().any(|path| {
769            path.as_bytes()
770                .rsplit(|byte| *byte == b'/')
771                .next()
772                .is_some_and(is_tsconfig_name)
773        })
774}

777    paths.iter().any(|path| path.as_bytes() == expected)
778}

792fn compose_decision(runs: &[ProviderRun]) -> Decision {
793    let evaluated = runs
794        .iter()
795        .filter(|run| run.skipped.is_none())
796        .collect::<Vec<_>>();
797    if evaluated.is_empty()
798        || evaluated.iter().any(|run| run.failure.is_some())
799        || evaluated
800            .iter()
801            .any(|run| assessment_status(run) != Some("complete"))
802        || evaluated.iter().any(|run| assessment_checked_nothing(run))
803    {
804        return Decision::Indeterminate;
805    }
806    if evaluated.iter().any(|run| assessment_has_diagnostics(run)) {
807        Decision::Deny
808    } else {
809        Decision::Allow
810    }
811}

813fn assessment_checked_nothing(run: &ProviderRun) -> bool {
814    let Some(assessment) = &run.assessment else {
815        return true;
816    };
817    let fast_files = assessment
818        .pointer("/evidence/0/data/filesRead")
819        .and_then(Value::as_u64);
820    let native_files = assessment
821        .pointer("/evidence/0/data/candidate/selectedSourceFiles")
822        .and_then(Value::as_u64);
823    fast_files.or(native_files) == Some(0)
824}

931fn append_run(output: &mut String, run: &ProviderRun) {
932    let name = user_provider_name(run.profile);
933    let label = format!("{} [{name}]", run.root);
934    if let Some(reason) = &run.skipped {
935        let _ = writeln!(output, "{label}: skipped ({reason})");
936        return;
937    }
938    if let Some(failure) = &run.failure {
939        let _ = writeln!(output, "{label}: unavailable: {failure}");
940        return;
941    }
942    let Some(assessment) = &run.assessment else {
943        return;
944    };
945    let status = assessment
946        .get("status")
947        .and_then(Value::as_str)
948        .unwrap_or("invalid");
949    let diagnostics = assessment
950        .get("diagnostics")
951        .and_then(Value::as_array)
952        .map_or(0, Vec::len);
953    let elapsed_ms = assessment
954        .pointer("/timing/elapsedMs")
955        .and_then(Value::as_u64)
956        .unwrap_or(0);
957    let _ = writeln!(
958        output,
959        "{label}: {status} ({diagnostics} diagnostics, {elapsed_ms} ms)"
960    );
961    append_diagnostics(output, assessment, &run.context);
962    if status != "complete"
963        && let Some(detail) = assessment
964            .pointer("/coverage/degraded/0/detail")
965            .and_then(Value::as_str)
966    {
967        let _ = writeln!(output, "  detail: {detail}");
968    }
969}

971fn append_diagnostics(output: &mut String, assessment: &Value, context: &str) {
972    let Some(diagnostics) = assessment.get("diagnostics").and_then(Value::as_array) else {
973        return;
974    };
975    if diagnostics.len() > MAX_HUMAN_DIAGNOSTICS {
976        append_diagnostic_summary(output, diagnostics);
977    }
978    let mut locations = BTreeSet::new();
979    let mut shown = 0_usize;
980    for diagnostic in diagnostics {
981        if diagnostics.len() > MAX_HUMAN_DIAGNOSTICS
982            && !locations.insert(diagnostic_location_key(diagnostic))
983        {
984            continue;
985        }
986        append_diagnostic(output, diagnostic, context);
987        shown = shown.saturating_add(1);
988        if shown == MAX_HUMAN_DIAGNOSTICS {
989            break;
990        }
991    }
992    if shown < diagnostics.len() {
993        let _ = writeln!(
994            output,
995            "  showing {shown} representative locations; --json retains all {} diagnostics",
996            diagnostics.len()
997        );
998    }
999}
1000
1001fn append_diagnostic(output: &mut String, diagnostic: &Value, context: &str) {
1002    let path = diagnostic
1003        .pointer("/location/path")
1004        .and_then(Value::as_str)
1005        .unwrap_or("_request");
1006    let line = diagnostic
1007        .pointer("/location/range/start/line")
1008        .and_then(Value::as_u64)
1009        .map_or(0, |line| line.saturating_add(1));
1010    let code = diagnostic
1011        .get("code")
1012        .and_then(Value::as_str)
1013        .unwrap_or("unknown");
1014    let message = diagnostic
1015        .get("message")
1016        .and_then(Value::as_str)
1017        .unwrap_or("diagnostic");
1018    let path = if context == "." {
1019        path.to_owned()
1020    } else {
1021        format!("{context}/{path}")
1022    };
1023    let _ = writeln!(output, "{path}:{line}: {code}: {message}");
1024    if let Some(help) = diagnostic.get("help").and_then(Value::as_str) {
1025        let _ = writeln!(output, "  help: {help}");
1026    }
1027}
1028
1029fn append_diagnostic_summary(output: &mut String, diagnostics: &[Value]) {
1030    let mut files = BTreeSet::new();
1031    let mut locations = BTreeSet::new();
1032    let mut rules = BTreeMap::<String, usize>::new();
1033    let mut hotspots = BTreeMap::<String, usize>::new();
1034    for diagnostic in diagnostics {
1035        let path = diagnostic
1036            .pointer("/location/path")
1037            .and_then(Value::as_str)
1038            .unwrap_or("_request")
1039            .to_owned();
1040        let code = diagnostic
1041            .get("code")
1042            .and_then(Value::as_str)
1043            .unwrap_or("unknown")
1044            .to_owned();
1045        files.insert(path.clone());
1046        locations.insert(diagnostic_location_key(diagnostic));
1047        *rules.entry(code).or_default() += 1;
1048        *hotspots.entry(path).or_default() += 1;
1049    }
1050    let _ = writeln!(
1051        output,
1052        "  summary: {} diagnostics at {} rule locations across {} files",
1053        diagnostics.len(),
1054        locations.len(),
1055        files.len()
1056    );
1057    append_ranked_counts(output, "rules", rules);
1058    append_ranked_counts(output, "hotspots", hotspots);
1059}
1060
1061fn append_ranked_counts(output: &mut String, label: &str, counts: BTreeMap<String, usize>) {
1062    let mut ranked = counts.into_iter().collect::<Vec<_>>();
1063    ranked.sort_by(|(left_name, left_count), (right_name, right_count)| {
1064        right_count
1065            .cmp(left_count)
1066            .then_with(|| left_name.cmp(right_name))
1067    });
1068    let summary = ranked
1069        .into_iter()
1070        .take(3)
1071        .map(|(name, count)| format!("{name}={count}"))
1072        .collect::<Vec<_>>()
1073        .join(", ");
1074    let _ = writeln!(output, "  top {label}: {summary}");
1075}
1076
1077fn diagnostic_location_key(diagnostic: &Value) -> (String, String, u64, u64) {
1078    (
1079        diagnostic
1080            .pointer("/location/path")
1081            .and_then(Value::as_str)
1082            .unwrap_or("_request")
1083            .to_owned(),
1084        diagnostic
1085            .get("code")
1086            .and_then(Value::as_str)
1087            .unwrap_or("unknown")
1088            .to_owned(),
1089        diagnostic
1090            .pointer("/location/range/start/line")
1091            .and_then(Value::as_u64)
1092            .unwrap_or(0),
1093        diagnostic
1094            .pointer("/location/range/start/char")
1095            .and_then(Value::as_u64)
1096            .unwrap_or(0),
1097    )
1098}

1118fn enforce(decision: Decision, advisory: bool) -> Result<()> {
1119    match decision {
1120        Decision::Allow => Ok(()),
1121        Decision::Deny if advisory => Ok(()),
1122        Decision::Deny => anyhow::bail!("host check found required intervention"),
1123        Decision::Indeterminate => anyhow::bail!("host check did not complete"),
1124    }
