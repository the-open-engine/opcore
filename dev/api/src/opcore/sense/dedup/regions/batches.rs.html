7// Ordered hash ranges keep every occurrence of an anchor together, regardless of path.
8// Splitting only the in-memory index preserves the existing global evaluation order.
9#[derive(Clone, Copy)]
10pub(super) struct Batch {
11    family: &'static str,
12    start: u128,
13    end: u128,
14    pub(super) limit: usize,
15}
16
17impl Batch {
18    pub(super) fn full(family: &'static str, limit: usize) -> Self {
19        Self {
20            family,
21            start: 0,
22            end: u128::MAX,
23            limit,
24        }
25    }
26
27    fn select(self, view: &RegionView<'_>) -> Selection {
28        let mut selection = Selection {
29            ranges: Vec::with_capacity(view.files.len()),
30            count: 0,
31            bounds: None,
32        };
33        for file in &view.files {
34            let hash = |index: &u16| {
35                anchor_hash(&file.facts.region_fingerprints.anchors[usize::from(*index)])
36            };
37            let range = if file.source.language.family() == self.family {
38                let start = file
39                    .anchor_order
40                    .partition_point(|index| hash(index) < self.start);
41                let end = file
42                    .anchor_order
43                    .partition_point(|index| hash(index) <= self.end);
44                start..end
45            } else {
46                0..0
47            };
48            if !range.is_empty() {
49                let min = hash(&file.anchor_order[range.start]);
50                let max = hash(&file.anchor_order[range.end - 1]);
51                selection.bounds = Some(
52                    selection
53                        .bounds
54                        .map_or((min, max), |(a, b)| (a.min(min), b.max(max))),
55                );
56            }
57            selection.count = selection.count.saturating_add(range.len());
58            selection.ranges.push(range);
59        }
60        selection
61    }
62
63    fn split(self, before: &Selection, after: &Selection) -> Option<(Self, Self)> {
64        let (min, max) = [before.bounds, after.bounds]
65            .into_iter()
66            .flatten()
67            .reduce(|(a, b), (c, d)| (a.min(c), b.max(d)))?;
68        if min == max {
69            return None;
70        }
71        let midpoint = min + (max - min) / 2;
72        Some((
73            Self {
74                end: midpoint,
75                ..self
76            },
77            Self {
78                start: midpoint + 1,
79                ..self
80            },
81        ))
82    }
83}

85pub(super) struct Selection {
86    pub(super) ranges: Vec<std::ops::Range<usize>>,
87    count: usize,
88    bounds: Option<(u128, u128)>,
89}
90
91impl Selection {
92    fn report_limit(
93        &self,
94        view: &RegionView<'_>,
95        limit: usize,
96        issues: &mut Vec<super::SenseIssue>,
97    ) -> bool {
98        if self.count <= limit {
99            return false;
100        }
101        let mut retained = 0usize;
102        let mut paths = Vec::new();
103        for (file, range) in view.files.iter().zip(&self.ranges) {
104            if range.len() > limit.saturating_sub(retained) {
105                paths.push(file.path.clone());
106            } else {
107                retained += range.len();
108            }
109        }
110        paths.sort();
111        push_limit_issue(
112            issues,
113            AffectedPaths {
114                count: paths.len(),
115                paths,
116            },
117            LimitIssueSpec {
118                code: "dedup_region_request_limit",
119                stage: DedupLimitStage::TokenRegionRequestCollection,
120                views: &[view.view],
121                limit,
122                processed: retained,
123                message: "one token-region anchor class exceeds the bounded batch",
124            },
125        );
126        true
127    }
128}

130pub(super) fn anchor_hash(anchor: &RegionFingerprint) -> u128 {
131    (u128::from(anchor.hash_one) << 64) | u128::from(anchor.hash_two)
132}

134pub(super) fn evaluate(
135    inputs: RegionInputs<'_>,
136    limit: usize,
137    views: &mut [RegionView<'_>; 2],
138    state: &mut EvaluationState,
139) {
140    let families = inputs
141        .after
142        .files()
143        .map(|file| file.language.family())
144        .collect::<BTreeSet<_>>();
145    for family in families {
146        let mut pending = vec![Batch::full(family, limit)];
147        while let Some(batch) = pending.pop() {
148            let split = evaluate_batch(inputs, batch, views, state);
149            if state.comparison.incomplete {
150                return;
151            }
152            if let Some((left, right)) = split {
153                pending.push(right);
154                pending.push(left);
155            }
156        }
157    }
158}

160fn evaluate_batch(
161    inputs: RegionInputs<'_>,
162    batch: Batch,
163    views: &mut [RegionView<'_>; 2],
164    state: &mut EvaluationState,
165) -> Option<(Batch, Batch)> {
166    let [before, after] = views;
167    let before_selection = batch.select(before);
168    let after_selection = batch.select(after);
169    if before_selection.count > batch.limit || after_selection.count > batch.limit {
170        if let Some(split) = batch.split(&before_selection, &after_selection) {
171            return Some(split);
172        }
173        // A single anchor class cannot be split without losing exact cardinality.
174        for (view, selection, coverage) in [
175            (&*before, &before_selection, &mut state.before),
176            (&*after, &after_selection, &mut state.after),
177        ] {
178            coverage.request_truncated |=
179                selection.report_limit(view, batch.limit, &mut state.issues);
180        }
181        state.comparison.incomplete = true;
182        return None;
183    }
184    before.index_batch(&before_selection);
185    after.index_batch(&after_selection);
186    let context = EvaluationContext {
187        inputs,
188        before,
189        after,
190    };
191    for (family, groups) in &after.groups {
192        for (anchor, occurrences) in groups {
193            state.evaluate_group(context, family, *anchor, occurrences);
194            if state.comparison.incomplete {
195                return None;
196            }
197        }
198    }
199    None
200}
