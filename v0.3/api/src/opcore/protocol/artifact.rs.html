1//! Stable, bounded hashing for executable artifacts.

3use std::{

10use sha2::{Digest as _, Sha256};

12pub(super) const MAX_EXECUTABLE_BYTES: u64 = 512 * 1024 * 1024;
13
14#[derive(Clone, Debug, Eq, PartialEq)]
15struct FileIdentity {
16    len: u64,
17    modified: Option<SystemTime>,
18    #[cfg(unix)]
19    device: u64,
20    #[cfg(unix)]
21    inode: u64,
22    #[cfg(unix)]
23    changed_seconds: i64,
24    #[cfg(unix)]
25    changed_nanoseconds: i64,
26}
27
28impl FileIdentity {
29    fn capture(metadata: &Metadata) -> Self {
30        #[cfg(unix)]
31        use std::os::unix::fs::MetadataExt as _;
32
33        Self {
34            len: metadata.len(),
35            modified: metadata.modified().ok(),
36            #[cfg(unix)]
37            device: metadata.dev(),
38            #[cfg(unix)]
39            inode: metadata.ino(),
40            #[cfg(unix)]
41            changed_seconds: metadata.ctime(),
42            #[cfg(unix)]
43            changed_nanoseconds: metadata.ctime_nsec(),
44        }
45    }
46}
47
48struct DigestSink(Sha256);
49
50impl DigestSink {
51    fn new() -> Self {
52        Self(Sha256::new())
53    }
54
55    fn finish(self) -> String {
56        format!("sha256:{}", hex::encode(self.0.finalize()))
57    }
58}
59
60impl io::Write for DigestSink {
61    fn write(&mut self, bytes: &[u8]) -> io::Result<usize> {
62        self.0.update(bytes);
63        Ok(bytes.len())
64    }
65
66    fn flush(&mut self) -> io::Result<()> {
67        Ok(())
68    }
69}

71struct BoundFile {
72    file: File,
73    identity: FileIdentity,
74}
75
76pub(super) fn digest_path(path: &Path) -> io::Result<String> {
77    let bound = bind_path(path)?;
78    digest_bound_path(path, bound)
79}
80
81fn digest_bound_path(path: &Path, bound: BoundFile) -> io::Result<String> {
82    let BoundFile { mut file, identity } = bound;
83    let digest = digest_open_file(&mut file, &identity)?;
84    validate_path_binding(path, &identity)?;
85    Ok(digest)
86}
87
88fn bind_path(path: &Path) -> io::Result<BoundFile> {
89    let before = fs::symlink_metadata(path)?;
90    validate_metadata(&before)?;
91    let identity = FileIdentity::capture(&before);
92    let file = open_no_follow(path)?;
93    let opened = file.metadata()?;
94    validate_metadata(&opened)?;
95    validate_stable(&identity, &opened)?;
96    Ok(BoundFile { file, identity })
97}
98
99fn digest_open_file(file: &mut File, identity: &FileIdentity) -> io::Result<String> {
100    let maximum = identity
101        .len
102        .checked_add(1)
103        .ok_or_else(|| io::Error::other("executable byte bound overflowed"))?;
104    let mut digest = DigestSink::new();
105    let total = io::copy(&mut file.take(maximum), &mut digest)?;
106    if total != identity.len {
107        return Err(io::Error::other(
108            "executable changed length while it was hashed",
109        ));
110    }
111    validate_stable(identity, &file.metadata()?)?;
112    Ok(digest.finish())
113}
114
115fn open_no_follow(path: &Path) -> io::Result<File> {
116    let mut options = OpenOptions::new();
117    options.read(true);
118    #[cfg(unix)]
119    {
120        use std::os::unix::fs::OpenOptionsExt as _;
121        options.custom_flags(libc::O_NOFOLLOW);
122    }
123    options.open(path)
124}
125
126fn validate_metadata(metadata: &Metadata) -> io::Result<()> {
127    if !metadata.file_type().is_symlink()
128        && metadata.is_file()
129        && metadata.len() <= MAX_EXECUTABLE_BYTES
130    {
131        Ok(())
132    } else {
133        Err(io::Error::other("executable is not a bounded regular file"))
134    }
135}
136
137fn validate_stable(before: &FileIdentity, after: &Metadata) -> io::Result<()> {
138    if before == &FileIdentity::capture(after) {
139        Ok(())
140    } else {
141        Err(io::Error::other("executable changed while it was hashed"))
142    }
143}
144
145fn validate_path_binding(path: &Path, opened: &FileIdentity) -> io::Result<()> {
146    let after = fs::symlink_metadata(path)?;
147    validate_metadata(&after)?;
148    validate_stable(opened, &after)
149}

151#[cfg(test)]
152mod tests {
153    use super::*;
154
155    #[test]
156    fn refuses_more_bytes_than_the_opened_file_identity_declared() {
157        let directory = tempfile::tempdir().unwrap();
158        let path = directory.path().join("artifact");
159        std::fs::write(&path, b"artifact").unwrap();
160        let mut file = File::open(&path).unwrap();
161        let metadata = file.metadata().unwrap();
162        std::fs::write(&path, b"artifact grew").unwrap();
163
164        assert!(digest_open_file(&mut file, &FileIdentity::capture(&metadata)).is_err());
165    }
166
167    #[cfg(unix)]
168    #[test]
169    fn refuses_atomic_path_replacement_after_open() {
170        let directory = tempfile::tempdir().unwrap();
171        let path = directory.path().join("artifact");
172        let replacement = directory.path().join("replacement");
173        std::fs::write(&path, b"original00").unwrap();
174        std::fs::write(&replacement, b"replaced00").unwrap();
175        let modified = path.metadata().unwrap().modified().unwrap();
176        File::options()
177            .write(true)
178            .open(&replacement)
179            .unwrap()
180            .set_times(std::fs::FileTimes::new().set_modified(modified))
181            .unwrap();
182
183        let bound = bind_path(&path).unwrap();
184        std::fs::rename(&replacement, &path).unwrap();
185        let result = digest_bound_path(&path, bound);
186
187        assert!(result.is_err());
188    }
