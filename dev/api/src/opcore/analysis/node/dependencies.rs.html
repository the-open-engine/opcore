21pub(super) fn extract(
22    parsed: &ParserReturn<'_>,
23    tokens: &[LexToken],
24    file: &SourceFile,
25) -> DependencyFacts {
26    if parsed.panicked || !parsed.diagnostics.is_empty() {
27        return crate::analysis::parser_failed_dependencies();
28    }
29    let mut collector = Collector::default();
30    for statement in &parsed.program.body {
31        collector.observe_statement(statement);
32    }
33    collector.observe_require_calls(parsed, file);
34    collector.observe_unsupported_loaders(tokens);
35    collector.finish()
36}
37
38#[derive(Default)]
39struct Collector {
40    references: Vec<DependencyReference>,
41    truncated: bool,
42}
43
44impl Collector {
45    fn observe_statement(&mut self, statement: &Statement<'_>) {
46        match statement {
47            Statement::ImportDeclaration(declaration) => self.push(
48                import_declaration_kind(declaration),
49                declaration.source.value.as_str(),
50            ),
51            Statement::ExportNamedDeclaration(declaration) => {
52                if let Some(source) = &declaration.source {
53                    self.push(export_named_kind(declaration), source.value.as_str());
54                }
55            }
56            Statement::ExportAllDeclaration(declaration) => self.push(
57                reference_kind(declaration.export_kind),
58                declaration.source.value.as_str(),
59            ),
60            Statement::TSImportEqualsDeclaration(declaration) => {
61                if let TSModuleReference::ExternalModuleReference(reference) =
62                    &declaration.module_reference
63                {
64                    self.push(
65                        DependencyReferenceKind::NodeUnsupportedDynamic,
66                        reference.expression.value.as_str(),
67                    );
68                }
69            }
70            _ => {}
71        }
72    }
73
74    fn observe_require_calls(&mut self, parsed: &ParserReturn<'_>, file: &SourceFile) {
75        let remaining = MAX_DEPENDENCY_FACTS_PER_FILE.saturating_sub(self.references.len());
76        let mut visitor = RequireVisitor::new(file, remaining);
77        visitor.visit_program(&parsed.program);
78        self.references.extend(visitor.references);
79        self.truncated |= visitor.truncated;
80    }
81
82    fn observe_unsupported_loaders(&mut self, tokens: &[LexToken]) {
83        for pair in tokens.windows(2) {
84            if pair[0].kind == Kind::Import && pair[1].kind == Kind::LParen {
85                self.push(DependencyReferenceKind::NodeUnsupportedDynamic, "");
86            }
87        }
88    }
89
90    fn push(&mut self, kind: DependencyReferenceKind, specifier: &str) {
91        if self.references.len() >= MAX_DEPENDENCY_FACTS_PER_FILE {
92            self.truncated = true;
93            return;
94        }
95        self.references.push(DependencyReference {
96            kind,
97            specifier: specifier.into(),
98            level: 0,
99        });
100    }
101
102    fn finish(mut self) -> DependencyFacts {
103        self.references.sort();
104        DependencyFacts {
105            status: if self.truncated {
106                DependencyExtractionStatus::Truncated
107            } else {
108                DependencyExtractionStatus::Complete
109            },
110            references: self.references,
111        }
112    }
113}

115struct RequireVisitor<'a> {
116    eligible_calls: HashSet<u32>,
117    references: Vec<DependencyReference>,
118    reference_limit: usize,
119    truncated: bool,
120    require_overridden: bool,
121    common_js_file: bool,
122    marker: std::marker::PhantomData<&'a ()>,
123}
124
125impl<'a> RequireVisitor<'a> {
126    fn new(file: &SourceFile, reference_limit: usize) -> Self {
127        Self {
128            eligible_calls: HashSet::new(),
129            references: Vec::new(),
130            reference_limit,
131            truncated: false,
132            require_overridden: false,
133            common_js_file: file
134                .path
135                .as_utf8()
136                .is_some_and(|path| path.as_bytes().ends_with(b".cjs"))
137                && file.language_mode.split(':').any(|part| part == "commonjs"),
138            marker: std::marker::PhantomData,
139        }
140    }
141
142    fn observe_top_level_statement(&mut self, statement: &Statement<'a>) {
143        match statement {
144            Statement::ExpressionStatement(statement) => {
145                self.mark_eligible_expression(&statement.expression);
146            }
147            Statement::VariableDeclaration(declaration) => {
148                for declarator in &declaration.declarations {
149                    if let Some(initializer) = &declarator.init {
150                        self.mark_eligible_expression(initializer);
151                    }
152                }
153            }
154            _ => {}
155        }
156    }
157
158    fn mark_eligible_expression(&mut self, expression: &Expression<'a>) {
159        if let Expression::CallExpression(call) = expression {
160            self.eligible_calls.insert(call.span.start);
161        }
162    }
163
164    fn literal_specifier(call: &CallExpression<'a>) -> Option<String> {
165        let [argument] = call.arguments.as_slice() else {
166            return None;
167        };
168        match argument {
169            oxc_ast::ast::Argument::StringLiteral(literal) => {
170                Some(literal.value.as_str().to_owned())
171            }
172            _ => None,
173        }
174    }
175
176    fn valid_common_js_specifier(specifier: &str) -> bool {
177        (specifier.starts_with("./") || specifier.starts_with("../"))
178            && specifier.as_bytes().ends_with(b".cjs")
179    }
180
181    fn push(&mut self, kind: DependencyReferenceKind, specifier: String) {
182        if self.references.len() >= self.reference_limit {
183            self.truncated = true;
184            return;
185        }
186        self.references.push(DependencyReference {
187            kind,
188            specifier,
189            level: 0,
190        });
191    }
192}

194impl<'a> Visit<'a> for RequireVisitor<'a> {
195    fn visit_program(&mut self, program: &oxc_ast::ast::Program<'a>) {
196        for statement in &program.body {
197            self.observe_top_level_statement(statement);
198        }
199        let mut override_visitor = RequireOverrideVisitor::default();
200        override_visitor.visit_program(program);
201        self.require_overridden = override_visitor.found;
202        walk::walk_program(self, program);
203    }
204
205    fn visit_call_expression(&mut self, call: &CallExpression<'a>) {
206        if matches!(&call.callee, Expression::Identifier(identifier) if identifier.name == "require")
207        {
208            let specifier = Self::literal_specifier(call).unwrap_or_default();
209            let eligible = self.common_js_file
210                && !self.require_overridden
211                && self.eligible_calls.contains(&call.span.start)
212                && Self::valid_common_js_specifier(&specifier);
213            self.push(
214                if eligible {
215                    DependencyReferenceKind::NodeRuntime
216                } else {
217                    DependencyReferenceKind::NodeUnsupportedDynamic
218                },
219                specifier,
220            );
221            if let Some(type_arguments) = &call.type_arguments {
222                self.visit_ts_type_parameter_instantiation(type_arguments);
223            }
224            self.visit_arguments(&call.arguments);
225            return;
226        }
227        walk::walk_call_expression(self, call);
228    }
229
230    fn visit_identifier_reference(&mut self, identifier: &oxc_ast::ast::IdentifierReference<'a>) {
231        if identifier.name == "require" {
232            self.push(
233                DependencyReferenceKind::NodeUnsupportedDynamic,
234                String::new(),
235            );
236        }
237        walk::walk_identifier_reference(self, identifier);
238    }
239}

241#[derive(Default)]
242struct RequireOverrideVisitor {
243    found: bool,
244}
245
246impl<'a> Visit<'a> for RequireOverrideVisitor {
247    fn visit_binding_identifier(&mut self, identifier: &BindingIdentifier<'a>) {
248        self.found |= identifier.name == "require";
249        walk::walk_binding_identifier(self, identifier);
250    }
251
252    fn visit_assignment_expression(&mut self, expression: &AssignmentExpression<'a>) {
253        let mut target = AssignmentTargetRequireVisitor::default();
254        target.visit_assignment_target(&expression.left);
255        self.found |= target.found;
256        walk::walk_assignment_expression(self, expression);
257    }
258
259    fn visit_for_statement_left(&mut self, left: &ForStatementLeft<'a>) {
260        if let Some(assignment_target) = left.as_assignment_target() {
261            let mut target = AssignmentTargetRequireVisitor::default();
262            target.visit_assignment_target(assignment_target);
263            self.found |= target.found;
264        }
265        walk::walk_for_statement_left(self, left);
266    }
267
268    fn visit_update_expression(&mut self, expression: &UpdateExpression<'a>) {
269        self.found |= matches!(
270            &expression.argument,
271            SimpleAssignmentTarget::AssignmentTargetIdentifier(identifier)
272                if identifier.name == "require"
273        );
274        walk::walk_update_expression(self, expression);
275    }
276}

278#[derive(Default)]
279struct AssignmentTargetRequireVisitor {
280    found: bool,
281}
282
283impl<'a> Visit<'a> for AssignmentTargetRequireVisitor {
284    fn visit_simple_assignment_target(&mut self, target: &SimpleAssignmentTarget<'a>) {
285        self.found |= matches!(
286            target,
287            SimpleAssignmentTarget::AssignmentTargetIdentifier(identifier)
288                if identifier.name == "require"
289        );
290    }
291
292    fn visit_assignment_target_rest(&mut self, rest: &oxc_ast::ast::AssignmentTargetRest<'a>) {
293        self.visit_assignment_target(&rest.target);
294    }
295
296    fn visit_assignment_target_with_default(
297        &mut self,
298        target: &oxc_ast::ast::AssignmentTargetWithDefault<'a>,
299    ) {
300        self.visit_assignment_target(&target.binding);
301    }
302
303    fn visit_assignment_target_property_identifier(
304        &mut self,
305        property: &oxc_ast::ast::AssignmentTargetPropertyIdentifier<'a>,
306    ) {
307        self.found |= property.binding.name == "require";
308    }
309
310    fn visit_assignment_target_property_property(
311        &mut self,
312        property: &oxc_ast::ast::AssignmentTargetPropertyProperty<'a>,
313    ) {
314        self.visit_assignment_target_maybe_default(&property.binding);
315    }

318fn import_declaration_kind(declaration: &ImportDeclaration<'_>) -> DependencyReferenceKind {
319    if declaration.import_kind == ImportOrExportKind::Type
320        || declaration.specifiers.as_ref().is_some_and(|specifiers| {
321            !specifiers.is_empty()
322                && specifiers.iter().all(|specifier| {
323                    matches!(
324                        specifier,
325                        ImportDeclarationSpecifier::ImportSpecifier(specifier)
326                            if specifier.import_kind == ImportOrExportKind::Type
327                    )
328                })
329        })
330    {
331        DependencyReferenceKind::NodeType
332    } else {
333        DependencyReferenceKind::NodeRuntime
334    }
335}
336
337fn export_named_kind(declaration: &ExportNamedDeclaration<'_>) -> DependencyReferenceKind {
338    if declaration.export_kind == ImportOrExportKind::Type
339        || (!declaration.specifiers.is_empty()
340            && declaration
341                .specifiers
342                .iter()
343                .all(|specifier| specifier.export_kind == ImportOrExportKind::Type))
344    {
345        DependencyReferenceKind::NodeType
346    } else {
347        DependencyReferenceKind::NodeRuntime
348    }
349}
350
351const fn reference_kind(kind: ImportOrExportKind) -> DependencyReferenceKind {
352    match kind {
353        ImportOrExportKind::Value => DependencyReferenceKind::NodeRuntime,
354        ImportOrExportKind::Type => DependencyReferenceKind::NodeType,
355    }
