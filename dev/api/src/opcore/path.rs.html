6pub const MAX_REPOSITORY_PATH_BYTES: usize = 4_096;

8#[derive(Clone, Eq, Hash, Ord, PartialEq, PartialOrd)]
9pub struct RepoPath(Vec<u8>);
10
11#[derive(Debug, Error, Eq, PartialEq)]
12pub enum PathError {
13    #[error("repository path is empty")]
14    Empty,
15    #[error("repository path must be relative")]
16    Absolute,
17    #[error("repository path contains an empty, dot, or parent component")]
18    InvalidComponent,
19    #[error("repository path contains NUL")]
20    Nul,
21    #[error("repository path exceeds {MAX_REPOSITORY_PATH_BYTES} bytes")]
22    TooLong,
23}
24
25impl RepoPath {
26    pub(crate) fn request_marker() -> Self {
27        Self(b"_request".to_vec())
28    }
29
30    /// Constructs a validated repository-relative Git path.
31    ///
32    /// # Errors
33    ///
34    /// Returns [`PathError`] when the byte path is empty, absolute, oversized, contains NUL, or
35    /// contains an ambiguous component.
36    pub fn new(bytes: impl Into<Vec<u8>>) -> Result<Self, PathError> {
37        let bytes = bytes.into();
38        validate(&bytes)?;
39        Ok(Self(bytes))
40    }
41
42    /// Constructs a validated repository path from protocol UTF-8.
43    ///
44    /// # Errors
45    ///
46    /// Returns [`PathError`] when the value contains backslashes or violates repository path
47    /// invariants.
48    pub fn from_protocol(value: &str) -> Result<Self, PathError> {
49        if value.as_bytes().contains(&b'\\') {
50            return Err(PathError::InvalidComponent);
51        }
52        Self::new(value.as_bytes())
53    }
54
55    #[must_use]
56    pub fn as_bytes(&self) -> &[u8] {
57        &self.0
58    }
59
60    #[must_use]
61    pub fn as_utf8(&self) -> Option<&str> {
62        std::str::from_utf8(&self.0).ok()
63    }
64
65    #[must_use]
66    pub fn to_path_buf(&self) -> PathBuf {
67        #[cfg(unix)]
68        {
69            use std::os::unix::ffi::OsStringExt;
70            PathBuf::from(std::ffi::OsString::from_vec(self.0.clone()))
71        }
72        #[cfg(not(unix))]
73        {
74            PathBuf::from(String::from_utf8_lossy(&self.0).into_owned())
75        }
76    }
77
78    #[must_use]
79    pub fn display_lossless(&self) -> String {
80        let mut output = String::new();
81        for &byte in &self.0 {
82            match byte {
83                b' '..=b'~' if byte != b'\\' => output.push(char::from(byte)),
84                b'\\' => output.push_str("\\\\"),
85                _ => {
86                    const HEX: &[u8; 16] = b"0123456789abcdef";
87                    output.push_str("\\x");
88                    output.push(char::from(HEX[usize::from(byte >> 4)]));
89                    output.push(char::from(HEX[usize::from(byte & 0x0f)]));
90                }
91            }
92        }
93        output
94    }
95}

97fn validate(bytes: &[u8]) -> Result<(), PathError> {
98    if bytes.is_empty() {
99        return Err(PathError::Empty);
100    }
101    if bytes.len() > MAX_REPOSITORY_PATH_BYTES {
102        return Err(PathError::TooLong);
103    }
104    if bytes.contains(&0) {
105        return Err(PathError::Nul);
106    }
107    if bytes.first() == Some(&b'/') || bytes.first() == Some(&b'\\') {
108        return Err(PathError::Absolute);
109    }
110    if bytes.len() >= 2 && bytes[1] == b':' && bytes[0].is_ascii_alphabetic() {
111        return Err(PathError::Absolute);
112    }
113    if bytes
114        .split(|byte| matches!(byte, b'/' | b'\\'))
115        .any(|part| part.is_empty() || part == b"." || part == b"..")
116    {
117        return Err(PathError::InvalidComponent);
118    }
119    Ok(())
120}

137impl Serialize for RepoPath {
138    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
139    where
140        S: Serializer,
141    {
142        self.as_utf8()
143            .map_or_else(|| self.display_lossless(), str::to_owned)
144            .serialize(serializer)

148impl<'de> Deserialize<'de> for RepoPath {
149    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>

151        D: Deserializer<'de>,

163    fn rejects_escape_and_ambiguous_components() {
164        for invalid in [
165            "", "/tmp/a", "../a", "a/../b", "a//b", "C:/a", "a\\.\\b", "a\\b",
166        ] {
167            assert!(RepoPath::from_protocol(invalid).is_err(), "{invalid}");
168        }

172    #[test]
173    fn deterministic_fuzz_paths_never_panic_or_escape_validation() {
174        let mut state = 0xd1b5_4a32_d192_ed03u64;
175        for case in 0..4_096usize {
176            let length = case % 257;
177            let bytes = (0..length)
178                .map(|_| {
179                    state = state
180                        .wrapping_mul(2_862_933_555_777_941_757)
181                        .wrapping_add(3_037_000_493);
182                    state.to_le_bytes()[5]
183                })
184                .collect::<Vec<_>>();
185            if let Ok(path) = RepoPath::new(bytes) {
186                assert!(!path.as_bytes().is_empty());
187                assert!(!path.as_bytes().starts_with(b"/"));
188                drop(path.display_lossless());
189                drop(path.to_path_buf());
190                drop(serde_json::to_vec(&path));
191            }
192        }
