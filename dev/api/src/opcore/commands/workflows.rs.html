25/// Run the built-in checks with repository and workflow settings.
26#[derive(Clone, Debug, Args)]
27pub struct RunArgs {
28    /// Post-edit checks worktree changes; pre-commit checks the full index; CI checks a commit.
29    #[arg(value_enum)]
30    pub workflow: Workflow,
31    /// Repository directory; nested paths resolve to the Git worktree root.
32    #[arg(long, default_value = ".")]
33    pub repo: PathBuf,
34    /// CI target commit or tree (default: HEAD); never reads worktree overlays.
35    #[arg(long, value_name = "REF")]
36    pub tree: Option<String>,
37    /// CI baseline for introduced Sense findings; fetch and supply the intended base commit.
38    #[arg(long, value_name = "REF")]
39    pub base: Option<String>,
40    /// Emit one structured report containing every selected check.
41    #[arg(long)]
42    pub json: bool,
43    /// Authorize selected native tools to execute trusted repository code with host access.
44    #[arg(long)]
45    pub allow_unsandboxed_native: bool,
46}
47
48impl RunArgs {
49    pub(crate) fn view(&self) -> ConfigView {
50        match self.workflow {
51            Workflow::PostEdit => ConfigView::Worktree,
52            Workflow::PreCommit => ConfigView::Index,
53            Workflow::Ci => ConfigView::Tree(self.tree.clone().unwrap_or_else(|| "HEAD".into())),
54        }
55    }
56
57    fn validate(&self) -> Result<()> {
58        if self.workflow == Workflow::Ci {
59            ensure!(
60                self.base.is_some(),
61                "CI Sense requires --base <commit>; fetch and supply the intended comparison base"
62            );
63        } else {
64            ensure!(
65                self.base.is_none() && self.tree.is_none(),
66                "--base and --tree apply only to the ci workflow"
67            );
68        }
69        Ok(())
70    }
71}

73pub(crate) struct WorkflowEvaluation {
74    pub(crate) verify: Assessment,
75    pub(crate) sense: SenseReport,
76    pub(crate) native: Option<HostEvaluation>,
77    pub(crate) configuration: PolicySnapshot,
78    rendered: String,
79}
80
81impl WorkflowEvaluation {
82    pub(crate) fn enforce(&self) -> Result<()> {
83        cli::enforce_status(self.verify.status, false)?;
84        let coverage = &self.configuration.policy.coverage;
85        let sense_result = sense_cli::enforce_report(
86            &self.sense,
87            false,
88            coverage.allow_partial,
89            coverage.allow_node_builtins,
90        );
91        if self.sense.status == SenseStatus::Partial && sense_result.is_err() {
92            let workflow = self
93                .configuration
94                .workflow
95                .context("workflow settings are unavailable")?;
96            anyhow::bail!(
97                "Sense requirements were not satisfied; rerun the same command with --json and inspect .opcore.json workflows.{}.coverage. Findings still require repair",
98                workflow.as_str()
99            );
100        }
101        sense_result?;
102        ensure!(
103            self.native.as_ref().is_none_or(|result| result.passed),
104            "required native checks did not pass; inspect the provider diagnostics and coverage"
105        );
106        Ok(())
107    }
108
109    fn report(&self, workflow: Workflow) -> Value {
110        json!({
111            "schema": "opcore.workflow.v1",
112            "workflow": workflow,
113            "status": if self.enforce().is_ok() { "accepted" } else { "requires_attention" },
114            "configuration": self.configuration.report_configuration(),
115            "verify": self.verify,
116            "sense": self.sense,
117            "native": self.native.as_ref().map(|result| &result.report),
118        })
119    }
120
121    fn render(&self, args: &RunArgs) -> Result<String> {
122        if args.json {
123            return bounded_json(&self.report(args.workflow));
124        }
125        let mut output = format!("opcore workflow: {}\n", args.workflow.as_str());
126        output.push_str(&cli::render_human(&self.verify));
127        output.push_str(&sense_cli::render_output(&self.sense, false)?);
128        if let Some(native) = &self.native {
129            output.push_str(&native.human);
130        }
131        ensure!(
132            output.len() <= MAX_ASSESSMENT_BYTES,
133            "workflow report exceeds the output byte limit"
134        );
135        Ok(output)
136    }
137}

139/// Executes a workflow and prints its combined result.
140///
141/// # Errors
142///
143/// Returns an error for invalid configuration, stale or unavailable coverage, or required findings.
144pub async fn run(args: RunArgs) -> Result<()> {
145    let outcome = run_inner(&args).await;
146    match outcome {
147        Ok(result) => {
148            println!("{}", result.rendered.trim_end());
149            result.enforce()
150        }
151        Err(error) => {
152            if args.json {
153                let report = json!({
154                    "schema": "opcore.workflow.v1",
155                    "workflow": args.workflow,
156                    "status": "incomplete",
157                    "error": format!("{error:#}"),
158                });
159                println!("{}", bounded_json(&report)?);
160            }
161            Err(error)
162        }
163    }
164}

166async fn run_inner(args: &RunArgs) -> Result<WorkflowEvaluation> {
167    args.validate()?;
168    let repository = GitRepository::discover(&args.repo).context("discover workflow repository")?;
169    evaluate(args, &repository).await
170}
171
172pub(crate) async fn evaluate(
173    args: &RunArgs,
174    repository: &GitRepository,
175) -> Result<WorkflowEvaluation> {
176    for attempt in 0..2 {
177        let configuration = PolicySnapshot::capture(repository, args.view(), Some(args.workflow))?;
178        ensure!(
179            configuration.policy.native.is_empty() || args.allow_unsandboxed_native,
180            "this workflow requires native checks; run in a trusted environment and pass --allow-unsandboxed-native"
181        );
182        let selected = repository.with_targets(&configuration.policy.targets);
183        let auxiliary = auxiliary_paths(&configuration)?;
184        let before =
185            selected.observation_with_auxiliary(&args.view(), args.base.as_deref(), &auxiliary)?;
186        let mut result = evaluate_once(args, repository, configuration).await?;
187        // Render before freshness validation so expensive serialization cannot hide a stale view.
188        result.rendered = result.render(args)?;
189        if is_current(args, repository, &before, &result)? {
190            return Ok(result);
191        }
192        ensure!(
193            attempt == 0,
194            "source or configuration changed during the workflow; rerun the same command"
195        );
196    }
197    anyhow::bail!("workflow freshness retry exhausted")
198}

200fn is_current(
201    args: &RunArgs,
202    repository: &GitRepository,
203    before: &str,
204    result: &WorkflowEvaluation,
205) -> Result<bool> {
206    let selected = repository.with_targets(&result.configuration.policy.targets);
207    let auxiliary = auxiliary_paths(&result.configuration)?;
208    let native_current = result
209        .native
210        .as_ref()
211        .map(|native| native.is_current(repository))
212        .transpose()?
213        .unwrap_or(true);
214    Ok(before
215        == selected.observation_with_auxiliary(&args.view(), args.base.as_deref(), &auxiliary)?
216        && result.configuration.is_current()?
217        && native_current)
218}

220fn auxiliary_paths(configuration: &PolicySnapshot) -> Result<BTreeSet<RepoPath>> {
221    let mut paths = configuration
222        .policy
223        .documentation
224        .bindings
225        .iter()
226        .map(|binding| binding.document.clone())
227        .collect::<BTreeSet<_>>();
228    paths.insert(RepoPath::from_protocol(POLICY_PATH)?);
229    Ok(paths)
230}

232async fn evaluate_once(
233    args: &RunArgs,
234    repository: &GitRepository,
235    configuration: PolicySnapshot,
236) -> Result<WorkflowEvaluation> {
237    let mut check = check_args(args);
238    let verify = cli::assess_with_policy(&check, repository, &configuration)
239        .await
240        .context("Verify could not complete")?;
241    let sense = sense_cli::assess(&sense_args(args), repository, &configuration)
242        .await
243        .context("Sense could not complete")?;
244    let native = if configuration.policy.native.is_empty() {
245        None
246    } else {
247        check.providers = configuration
248            .policy
249            .native
250            .iter()
251            .map(|provider| match provider {
252                NativeProvider::RustNative => CheckProvider::RustNative,
253                NativeProvider::NodeNative => CheckProvider::NodeNative,
254                NativeProvider::PythonNative => CheckProvider::PythonNative,
255            })
256            .collect();
257        check.comparison = Some(CheckComparison::All);
258        Some(host_cli::evaluate(&check, repository, &configuration).await?)
259    };
260    Ok(WorkflowEvaluation {
261        verify,
262        sense,
263        native,
264        configuration,
265        rendered: String::new(),
266    })
267}

269fn check_args(args: &RunArgs) -> CheckArgs {
270    CheckArgs {
271        repo: args.repo.clone(),
272        changed: args.workflow == Workflow::PostEdit,
273        staged: args.workflow == Workflow::PreCommit,
274        all: args.workflow != Workflow::PostEdit,
275        tree: match args.view() {
276            ConfigView::Tree(tree) => Some(tree),
277            _ => None,
278        },
279        workflow: Some(args.workflow),
280        allow_unsandboxed_native: args.allow_unsandboxed_native,
281        ..CheckArgs::default()
282    }
283}

285fn sense_args(args: &RunArgs) -> sense_cli::SenseArgs {
286    sense_cli::SenseArgs {
287        repo: args.repo.clone(),
288        json: false,
289        advisory: false,
290        allow_partial: false,
291        allow_node_builtins: false,
292        staged: args.workflow == Workflow::PreCommit,
293        hypothetical: None,
294        workflow: Some(args.workflow),
295        tree: match args.view() {
296            ConfigView::Tree(tree) => Some(tree),
297            _ => None,
298        },
299        base: args.base.clone(),
300    }

303fn bounded_json(value: &Value) -> Result<String> {
304    let output = serde_json::to_string(value)?;
305    ensure!(
306        output.len() <= MAX_ASSESSMENT_BYTES,
307        "workflow report exceeds the output byte limit"
