47fn validate_initialize_metadata(result: &Value) -> Result<()> {
48    if let Some(roles) = result.get("roles") {
49        require_exact_strings(Some(roles), &["judge"], "provider roles")?;
50    }
51    if let Some(provenance) = result.get("provenance") {
52        let provenance = required_object(
53            provenance,
54            &["publisher", "signature"],
55            &[],
56            "provider provenance",
57        )?;
58        if let Some(publisher) = provenance.get("publisher") {
59            require_equal_string(Some(publisher), "the-open-engine", "provider publisher")?;
60        }
61        if let Some(signature) = provenance.get("signature") {
62            require_nonempty_string(Some(signature), "provider signature")?;
63        }
64    }
65    Ok(())
66}

68fn validate_server_info(result: &Value, profile: ProviderProfile) -> Result<()> {

70        result.get("serverInfo").unwrap_or(&Value::Null),
71        &["name", "version", "fingerprint"],

73        "provider serverInfo",

76    require_equal_string(

189    mut assessment: Value,
190    request: &HostRequest,
191    profile: ProviderProfile,
192    read_blobs: &BTreeSet<String>,
193) -> Result<Value> {

232fn validate_assessment_content(
233    object: &serde_json::Map<String, Value>,
234    request: &HostRequest,
235    profile: ProviderProfile,
236) -> Result<()> {
237    let status = assessment_status(object.get("status"))?;
238    let expected_coverage = expected_coverage_part(request, profile)?;
239    let diagnostic_paths = requested_diagnostic_paths(request)?;
240    validate_diagnostics(
241        object.get("diagnostics"),
242        profile,
243        &expected_coverage["comparison"],
244        &diagnostic_paths,
245    )?;
246    validate_evidence(object.get("evidence"))?;
247    validate_coverage(
248        object.get("coverage"),
249        status,
250        &expected_coverage,
251        profile.provider_id(),
252    )?;
253    Ok(())
254}

269fn assessment_status(value: Option<&Value>) -> Result<&str> {
270    let status = value
271        .and_then(Value::as_str)
272        .context("provider assessment status must be a string")?;
273    if ![
274        "complete",
275        "incomplete",
276        "unsupported",
277        "error",
278        "cancelled",
279    ]
280    .contains(&status)
281    {
282        bail!("provider assessment status is unknown");
283    }
284    Ok(status)

287fn expected_coverage_part(request: &HostRequest, profile: ProviderProfile) -> Result<Value> {
288    let params = request
289        .evaluation_params
290        .as_object()
291        .context("check/evaluate params must remain an object")?;
292    let sources = request_strings(
293        params.get("diagnosticSources"),
294        profile.provider_id(),
295        "requested diagnosticSources",
296    )?;
297    let rules = request_strings(params.get("rules"), profile.rule(), "requested rules")?;
298    let scope = params
299        .get("scope")
300        .context("check/evaluate scope is missing")?;
301    let comparison = params
302        .get("comparison")
303        .context("check/evaluate comparison is missing")?;
304    Ok(json!({
305        "scope": scope,
306        "diagnosticSources": sources,
307        "rules": rules,
308        "comparison": comparison
309    }))
310}

312fn request_strings(value: Option<&Value>, default: &str, label: &str) -> Result<Vec<String>> {
313    let Some(value) = value else {
314        return Ok(vec![default.to_owned()]);
315    };
316    let values = string_set(Some(value), label)?;
317    if values.is_empty() {
318        Ok(vec![default.to_owned()])
319    } else {
320        Ok(values.into_iter().collect())
321    }
322}

324fn validate_diagnostics(
325    value: Option<&Value>,
326    profile: ProviderProfile,
327    comparison: &Value,
328    allowed_paths: &BTreeSet<RepoPath>,
329) -> Result<()> {
330    let diagnostics = value
331        .and_then(Value::as_array)
332        .context("provider diagnostics must be an array")?;
333    if diagnostics.len() > MAX_DIAGNOSTICS {
334        bail!("provider diagnostics exceed {MAX_DIAGNOSTICS} findings");
335    }
336    for diagnostic in diagnostics {
337        validate_diagnostic(diagnostic, profile, comparison, allowed_paths)?;
338    }
339    Ok(())
340}

418fn validate_introduced(value: Option<&Value>, comparison: &Value) -> Result<()> {
419    match comparison.as_str() {
420        Some("introduced") if value.and_then(Value::as_bool) == Some(true) => Ok(()),
421        Some("introduced") => bail!("introduced diagnostic must be marked introduced"),
422        Some("all") if value.is_none() || value.and_then(Value::as_bool) == Some(false) => Ok(()),
423        Some("all") => bail!("all-comparison diagnostic cannot be marked introduced"),
424        Some("resolved") => bail!("unsupported resolved assessment cannot contain diagnostics"),
425        _ => bail!("provider assessment comparison is invalid"),
426    }
427}

429fn validate_location(value: Option<&Value>, allowed_paths: &BTreeSet<RepoPath>) -> Result<()> {
430    let location = required_object(
431        value.unwrap_or(&Value::Null),
432        &["path", "range"],
433        &["path"],
434        "diagnostic location",
435    )?;
436    let path = location
437        .get("path")
438        .and_then(Value::as_str)
439        .context("diagnostic location path must be a string")?;
440    let path = RepoPath::from_protocol(path).context("diagnostic location path is invalid")?;
441    if !allowed_paths.contains(&path) {
442        bail!("diagnostic location is outside the requested candidate scope");
443    }
444    if let Some(range) = location.get("range") {
445        validate_range(range)?;
446    }
447    Ok(())
448}

450fn requested_diagnostic_paths(request: &HostRequest) -> Result<BTreeSet<RepoPath>> {
451    let mut candidate = request
452        .workspace
453        .entries
454        .keys()
455        .cloned()
456        .collect::<BTreeSet<_>>();
457    let changes = request
458        .evaluation_params
459        .pointer("/changeset/changes")
460        .and_then(Value::as_array)
461        .context("check/evaluate changes must be an array")?;
462    let mut changed_after = BTreeSet::new();
463    for change in changes {
464        apply_candidate_change(change, &mut candidate, &mut changed_after)?;
465    }
466    select_requested_paths(
467        request.evaluation_params.get("scope"),
468        candidate,
469        changed_after,
470    )
471}

473fn apply_candidate_change(
474    change: &Value,
475    candidate: &mut BTreeSet<RepoPath>,
476    changed_after: &mut BTreeSet<RepoPath>,
477) -> Result<()> {
478    let (object, path, kind) = parse_candidate_change(change)?;
479    match kind {
480        "create" | "modify" => add_candidate_path(candidate, changed_after, path),
481        "delete" => {
482            candidate.remove(&path);
483        }
484        "rename" => apply_candidate_rename(object, candidate, changed_after, path)?,
485        _ => bail!("check/evaluate change kind is invalid"),
486    }
487    Ok(())
488}
489
490fn parse_candidate_change(
491    change: &Value,
492) -> Result<(&serde_json::Map<String, Value>, RepoPath, &str)> {
493    let object = change
494        .as_object()
495        .context("check/evaluate change must be an object")?;
496    let path = object
497        .get("path")
498        .and_then(Value::as_str)
499        .context("check/evaluate change path must be a string")?;
500    let path = RepoPath::from_protocol(path).context("check/evaluate change path is invalid")?;
501    let kind = object
502        .get("kind")
503        .and_then(Value::as_str)
504        .context("check/evaluate change kind is invalid")?;
505    Ok((object, path, kind))
506}

508fn add_candidate_path(
509    candidate: &mut BTreeSet<RepoPath>,
510    changed_after: &mut BTreeSet<RepoPath>,
511    path: RepoPath,
512) {
513    candidate.insert(path.clone());
514    changed_after.insert(path);
515}
516
517fn apply_candidate_rename(
518    object: &serde_json::Map<String, Value>,
519    candidate: &mut BTreeSet<RepoPath>,
520    changed_after: &mut BTreeSet<RepoPath>,
521    path: RepoPath,
522) -> Result<()> {
523    let from = object
524        .get("from")
525        .and_then(Value::as_str)
526        .context("rename change requires from")?;
527    let from = RepoPath::from_protocol(from).context("rename source path is invalid")?;
528    candidate.remove(&from);
529    add_candidate_path(candidate, changed_after, path);
530    Ok(())
531}

533fn select_requested_paths(
534    scope: Option<&Value>,
535    candidate: BTreeSet<RepoPath>,
536    changed_after: BTreeSet<RepoPath>,
537) -> Result<BTreeSet<RepoPath>> {
538    match scope {
539        Some(Value::String(scope)) if scope == "workspace" => Ok(candidate),
540        Some(Value::String(scope)) if scope == "changeset" => Ok(changed_after),
541        Some(Value::Object(scope)) => requested_path_subset(scope, &candidate),
542        _ => bail!("check/evaluate scope is invalid"),
543    }
544}
545
546fn requested_path_subset(
547    scope: &serde_json::Map<String, Value>,
548    candidate: &BTreeSet<RepoPath>,
549) -> Result<BTreeSet<RepoPath>> {
550    let paths = scope
551        .get("paths")
552        .and_then(Value::as_array)
553        .context("path scope requires paths")?;
554    let mut allowed = BTreeSet::new();
555    for path in paths {
556        let path = path
557            .as_str()
558            .context("path scope entries must be strings")?;
559        let path = RepoPath::from_protocol(path).context("path scope entry is invalid")?;
560        if candidate.contains(&path) {
561            allowed.insert(path);
562        }
563    }
564    Ok(allowed)
565}

578fn validate_position(value: Option<&Value>, label: &str) -> Result<()> {
579    let position = required_object(

581        &["line", "char"],
582        &["line", "char"],
583        label,
584    )?;
585    if position.get("line").and_then(Value::as_u64).is_none()
586        || position.get("char").and_then(Value::as_u64).is_none()
587    {
588        bail!("{label} line and char must be non-negative integers");

593fn validate_evidence(value: Option<&Value>) -> Result<()> {
594    let Some(value) = value else {
595        return Ok(());
596    };
597    let evidence = value
598        .as_array()
599        .context("provider evidence must be an array")?;
600    for item in evidence {
601        let item = required_object(
602            item,
603            &["kind", "id", "digest", "uri", "message", "data"],
604            &["kind"],
605            "provider evidence item",
606        )?;
607        require_one_of_strings(
608            item.get("kind"),
609            &["trace", "artifact", "metric", "message", "reference"],
610            "provider evidence kind",
611        )?;
612        for field in ["id", "digest", "uri", "message"] {
613            if let Some(value) = item.get(field) {
614                require_nonempty_string(Some(value), "provider evidence string")?;
615            }
616        }
617        if item.get("data").is_some_and(|data| !data.is_object()) {
618            bail!("provider evidence data must be an object");
619        }
620    }
621    Ok(())
622}

624fn validate_coverage(

626    status: &str,

651            "unsupported",
652            "exhaustive",
653            "truncated",
654        ],
655        "provider coverage",

659fn validate_coverage_claims(
660    coverage: &serde_json::Map<String, Value>,
661    expected: &Value,
662    provider_id: &str,
663) -> Result<()> {
664    validate_coverage_part(coverage.get("requested"), "requested coverage")?;
665    validate_coverage_part(coverage.get("covered"), "covered coverage")?;
666    if coverage.get("requested") != Some(expected) {
667        bail!("provider requested coverage does not echo the host request");
668    }
669    validate_degradations(coverage.get("degraded"), "coverage degraded", provider_id)?;
670    validate_degradations(
671        coverage.get("unsupported"),
672        "coverage unsupported",
673        provider_id,
674    )?;
675    Ok(())
676}

678fn coverage_flags(coverage: &serde_json::Map<String, Value>) -> Result<(bool, bool)> {
679    let exhaustive = coverage
680        .get("exhaustive")
681        .and_then(Value::as_bool)
682        .context("coverage exhaustive must be a boolean")?;
683    let truncated = coverage
684        .get("truncated")
685        .and_then(Value::as_bool)
686        .context("coverage truncated must be a boolean")?;
687    Ok((exhaustive, truncated))
688}

690fn validate_coverage_status(
691    coverage: &serde_json::Map<String, Value>,
692    expected: &Value,
693    status: &str,
694    exhaustive: bool,
695    truncated: bool,
696) -> Result<()> {
697    if status == "complete" {
698        validate_complete_coverage(coverage, expected, exhaustive, truncated)?;
699    } else if exhaustive {
700        bail!("non-complete assessment cannot claim exhaustive coverage");
701    }
702    Ok(())
703}
704
705fn validate_complete_coverage(
706    coverage: &serde_json::Map<String, Value>,
707    expected: &Value,
708    exhaustive: bool,
709    truncated: bool,
710) -> Result<()> {
711    if !exhaustive || truncated || coverage.get("covered") != Some(expected) {
712        bail!("complete assessment did not cover the exact requested scope");
713    }
714    for field in ["degraded", "unsupported"] {
715        if !coverage
716            .get(field)
717            .and_then(Value::as_array)
718            .is_some_and(Vec::is_empty)
719        {
720            bail!("complete assessment contains degraded coverage");
721        }
722    }
723    Ok(())
724}

743fn validate_scope(value: Option<&Value>, label: &str) -> Result<()> {
744    if value
745        .and_then(Value::as_str)
746        .is_some_and(|scope| matches!(scope, "changeset" | "workspace"))
747    {
748        return Ok(());
749    }
750    let scope = required_object(value.unwrap_or(&Value::Null), &["paths"], &["paths"], label)?;
751    let paths = scope
752        .get("paths")
753        .and_then(Value::as_array)
754        .context("path scope paths must be an array")?;
755    if paths.is_empty() {
756        bail!("path scope must not be empty");
757    }
758    let mut unique = BTreeSet::new();
759    for path in paths {
760        let path = path.as_str().context("path scope entry must be a string")?;
761        let path = RepoPath::from_protocol(path).context("path scope entry is invalid")?;
762        if !unique.insert(path) {
763            bail!("path scope contains a duplicate path");
764        }
765    }
766    Ok(())
767}

856fn validate_freshness(
857    value: Option<&Value>,
858    request: &HostRequest,
859    read_blobs: &BTreeSet<String>,
860) -> Result<()> {
861    let freshness = required_object(
862        value.unwrap_or(&Value::Null),
863        &["baseline", "changesetDigest", "blobs"],
864        &["baseline", "changesetDigest", "blobs"],
865        "provider validAsOf",
866    )?;
867    if freshness.get("baseline") != Some(request.workspace.baseline()) {
868        bail!("provider assessment echoed a stale baseline");
869    }
870    let changeset = request
871        .evaluation_params
872        .get("changeset")
873        .context("check/evaluate changeset is missing")?;
874    let expected_digest = format!(
875        "sha256:{}",
876        hex::encode(Sha256::digest(serde_json::to_vec(changeset)?))
877    );
878    require_equal_string(
879        freshness.get("changesetDigest"),
880        &expected_digest,
881        "assessment changeset digest",
882    )?;
883    let cited = string_set(freshness.get("blobs"), "assessment blob read set")?;
884    if &cited != read_blobs {
885        bail!("provider assessment blob read set does not match host callbacks");
886    }
887    Ok(())
888}

964pub(super) fn validate_timing(value: Option<&Value>) -> Result<()> {
965    let timing = required_object(
966        value.unwrap_or(&Value::Null),
967        &["startedAt", "endedAt", "elapsedMs"],
968        &[],
969        "assessment timing",
970    )?;
971    let has_elapsed = if let Some(elapsed) = timing.get("elapsedMs") {
972        if elapsed.as_u64().is_none() {
973            bail!("assessment elapsedMs must be a non-negative integer");
974        }
975        true
976    } else {
977        false
978    };
979    for field in ["startedAt", "endedAt"] {
980        if let Some(value) = timing.get(field) {
981            require_timestamp(Some(value), "assessment timestamp")?;
982        }
983    }
984    if has_elapsed {
985        return Ok(());
986    }
987    require_timestamp(timing.get("startedAt"), "assessment startedAt")?;
988    require_timestamp(timing.get("endedAt"), "assessment endedAt")
989}

1019        "decision",

1057fn required_object<'a>(
1058    value: &'a Value,
1059    allowed: &[&str],
1060    required: &[&str],
1061    label: &str,
1062) -> Result<&'a serde_json::Map<String, Value>> {
1063    let object = exact_object(value, allowed, label)?;
1064    if let Some(field) = required.iter().find(|field| !object.contains_key(**field)) {
1065        bail!("{label} omitted required field {field}");
1066    }
1067    Ok(object)
1068}

1070fn string_set(value: Option<&Value>, label: &str) -> Result<BTreeSet<String>> {
1071    let values = value
1072        .and_then(Value::as_array)
1073        .with_context(|| format!("{label} must be an array"))?;
1074    let mut unique = BTreeSet::new();
1075    for value in values {
1076        let value = value
1077            .as_str()
1078            .with_context(|| format!("{label} entries must be strings"))?;
1079        if value.is_empty() {
1080            bail!("{label} entries must not be empty");
1081        }
1082        if !unique.insert(value.to_owned()) {
1083            bail!("{label} contains a duplicate entry");
1084        }
1085    }
1086    Ok(unique)
1087}

1089fn require_exact_strings(value: Option<&Value>, expected: &[&str], label: &str) -> Result<()> {
1090    let actual = string_set(value, label)?;
1091    let expected = expected
1092        .iter()
1093        .map(|value| (*value).to_owned())
1094        .collect::<BTreeSet<_>>();
1095    if actual != expected {
1096        bail!("{label} does not match the bundled profile");
1097    }
1098    Ok(())
1099}
