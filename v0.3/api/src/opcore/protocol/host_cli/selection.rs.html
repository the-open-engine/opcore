10    pub(super) fn for_profile(
11        &self,
12        profile: ProviderProfile,
13        args: &CheckArgs,
14        policy: &PolicySnapshot,
15    ) -> Result<Self> {
16        let configured = match profile {
17            ProviderProfile::Fast => None,
18            ProviderProfile::RustNative => Some(&policy.policy.providers.rust_native.roots),
19            ProviderProfile::NodeNative => Some(&policy.policy.providers.node_native.roots),
20            ProviderProfile::PythonNative => Some(&policy.policy.providers.python_native.roots),
21        };
22        let roots = if args.roots.is_empty() {
23            configured.map_or_else(
24                || Ok(self.roots.clone()),
25                |roots| {
26                    anyhow::ensure!(
27                        !roots.is_empty(),
28                        "{} roots is empty; select a project root or remove the provider from this workflow",
29                        profile.provider_id()
30                    );
31                    selected_roots(roots)
32                },
33            )?
34        } else {
35            self.roots.clone()
36        };
37        let (excluded_roots, roots) = roots.into_iter().partition(|root: &WorkspaceRoot| {
38            root.prefix.as_ref().is_some_and(|prefix| {
39                policy
40                    .policy
41                    .targets
42                    .exclude
43                    .iter()
44                    .any(|excluded| path_has_prefix(prefix.as_bytes(), excluded.as_bytes()))
45            })
46        });
47        Ok(Self {
48            profiles: vec![profile],
49            roots,
50            excluded_roots,
51            skip_inapplicable: self.skip_inapplicable,
52            comparison: self.comparison,
53        })
54    }
55}

84pub(super) fn validate_native_exclusions(
85    paths: &BTreeSet<RepoPath>,
86    selection: &HostSelection,
87    targets: &TargetPolicy,
88) -> Result<()> {
89    if !selection.profiles.iter().any(|profile| profile.is_native()) {
90        return Ok(());
91    }
92    for excluded in &targets.exclude {
93        let affected = paths.iter().find(|path| {
94            path_has_prefix(path.as_bytes(), excluded.as_bytes())
95                && selection
96                    .roots
97                    .iter()
98                    .any(|root| path_in_workspace_root(path, root))
99        });
100        anyhow::ensure!(
101            affected.is_none(),
102            concat!(
103                "native project contains excluded input {}; the compiler requires its complete project context. ",
104                "Select separate projects with providers.<name>.roots or configure the compiler's own source selection"
105            ),
106            affected.map_or_else(String::new, ToString::to_string)
107        );
108    }
109    Ok(())
110}

112pub(super) fn selected_profiles(args: &CheckArgs) -> Result<(Vec<ProviderProfile>, bool)> {
113    if args.native {
114        return Ok((
115            vec![ProviderProfile::Fast, ProviderProfile::RustNative],
116            true,
117        ));
118    }
119    if args.providers.is_empty() {
120        return Ok((vec![ProviderProfile::Fast], false));
121    }
122    let all = args.providers.contains(&CheckProvider::All);
123    anyhow::ensure!(
124        !all || args.providers.len() == 1,
125        "--providers all cannot be combined with another provider"
126    );
127    if all {
128        return Ok((ProviderProfile::ALL.to_vec(), true));
129    }
130    let mut profiles = Vec::new();
131    for selection in &args.providers {
132        let profile = selection
133            .profile()
134            .context("provider selection unexpectedly omitted its profile")?;
135        if !profiles.contains(&profile) {
136            profiles.push(profile);
137        }
138    }
139    Ok((profiles, false))
140}

142pub(super) fn selected_roots(values: &[String]) -> Result<Vec<WorkspaceRoot>> {
143    if values.is_empty() {
144        return Ok(vec![WorkspaceRoot {
145            label: ".".into(),
146            prefix: None,
147        }]);
148    }
149    let mut roots = Vec::new();
150    for value in values {
151        let normalized = value.trim_end_matches('/');
152        let root = if normalized == "." {
153            WorkspaceRoot {
154                label: ".".into(),
155                prefix: None,
156            }
157        } else {
158            let prefix = RepoPath::from_protocol(normalized)
159                .with_context(|| format!("validate selected root {value}"))?;
160            WorkspaceRoot {
161                label: prefix.to_string(),
162                prefix: Some(prefix),
163            }
164        };
165        anyhow::ensure!(
166            !roots.iter().any(|existing| roots_overlap(existing, &root)),
167            "selected roots overlap at {}",
168            root.label
169        );
170        roots.push(root);
171    }
172    Ok(roots)
173}

175pub(super) fn roots_overlap(left: &WorkspaceRoot, right: &WorkspaceRoot) -> bool {
176    match (&left.prefix, &right.prefix) {
177        (None, _) | (_, None) => true,
178        (Some(left), Some(right)) => {
179            path_has_prefix(left.as_bytes(), right.as_bytes())
180                || path_has_prefix(right.as_bytes(), left.as_bytes())
181        }
182    }
183}

185pub(super) fn rebase_path(path: &RepoPath, root: &WorkspaceRoot) -> Option<RepoPath> {
186    let Some(prefix) = &root.prefix else {
187        return Some(path.clone());
188    };
189    let suffix = path.as_bytes().strip_prefix(prefix.as_bytes())?;
190    let suffix = suffix.strip_prefix(b"/")?;
191    RepoPath::new(suffix.to_vec()).ok()
192}

202pub(super) fn project_context_from_paths(
203    paths: &BTreeSet<RepoPath>,
204    root: &WorkspaceRoot,
205    profile: ProviderProfile,
206) -> WorkspaceRoot {
207    match profile {
208        ProviderProfile::NodeNative => ancestor_roots(root)
209            .into_iter()
210            .find(|candidate| node_context(paths, candidate))
211            .unwrap_or_else(|| root.clone()),
212        ProviderProfile::PythonNative => ancestor_roots(root)
213            .into_iter()
214            .find(|candidate| python_context(paths, candidate))
215            .unwrap_or_else(|| root.clone()),
216        ProviderProfile::RustNative => WorkspaceRoot {
217            label: ".".into(),
218            prefix: None,
219        },
220        ProviderProfile::Fast => root.clone(),
221    }
222}

224pub(super) fn project_capture_roots(
225    paths: &BTreeSet<RepoPath>,
226    selection: &HostSelection,
227) -> Vec<RepoPath> {
228    let mut roots = Vec::new();
229    for profile in &selection.profiles {
230        for root in &selection.roots {
231            let context = if *profile == ProviderProfile::RustNative {
232                root.clone()
233            } else {
234                project_context_from_paths(paths, root, *profile)
235            };
236            let Some(prefix) = context.prefix else {
237                return Vec::new();
238            };
239            roots.push(prefix);
240        }
241    }
242    roots.sort();
243    roots.dedup();
244    let mut minimal: Vec<RepoPath> = Vec::new();
245    for root in roots {
246        if minimal
247            .iter()
248            .all(|parent| !path_has_prefix(root.as_bytes(), parent.as_bytes()))
249        {
250            minimal.push(root);
251        }
252    }
253    minimal
254}

256pub(super) fn ancestor_roots(root: &WorkspaceRoot) -> Vec<WorkspaceRoot> {
257    let Some(prefix) = &root.prefix else {
258        return vec![root.clone()];
259    };
260    let mut candidates = Vec::new();
261    let mut current = prefix.as_bytes();
262    loop {
263        candidates.push(WorkspaceRoot {
264            label: String::from_utf8_lossy(current).into_owned(),
265            prefix: RepoPath::new(current.to_vec()).ok(),
266        });
267        let Some(index) = current.iter().rposition(|byte| *byte == b'/') else {
268            break;
269        };
270        current = &current[..index];
271    }
272    candidates.push(WorkspaceRoot {
273        label: ".".into(),
274        prefix: None,
275    });
276    candidates
277}

279pub(super) fn node_context(paths: &BTreeSet<RepoPath>, root: &WorkspaceRoot) -> bool {
280    has_context_file(paths, root, b"package.json")
281        && (has_context_file(paths, root, b"package-lock.json")
282            || has_context_file(paths, root, b"npm-shrinkwrap.json"))
283        && paths
284            .iter()
285            .any(|path| context_filename(path, root).is_some_and(is_tsconfig_name))
286}

288pub(super) fn python_context(paths: &BTreeSet<RepoPath>, root: &WorkspaceRoot) -> bool {
289    [
290        b"pyproject.toml".as_slice(),
291        b"pyrightconfig.json",
292        b"mypy.ini",
293        b"setup.cfg",
294        b"tox.ini",
295    ]
296    .iter()
297    .any(|name| has_context_file(paths, root, name))
298}

300pub(super) fn has_context_file(
301    paths: &BTreeSet<RepoPath>,
302    root: &WorkspaceRoot,

305    paths

307        .any(|path| context_filename(path, root) == Some(name))

310pub(super) fn context_filename<'a>(path: &'a RepoPath, root: &WorkspaceRoot) -> Option<&'a [u8]> {
311    let bytes = path.as_bytes();
312    match &root.prefix {
313        Some(prefix) => bytes
314            .strip_prefix(prefix.as_bytes())?
315            .strip_prefix(b"/")
316            .filter(|suffix| !suffix.contains(&b'/')),
317        None => (!bytes.contains(&b'/')).then_some(bytes),
318    }
319}

321pub(super) fn is_tsconfig_name(name: &[u8]) -> bool {
322    name == b"tsconfig.json" || (name.starts_with(b"tsconfig.") && name.ends_with(b".json"))
323}

325pub(super) fn workspace_mapping(
326    capture: &GitNativeCapture,
327    focus: &WorkspaceRoot,
328    context: &WorkspaceRoot,
329    profile: ProviderProfile,
330) -> Result<std::collections::BTreeMap<RepoPath, RepoPath>> {
331    let mut mapping = std::collections::BTreeMap::new();
332    let mut destinations = BTreeSet::new();
333    for path in capture.paths.keys() {
334        let include = (profile.is_native() && path_in_workspace_root(path, context))
335            || path_in_workspace_root(path, focus)
336            || context_filename(path, context)
337                .is_some_and(|name| provider_context_file(profile, name));
338        if !include {
339            continue;
340        }
341        let destination = rebase_path(path, context)
342            .with_context(|| format!("map {path} into project context {}", context.label))?;
343        anyhow::ensure!(
344            destinations.insert(destination.clone()),
345            "project context maps multiple files onto {destination}"
346        );
347        mapping.insert(path.clone(), destination);
348    }
349    Ok(mapping)
350}

352pub(super) fn relative_root(root: &WorkspaceRoot, context: &WorkspaceRoot) -> Option<RepoPath> {
353    match (&root.prefix, &context.prefix) {
354        (Some(root), None) => Some(root.clone()),
355        (Some(root), Some(context)) if root != context => root
356            .as_bytes()
357            .strip_prefix(context.as_bytes())?
358            .strip_prefix(b"/")
359            .and_then(|path| RepoPath::new(path.to_vec()).ok()),
360        _ => None,
361    }
362}

364pub(super) fn path_in_workspace_root(path: &RepoPath, root: &WorkspaceRoot) -> bool {
365    root.prefix
366        .as_ref()
367        .is_none_or(|prefix| path_has_prefix(path.as_bytes(), prefix.as_bytes()))
368}

370pub(super) fn provider_context_file(profile: ProviderProfile, name: &[u8]) -> bool {
371    match profile {
372        ProviderProfile::NodeNative => {
373            matches!(
374                name,
375                b"package.json" | b"package-lock.json" | b"npm-shrinkwrap.json" | b".npmrc"
376            ) || is_tsconfig_name(name)
377        }
378        ProviderProfile::PythonNative => matches!(
379            name,
380            b"pyproject.toml" | b"pyrightconfig.json" | b"mypy.ini" | b"setup.cfg" | b"tox.ini"
381        ),
382        ProviderProfile::Fast | ProviderProfile::RustNative => false,
383    }
384}

386pub(super) fn path_has_prefix(path: &[u8], prefix: &[u8]) -> bool {
387    path == prefix
388        || path
389            .strip_prefix(prefix)
390            .is_some_and(|suffix| suffix.starts_with(b"/"))
